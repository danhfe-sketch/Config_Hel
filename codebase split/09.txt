                'changed_parameters': [p for p in updated_parameters if p['changed']],
                'unchanged_parameters': [p for p in updated_parameters if not p['changed']],
                'session_id': f"config_read_{timestamp.strftime('%Y%m%d_%H%M%S')}"
            }
            
            # Store in instance for potential export
            if not hasattr(self, 'update_sessions'):
                self.update_sessions = []
            
            self.update_sessions.append(session_info)
            
            # Keep only the last 10 sessions to avoid memory issues
            if len(self.update_sessions) > 10:
                self.update_sessions = self.update_sessions[-10:]
                
        except Exception as e:
            self._log_message(f"WARNING: Failed to store update session info: {str(e)}")

    def _update_parameter_via_specification(self, section_name: str, param_name: str, value: Any) -> bool:
        """
        Update a parameter using the specification-based method.
        
        Args:
            section_name: Name of the parameter section
            param_name: Name of the parameter
            value: Value to set
            
        Returns:
            bool: True if update was successful, False otherwise
        """
        try:
            # Check if specification factory is available
            if not hasattr(self.app.parameters_frame, 'parameter_specification_factory'):
                self._log_message(f"DEBUG: No specification factory available for {section_name}.{param_name}")
                return False
            
            # Check if section widgets are available
            if not hasattr(self.app.parameters_frame, 'section_widgets'):
                self._log_message(f"DEBUG: No section widgets available for {section_name}.{param_name}")
                return False
            
            factory = self.app.parameters_frame.parameter_specification_factory
            section_widgets = self.app.parameters_frame.section_widgets
            
            # Check if the section exists
            if section_name not in section_widgets:
                self._log_message(f"DEBUG: Section '{section_name}' not found in section widgets")
                return False
            
            # Check if the parameter exists in the section
            section_widget = section_widgets[section_name]
            if param_name not in section_widget.parameters:
                self._log_message(f"DEBUG: Parameter '{param_name}' not found in section '{section_name}'")
                return False
            
            # Get current value for comparison
            current_value = factory.get_parameter_value(section_name, param_name, section_widgets)
            
            # Set the new value
            def update_parameter():
                try:
                    factory.set_parameter_value(section_name, param_name, value, section_widgets)
                    
                    # Verify the update
                    new_value = factory.get_parameter_value(section_name, param_name, section_widgets)
                    
                    # Log the update
                    self._log_message(f"DEBUG: Parameter {section_name}.{param_name} updated: '{current_value}' -> '{new_value}'")
                    
                    # Add visual indicator if available
                    self._add_visual_update_indicator(section_name, param_name, current_value != new_value)
                    
                except Exception as e:
                    self._log_message(f"ERROR: Failed to set parameter {section_name}.{param_name}: {str(e)}")
                    raise
            
            # Schedule the update on the main thread
            self.app.after(0, update_parameter)
            return True
            
        except Exception as e:
            self._log_message(f"ERROR: Failed to update parameter {section_name}.{param_name} via specification: {str(e)}")
            return False

    def _update_widget_value(self, widget_name: str, value: Any):
        """
        Update a specific widget value using legacy method.
        
        Args:
            widget_name: Name of the widget to update
            value: Value to set
        """
        try:
            self._log_message(f"DEBUG: Attempting legacy widget update for {widget_name} = {value}")
            
            # Try parameter_widgets first (legacy approach)
            if hasattr(self.app.parameters_frame, 'parameter_widgets'):
                widgets = self.app.parameters_frame.parameter_widgets
                if widget_name in widgets:
                    widget = widgets[widget_name]
                    if hasattr(widget, 'set_value'):
                        def update_widget():
                            try:
                                widget.set_value(value)
                                self._log_message(f"DEBUG: Updated {widget_name} to {value} via parameter_widgets")
                            except Exception as e:
                                self._log_message(f"ERROR: Failed to update {widget_name} via parameter_widgets: {str(e)}")
                        
                        self.app.after(0, update_widget)
                        return
                    else:
                        self._log_message(f"DEBUG: Widget {widget_name} found but has no set_value method")
                else:
                    self._log_message(f"DEBUG: Widget {widget_name} not found in parameter_widgets")
            else:
                self._log_message(f"DEBUG: No parameter_widgets attribute found")
            
            # If legacy approach failed, log the issue
            self._log_message(f"WARNING: Could not update widget {widget_name} - widget not found or not accessible")
            
        except Exception as e:
            self._log_message(f"ERROR: Failed to update widget {widget_name}: {str(e)}")

    def read_specific_parameter(self, param_name: str, command: str = None, 
                              command_type: str = None, max_retries: int = 2) -> ParameterReadResult:
        """
        Read a specific parameter from the device with retry logic.
        
        Args:
            param_name: Name of the parameter to read
            command: AT command to send (optional, will be looked up if not provided)
            command_type: Type of command for response parsing
            max_retries: Maximum number of retry attempts for failed reads
            
        Returns:
            ParameterReadResult with the reading result
        """
        if command is None:
            # Look up the command for this parameter
            if param_name in self.parameter_query_commands:
                query_info = self.parameter_query_commands[param_name]
                if isinstance(query_info, tuple):
                    command, command_type = query_info
                else:
                    return ParameterReadResult(
                        parameter_name=param_name,
                        success=False,
                        value=None,
                        error_message=f"Invalid command configuration for {param_name}",
                        raw_response=""
                    )
            else:
                return ParameterReadResult(
                    parameter_name=param_name,
                    success=False,
                    value=None,
                    error_message=f"Unknown parameter: {param_name}",
                    raw_response=""
                )
        
        last_error = None
        last_response = ""
        
        # Attempt to read the parameter with retries
        for attempt in range(max_retries + 1):
            try:
                if attempt > 0:
                    # Log retry attempt
                    self._log_message(f"INFO: Retrying {param_name} read (attempt {attempt + 1}/{max_retries + 1})")
                    # Add a small delay between retries
                    time.sleep(0.5)
                
                # Send the command and wait for response
                # Use longer timeout for TIMER command as it may take more time to respond
                timeout = 8.0 if command == 'TIMER' else 5.0
                response = self._send_command_and_wait(command, timeout=timeout)
                
                if response is None:
                    last_error = self.i18n.get('config_read_timeout', 'Command timeout')
                    continue
                
                last_response = response
                
                # Parse the response
                parsed_response = self.parse_device_response(command, response, command_type)

                # Diagnostic logs: capture parser output right after parsing
                try:
                    self._log_message(f"DEBUG: after_parse - param_name={param_name}, command={command}, command_type={command_type}, parsed_success={parsed_response.success}, parsed_values_keys={list(parsed_response.parsed_values.keys()) if isinstance(parsed_response.parsed_values, dict) else []}")
                except Exception as _e:
                    self._log_message(f"DEBUG: after_parse - failed to log parsed response: {_e}")
                if parsed_response.success and parsed_response.parsed_values:
                    # Extract the main value for this parameter
                    value = self._extract_parameter_value(param_name, parsed_response.parsed_values)
                    
                    # Log successful read after retries
                    if attempt > 0:
                        self._log_message(f"INFO: Successfully read {param_name} after {attempt + 1} attempts")
                    
                    return ParameterReadResult(
                        parameter_name=param_name,
                        success=True,
                        value=value,
                        error_message=None,
                        raw_response=response
                    )
                else:
                    last_error = parsed_response.error_message or "Failed to parse response"
                    
            except Exception as e:
                last_error = str(e)
                self._log_message(f"WARNING: Exception during {param_name} read attempt {attempt + 1}: {last_error}")
        
        # All attempts failed, return the final error
        final_error_msg = self._get_specific_error_message(param_name, last_error or "Unknown error")
        
        return ParameterReadResult(
            parameter_name=param_name,
            success=False,
            value=None,
            error_message=final_error_msg,
            raw_response=last_response
        )

    def parse_device_response(self, command: str, response: str, command_type: str) -> ParsedResponse:
        """
        Parse device response based on the command type.
        
        Args:
            command: The command that was sent
            response: The device response string
            command_type: The type of command for parsing
            
        Returns:
            ParsedResponse with parsed information
        """
        parsed_values = {}
        error_message = None
        success = False
        
        try:
            # Get the appropriate pattern for this command type
            pattern = self.response_patterns.get(command_type)
            
            if not pattern:
                error_message = f"No parsing pattern defined for command type: {command_type}"
                return ParsedResponse(command, False, {}, response, error_message)
            
            # Apply pattern matching
            match = re.search(pattern, response)
            
            if match:
                success = True
                
                # Parse based on command type
                if command_type == 'get_imei':
                    parsed_values['imei'] = match.group(1)
                    
                elif command_type == 'get_iccid':
                    parsed_values['iccid'] = match.group(1)
                    
                elif command_type == 'get_version':
                    parsed_values['version'] = match.group(1).strip()
                    
                elif command_type == 'get_status':
                    # Parse status response with multiple key:value pairs
                    status_text = match.group(1)
                    for section in status_text.split(';'):
                        if ':' in section:
                            key, value = section.split(':', 1)
                            parsed_values[key.strip().lower()] = value.strip()
                
                # Network Parameters
                elif command_type == 'get_apn':
                    parsed_values['apn'] = match.group(1)
                    parsed_values['user'] = match.group(2)
                    parsed_values['password'] = match.group(3).rstrip('#')
                    
                elif command_type == 'get_server':
                    parsed_values['server_type'] = match.group(1)
                    parsed_values['address'] = match.group(2)
                    parsed_values['port'] = match.group(3)
                    parsed_values['protocol'] = match.group(4)
                
                # Time Parameters
                elif command_type == 'get_timer':
                    # Handle multiple response formats
                    if match.group(1) and match.group(2):  # TIMER:x,y format

    def _parse_all_parameters_response(self, response: str, parsed_values: Dict[str, Any]) -> None:
        """Parse the comprehensive PARAM# response."""
        # Example: "IMEI:357457040000821;TIMER:20,300;SEMDS:3;HBT:180Sec;Defense:3;"
        for section in response.split(';'):
            if ':' in section:
                key, value = section.split(':', 1)
                key = key.strip()
                value = value.strip()
                
                if key == 'IMEI':
                    parsed_values['imei'] = value
                elif key == 'TIMER':
                    parts = value.split(',')
                    if len(parts) >= 2:
                        parsed_values['timer_ign_on'] = parts[0]
                        parsed_values['timer_ign_off'] = parts[1]
                elif key == 'HBT':
                    # Handle format like "180Sec"
                    hbt_value = re.sub(r'[^\d]', '', value)  # Extract only digits
                    parsed_values['heartbeat_interval'] = hbt_value
                elif key == 'Defense':
                    parsed_values['defense_time'] = value

    def _send_command_and_wait(self, command: str, timeout: float = 5.0) -> Optional[str]:
        """
        Send a command and wait for a response.
        
        Args:
            command: The raw command to send (will be formatted with ATSMS prefix)
            timeout: Timeout in seconds
            
        Returns:
            The device response or None if timeout
        """
        # Check if we have access to the serial service
        if not hasattr(self.app, 'serial_service'):
            self._log_message("ERROR: Cannot send command - serial_service not found")
            return None
        
        try:
            # Format the command with ATSMS prefix for proper LT32Pro USB communication
            if self.command_formatter.is_read_command_supported(command):
                formatted_command = self.command_formatter.format_read_command(command)
                self._log_message(f"DEBUG: Formatted command '{command}' as '{formatted_command}'")
            else:
                # Fallback for unsupported commands - add ATSMS prefix manually
                formatted_command = f"ATSMS:{command}"
                self._log_message(f"DEBUG: Using fallback formatting for '{command}' as '{formatted_command}'")
            
            # Validate command syntax
            self.command_formatter.validate_command_syntax(formatted_command)
            
        except Exception as e:
            self._log_message(f"ERROR: Failed to format command '{command}': {str(e)}")
            return None
        
        # Set up response capture
        self.waiting_for_response = True
        self.captured_response = None
        self.response_timeout = False
        
        # Send the formatted command
        success = self.app.serial_service.send_command(formatted_command, command)
        if not success:
            self._log_message(f"ERROR: Failed to send formatted command: {formatted_command}")
            self.waiting_for_response = False
            return None
        
        self._log_message(f"DEBUG: Successfully sent formatted command: {formatted_command}")
        
        # Wait for a response
        start_time = time.time()
        while time.time() - start_time < timeout:
            if self.captured_response is not None:
                response = self.captured_response
                self.waiting_for_response = False
                self.captured_response = None
                self._log_message(f"DEBUG: Received response for '{command}': {response}")
                return response
            
            # Sleep for a short time
            time.sleep(0.1)
        
        # Timeout
        self.waiting_for_response = False
        self.response_timeout = True
        self._log_message(f"WARNING: Timeout waiting for response to '{command}' (formatted as '{formatted_command}')")
        return None
    
    def capture_device_response(self, response: str) -> None:
        """
        Capture a device response for the configuration reader.
        
        Args:
            response: The device response string
        """
        # Always log the response for debugging
        self._log_message(f"DEBUG: Received device response: '{response}'")
        
        # Check for cancellation before capturing response
        if self.cancel_requested:
            self._log_message(f"DEBUG: Ignoring response due to cancellation: '{response}'")
            return
        
        # Capture response if we're waiting for one
        if self.waiting_for_response:
            if self.captured_response is None:
                self.captured_response = response.strip()
                self._log_message(f"DEBUG: Captured response for processing: '{self.captured_response}'")
            else:
                # We already have a response - this might be a duplicate or delayed response
                self._log_message(f"DEBUG: Already have captured response '{self.captured_response}', received additional response: '{response}'")
                # Don't overwrite the first response to avoid timing issues
        else:
            # Log responses we're not expecting (helps with debugging)
            self._log_message(f"DEBUG: Ignoring unsolicited response (not waiting): '{response}'")

        """
        retry_count = 0
        last_error = None
        
        while retry_count <= max_retries:
            # Check for cancellation
            if self.cancel_requested:
                return ParameterReadResult(
                    parameter_name=param_name,
                    success=False,
                    value=None,
                    error_message="Operation canceled by user",
                    raw_response=""
                )
            
            try:
                # Validate connection before each attempt
                if self.parameter_error_handler and retry_count > 0:
                    validation_result = self.parameter_error_handler.validate_connection_before_operation(f"reading {param_name}")
                    if not validation_result.is_valid:
                        return ParameterReadResult(
                            parameter_name=param_name,
                            success=False,
                            value=None,
                            error_message=f"Connection validation failed: {validation_result.error_message}",
                            raw_response=""
                        )
                
                # Format and send command with timeout handling
                start_time = time.time()
                
                try:
                    # Check for cancellation before sending command
                    if self.cancel_requested:
                        return ParameterReadResult(
                            parameter_name=param_name,
                            success=False,
                            value=None,
                            error_message="Operation canceled by user",
                            raw_response=""
                        )
                    
                    # Format the command with ATSMS prefix for proper LT32Pro USB communication
                    if self.command_formatter.is_read_command_supported(command):
                        formatted_command = self.command_formatter.format_read_command(command)
                        self._log_message(f"DEBUG: Formatted command '{command}' as '{formatted_command}' for {param_name}")
                    else:
                        # Fallback for unsupported commands - add ATSMS prefix manually
                        formatted_command = f"ATSMS:{command}#"
                        self._log_message(f"DEBUG: Using fallback formatting for '{command}' as '{formatted_command}' for {param_name}")
                    
                    # Validate command syntax
                    self.command_formatter.validate_command_syntax(formatted_command)
                    
                    # Send the formatted command
                    success = self.app.serial_service.send_command(formatted_command, command_type)
                    if not success:
                        raise RuntimeError(f"Failed to send formatted command: {formatted_command}")
                    
                    self._log_message(f"DEBUG: Successfully sent formatted command for {param_name}: {formatted_command}")
                    
                except Exception as e:
                    return ParameterReadResult(
                        parameter_name=param_name,
                        success=False,
                        value=None,
                        error_message=f"Command formatting error: {str(e)}",
                        raw_response=""
                    )
                
                # Wait for response with timeout
                response = self._wait_for_response_with_timeout(param_name, command, timeout=10.0)
                read_duration = time.time() - start_time
                
                # Check for cancellation after waiting for response
                if self.cancel_requested:
                    return ParameterReadResult(
                        parameter_name=param_name,
                        success=False,
                        value=None,
                        error_message="Operation canceled by user",
                        raw_response=""
                    )
                
                if response is None:
                    # Handle timeout
                    if self.parameter_error_handler:
                        error = self.parameter_error_handler.handle_communication_timeout(
                            param_name, command, read_duration
                        )
                        last_error = error
                        
                        # Check if we should retry
                        if retry_count < max_retries and read_duration < 15.0:  # Only retry for shorter timeouts
                            retry_count += 1
                            self._log_message(f"INFO: Retrying {param_name} (attempt {retry_count + 1}/{max_retries + 1})")
                            time.sleep(1.0)  # Brief delay before retry
                            continue
                    return ParameterReadResult(
                        parameter_name=param_name,
                        success=False,
                        value=None,
                        error_message=f"Communication timeout after {read_duration:.1f}s",
                        raw_response=""
                    )
                
                # Parse response
                parsed_response = self.parse_device_response(param_name, response, command_type)
                
                if parsed_response.success:
                    # Successful read
                    value = self._extract_parameter_value(param_name, parsed_response.parsed_values)
                    return ParameterReadResult(
                        parameter_name=param_name,
                        success=True,
                        value=value,
                        error_message=None,
                        raw_response=response
                    )
                else:
                    # Handle parsing error
                    if self.parameter_error_handler:
                        error = self.parameter_error_handler.handle_invalid_response(
                            param_name, command, response, 
                            expected_format=self._get_expected_format(command_type),
                            parsing_error=parsed_response.error_message or "Unknown parsing error"
                        )
                        last_error = error
                        
                        # Check if we should retry for parsing errors
                        if retry_count < max_retries and "format" not in parsed_response.error_message.lower():
                            retry_count += 1
                            self._log_message(f"INFO: Retrying {param_name} due to parsing error (attempt {retry_count + 1}/{max_retries + 1})")
                            time.sleep(0.5)  # Brief delay before retry
                            continue
                    
                    return ParameterReadResult(
                        parameter_name=param_name,
                        success=False,
                        value=None,
                        error_message=f"Invalid response: {parsed_response.error_message}",
                        raw_response=response
                    )
                    
            except Exception as e:
                # Handle unexpected errors
                if self.parameter_error_handler:
                    error_details = {
                        'exception': str(e),
                        'command_sent': command,
                        'retry_count': retry_count
                    }
                    error = self.parameter_error_handler.handle_parameter_operation_error(
                        'read', param_name, error_details
                    )
                    last_error = error
                
                if retry_count < max_retries:
                    retry_count += 1
                    self._log_message(f"WARNING: Exception reading {param_name}, retrying (attempt {retry_count + 1}/{max_retries + 1}): {str(e)}")
                    time.sleep(1.0)
                    continue
                else:
                    return ParameterReadResult(
                        parameter_name=param_name,
                        success=False,
                        value=None,
                        error_message=f"Unexpected error: {str(e)}",
                        raw_response=""
                    )
        
        # All retries exhausted
        error_msg = f"Failed after {max_retries + 1} attempts"
        if last_error:
            error_msg += f": {last_error.error_message}"
        
        return ParameterReadResult(
            parameter_name=param_name,
            success=False,
            value=None,
            error_message=error_msg,
            raw_response=""
        )
    
    def _wait_for_response_with_timeout(self, param_name: str, command: str, timeout: float = 10.0) -> Optional[str]:
        """
        Wait for device response with timeout and enhanced error handling.
        
        Args:
            param_name: Name of parameter being read
            command: Command that was sent
            timeout: Timeout in seconds
            
        Returns:
            Response string or None if timeout or cancellation
        """
        start_time = time.time()
        self.waiting_for_response = True
        self.captured_response = None
        self.response_timeout = False
        
        # Wait for response with periodic connection checks and cancellation checks
        while time.time() - start_time < timeout:
            # Check for cancellation
            if self.cancel_requested:
                self._log_message(f"INFO: Cancellation requested while waiting for {param_name} response")
                self.waiting_for_response = False
                self.response_timeout = True
                self.captured_response = None
                return None
            
            # Check if we have a captured response - this is the primary success condition
            if self.captured_response is not None:
                response = self.captured_response
                elapsed_time = time.time() - start_time
                self._log_message(f"DEBUG: Successfully received response for {param_name} in {elapsed_time:.2f}s: '{response}'")
                self.waiting_for_response = False
                self.captured_response = None
                return response
            
            # Check if connection is still valid every 2 seconds
            elapsed = time.time() - start_time
            if elapsed > 2.0 and int(elapsed) % 2 == 0:
                if not hasattr(self.app, 'serial_service') or not self.app.serial_service.is_connected:
                    self._log_message(f"WARNING: Connection lost while waiting for {param_name} response")
                    break
            
            time.sleep(0.05)  # Reduced sleep time for faster response detection
        
        # Final check for response before declaring timeout
        if self.captured_response is not None:
            response = self.captured_response
            elapsed_time = time.time() - start_time
            self._log_message(f"DEBUG: Response received just before timeout for {param_name} in {elapsed_time:.2f}s: '{response}'")
            self.waiting_for_response = False
            self.captured_response = None
            return response
        
        # Check for cancellation one more time before declaring timeout
        if self.cancel_requested:
            self._log_message(f"INFO: Cancellation requested before timeout for {param_name} response")
            self.waiting_for_response = False
            self.response_timeout = True
            self.captured_response = None
            return None
        
        # Timeout occurred
        elapsed_time = time.time() - start_time
        self._log_message(f"WARNING: Timeout waiting for {param_name} response after {elapsed_time:.2f}s (threshold: {timeout}s)")
        self.waiting_for_response = False
        self.response_timeout = True
        self.captured_response = None
        return None
    
    def _get_expected_format(self, command_type: str) -> str:
        """
        Get the expected response format for a command type.
        
        Args:
            command_type: Type of command
            
        Returns:
            Expected format string
        """
        format_map = {
            'get_apn': 'APN:server,user,password',
            'get_server': 'SERVER:type,address,port,protocol',
            'get_timer': 'TIMER:moving_time,stopped_time',
            'get_hbt': 'HBT:moving_heartbeat,stopped_heartbeat',
            'get_protocol': 'GT06SEL=value',
            'get_tcp_ack': 'POSBYACK=value',
            'get_accline': 'ACCLINE=value',
            'get_acc_cfg': 'ACC_CFG=type-on_voltage-off_voltage'
        }
        return format_map.get(command_type, 'Unknown format')

    def _log_message(self, message: str) -> None:
        """Log a message to the application log."""
        if hasattr(self.app, 'log_frame'):
            self.app.log_frame.log_message(message + "\n")
        else:
            print(message)

    def read_parameter_section(self, section_name: str, progress_callback: Optional[Callable] = None,
                             completion_callback: Optional[Callable] = None) -> None:
        """
        Read all parameters from a specific section.
        
        Args:
            section_name: Name of the parameter section to read
            progress_callback: Optional callback for progress updates
            completion_callback: Optional callback when reading is complete
        """
        if self.is_reading:
            error_msg = self.i18n.get('config_read_already_in_progress', 
                                    'Configuration reading is already in progress')
            if completion_callback:
                result = ConfigurationReadResult(
                    success=False,
                    parameters_read=[],
                    total_parameters=0,
                    failed_parameters=[],
                    read_duration=0.0,
                    timestamp=datetime.now()
                )
                completion_callback(result, error_msg)
            return

        # Get section parameters
        section_commands = self._get_section_parameter_commands(section_name)
        if not section_commands:
            error_msg = f"Unknown parameter section: {section_name}"
            if completion_callback:
                result = ConfigurationReadResult(
                    success=False,
                    parameters_read=[],
                    total_parameters=0,
                    failed_parameters=[],
                    read_duration=0.0,
                    timestamp=datetime.now()
                )
                completion_callback(result, error_msg)
            return

        self.progress_callback = progress_callback
        self.completion_callback = completion_callback
        self.cancel_requested = False
        self.is_reading = True

        # Start reading in a separate thread
        self.read_thread = threading.Thread(
            target=self._read_section_thread, 
            args=(section_name, section_commands), 
            daemon=True
        )
        self.read_thread.start()

    def _read_section_thread(self, section_name: str, section_commands: List[tuple]) -> None:
        """Thread function for reading a parameter section."""
        start_time = time.time()
        parameters_read = []
        failed_parameters = []
        
        try:
            self._log_message(f"INFO: Starting {section_name} section read with {len(section_commands)} commands")
            
            for i, (param_name, command, command_type) in enumerate(section_commands):
                if self.cancel_requested:
                    self._log_message(f"INFO: {section_name} section reading canceled by user")
                    break
                
                # Update progress
                status_text = f"Reading {param_name}..."
                if self.progress_callback:
                    self.app.after(0, self.progress_callback, i + 1, len(section_commands), status_text)
                
                # Read the parameter
                result = self.read_specific_parameter(param_name, command, command_type)
                parameters_read.append(result)
                
                if not result.success:
                    failed_parameters.append(param_name)
                    self._log_message(f"WARNING: Failed to read {param_name}: {result.error_message}")
                else:
                    self._log_message(f"INFO: Successfully read {param_name}: {result.value}")
                
                # Add delay between commands
                if i < len(section_commands) - 1:
                    time.sleep(0.5)
            
            # Calculate results
            read_duration = time.time() - start_time
            successful_reads = len(section_commands) - len(failed_parameters)
            success = successful_reads > 0
            
            result = ConfigurationReadResult(
                success=success,
                parameters_read=parameters_read,
                total_parameters=len(section_commands),
                failed_parameters=failed_parameters,
                read_duration=read_duration,
                timestamp=datetime.now()
            )
            
            # Update UI with read parameters
            if success:
                self._update_ui_with_parameters(parameters_read)
            
            # Call completion callback
            if self.completion_callback:
                error_msg = None
                if failed_parameters:
                    error_msg = f"Failed to read {len(failed_parameters)} parameters from {section_name}"
                self.completion_callback(result, error_msg)
            
            self._log_message(f"INFO: {section_name} section reading completed in {read_duration:.2f} seconds")
            
        except Exception as e:
            error_msg = f"Unexpected error during {section_name} section reading: {str(e)}"
            self._log_message(f"ERROR: {error_msg}")
            
            result = ConfigurationReadResult(
                success=False,
                parameters_read=parameters_read,
                total_parameters=len(section_commands),
                failed_parameters=failed_parameters,
                read_duration=time.time() - start_time,
                timestamp=datetime.now()
            )
            
            if self.completion_callback:
                self.completion_callback(result, error_msg)
        
        finally:
            self.is_reading = False

    def get_specific_error_message(self, param_name: str, error_type: str) -> str:
        """
        Get a specific error message for a parameter and error type.
        
        Args:
            param_name: Name of the parameter
            error_type: Type of error (timeout, parse_error, connection_error, etc.)
            
        Returns:
            Localized error message
        """
        error_messages = {
            'timeout': self.i18n.get('param_read_timeout', 
                                   'Timeout reading {param}: Device did not respond within expected time'),
            'parse_error': self.i18n.get('param_parse_error', 
                                       'Parse error reading {param}: Device response format was unexpected'),
            'connection_error': self.i18n.get('param_connection_error', 
                                            'Connection error reading {param}: Device connection was lost'),
            'invalid_response': self.i18n.get('param_invalid_response', 
                                            'Invalid response reading {param}: Device returned an error or unexpected data'),
            'command_error': self.i18n.get('param_command_error', 
                                         'Command error reading {param}: Failed to send command to device')
        }
        
        message_template = error_messages.get(error_type, 
                                            self.i18n.get('param_unknown_error', 
                                                        'Unknown error reading {param}'))
        
        return message_template.format(param=param_name)

    def cancel_reading(self) -> None:
        """Cancel the current configuration reading operation with enhanced feedback."""
        if self.is_reading:
            self.cancel_requested = True
            self._log_message("INFO: Configuration reading cancellation requested by user")
            
            # Update progress dialog to show cancellation in progress
            if self.progress_dialog_id and hasattr(self.app, 'operation_confirmation_service'):
                try:
                    cancel_status = (f"Cancelling operation...\n"
                                   f"Current parameter: {getattr(self, 'current_parameter_name', 'Unknown')}\n"
                                   f"Completed: {getattr(self, 'successful_reads_count', 0)} successful, {getattr(self, 'failed_reads_count', 0)} failed")
                    
                    self.app.after(0, self.app.operation_confirmation_service.update_progress,
                                  self.progress_dialog_id, getattr(self, 'current_parameter_index', 0), cancel_status)
                except Exception as e:
                    self._log_message(f"WARNING: Failed to update progress dialog during cancellation: {str(e)}")
            
            # Cancel any pending serial operations immediately
            self._cancel_serial_operations()
            
            # Also cancel any waiting for response
            self.waiting_for_response = False
            self.response_timeout = True
            
            # If we have access to the serial service, cancel its operations too
            if hasattr(self.app, 'serial_service'):
                self.app.serial_service.waiting_for_response = False
                self.app.serial_service.response_timeout = True
                self.app.serial_service.captured_response = None
            
            # Wait for the thread to finish gracefully
            if hasattr(self, 'read_thread') and self.read_thread and self.read_thread.is_alive():
                self.read_thread.join(timeout=3.0)  # Wait up to 3 seconds for graceful shutdown
            
            self.is_reading = False
            successful_count = getattr(self, 'successful_reads_count', 0)
            self._log_message(f"INFO: Configuration reading cancelled - {successful_count} parameters read successfully before cancellation")
        else:
            self._log_message("INFO: No active configuration reading operation to cancel")
    
    def _cancel_serial_operations(self) -> None:
        """Cancel any ongoing serial operations."""
        try:
            # Clear any pending responses
            self.waiting_for_response = False
            self.captured_response = None
            self.response_timeout = True  # Set to True to indicate timeout/cancellation
            
            # If we have access to the serial service, clear its state too
            if hasattr(self.app, 'serial_service'):
                self.app.serial_service.waiting_for_response = False
                self.app.serial_service.response_timeout = True
                if hasattr(self.app.serial_service, 'response_buffer'):
                    self.app.serial_service.response_buffer.clear()
                # Clear any captured response
                self.app.serial_service.captured_response = None
            
            self._log_message("INFO: Cancelled pending serial communications and response capture")
        except Exception as e:
            self._log_message(f"WARNING: Failed to cancel serial operations: {str(e)}")

    def export_parameter_update_log(self, file_path: str = None) -> str:
        """
        Export parameter update log to a file or return as string.
        
        Args:
            file_path: Optional file path to save the log
            
        Returns:
            Log content as string
        """
        try:
            log_entries = self.get_parameter_update_log()
            update_sessions = self.get_update_sessions()
            conditional_updates = self.get_conditional_update_log()
            
            # Build log content
            log_lines = []
            log_lines.append("# Parameter Update Log Export")
            log_lines.append(f"# Generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
            log_lines.append("")
            
            # Add session summary
            if update_sessions:
                log_lines.append("## Update Sessions Summary")
                for session in update_sessions:
                    log_lines.append(f"Session: {session['session_id']}")
                    log_lines.append(f"  Timestamp: {session['timestamp'].strftime('%Y-%m-%d %H:%M:%S')}")
                    log_lines.append(f"  Total Parameters: {session['total_parameters']}")
                    log_lines.append(f"  Changed: {len(session['changed_parameters'])}")
                    log_lines.append(f"  Unchanged: {len(session['unchanged_parameters'])}")
                    log_lines.append("")
            
            # Add detailed parameter updates
            if log_entries:
                log_lines.append("## Detailed Parameter Updates")
                for entry in log_entries:
                    timestamp_str = entry['timestamp'].strftime('%Y-%m-%d %H:%M:%S.%f')[:-3]
                    status = "CHANGED" if entry['changed'] else "UNCHANGED"
                    old_str = self._format_value_for_logging(entry['old_value'])
                    new_str = self._format_value_for_logging(entry['new_value'])
                    
                    log_lines.append(f"[{timestamp_str}] {entry['parameter']}: {status}")
                    log_lines.append(f"  Old: {old_str}")
                    log_lines.append(f"  New: {new_str}")
                    log_lines.append("")
            
            # Add conditional field updates
            if conditional_updates:
                log_lines.append("## Conditional Field Updates")
                for session in conditional_updates:
                    log_lines.append(f"Session: {session['session_id']}")
                    log_lines.append(f"  Timestamp: {session['timestamp'].strftime('%Y-%m-%d %H:%M:%S')}")
                    for update in session['updates']:
                        visibility = "SHOWN" if update['visible'] else "HIDDEN"
                        log_lines.append(f"  {update['field']}: {visibility} (trigger: {update['trigger']}={update['trigger_value']})")
                    log_lines.append("")
            
            log_content = "\n".join(log_lines)
            
            # Save to file if path provided
            if file_path:
                with open(file_path, 'w', encoding='utf-8') as f:
                    f.write(log_content)
                self._log_message(f"INFO: Parameter update log exported to {file_path}")
            
            return log_content
            
        except Exception as e:
            error_msg = f"ERROR: Failed to export parameter update log: {str(e)}"
            self._log_message(error_msg)
            return error_msg

    def get_update_statistics(self) -> Dict[str, Any]:
        """
        Get statistics about parameter updates.
        
        Returns:
            Dictionary with update statistics
        """
        try:
            log_entries = self.get_parameter_update_log()
            update_sessions = self.get_update_sessions()
            
            if not log_entries:
                return {
                    'total_updates': 0,
                    'changed_updates': 0,
                    'unchanged_updates': 0,
                    'unique_parameters': 0,
                    'sessions': 0,
                    'last_update': None
                }
            
            changed_count = sum(1 for entry in log_entries if entry['changed'])

    def _get_device_info(self) -> Dict[str, str]:
        """
        Get device information for profile metadata.
        
        Returns:
            Dictionary with device information
        """
        device_info = {}
        
        try:
            # Try to get device info from serial service
            if hasattr(self.app, 'serial_service') and self.app.serial_service:
                if hasattr(self.app.serial_service, 'device_info'):
                    device_info.update(self.app.serial_service.device_info)
                
                # Try to get port information
                if hasattr(self.app.serial_service, 'port'):
                    device_info['port'] = str(self.app.serial_service.port)
                
                # Try to get baudrate
                if hasattr(self.app.serial_service, 'baudrate'):
                    device_info['baudrate'] = str(self.app.serial_service.baudrate)
            
            # Try to get additional info from configuration reader
            if hasattr(self.app, 'configuration_reader_service') and self.app.configuration_reader_service:
                config_reader = self.app.configuration_reader_service
                
                # Try to read device identification parameters
                try:
                    # Common device identification parameters
                    id_params = ['imei', 'serial_number', 'device_id', 'firmware_version', 'hardware_version']
                    
                    for param in id_params:
                        try:
                            result = config_reader.read_specific_parameter(param, f"AT+{param.upper()}", "query", max_retries=1)
                            if result.success and result.value:
                                device_info[param] = str(result.value)
                        except Exception:
                            continue  # Skip failed parameter reads
                            
                except Exception as e:
                    self.logger.debug(f"Could not read device identification parameters: {e}")
            
            # Add timestamp
            device_info['read_timestamp'] = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
            
            return device_info
            
        except Exception as e:
            self.logger.error(f"Error getting device info: {str(e)}")
            return {'error': str(e), 'read_timestamp': datetime.now().strftime('%Y-%m-%d %H:%M:%S')}

            all_commands = config_reader._get_all_parameter_commands()
            
            if not all_commands:
                self.logger.error("No parameter commands available")
                return None
            
            parameters = {}
            successful_reads = 0
            total_commands = len(all_commands)
            
            self.logger.info(f"Reading {total_commands} parameters from device...")
            
            # Update progress dialog with actual parameter count
            if progress_dialog:
                progress_dialog.total_steps = total_commands + 10  # Extra steps for setup/cleanup
                init_msg = self._get_localized_message("initializing_device_read", "Initializing device read...")
                progress_dialog.update_progress(5, init_msg)
            
            # Read each parameter individually with progress updates
            for i, (param_name, command, command_type) in enumerate(all_commands):
                try:
                    # Update progress
                    if progress_dialog:
                        read_msg = self._get_localized_message(
                            "reading_parameter", 
                            "Reading parameter: {param_name}"
                        ).format(param_name=param_name)
                        progress_dialog.update_progress(10 + i, read_msg)
                    
                    # Use the existing parameter reading method
                    result = config_reader.read_specific_parameter(param_name, command, command_type, max_retries=1)
                    
                    if result.success and result.value is not None:
                        parameters[param_name] = str(result.value)
                        successful_reads += 1
                        self.logger.debug(f"Successfully read {param_name}: {result.value}")
                    else:
                        self.logger.warning(f"Failed to read parameter {param_name}: {result.error_message}")
                        # Store empty value for failed parameters to maintain structure
                        parameters[param_name] = ""
                        
                except Exception as param_error:
                    self.logger.warning(f"Exception reading parameter {param_name}: {str(param_error)}")
                    parameters[param_name] = ""
            
            # Final progress update
            if progress_dialog:
                complete_msg = self._get_localized_message(
                    "device_read_complete", 
                    "Device read complete: {successful}/{total} parameters"
                ).format(successful=successful_reads, total=total_commands)
                progress_dialog.update_progress(total_commands + 5, complete_msg)
            
            # Consider successful if we read at least some parameters
            if successful_reads > 0:
                self.logger.info(f"Successfully read {successful_reads}/{total_commands} parameters from device")
                return parameters
            else:
                self.logger.error("Failed to read any parameters from device")
                return None

            self.logger.error(f"Error reading device parameters: {str(e)}")

                self.logger.debug(f"Successfully sent {param_name}: {param_value}")
                return True
            else:
                self.logger.warning(f"Device response for {param_name}: {response}")

    def _handle_multi_field_parameter(self, param_name: str, field_values: Dict[str, Any]) -> Dict[str, Any]:
        """
        A simple handler for multi-field parameters that prepares a dictionary
        for template formatting.
        """
        command_params = {}
        
        # Provide default values for jammer_relay_action
        if param_name == 'jammer_relay_action':
            field_values.setdefault('param_d', '0')
            field_values.setdefault('param_e', '0')
            field_values.setdefault('param_f', '0')
            field_values.setdefault('param_g', 'Manter ativada')

        for field_name, field_value in field_values.items():
            template_key = f"{param_name}_{field_name}"
            
            if field_name == 'enabled':
                mapped_value = 'ON' if field_value else 'OFF'
            elif field_name == 'alert_type':
                mapped_value = self._map_value('alert_type', field_value)
            elif field_name == 'output_operation':
                mapped_value = self._map_value('jammer_output_operation', field_value)
            elif field_name == 'param_g':
                mapped_value = self._map_value('param_g', field_value)
            else:
                mapped_value = field_value
            
            command_params[template_key] = mapped_value

        # Special handling for server_settings type
        if param_name == 'server_settings':
            address = field_values.get('address', '')
            command_params['server_settings_type'] = self._determine_server_type(address)

    def get_command(self, param_name: str, value: Any, context: Dict[str, Any] = None,
                   confirm_destructive: bool = True, validate_command: bool = True) -> str:
        """
        Get the formatted AT command for a parameter with validation.

        Args:
            param_name: The parameter name
            value: The parameter value
            context: Optional context with additional parameter values needed for the command
            confirm_destructive: Whether to confirm destructive operations
            validate_command: Whether to validate the command before returning it

        Returns:
            The formatted AT command string

        Raises:
            ValueError: If command generation fails or validation fails
        """
        # Input validation
        if not param_name:
            raise ValueError("Parameter name cannot be empty")
        
        if not isinstance(param_name, str):
            raise ValueError(f"Parameter name must be a string, got {type(param_name).__name__}")
        
        self.logger.debug(f"Generating command for parameter: {param_name} with value: {value}")
        
        # Log context if provided
        if context:
            self.logger.debug(f"Command context: {context}")
        
        try:
            # Check if this is a destructive operation that requires confirmation
            if confirm_destructive and param_name in [
                'factory_reset', 'reset_device', 'format_device', 'clear_memory'
            ]:
                self.logger.warning(f"Destructive operation requested: {param_name}")
                
                # Map param_name to operation type
                operation_type_map = {
                    'factory_reset': 'OPERATION_FACTORY_RESET',
                    'reset_device': 'OPERATION_DEVICE_RESET',
                    'format_device': 'OPERATION_FORMAT_DEVICE',
                    'clear_memory': 'OPERATION_CLEAR_MEMORY'
                }

                # Get the app instance if available in context
                app = context.get('app') if context else None

                # If we have the app and operation confirmation service, confirm the operation
                if app and hasattr(app, 'operation_confirmation_service'):
                    operation_type = getattr(app.operation_confirmation_service, operation_type_map[param_name])
                    if not app.operation_confirmation_service.confirm_operation(operation_type):
                        self.logger.info(f"Destructive operation {param_name} was canceled by user")
                        raise ValueError(f"Operation {param_name} was canceled by user")
                    else:
                        self.logger.info(f"Destructive operation {param_name} confirmed by user")
                else:
                    self.logger.warning(f"No confirmation service available for destructive operation: {param_name}")

            # Check if this is a multi-field parameter that needs special handling
            multi_field_params = [
                'jammer_relay_action', 'main_power_alert', 'backup_battery_alert',
                'angle_report', 'speed_alert', 'moving_alert', 'vibration_window',
                'apn_settings', 'server_settings', 'report_intervals', 'heartbeat_intervals',
                'ignition_type', 'vibration_sensitivity', 'jammer_detection', 'motion_sensor_alert'
            ]
            
            command = None  # Initialize command variable
            
            if param_name in multi_field_params:
                self.logger.debug(f"Processing multi-field parameter: {param_name}")
                
                # Handle multi-field parameters using the specialized method
                if isinstance(value, dict):
                    # Value is already a dictionary of field values
                    field_values = value
                else:
                    # Convert single value to dictionary format for compatibility
                    self.logger.debug(f"Converting single value to dictionary format for {param_name}")
                    field_values = {'value': value}
                    if context:
                        field_values.update(context)
                
                # Use the multi-field parameter handler
                if param_name == 'moving_alert':
                    command_params = field_values.copy()
                    if 'enabled' in command_params:
                        command_params['enabled'] = 'ON' if command_params['enabled'] else 'OFF'
                    if 'alert_type' in command_params:
                        alert_type_map = self.value_mappings.get('alert_type', {})
                        if command_params['alert_type'] in alert_type_map:
                            command_params['alert_type'] = alert_type_map[command_params['alert_type']]
                elif param_name == 'vibration_window':
                    command_params = field_values.copy()
                    if 'threshold' in command_params:
                        command_params['value'] = command_params.pop('threshold')
                else:
                    command_params = self._handle_multi_field_parameter(param_name, field_values)
                
                # Get the command template
                template = self.command_templates.get(param_name)
                if not template:
                    available_templates = list(self.command_templates.keys())
                    raise ValueError(f"Unknown parameter: {param_name}. Available parameters: {available_templates}")
                
                self.logger.debug(f"Using template for {param_name}: {template}")
                
                # Special-case: when certain multi-field parameters are disabled, the
                # device expects a minimal command with only the state (e.g. "JAMMER,OFF#",
                # "SPEED,OFF#") instead of the full expanded template.
                try:
                    if param_name == 'jammer_detection' and command_params.get('state') == 'OFF':
                        # Generate minimal jammer command when disabled
                        command = f"JAMMER,{command_params.get('state')}#"
                        self.logger.debug(f"Using minimal command for disabled jammer_detection: {command}")
                    elif param_name == 'speed_alert' and command_params.get('enabled') == 'OFF':
                        # Generate minimal speed command when disabled
                        command = f"SPEED,OFF#"
                        self.logger.debug(f"Using minimal command for disabled speed_alert: {command}")
                    elif param_name == 'backup_battery_alert' and command_params.get('state') == 'OFF':
                        # Generate minimal backup battery command when disabled
                        command = "BATALM,OFF#"
                        self.logger.debug(f"Using minimal command for disabled backup_battery_alert: {command}")
                    elif param_name == 'backup_battery_alert':
                        # Enabled backup battery -> include mode (A)
                        mode = command_params.get('mode', '0')
                        # ensure numeric mode
                        try:
                            mode_int = int(mode)
                        except Exception:
                            mode_int = 0
                        if mode_int not in (0, 1):
                            mode_int = 0
                        command = f"BATALM,ON,{mode_int}#"
                        self.logger.debug(f"Using full command for backup_battery_alert: {command}")
                    elif param_name == 'motion_sensor_alert':
                        command = self._generate_motion_sensor_alert_command(field_values, validate_command)
                    else:
                        # Format the command using the mapped parameters
                        command = template.format(**command_params)
                        self.logger.debug(f"Formatted command for {param_name}: {command}")
                except KeyError as e:
                    missing_key = str(e).strip("'")
                    available_keys = list(command_params.keys())
                    raise ValueError(f"Missing required parameter: {missing_key} for {param_name}. Available parameters: {available_keys}")
                except Exception as e:
                    raise ValueError(f"Error formatting command template for {param_name}: {str(e)}")
            else:
                self.logger.debug(f"Processing single-field parameter: {param_name}")
                
                # Handle single-field parameters using the existing approach
                # Apply value mapping if needed
                try:
                    mapped_value = self._map_value(param_name, value)
                    self.logger.debug(f"Mapped value for {param_name}: {value} -> {mapped_value}")
                except Exception as e:
                    raise ValueError(f"Error mapping value for {param_name}: {str(e)}")

                # Get the command template
                template = self.command_templates.get(param_name)
                if not template:
                    available_templates = list(self.command_templates.keys())
                    raise ValueError(f"Unknown parameter: {param_name}. Available parameters: {available_templates}")

                self.logger.debug(f"Using template for {param_name}: {template}")

                # Prepare format arguments
                format_args = {'value': mapped_value}
                if context:
                    format_args.update(context)

                self.logger.debug(f"Format arguments for {param_name}: {format_args}")

                # Format the command
                try:
                    command = template.format(**format_args)
                    self.logger.debug(f"Formatted command for {param_name}: {command}")
                except KeyError as e:
                    missing_key = str(e).strip("'")
                    available_keys = list(format_args.keys())
                    raise ValueError(f"Missing required context parameter: {missing_key} for {param_name}. Available parameters: {available_keys}")
                except Exception as e:
                    raise ValueError(f"Error formatting command template for {param_name}: {str(e)}")
            
            # Add the command prefix for both single-field and multi-field parameters
            full_command = f"{self.command_prefix}{command}"
            self.logger.debug(f"Full command with prefix: {full_command}")
            
            # Validate the command using CommandValidationGateway if requested
            if validate_command and param_name not in ['vibration_window', 'network_select']:
                self.logger.debug(f"Validating command for {param_name}")
                
                validation_context = ValidationContext(
                    command_type=self._determine_command_type_from_param(param_name),
                    source='device_command_mapper',
                    timestamp=datetime.now()
                )
                
                validation_result = self.validation_gateway.validate_command_before_send(
                    full_command, validation_context
                )
                
                if not validation_result.is_valid:
                    error_msg = f"Command validation failed for {param_name}: {validation_result.error_message}"
                    if validation_result.suggestions:
                        error_msg += f". Suggestions: {'; '.join(validation_result.suggestions)}"
                    
                    self.logger.error(f"Command blocked by validation: {full_command}")
                    self.validation_gateway.log_blocked_command(validation_result, validation_context)
                    
                    raise ValueError(error_msg)
                
                self.logger.debug(f"Command validation passed for {param_name}: {full_command}")
            else:
                self.logger.debug(f"Command validation skipped for {param_name}")
            
            # Remove trailing '#' for SZCS commands during profile loading
            if full_command.startswith("ATSMS:SZCS#") and full_command.endswith('#'):
                # Check if this is a multi-part command like ACCLINE#ACC_CFG (these should keep their structure)
                command_body = full_command[6:]  # Remove "ATSMS:" prefix
                # Multi-part commands have multiple '=' signs or specific patterns like ACCLINE#ACC_CFG
                if 'ACCLINE=' in command_body and 'ACC_CFG=' in command_body:
                    # This is a multi-part ignition command, don't remove trailing '#'
                    self.logger.debug(f"Keeping multi-part SZCS command structure: {full_command}")
                else:
                    # This is a single SZCS command with trailing '#', remove it
                    full_command = full_command[:-1]
                    self.logger.debug(f"Removed trailing '#' for SZCS command: {full_command}")
            
            self.logger.info(f"Successfully generated command for {param_name}: {full_command}")
            return full_command
            
        except ValueError:
            # Re-raise ValueError exceptions as they contain user-friendly error messages
            raise
        except Exception as e:
            # Catch any other unexpected exceptions and provide detailed error information
            self.logger.error(f"Unexpected error generating command for {param_name}: {str(e)}")
            raise ValueError(f"Failed to generate command for parameter {param_name}: {str(e)}")

    def get_parameter_command_variations(self, section: str, param_name: str, values: Dict[str, Any]) -> List[str]:
        """
        Generate command variations for a parameter with multiple fields.
        
        This method handles both single-field and multi-field parameters and returns
        a list of commands for the given parameter.
        
        Args:
            section: The parameter section name
            param_name: The parameter name
            values: Dictionary of parameter field values
            
        Returns:
            List of commands for the parameter
            
        Raises:
            ValueError: If parameter is unknown or command generation fails
        """
        # Input validation
        if not section:
            raise ValueError("Section name cannot be empty")
        
        if not param_name:
            raise ValueError("Parameter name cannot be empty")
        
        if not isinstance(values, dict):
            raise ValueError(f"Values must be a dictionary, got {type(values).__name__}")
        
        self.logger.debug(f"Generating command variations for {section}.{param_name} with values: {values}")
        
        try:
            # Generate the command using the existing get_command method
            command = self.get_command(param_name, values, context={'section': section})
            
            self.logger.debug(f"Successfully generated command for {section}.{param_name}: {command}")
            return [command]
            
        except ValueError as e:
            # Provide detailed error message for unknown parameters
            if "Unknown parameter" in str(e):
                available_params = list(self.command_templates.keys())
                error_msg = f"Unknown parameter: {section}.{param_name}. Available parameters: {available_params}"
                self.logger.error(error_msg)
                raise ValueError(error_msg)
            else:
                error_msg = f"Command generation failed for {section}.{param_name}: {str(e)}"
                self.logger.error(error_msg)
                raise ValueError(error_msg)
        except Exception as e:
            # Catch any other unexpected exceptions
            error_msg = f"Unexpected error generating command variations for {section}.{param_name}: {str(e)}"
            self.logger.error(error_msg)
            raise ValueError(error_msg)

    def _map_value(self, param_name: str, value: Any) -> str:
        """
        Map parameter values to their command equivalents if needed.
        
        Args:
            param_name: Name of the parameter
            value: Value to map
            
        Returns:
            Mapped value as string
            
        Raises:
            ValueError: If value mapping fails
        """
        try:
            if param_name == 'backup_battery_protection_voltage':
                try:
                    return str(int(float(value)))
                except (ValueError, TypeError):
                    self.logger.warning(f"Could not convert backup_battery_protection_voltage value '{value}' to int.")

            # Special handling for boolean values in specific parameters
            if param_name == 'speed_buzzer' and isinstance(value, bool):
                # Map boolean to 1/0 for speed buzzer
                mapped_value = '1' if value else '0'
                self.logger.debug(f"Mapped boolean value for {param_name}: {value} -> {mapped_value}")
                return mapped_value
            
            # Special handling for sleep mode values
            if param_name == 'sleep_mode':
                # Use the specification mapping for sleep mode
                from .parameter_specification import get_command_value_mapping
                sleep_mapping = get_command_value_mapping('sleep_mode')
                if sleep_mapping and value in sleep_mapping:
                    mapped_value = str(sleep_mapping[value])
                    self.logger.debug(f"Mapped sleep_mode value: {value} -> {mapped_value}")
                    return mapped_value
                # Fallback to internal mapping
                elif value in self.value_mappings.get('sleep_mode', {}):
                    mapped_value = str(self.value_mappings['sleep_mode'][value])
                    self.logger.debug(f"Mapped sleep_mode value (fallback): {value} -> {mapped_value}")
                    return mapped_value
            
            # Check if parameter has value mappings defined
            if param_name in self.value_mappings:
                mapping = self.value_mappings[param_name]
                
                if value in mapping:
                    mapped_value = mapping[value]
                    self.logger.debug(f"Mapped value for {param_name}: {value} -> {mapped_value}")
                    return str(mapped_value)
                else:
                    # Value not found in mapping - this might be an error
                    available_values = list(mapping.keys())
                    self.logger.warning(f"Value '{value}' not found in mapping for {param_name}. Available values: {available_values}")
                    
                    # For some parameters, unmapped values might be valid (e.g., numeric values)
                    # Log the issue but continue with the original value
                    if isinstance(value, (int, float, str)):
                        self.logger.debug(f"Using original value for {param_name}: {value}")
                        return str(value)
                    else:
                        raise ValueError(f"Invalid value '{value}' for parameter {param_name}. Valid values: {available_values}")
            else:
                # No mapping defined for this parameter - use value directly
                self.logger.debug(f"No value mapping defined for {param_name}, using original value: {value}")
                return str(value)
                
        except Exception as e:
            self.logger.error(f"Error mapping value for parameter {param_name}: {str(e)}")
            raise ValueError(f"Failed to map value '{value}' for parameter {param_name}: {str(e)}")

    def _determine_command_type_from_param(self, param_name: str) -> str:
        """
        Determine the command type based on the parameter name.
        
        Args:
            param_name: The parameter name
            
        Returns:
            String identifying the command type
        """
        if param_name.startswith('apn_'):
            return 'apn'
        elif param_name.startswith('server_'):
            return 'server'
        elif param_name.startswith('timer_'):
            return 'timer'
        elif param_name.startswith('heartbeat_'):
            return 'heartbeat'
        elif param_name in ['vibration', 'sensitivity', 'gt06_protocol', 'network_select', 
                           'sleep_mode', 'sleep_time', 'ignition_type',
                           'angle_reporting', 'blind_speed', 'curve_sensor', 'lock_type']:
            return 'szcs'
        else:
            return 'unknown'

    def validate_command(self, command: str) -> Tuple[bool, str]:
        """
        Validate a command string for proper format.

        Args:
            command: The command string to validate

        Returns:
            Tuple of (is_valid, error_message)
        """
        # Check if command starts with prefix
        if not command.startswith(self.command_prefix):
            return False, f"Command must start with {self.command_prefix}"

        # Check for common format issues
        if not re.match(r'^ATSMS:[A-Z0-9#,_=:]+.*', command):
            return False, "Command format is invalid"

        # Check for specific command patterns
        command_body = command[len(self.command_prefix):]

        # Check APN command format
        if command_body.startswith("APN,"):
            parts = command_body.split(',')
            if len(parts) < 2 or not command_body.endswith('#'):
                return False, "Invalid APN command format. Should be APN,server,user,password#"

        # Check SERVER command format
        elif command_body.startswith("SERVER,"):
            parts = command_body.split(',')
            if len(parts) < 5 or not command_body.endswith('#'):
                return False, "Invalid SERVER command format. Should be SERVER,type,address,port,protocol#"

        # Check TIMER command format
        elif command_body.startswith("TIMER,"):
            parts = command_body.split(',')
            if len(parts) < 3 or not command_body.endswith('#'):
                return False, "Invalid TIMER command format. Should be TIMER,ign_on,ign_off#"

        # Check HBT command format
        elif command_body.startswith("HBT,"):
            parts = command_body.split(',')
            if len(parts) < 2 or not command_body.endswith('#'):
                return False, "Invalid HBT command format. Should be HBT,ign_on,ign_off#"

        # Check SZCS command format
        elif command_body.startswith("SZCS#"):
            if not re.match(r'^SZCS#[A-Z0-9_]+=.+#', command_body):
                return False, "Invalid SZCS command format. Should be SZCS#PARAM=value#"

        # All checks passed
        return True, ""

    def parse_device_response(self, response: str, command_type: str) -> Dict[str, Any]:
        """
        Parse device response based on the command type.

        Args:
            response: The device response string
            command_type: The type of command that was sent

        Returns:
            Dictionary with parsed information
        """
        result = {}

        # Handle different response types based on LT32Pro documentation
        if command_type == 'get_imei' or 'IMEI' in response:
            # Example responses:
            # "READOK: IMEI=868022038821794"
            # "IMEI: 868022038821794"
            # "IMEI:868022038821794"
            match = re.search(r'IMEI[=:]\s*(\d+)', response)
            if match:
                result['imei'] = match.group(1)

        elif command_type == 'get_teriid' or 'TERIID' in response:
            # Example responses:
            # "READOK: TERIID=864943044534994"
            # "TERIID: 864943044534994"
            # "TERIID:864943044534994"
            match = re.search(r'TERIID[=:]\s*(\d+)', response)
            if match:
                result['teriid'] = match.group(1)

        elif command_type == 'get_iccid' or 'ICCID' in response:
            # Example responses:
            # "READOK: ICCID=89860000192397786063"
            # "ICCID: 89860000192397786063"
            # "ICCID:89860000192397786063"
            match = re.search(r'ICCID[=:]\s*(\d+)', response)
            if match:
                result['iccid'] = match.group(1)

        elif command_type == 'get_imsi':
            # Example response: "READOK: IMSI=234588563274918"
            match = re.search(r'IMSI=(\d+)', response)
            if match:
                result['imsi'] = match.group(1)

        elif command_type == 'get_status':
            # Example response: "Battery:4.23V;GPRS:Offline;GSMSignalLevel:25;ACC:OFF;GPS:OFF;Defense:ON;"
            # Parse each section separated by semicolons
            for section in response.split(';'):
                if ':' in section:
                    key, value = section.split(':', 1)
                    result[key.strip()] = value.strip()

        elif command_type == 'get_version':
            # Example response: "W16A_10D6K_B44_V6.21 2024 11 15"
            result['version'] = response.strip()

        elif command_type == 'get_all_params':
            # Example response: "IMEI:357457040000821;TIMER:20,300;SEMDS:3;HBT:180Sec;Defense:3;"
            # Parse each section separated by semicolons
            for section in response.split(';'):
                if ':' in section:
                    key, value = section.split(':', 1)
                    result[key.strip()] = value.strip()

        # Handle bare device responses like "ACCOFFALM,OFF" (some firmwares reply without SETOK/READOK)
        elif re.search(r'^\s*ACCOFFALM\b', response, re.IGNORECASE):
            # Example device responses: "ACCOFFALM,OFF" or "ACCOFFALM OFF" or "ACCOFFALM:OFF"
            match = re.search(r'ACCOFFALM[,=:\s]*\s*(ON|OFF|\d+)', response, re.IGNORECASE)
            if match:
                param_value = match.group(1).upper()
                result['parameter'] = 'ACCOFFALM'
                result['value'] = param_value
                result['status'] = 'success'
                try:
                    self.logger.debug(f"Parsed bare ACCOFFALM response: raw={response!r} -> value={param_value}")
                except Exception:
                    print(f"DEBUG: Parsed bare ACCOFFALM response: raw={response!r} -> value={param_value}")
        # Handle command responses with SETOK prefix
        elif response.startswith('SETOK:'):
            # Example: "SETOK: VIBL=3"
            # Extract the parameter name and value
            match = re.search(r'SETOK:\s+([A-Z0-9_]+)=(.+)', response)
            if match:
                param_name = match.group(1)
                param_value = match.group(2)
                result['parameter'] = param_name
                result['value'] = param_value
                result['status'] = 'success'

        # Handle APN responses
        elif response.startswith('APN:'):
            # Example: "APN:cmnet,,"
            parts = response[4:].split(',')
            if len(parts) >= 1:
                result['apn_server'] = parts[0]
            if len(parts) >= 2:
                result['apn_user'] = parts[1]
            if len(parts) >= 3:
                result['apn_password'] = parts[2].rstrip('#')

        # Handle SERVER responses
        elif response.startswith('SERVER:'):
            # Example: "SERVER:0,58.61.154.247,7018,0"
            parts = response[7:].split(',')
            if len(parts) >= 4:
                result['server_type'] = parts[0]  # 0 for IP, 1 for domain
                result['server_address'] = parts[1]
                result['server_port'] = parts[2]
                result['server_protocol'] = parts[3]  # 0 for TCP

        # Handle TIMER responses
        elif response.startswith('TIMER ACC ON:'):
            # Example: "TIMER ACC ON:30s,ACC OFF:300s"
            match = re.search(r'TIMER ACC ON:(\d+)s,ACC OFF:(\d+)s', response)
            if match:
                result['timer_ign_on'] = match.group(1)
                result['timer_ign_off'] = match.group(2)

        # Handle HBT (Heartbeat) responses
        elif response.startswith('HBT ACC ON:'):
            # Example: "HBT ACC ON:180s,ACC OFF:180s"
            match = re.search(r'HBT ACC ON:(\d+)s,ACC OFF:(\d+)s', response)
            if match:
                result['heartbeat_ign_on'] = match.group(1)
                result['heartbeat_ign_off'] = match.group(2)

        elif response.startswith('JAMRELAY:'):
            # Example: "JAMRELAY:0,5,20,1,1,1,0"
            parts = response[len('JAMRELAY:'):].split(',')
            if len(parts) == 7:
                result['parameter'] = 'jammer_relay_action'
                result['value'] = {
                    'output_operation': parts[0],
                    'confirmation_time': parts[1],
                    'speed_limit': parts[2],
                    'd_parameter': parts[3],
                    'e_parameter': parts[4],
                    'f_parameter': parts[5],
                    'g_parameter': parts[6]
                }

        # Handle DEFENSE responses
        elif response.startswith('DEFENSE:'):
            # Example: "DEFENSE:3"
            match = re.search(r'DEFENSE:(\d+)', response)
            if match:
                result['defense_time'] = match.group(1)

        # Handle RELAY responses
        elif response.startswith('RELAY ') and response.endswith(' OK'):
            # Example: "RELAY 1 OK"
            match = re.search(r'RELAY (\d) OK', response)
            if match:
                result['relay_state'] = match.group(1)
                result['status'] = 'success'

        # Handle generic OK responses
        elif ' OK' in response:
            result['status'] = 'success'
            result['message'] = response

        return result

    def get_bulk_commands(self, parameters: Dict[str, Any], validate_commands: bool = True) -> List[str]:
        """
        Generate a list of commands for bulk sending based on LT32Pro documentation.
        This method processes all flattened parameters from the UI and generates appropriate commands.

        Args:
            parameters: Dictionary of flattened parameter names and values from UI
            validate_commands: Whether to validate commands before including them

        Returns:
            List of formatted commands in the correct sequence

        Raises:
            ValueError: If command validation fails for critical parameters
        """
        commands = []
        validation_errors = []
        validation_warnings = []
        processed_params = set()  # Track which parameters we've already processed

        self.logger.info(f"Starting bulk command generation for {len(parameters)} parameters")
        self.logger.info(f"Parameters keys: {list(parameters.keys())}")

        # Remove UI-only parameters that do not map to device commands
        params_to_ignore = []
        for p in params_to_ignore:
            if p in parameters:
                del parameters[p]

        # Pre-process to group flattened compound parameters
        compound_group_names = [
            'apn_settings', 'server_settings', 'report_intervals', 'heartbeat_intervals',
            'ignition_type', 'speed_alert', 'main_power_alert', 'angle_report',
            'vibration_sensitivity', 'jammer_relay_action', 'moving_alert', 'vibration_window',
            'motion_sensor_alert'
        ]
        
        group_to_section = {
            'apn_settings': 'network_parameters', 'server_settings': 'network_parameters',
            'report_intervals': 'time_parameters', 'heartbeat_intervals': 'time_parameters',
            'ignition_type': 'ignition', 'speed_alert': 'speed',
            'main_power_alert': 'battery_alerts', 'angle_report': 'angle',
            'vibration_sensitivity': 'vibration_sensor', 'jammer_relay_action': 'jammer_detection',
            'moving_alert': 'alerts',
            'vibration_window': 'vibration_sensor',
            'motion_sensor_alert': 'movement_alerts'
        }

        new_params = {}
        keys_to_remove = []

        for key, value in parameters.items():
            if '.' in key:
                try:
                    group_name, field_name = key.split('.', 1)
                    if group_name in compound_group_names:
                        section = group_to_section[group_name]
                        new_key = f"{section}.{group_name}"
                        if new_key not in new_params:
                            new_params[new_key] = {}
                        new_params[new_key][field_name] = value
                        keys_to_remove.append(key)
                except ValueError:
                    # Handle cases where split doesn't produce 2 values, though '.' is present
                    pass

        for key in keys_to_remove:
            if key in parameters:
                del parameters[key]
        parameters.update(new_params)
        
        # Log all parameters for debugging
        self.logger.debug(f"All parameters received: {list(parameters.keys())}")
        for param_name, param_value in parameters.items():
            self.logger.debug(f"Parameter: {param_name} = {param_value} (type: {type(param_value).__name__})")

        # Group flattened parameters by their section and parameter name
        grouped_params = {}
        single_params = {}
        
        for param_name, param_value in parameters.items():
            if '.' in param_name:
                # This is a flattened parameter with section.parameter_name format
                section, param = param_name.split('.', 1)
                if section not in grouped_params:
                    grouped_params[section] = {}
                grouped_params[section][param] = param_value
            else:
                # This is a single parameter
                single_params[param_name] = param_value
        
        self.logger.debug(f"Grouped parameters: {grouped_params}")
        self.logger.debug(f"Single parameters: {single_params}")

        # Process compound commands from grouped parameters
        compound_groups = [
            # Network Configuration (Critical - send first)
            {
                'name': 'apn_settings',
                'section': 'network_parameters',
                'params': ['apn', 'user', 'password'],
                'required': ['apn'],  # At minimum we need the server
                'method': self._generate_apn_command
            },
            {
                'name': 'server_settings',
                'section': 'network_parameters',
                'params': ['address', 'port'],
                'required': ['address', 'port'],  # Both required
                'method': self._generate_server_command
            },
            
            # Timing Configuration
            {
                'name': 'report_intervals',
                'section': 'time_parameters',
                'params': ['moving_time', 'stopped_time'],
                'required': ['moving_time', 'stopped_time'],  # Both required
                'method': self._generate_timer_command
            },
            {
                'name': 'heartbeat_intervals',
                'section': 'time_parameters',
                'params': ['moving_heartbeat', 'stopped_heartbeat'],
                'required': ['moving_heartbeat', 'stopped_heartbeat'],  # Both required
                'method': self._generate_heartbeat_command
            },
            
            # Ignition Configuration
            {
                'name': 'ignition_type',
                'section': 'ignition',
                'params': ['ignition_mode', 'on_voltage', 'off_voltage'],
                'required': ['ignition_mode'],  # Mode is required, voltages are optional
                'method': self._generate_ignition_command
            },
            
            # Speed Alert Configuration
            {
                'name': 'speed_alert',
                'section': 'speed',
                'params': ['enabled', 'detection_time', 'speed_limit', 'alert_type'],
                'required': ['enabled'],  # At minimum we need enabled status
                'method': self._generate_speed_alert_command
            },
            
            # Power Alert Configuration
            {
                'name': 'main_power_alert',
                'section': 'battery_alerts',
                'params': ['enabled', 'alert_type', 'detection_time', 'min_charge_time'],
                'required': ['enabled'],
                'method': self._generate_main_power_alert_command
            },
            
            # Angle Report Configuration
            {
                'name': 'angle_report',
                'section': 'angle',
                'params': ['enabled', 'min_speed', 'angle_variation'],
                'required': ['enabled'],
                'method': self._generate_angle_report_command
            },
            
            # Vibration Configuration
            {
                'name': 'vibration_sensitivity',
                'section': 'vibration_sensor',
                'params': ['level'],
                'required': ['level'],
                'method': self._generate_vibration_command
            },
            
            # Jammer Relay Configuration
            {
                'name': 'jammer_relay_action',
                'section': 'jammer_detection',
                'params': ['output_operation', 'confirmation_time', 'speed_limit', 'param_d', 'param_e', 'param_f', 'param_g'],
                'required': ['output_operation'],
                'method': self._generate_jammer_relay_command
            },
            {
                'name': 'motion_sensor_alert',
                'section': 'movement_alerts',
                'params': ['enabled', 'alert_type'],
                'required': [],
                'method': self._generate_motion_sensor_alert_command
            }
        ]
        
        # Process compound commands first
        for group in compound_groups:
            self.logger.debug(f"Processing compound group: {group['name']}" )
            
            # Check if we have the section and required parameters for this group
            section = group.get('section')
            if section not in grouped_params:
                self.logger.debug(f"Section {section} not found in grouped parameters")
                continue
                
            section_params = grouped_params[section]
            param_name = group['name']
            
            if param_name not in section_params:
                self.logger.debug(f"Parameter {param_name} not found in section {section}")
                continue
                
            # Extract the parameter values
            param_values = section_params[param_name]
            if not isinstance(param_values, dict):
                self.logger.debug(f"Parameter {param_name} is not a dictionary: {type(param_values).__name__}")
                continue
            
            # Check if we have the required parameters for this group
            has_required = all(param in param_values and param_values[param] is not None and param_values[param] != '' 
                             for param in group['required'])
            
            self.logger.debug(f"Group {group['name']} has required params: {has_required}")
            
            if has_required:
                try:
                    # Collect all available parameters for this group
                    group_params = {}
                    for param in group['params']:
                        if param in param_values and param_values[param] is not None and param_values[param] != '':
                            group_params[param] = param_values[param]
                            self.logger.debug(f"Added param to group: {param} = {param_values[param]}")
                    
                    self.logger.debug(f"Group params collected: {group_params}")
                    
                    if group_params:  # Only process if we have some parameters
                        command = group['method'](group_params, validate_commands)
                        if command:
                            # Remove trailing '#' for SZCS commands during profile loading
                            if command.startswith("ATSMS:SZCS#") and command.endswith('#'):
                                command = command[:-1]
                                self.logger.debug(f"Removed trailing '#' for SZCS command in bulk operation: {command}")
                            commands.append(command)
                            self.logger.info(f"Added bulk command for {group['name']}: {command}")
                            # Mark these parameters as processed
                            processed_params.add(f"{section}.{param_name}")
                            self.logger.debug(f"Marked param as processed: {section}.{param_name}")
                        else:
                            self.logger.warning(f"No command generated for group: {group['name']}")
                    else:
                        self.logger.warning(f"No group params collected for: {group['name']}")
                            
                except ValueError as e:
                    if validate_commands and group['name'] in ['apn_settings', 'server_settings']:
                        error_msg = f"{group['name'].upper()} command validation failed: {str(e)}"
                        self.logger.error(error_msg)
                        validation_errors.append(f"{group['name']}: {error_msg}")
                    else:
                        warning_msg = f"Command validation failed for {group['name']}: {str(e)}"
                        validation_warnings.append(f"{group['name']}: {warning_msg}")
                        self.logger.warning(f"Non-critical parameter validation failed: {group['name']}")
                        
                except Exception as e:
                    self.logger.error(f"Unexpected error generating command for {group['name']}: {str(e)}")
                    if validate_commands and group['name'] in ['apn_settings', 'server_settings']:
                        validation_errors.append(f"{group['name']}: {str(e)}")
            else:
                # Log which required parameters are missing
                missing_params = [param for param in group['required'] 
                                 if param not in param_values or param_values[param] is None or param_values[param] == '']
                self.logger.debug(f"Group {group['name']} missing required params: {missing_params}")
        
        # Process single parameters that weren't part of compound commands
        individual_params = [
            'protocol', 'tcp_ack', 'speed_buzzer', 'sleep_mode', 'sleep_time',
            'backup_battery_alert', 'jammer_detection', 'odometer', 'hourmeter',
            'network_settings', 'defense_settings', 'source_off_type', 'relay_control', 'sends_control',
            'distance_movement_action', 'acc_on_alert', 'acc_off_alert', 'blind_area_speed'
        ]
        
        self.logger.info(f"Processing {len(individual_params)} individual parameters")
        self.logger.info(f"Individual parameters list: {individual_params}")
        self.logger.info(f"Single parameters keys: {list(single_params.keys())}")
        
        for param_name in individual_params:
            if param_name in single_params and param_name not in processed_params:
                param_value = single_params[param_name]
                
                self.logger.debug(f"Processing individual param: {param_name} = {param_value}")
                
                # Skip None values and empty strings
                if param_value is None or param_value == '':
                    self.logger.debug(f"Skipping individual param {param_name}: empty value")
                    continue
                    
                try:
                    command = self._generate_individual_command(param_name, param_value, validate_commands)
                    if command:
                        # Remove trailing '#' for SZCS commands during profile loading
                        if command.startswith("ATSMS:SZCS#") and command.endswith('#'):
                            command = command[:-1]
                            self.logger.debug(f"Removed trailing '#' for SZCS command in bulk operation: {command}")
                        commands.append(command)
                        self.logger.info(f"Added bulk command for {param_name}: {command}")
                        processed_params.add(param_name)
                    else:
                        self.logger.warning(f"No command generated for parameter: {param_name}")
                        
                except ValueError as e:
                    if validate_commands:
                        warning_msg = f"Command validation failed for {param_name}: {str(e)}"
                        validation_warnings.append(f"{param_name}: {warning_msg}")
                        self.logger.warning(f"Non-critical parameter validation failed: {param_name}")
                    else:
                        self.logger.warning(f"Command generation failed for {param_name}: {str(e)}")
                        
                except Exception as e:
                    self.logger.error(f"Unexpected error generating command for {param_name}: {str(e)}")
            else:
                if param_name in processed_params:
                    self.logger.debug(f"Skipping individual param {param_name}: already processed")
                elif param_name not in single_params:
                    self.logger.debug(f"Skipping individual param {param_name}: not in parameters")
        
        # Process any remaining parameters that weren't explicitly handled
        remaining_params = set(parameters.keys()) - processed_params
        if remaining_params:
            self.logger.info(f"Processing {len(remaining_params)} additional parameters: {remaining_params}")
            for param_name in remaining_params:
                param_value = parameters[param_name]
                
                self.logger.debug(f"Processing additional param: {param_name} = {param_value}")
                
                # Skip None values, empty strings, and metadata parameters
                if param_value is None or param_value == '' or param_name in ['profile_name', 'created_date', 'device_imei', 'version']:
                    self.logger.debug(f"Skipping additional param {param_name}: empty or metadata")
                    continue
                    
                try:
                    # Try to generate a command using the generic get_command method
                    short_param_name = param_name.split('.')[-1]
                    command = self.get_command(short_param_name, param_value, validate_command=validate_commands)
                    if command:
                        # Remove trailing '#' for SZCS commands during profile loading
                        if command.startswith("ATSMS:SZCS#") and command.endswith('#'):
                            command = command[:-1]
                            self.logger.debug(f"Removed trailing '#' for SZCS command in bulk operation: {command}")
                        commands.append(command)
                        self.logger.info(f"Added bulk command for additional parameter {param_name}: {command}")
                        processed_params.add(param_name)
                    else:
                        self.logger.debug(f"No command generated for additional parameter {param_name}")
                        
                except Exception as e:
                    self.logger.debug(f"Could not generate command for additional parameter {param_name}: {str(e)}")
        
        # Log which parameters were not processed
        unprocessed_params = set(parameters.keys()) - processed_params
        if unprocessed_params:
            self.logger.warning(f"Unprocessed parameters ({len(unprocessed_params)}): {unprocessed_params}")
            for param_name in unprocessed_params:
                param_value = parameters[param_name]
                self.logger.warning(f"Unprocessed param: {param_name} = {param_value} (type: {type(param_value).__name__})")
        
        # Check if we have critical validation errors that should block the operation
        if validation_errors:
            error_summary = f"Critical parameter validation failed: {'; '.join(validation_errors)}"
            self.logger.error(error_summary)
            raise ValueError(error_summary)
        
        # Log validation warnings for non-critical parameters
        if validation_warnings:
            warning_summary = f"Non-critical parameter validation warnings: {'; '.join(validation_warnings)}"
            self.logger.warning(warning_summary)

        self.logger.info(f"Generated {len(commands)} bulk commands from {len(parameters)} parameters")
        return commands
    
    # Format methods for bulk command generation
    def format_apn_command(self, apn: str, user: str = "", password: str = "") -> str:
        """Format APN configuration command."""
        if hasattr(self, 'command_validation_gateway') and self.command_validation_gateway:
            try:
                command = f"ATSMS:APN,{apn},{user},{password}#"
                self.command_validation_gateway.validate_command(command, source="bulk_commands")
                return command
            except ValueError as e:
                raise ValueError(f"APN command validation failed: {str(e)}")
        return f"ATSMS:APN,{apn},{user},{password}#"
    
    def _determine_server_type(self, address: str) -> str:
        """Determine server type: 0 for IP address, 1 for domain name."""
        import re
        # Check if address is an IP address (IPv4)
        ip_pattern = r'^(\d{1,3}\.){3}\d{1,3}$'
        if re.match(ip_pattern, address):
            return '0'  # IP address
        else:
            return '1'  # Domain name
    
    def format_server_command(self, address: str, port: str) -> str:
        """Format server configuration command."""
        # Determine server type (0 for IP, 1 for domain)
        server_type = self._determine_server_type(address)
        command = f"ATSMS:SERVER,{server_type},{address},{port},0#"
        
        if hasattr(self, 'command_validation_gateway') and self.command_validation_gateway:
            try:
                self.command_validation_gateway.validate_command(command, source="bulk_commands")
            except ValueError as e:
                raise ValueError(f"Server command validation failed: {str(e)}")
        return command

        command = f"ATSMS:TIMER,{moving_time},{stopped_time}#"
        
        if hasattr(self, 'command_validation_gateway') and self.command_validation_gateway:
