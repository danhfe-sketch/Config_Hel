    def __init__(self, serial_service, parameter_ui_list_path: str):
        """
        Initialize the Universal Profile Manager.
        
        Args:
            serial_service: The serial communication service instance
            parameter_ui_list_path: Path to the Parameter UI List markdown file
        """
        self.serial_service = serial_service
        self.parameter_ui_list_path = parameter_ui_list_path
        
        # Initialize core services
        self.spec_parser = ParameterSpecificationParser(parameter_ui_list_path)
        self.profile_service = UniversalProfileService(self.spec_parser)
        self.device_manager = DeviceCommunicationManager(serial_service, self.spec_parser)
        
        # Current profile state
        self._current_profile: Optional[UniversalProfile] = None
        self._operation_cancelled = False
        self._current_operation: Optional[str] = None
    
    async def load_from_device(self, progress_callback: Optional[Callable] = None) -> ProfileOperationResult:
        """
        Load the current configuration from the LT32 PRO device into a profile.
        
        This method reads all supported parameters from the device and creates
        a universal profile with the current device configuration.
        
        Args:
            progress_callback: Optional callback for progress updates
                              Called with (current_step, total_steps, step_description, success)
        
        Returns:
            ProfileOperationResult: Result of the load operation including the profile
            
        Raises:
            ConnectionError: If device is not connected
        """
        start_time = time.time()
        self._operation_cancelled = False
        self._current_operation = "load_from_device"
        
        result = ProfileOperationResult(
            operation_type=OperationType.LOAD_FROM_DEVICE,
            success=False,
            started_at=datetime.now()
        )
        
        try:
            # Check device connection
            if not self.serial_service.is_connected:
                result.add_error("Device is not connected")
                return result
            
            # Update progress
            if progress_callback:
                progress_callback(0, 100, "Starting device read operation", None)
            
            # Read all parameters from device
            try:
                device_responses = await self.device_manager.read_all_parameters(
                    progress_callback=self._create_device_progress_callback(progress_callback, 10, 90)
                )
                
                if self._operation_cancelled:
                    result.add_error("Operation was cancelled")
                    return result
                
                if not device_responses:
                    result.add_error("No parameters could be read from device")
                    return result
                
                # Update progress
                if progress_callback:
                    progress_callback(90, 100, "Creating profile from device data", None)
                
                # Create profile from device responses
                profile_name = f"Device Profile {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}"
                profile = self.profile_service.create_profile_from_device(device_responses, profile_name)
                
                # Validate the created profile
                validation_result = self.profile_service.validate_profile(profile)
                
                # Add validation warnings to result
                for warning in validation_result.warnings:
                    result.add_warning(warning)
                
                if not validation_result.is_valid:
                    for error in validation_result.errors:
                        result.add_error(f"Profile validation: {error}")
                
                # Set the current profile
                self._current_profile = profile
                result.profile = profile
                result.success = True
                
                # Add parameter results
                for param_name, response in device_responses.items():
                    param_result = ParameterOperationResult(
                        parameter_name=param_name,
                        success=True,
                        value=profile.get_parameter_value(param_name),
                        device_response=response
                    )
                    result.add_parameter_result(param_result)
                
                # Final progress update
                if progress_callback:
                    progress_callback(100, 100, f"Successfully loaded {len(device_responses)} parameters", True)
                
                result.add_detail("device_responses_count", len(device_responses))
                result.add_detail("profile_name", profile_name)
                
            except Exception as e:
                result.add_error(f"Failed to read from device: {str(e)}")
                
        except Exception as e:
            result.add_error(f"Unexpected error during load from device: {str(e)}")
            
        finally:
            result.duration_seconds = time.time() - start_time
            result.completed_at = datetime.now()
            self._current_operation = None
        
        return result
    
    async def save_to_device(self, progress_callback: Optional[Callable] = None) -> ProfileOperationResult:
        """
        Save the current profile configuration to the LT32 PRO device.
        
        This method validates the current profile and writes all parameters
        to the device using the appropriate commands.
        
        Args:
            progress_callback: Optional callback for progress updates
                              Called with (current_step, total_steps, step_description, success)
        
        Returns:
            ProfileOperationResult: Result of the save operation
            
        Raises:
            ConnectionError: If device is not connected
            ValueError: If no profile is loaded or profile is invalid
        """
        start_time = time.time()
        self._operation_cancelled = False
        self._current_operation = "save_to_device"
        
        result = ProfileOperationResult(
            operation_type=OperationType.SAVE_TO_DEVICE,
            success=False,
            started_at=datetime.now()
        )
        
        try:
            # Check device connection
            if not self.serial_service.is_connected:
                result.add_error("Device is not connected")
                return result
            
            # Check if we have a current profile
            if not self._current_profile:
                result.add_error("No profile loaded to save to device")
                return result
            
            # Update progress
            if progress_callback:
                progress_callback(0, 100, "Validating profile before save", None)
            
            # Validate profile before saving
#            validation_result = self.profile_service.validate_profile(self._current_profile)
            
            # Add validation warnings to result
            # for warning in validation_result.warnings:
            #     result.add_warning(warning)
            
            # if not validation_result.is_valid:
            #     for error in validation_result.errors:
            #         result.add_error(f"Profile validation failed: {error}")
            #     return result
            
            # Update progress
            if progress_callback:
                progress_callback(10, 100, "Preparing parameters for device", None)
            
            # Prepare parameters for device writing
            parameters_to_write = {}
            complex_param_values = {}
            all_specs = self.spec_parser.parse_specifications()

            for param_name, param_value_obj in self._current_profile.parameters.items():
                # Normalize parameter name
                base_param_name = param_name.split('.')[-1]
                spec = all_specs.get(base_param_name)
                
                if spec:
                    if spec.fields:
                        field_name = param_name.split('.')[-1]
                        if base_param_name not in complex_param_values:
                            complex_param_values[base_param_name] = {}
                        complex_param_values[base_param_name][field_name] = param_value_obj.value
                    else:
                        parameters_to_write[base_param_name] = param_value_obj.value
            
            parameters_to_write.update(complex_param_values)

            # Debugging: Check odometer and hourmeter values before sending
            if 'odometer' in parameters_to_write:
                print(f"DEBUG: Odometer in parameters_to_write: {parameters_to_write['odometer']}")
            if 'hourmeter' in parameters_to_write:
                print(f"DEBUG: Hourmeter in parameters_to_write: {parameters_to_write['hourmeter']}")

            if not parameters_to_write:
                result.add_warning("No parameters to write to device")
                result.success = True
                return result
            
            # Write parameters to device
            try:
                write_results = await self.device_manager.write_all_parameters(
                    parameters_to_write,
                    progress_callback=self._create_device_progress_callback(progress_callback, 20, 90)
                )
                
                if self._operation_cancelled:
                    result.add_error("Operation was cancelled")
                    return result
                
                # Process write results
                successful_writes = 0
                failed_writes = 0
                
                for param_name, success in write_results.items():
                    param_result = ParameterOperationResult(
                        parameter_name=param_name,
                        success=success,
                        value=parameters_to_write.get(param_name)
                    )
                    
                    if not success:
                        param_result.error_message = "Failed to write to device"
                        failed_writes += 1
                    else:
                        successful_writes += 1
                    
                    result.add_parameter_result(param_result)
                
                # Update progress
                if progress_callback:
                    progress_callback(90, 100, "Finalizing save operation", None)
                
                # Determine overall success
                if successful_writes > 0:
                    result.success = True
                    if failed_writes > 0:
                        result.add_warning(f"Partial success: {successful_writes} parameters written, {failed_writes} failed")
                else:
                    result.add_error("No parameters could be written to device")
                
                # Final progress update
                if progress_callback:
                    status_msg = f"Saved {successful_writes}/{len(parameters_to_write)} parameters"
                    progress_callback(100, 100, status_msg, successful_writes > 0)
                
                result.add_detail("parameters_to_write", len(parameters_to_write))
                result.add_detail("successful_writes", successful_writes)
                result.add_detail("failed_writes", failed_writes)
                
            except Exception as e:
                result.add_error(f"Failed to write to device: {str(e)}")
                
        except Exception as e:
            result.add_error(f"Unexpected error during save to device: {str(e)}")
            
        finally:
            result.duration_seconds = time.time() - start_time
            result.completed_at = datetime.now()
            self._current_operation = None
        
        return result

    async def save_to_file(self, file_path: str, profile_name: Optional[str] = None) -> ProfileOperationResult:
        """
        Save the current profile to a JSON file.
        
        Args:
            file_path: Path where to save the profile file
            profile_name: Optional new name for the profile
        
        Returns:
            ProfileOperationResult: Result of the save operation
        """
        start_time = time.time()
        self._current_operation = "save_to_file"
        
        result = ProfileOperationResult(
            operation_type=OperationType.SAVE_TO_FILE,
            success=False,
            started_at=datetime.now()
        )
        
        try:
            # Check if we have a current profile
            if not self._current_profile:
                result.add_error("No profile loaded to save to file")
                return result
            
            # Create a copy of the profile for saving
            profile_to_save = self._current_profile.clone()
            
            # Update profile name if provided
            if profile_name:
                profile_to_save.name = profile_name
            
            # Update modification timestamp
            profile_to_save.modified_at = datetime.now()
            
            # Validate profile before saving
            validation_result = self.profile_service.validate_profile(profile_to_save)
            
            # Add validation warnings to result
            for warning in validation_result.warnings:
                result.add_warning(warning)
            
            if not validation_result.is_valid:
                for error in validation_result.errors:
                    result.add_error(f"Profile validation failed: {error}")
                return result
            
            # Save profile using the profile service
            try:
                success = self.profile_service.save_profile_to_file(profile_to_save, file_path)
                
                if success:
                    result.success = True
                    result.profile = profile_to_save
                    
                    # Add parameter results
                    for param_name, param_value in profile_to_save.parameters.items():
                        param_result = ParameterOperationResult(
                            parameter_name=param_name,
                            success=True,
                            value=param_value.value
                        )
                        result.add_parameter_result(param_result)
                    
                    result.add_detail("file_path", file_path)
                    result.add_detail("profile_name", profile_to_save.name)
                    result.add_detail("parameters_saved", len(profile_to_save.parameters))
                else:
                    result.add_error("Failed to save profile to file")
                    
            except Exception as e:
                result.add_error(f"Failed to save profile file: {str(e)}")
                
        except Exception as e:
            result.add_error(f"Unexpected error during save to file: {str(e)}")
            
        finally:
            result.duration_seconds = time.time() - start_time
            result.completed_at = datetime.now()
            self._current_operation = None
        
        return result

    def populate_ui_fields(self, ui_field_values: Dict[str, Any]) -> ValidationResult:
        """
        Populate UI fields from the current profile with enhanced visibility and dependency handling.
        
        Args:
            ui_field_values: Dictionary to populate with UI field values
        
        Returns:
            ValidationResult: Result of the population operation
        """
        result = ValidationResult(is_valid=True)
        
        if not self._current_profile:
            result.add_error("No profile loaded to populate UI fields")
            return result
        
        try:
            # Clear existing UI values
            ui_field_values.clear()
            
            # Create context for visibility evaluation
            context = {name: pv.value for name, pv in self._current_profile.parameters.items()}
            
            # Track parameters that were skipped due to visibility
            skipped_parameters = []
            populated_parameters = []
            
            # Populate UI fields from profile parameters
            for param_name, param_value in self._current_profile.parameters.items():
                try:
                    # Get parameter specification for UI mapping
                    param_spec = self.spec_parser.get_parameter_spec(param_name)
                    
                    if param_spec:
                        # Check if parameter should be visible
                        if self._evaluate_visibility_condition(param_spec.visibility_condition, context):
                            # Handle complex parameters with multiple fields
                            if param_spec.fields:
                                # Complex parameter - populate individual fields
                                for field in param_spec.fields:
                                    field_name = f"{param_name}.{field.name}"
                                    if hasattr(param_value.value, field.name):
                                        field_value = getattr(param_value.value, field.name)
                                        ui_field_values[field_name] = field_value
                                        populated_parameters.append(field_name)
                                    elif isinstance(param_value.value, dict) and field.name in param_value.value:
                                        ui_field_values[field_name] = param_value.value[field.name]
                                        populated_parameters.append(field_name)
                            else:
                                # Simple parameter
                                ui_field_values[param_name] = param_value.value
                                populated_parameters.append(param_name)
                        else:
                            skipped_parameters.append(param_name)
                            result.add_warning(f"Parameter '{param_name}' is not visible with current values")
                    else:
                        # Parameter not found in specifications - include it anyway
                        ui_field_values[param_name] = param_value.value
                        populated_parameters.append(param_name)
                        result.add_warning(f"No specification found for parameter '{param_name}'")
                        
                except Exception as e:
                    result.add_warning(f"Failed to populate parameter '{param_name}': {str(e)}")
            
            # Validate field dependencies after population
            dependency_result = self._validate_field_dependencies(ui_field_values)
            if not dependency_result.is_valid:
                result.warnings.extend(dependency_result.warnings)
                result.errors.extend(dependency_result.errors)
            
            result.add_detail("parameters_populated", len(populated_parameters))
            result.add_detail("parameters_skipped", len(skipped_parameters))
            result.add_detail("populated_parameters", populated_parameters)
            result.add_detail("skipped_parameters", skipped_parameters)
            
        except Exception as e:
            result.is_valid = False
            result.add_error(f"Failed to populate UI fields: {str(e)}")
        
        return result
    
    def extract_ui_field_values(self, ui_field_values: Dict[str, Any]) -> ValidationResult:
        """
        Extract UI field values and update the current profile with enhanced validation and dependency handling.
        
        Args:
            ui_field_values: Dictionary of UI field values
        
        Returns:
            ValidationResult: Result of the extraction operation
        """
        result = ValidationResult(is_valid=True)
        
        try:
            # Validate field dependencies first
            dependency_result = self._validate_field_dependencies(ui_field_values)
            if not dependency_result.is_valid:
                result.warnings.extend(dependency_result.warnings)
                result.errors.extend(dependency_result.errors)
            
            # Track processing results
            processed_parameters = []
            failed_parameters = []
            updated_dependencies = []
            
            # Create a new profile if none exists
            if not self._current_profile:
                profile_name = f"UI Profile {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}"
                self._current_profile = self.profile_service.create_profile_from_ui(ui_field_values, profile_name)
                processed_parameters = list(ui_field_values.keys())
            else:
                # Update existing profile with UI values
                for param_name, value in ui_field_values.items():
                    try:
                        # Get parameter specification for validation
                        param_spec = self.spec_parser.get_parameter_spec(param_name)
                        
                        # Check if this is a complex parameter field first
                        if '.' in param_name:
                            # Handle complex parameter field
                            base_param, field_name = param_name.split('.', 1)
                            self._set_complex_parameter_field(base_param, field_name, value)
                            processed_parameters.append(param_name)
                        elif param_spec:
                            # Validate individual parameter value
                            param_validation = param_spec.validate_value(value)
                            
                            if param_validation.is_valid:
                                # Simple parameter
                                old_value = self._current_profile.get_parameter_value(param_name)
                                self._current_profile.set_parameter_value(param_name, value, "user")
                                
                                # Check if this parameter affects other parameters
                                if old_value != value:
                                    dependencies = self._update_dependent_fields(param_name, value, ui_field_values)
                                    updated_dependencies.extend(dependencies)
                                
                                processed_parameters.append(param_name)
                            else:
                                # Use default value for invalid parameters
                                default_value = param_spec.get_default_value()
                                if default_value is not None:
                                    self._current_profile.set_parameter_value(param_name, default_value, "default")
                                    result.add_warning(f"Invalid value for '{param_name}': {param_validation.errors[0]}. Using default: {default_value}")
                                    processed_parameters.append(param_name)
                                else:
                                    failed_parameters.append(param_name)
                                    result.add_error(f"Invalid value for '{param_name}': {param_validation.errors[0]}")
                        else:
                            # Parameter not in specifications - set it anyway with warning
                            self._current_profile.set_parameter_value(param_name, value, "user")
                            processed_parameters.append(param_name)
                            result.add_warning(f"No specification found for parameter '{param_name}'")
                            
                    except Exception as e:
                        failed_parameters.append(param_name)
                        result.add_error(f"Failed to process parameter '{param_name}': {str(e)}")
            
            # Validate the updated profile
            validation_result = self.profile_service.validate_profile(self._current_profile)
            
            # Merge validation results
            result.warnings.extend(validation_result.warnings)
            if not validation_result.is_valid:
                result.errors.extend(validation_result.errors)
                result.is_valid = False
            
            result.add_detail("parameters_extracted", len(processed_parameters))
            result.add_detail("parameters_failed", len(failed_parameters))
            result.add_detail("dependencies_updated", len(updated_dependencies))
            result.add_detail("processed_parameters", processed_parameters)
            result.add_detail("failed_parameters", failed_parameters)
            result.add_detail("updated_dependencies", updated_dependencies)
            result.add_detail("profile_name", self._current_profile.name)
            
        except Exception as e:
            result.add_error(f"Failed to extract UI field values: {str(e)}")
        
        return result

    def get_visible_parameters(self, context: Optional[Dict[str, Any]] = None) -> List[str]:
        """
        Get list of parameters that should be visible given the current context.
        
        Args:
            context: Optional context dictionary. If None, uses current profile values.
        
        Returns:
            List[str]: List of visible parameter names
        """
        if context is None and self._current_profile:
            context = {name: pv.value for name, pv in self._current_profile.parameters.items()}
        elif context is None:
            context = {}
        
        visible_params = []
        
        try:
            all_params = self.spec_parser.get_all_parameters()
            
            for param_name in all_params:
                param_spec = self.spec_parser.get_parameter_spec(param_name)
                if param_spec:
                    if self._evaluate_visibility_condition(param_spec.visibility_condition, context):
                        visible_params.append(param_name)
                else:
                    # Include parameters without specifications
                    visible_params.append(param_name)
        
        except Exception:
            # If visibility evaluation fails, return all parameters
            visible_params = self.get_supported_parameters()
        
        return visible_params
    
    def get_parameter_dependencies(self, parameter_name: str) -> Dict[str, List[str]]:
        """
        Get dependency information for a parameter.
        
        Args:
            parameter_name: Name of the parameter to check
        
        Returns:
            Dict containing 'depends_on' and 'affects' lists
        """
        result = {
            'depends_on': [],
            'affects': []
        }
        
        try:
            # Get what this parameter depends on
            result['depends_on'] = self._get_field_dependencies(parameter_name)
            
            # Get what parameters this one affects
            all_params = self.spec_parser.get_all_parameters()
            for param_name in all_params:
                dependencies = self._get_field_dependencies(param_name)
                if parameter_name in dependencies:
                    result['affects'].append(param_name)
        except Exception:
            pass
        
        return result
    
    def validate_ui_field_values(self, ui_field_values: Dict[str, Any]) -> ValidationResult:
        """
        Validate UI field values without updating the profile.
        
        Args:
            ui_field_values: Dictionary of UI field values to validate
        
        Returns:
            ValidationResult: Validation result with detailed feedback
        """
        result = ValidationResult(is_valid=True)
        
        try:
            # Validate field dependencies
            dependency_result = self._validate_field_dependencies(ui_field_values)
            result.warnings.extend(dependency_result.warnings)
            result.errors.extend(dependency_result.errors)
            
            # Validate individual field values
            valid_fields = []
            invalid_fields = []
            
            for field_name, field_value in ui_field_values.items():
                param_spec = self.spec_parser.get_parameter_spec(field_name)
                
                if param_spec:
                    field_validation = param_spec.validate_value(field_value)
                    
                    if field_validation.is_valid:
                        valid_fields.append(field_name)
                    else:
                        invalid_fields.append(field_name)
                        result.add_error(f"Invalid value for '{field_name}': {field_validation.errors[0]}")
                else:
                    # No specification - assume valid but warn
                    valid_fields.append(field_name)
                    result.add_warning(f"No specification found for field '{field_name}'")
            
            if invalid_fields:
                result.is_valid = False
            
            result.add_detail("valid_fields", valid_fields)
            result.add_detail("invalid_fields", invalid_fields)
            result.add_detail("total_fields", len(ui_field_values))
            
        except Exception as e:
            result.add_error(f"Failed to validate UI field values: {str(e)}")
            result.is_valid = False
        
        return result

    def _evaluate_visibility_condition(self, condition: Optional[str], context: Dict[str, Any]) -> bool:
        """
        Evaluate a visibility condition against the current parameter context.
        
        Args:
            condition: Visibility condition string (e.g., "ignition_mode == 'Virtual por bateria'")
            context: Dictionary of parameter values for evaluation
        
        Returns:
            bool: True if the condition is met or no condition exists
        """
        if not condition:
            return True
        
        try:
            # Simple condition parsing - supports basic equality and boolean checks
            condition = condition.strip()
            
            # Handle boolean conditions
            if condition in context:
                return bool(context[condition])
            
            # Handle equality conditions
            if '==' in condition:
                left, right = condition.split('==', 1)
                left = left.strip()
                right = right.strip().strip("'\"")
                
                if left in context:
                    return str(context[left]) == right
            
            # Handle boolean true/false conditions
            if '== true' in condition:
                param_name = condition.replace('== true', '').strip()
                if param_name in context:
                    return bool(context[param_name])
            
            if '== false' in condition:
                param_name = condition.replace('== false', '').strip()
                if param_name in context:
                    return not bool(context[param_name])
            
            # Default to visible if condition cannot be evaluated
            return True
            
        except Exception:
            # If condition evaluation fails, default to visible
            return True
    
    def _update_dependent_fields(self, changed_field: str, new_value: Any, ui_field_values: Dict[str, Any]) -> List[str]:
        """
        Update fields that depend on the changed field.
        
        Args:
            changed_field: Name of the field that changed
            new_value: New value of the changed field
            ui_field_values: Dictionary of all UI field values
        
        Returns:
            List[str]: List of field names that were updated
        """
        updated_fields = []
        
        try:
            # Get all parameter specifications to check for dependencies
            all_params = self.spec_parser.get_all_parameters()
            
            for param_name in all_params:
                param_spec = self.spec_parser.get_parameter_spec(param_name)
                if param_spec and param_spec.visibility_condition:
                    # Check if this parameter's visibility depends on the changed field
                    if changed_field in param_spec.visibility_condition:
                        # Update the context and re-evaluate visibility
                        context = dict(ui_field_values)
                        context[changed_field] = new_value
                        
                        is_visible = self._evaluate_visibility_condition(param_spec.visibility_condition, context)
                        
                        # If parameter becomes invisible, remove it from UI values
                        if not is_visible and param_name in ui_field_values:
                            del ui_field_values[param_name]
                            updated_fields.append(param_name)
                        # If parameter becomes visible and has a default, add it
                        elif is_visible and param_name not in ui_field_values:
                            default_value = param_spec.get_default_value()
                            if default_value is not None:
                                ui_field_values[param_name] = default_value
                                updated_fields.append(param_name)
            
        except Exception:
            # If dependency update fails, continue without updates
            pass
        
        return updated_fields
    
    def _get_field_dependencies(self, field_name: str) -> List[str]:
        """
        Get the list of fields that the given field depends on.
        
        Args:
            field_name: Name of the field to check dependencies for
        
        Returns:
            List[str]: List of field names that this field depends on
        """
        dependencies = []
        
        try:
            param_spec = self.spec_parser.get_parameter_spec(field_name)
            if param_spec and param_spec.visibility_condition:
                # Simple dependency extraction from visibility condition
                condition = param_spec.visibility_condition
                
                # Extract parameter names from condition
                import re
                # Find parameter names (alphanumeric with underscores)
                param_matches = re.findall(r'\b([a-zA-Z_][a-zA-Z0-9_]*)\b', condition)
                
                # Filter out keywords and operators
                keywords = {'true', 'false', 'and', 'or', 'not', 'in', 'is'}
                dependencies = [match for match in param_matches if match not in keywords]
        
        except Exception:
            pass
        
        return dependencies
    
    def _validate_field_dependencies(self, ui_field_values: Dict[str, Any]) -> ValidationResult:
        """
        Validate that all field dependencies are satisfied.
        
        Args:
            ui_field_values: Dictionary of UI field values
        
        Returns:
            ValidationResult: Result of dependency validation
        """
        result = ValidationResult(is_valid=True)
        
        try:
            # Check each field's dependencies
            for field_name, field_value in ui_field_values.items():
                dependencies = self._get_field_dependencies(field_name)
                
                for dependency in dependencies:
                    if dependency not in ui_field_values:
                        result.add_warning(f"Field '{field_name}' depends on '{dependency}' which is not present")
                    else:
                        # Check if the dependency relationship is valid
                        param_spec = self.spec_parser.get_parameter_spec(field_name)
                        if param_spec and param_spec.visibility_condition:
                            is_visible = self._evaluate_visibility_condition(
                                param_spec.visibility_condition, ui_field_values
                            )
                            
                            if not is_visible:
                                result.add_error(f"Field '{field_name}' should not be visible with current dependency values. "
                                               f"Condition: {param_spec.visibility_condition}")
        
        except Exception as e:
            result.add_warning(f"Failed to validate field dependencies: {str(e)}")
        
        return result

    def _parse_device_response(self, param_spec, response: str) -> Any:
        """
        Parse a device response according to parameter specification.
        """
        if not response:
            return param_spec.get_default_value()

        raw_value = response.strip()

        # For complex parameters with fields, parse into a dictionary
        if param_spec.fields:
            # Special handling for server_settings
            if param_spec.name == 'server_settings':
                result = {}
                parts = raw_value.split(':')[-1].split(',')
                
                port_field = None
                for field in param_spec.fields:
                    if field.name == 'port':
                        port_field = field
                        break

                if len(parts) >= 3 and port_field:
                    result['address'] = parts[1]
                    result['port'] = self._convert_field_value(port_field, parts[2])
                else:
                    result['address'] = ''
                    if port_field:
                        result['port'] = port_field.get_default_value()
                    else:
                        result['port'] = 0 # fallback default
                return result

            # Special handling for ignition_type due to its unique response format
            if param_spec.name == 'ignition_type':
                result = {}
                # Default ignition mode mapping based on response
                ignition_mode_value = 'F├¡sica (Physical)'
                if 'ACCLINE=0' in raw_value:
                    ignition_mode_value = 'Virtual por aceler├┤metro (Virtual by accelerometer)'
                # Note: Only include voltage fields if battery mode is active
                result['ignition_mode'] = ignition_mode_value
                return result

            # Extract the value part of the response (e.g., from "APN:val1,val2")
            if ':' in raw_value:
                raw_value = raw_value.split(':', 1)[-1].strip()
            
            parts = raw_value.split(',')
            provisional = {}
            for i, field in enumerate(param_spec.fields):
                if i < len(parts):
                    # Convert each part to the correct field type
                    provisional[field.name] = self._convert_field_value(field, parts[i])
                else:
                    provisional[field.name] = field.get_default_value()

            # Filter out fields that should not be visible given the current context
            visible_field_names = {f.name for f in param_spec.get_visible_fields(provisional)}
            filtered: Dict[str, Any] = {}
            for field in param_spec.fields:
                if field.visibility_condition is None or field.name in visible_field_names:
                    filtered[field.name] = provisional[field.name]
            return filtered

        # For simple parameters, extract the single value
        last_delim = max(raw_value.rfind('='), raw_value.rfind(':'))
        if last_delim != -1:
            raw_value = raw_value[last_delim + 1:].strip()
            
        return self._convert_and_map_value(param_spec, raw_value)

    def _clean_time_parameter_field(self, field, raw_value: str) -> str:
        """

        Args:

    def _extract_numeric_value_from_time_string(self, time_string: str) -> str:
        """
        Extract numeric value from time parameter strings.
        
        Examples:
        - "60s" -> "60"
        - "ACC OFF:120s" -> "120"
        - "180" -> "180"
        
        Args:
            time_string: Time string that may contain units and prefixes
            
        Returns:
            str: Numeric value only
        """
        if not time_string:
            return ""
        
        # Extract all digits from the string
        import re
        digits = re.findall(r'\d+', str(time_string))
        
        # Return the last number found (handles cases like "ACC OFF:120s")

    def _convert_field_value(self, field, raw_value: str) -> Any:
        """Convert a raw field value to the appropriate type and map to options."""
        # Apply time parameter cleaning for time-related fields
        cleaned_value = self._clean_time_parameter_field(field, raw_value)
        
        # First, convert to a Python type
        typed_value = cleaned_value
        if field.ui_element.value == "number_input":
            try:
                typed_value = int(cleaned_value)
            except (ValueError, TypeError):
                try:
                    typed_value = float(cleaned_value)
                except (ValueError, TypeError):
                    # For validation errors like port:0, we keep the value for now
                    typed_value = cleaned_value
        elif field.ui_element.value == "toggle_switch":
            typed_value = cleaned_value.lower() in ("true", "1", "on", "yes", "enabled")

        # Then, map to UI option string if applicable
        if field.options:
            if field.ui_element.value == "dropdown":
                try:
                    index = int(str(typed_value).strip())
                    # Clamp out-of-range indices to nearest valid option
                    if index < 0:
                        index = 0
                    if index >= len(field.options):
                        index = len(field.options) - 1
                    return field.options[index]
                except (ValueError, TypeError):
                    if typed_value in field.options:
                        return typed_value
            elif field.ui_element.value == "toggle_switch":
                # This mapping is for UI display, the validator should handle the boolean
                if typed_value is True and len(field.options) > 0:
                    return field.options[0]
                elif typed_value is False and len(field.options) > 1:
                    return field.options[1]
        
        return typed_value

    def _convert_and_map_value(self, param_spec, raw_value: str) -> Any:
        """Convert a raw parameter value to the appropriate type and map to UI options if needed."""
        # First, convert to a Python type (int, bool, etc.)
        typed_value = self._convert_simple_value(param_spec, raw_value)

        # Then, map this typed value to the corresponding UI option string if applicable
        if param_spec.options:
            if param_spec.ui_element and param_spec.ui_element.value == "dropdown":
                try:
                    # Device value is an index for the options list
                    index = int(str(typed_value).strip())
                    # Clamp out-of-range indices to nearest valid option
                    if index < 0:
                        index = 0
                    if index >= len(param_spec.options):
                        index = len(param_spec.options) - 1
                    return param_spec.options[index]
                except (ValueError, TypeError):
                    # If it's not an index, it might be the value itself
                    if typed_value in param_spec.options:
                        return typed_value
            
            elif param_spec.ui_element and param_spec.ui_element.value == "toggle_switch":
                # Map boolean to "Enabled" / "Disabled" based on options
                if typed_value is True and "Enabled" in param_spec.options:
                    return "Enabled"
                elif typed_value is False and "Disabled" in param_spec.options:
                    return "Disabled"

        # For other cases, or if mapping fails, return the raw value or the typed value
        # Let's return the typed_value, as it's more likely to be what validation expects if no options are matched
        return typed_value

    def _convert_simple_value(self, param_spec, raw_value: str) -> Any:

    def validate_profile(self, profile: UniversalProfile) -> ValidationResult:
        """
        Validate a profile against current parameter specifications.
        
        Args:
            profile: Profile to validate
            
        Returns:
            ValidationResult: Comprehensive validation result
        """
        result = ValidationResult(is_valid=True)
        
        # Basic profile validation
        if not profile.name:
            result.add_error("Profile name is required")
        
        if not profile.version:
            result.add_error("Profile version is required")
        
        # Validate individual parameters
        for param_name, param_value in profile.parameters.items():
            # Try exact lookup first; if not found, try the short parameter name after the last dot
            param_spec = self.spec_parser.get_parameter_spec(param_name)
            lookup_name = param_name
            if not param_spec:
                short_name = param_name.split('.')[-1]
                param_spec = self.spec_parser.get_parameter_spec(short_name)
                if param_spec:
                    lookup_name = short_name

            if param_spec:
                value_to_validate = param_value.value

                # If the profile stored a dict like {'tcp_ack': 'Disabled'} for a single-field parameter,
                # unwrap it to the inner value (common when loading device dump outputs).
                if not param_spec.fields and isinstance(value_to_validate, dict):
                    # If dict has a single key that matches the short parameter name, extract it
                    if len(value_to_validate) == 1:
                        sole_key = next(iter(value_to_validate))
                        if sole_key == lookup_name or sole_key.endswith(lookup_name):
                            value_to_validate = value_to_validate[sole_key]

                # If parameter has conditional fields and value is a dict, attempt coercion then validate visible fields
                if param_spec.fields and isinstance(value_to_validate, dict):
                    # Normalize fields container
                    fields_list = param_spec.fields
                    if isinstance(fields_list, dict):
                        fields_list = list(fields_list.values())
                    coerced = dict(value_to_validate)  # shallow copy to avoid mutating original

                    # Coerce common device dump formats to expected UI types
                    for field in fields_list:
                        field_name = getattr(field, "name", None) or (field.get("name") if isinstance(field, dict) else None)
                        if not field_name or field_name not in coerced:
                            continue
                        v = coerced[field_name]

                        ui_type = getattr(field, "ui_element", None) or (field.get("ui_element") if isinstance(field, dict) else None)
                        options = getattr(field, "options", None) or (field.get("options") if isinstance(field, dict) else None) or []

                        # Coerce numeric strings to ints/floats for number_input
                        if ui_type == "number_input" and isinstance(v, str):
                            s = v.strip()
                            try:
                                if s.isdigit():
                                    coerced[field_name] = int(s)
                                else:
                                    coerced[field_name] = float(s)
                            except Exception:
                                coerced[field_name] = v

                        # Map booleans/numeric flags to option labels for toggle_switch
                        if ui_type == "toggle_switch" and options:
                            opts = [opt.strip() for opt in options]
                            if isinstance(v, bool):
                                coerced[field_name] = opts[0] if v else (opts[1] if len(opts) > 1 else opts[0])
                            elif isinstance(v, (int, float)):
                                coerced[field_name] = opts[0] if int(v) == 1 else (opts[1] if len(opts) > 1 else opts[0])
                            elif isinstance(v, str) and v.strip().lower() in ("true", "false", "1", "0", "enabled", "disabled"):
                                s = v.strip().lower()
                                if s in ("true", "1", "enabled"):
                                    coerced[field_name] = opts[0]
                                else:
                                    coerced[field_name] = opts[1] if len(opts) > 1 else opts[0]

                    # If numeric fields coerced to 0 but validation expects min>0, replace with default instead of failing
                    try:
                        for field in fields_list:
                            field_name = getattr(field, "name", None) or (field.get("name") if isinstance(field, dict) else None)
                            if not field_name or field_name not in coerced:
                                continue
                            validation_rule = getattr(field, "validation", None) or (field.get("validation") if isinstance(field, dict) else None) or ""
                            val = coerced.get(field_name)
                            if isinstance(val, (int, float)) and int(val) == 0 and validation_rule and "-" in validation_rule:
                                parts = validation_rule.split("-", 1)
                                try:
                                    min_val = int(parts[0])
                                    if min_val > 0:
                                        # replace with the field's default if available
                                        default = getattr(field, "get_default_value", None)
                                        if callable(default):
                                            coerced[field_name] = default()
                                        else:
                                            # try attribute default or leave removed
                                            fld_def = getattr(field, "default", None) or None
                                            if fld_def is not None:
                                                coerced[field_name] = fld_def
                                            else:
                                                coerced.pop(field_name, None)
                                except Exception:
                                    pass
                    except Exception:
                        pass

                    # Evaluate visible fields based on coerced context
                    context = coerced.copy()
                    try:
                        visible_fields = {f.name for f in param_spec.get_visible_fields(context)}
                    except Exception:
                        visible_fields = set(coerced.keys())
                    filtered_value = {k: v for k, v in coerced.items() if k in visible_fields}

                    # Validate using the filtered/coerced structure
                    try:
                        param_validation = param_spec.validate_value(filtered_value)
                    except Exception as e:
                        pv = ValidationResult(is_valid=False)
                        pv.add_error(f"Validation exception: {e}")
                        param_validation = pv
                else:
                    # Handle single-value toggle switches stored as booleans by mapping them to option labels
                    ui_type = getattr(param_spec, "ui_element", None)
                    if ui_type == "toggle_switch" and isinstance(value_to_validate, bool):
                        options = getattr(param_spec, "options", []) or []
                        opts = [opt.strip() for opt in options]
                        if opts:
                            value_to_validate = opts[0] if value_to_validate else (opts[1] if len(opts) > 1 else opts[0])

                    # Normal validation (guard against exceptions)
                    try:
                        param_validation = param_spec.validate_value(value_to_validate)
                    except Exception as e:
                        pv = ValidationResult(is_valid=False)
                        pv.add_error(f"Validation exception: {e}")
                        param_validation = pv

                if not param_validation.is_valid:
                    for error in param_validation.errors:
                        result.add_error(f"Parameter '{lookup_name}': {error}")
                
                for warning in param_validation.warnings:
                    result.add_warning(f"Parameter '{lookup_name}': {warning}")
                
                # Update parameter's validation status
                param_value.set_validation_result(param_validation)
            else:
                # Parameter not found in current specifications
                result.add_warning(f"Parameter '{param_name}' not found in current specifications")
        
        # Check for missing required parameters
        all_param_names = self.spec_parser.get_all_parameters()
        for param_name in all_param_names:
            if param_name not in profile.parameters:
                param_spec = self.spec_parser.get_parameter_spec(param_name)
                if param_spec:
                    # Add default value for missing parameter
                    default_value = param_spec.get_default_value()
                    profile.set_parameter_value(param_name, default_value, "default")
                    result.add_warning(f"Added default value for missing parameter '{param_name}'")
        
        # Validate parameter dependencies and visibility conditions
        self._validate_parameter_dependencies(profile, result)
        
        return result
    
    def _validate_parameter_dependencies(self, profile: UniversalProfile, result: ValidationResult) -> None:
        """
        Validate parameter dependencies and visibility conditions.
        
        Args:
            profile: Profile to validate
            result: Validation result to update
        """
        # Create context for visibility checks
        context = {}
        for param_name, param_value in profile.parameters.items():
            context[param_name] = param_value.value
        
        # Check visibility conditions for all parameters
        for param_name, param_value in profile.parameters.items():
            param_spec = self.spec_parser.get_parameter_spec(param_name)
            if param_spec and not param_spec.is_visible(context):
                result.add_warning(f"Parameter '{param_name}' should not be visible with current values")
        
        # Check field visibility for complex parameters
        for param_name, param_value in profile.parameters.items():
            param_spec = self.spec_parser.get_parameter_spec(param_name)
            if param_spec and param_spec.fields and isinstance(param_value.value, dict):
                visible_fields = param_spec.get_visible_fields(param_value.value)
                visible_field_names = {field.name for field in visible_fields}
                
                for field_name in param_value.value.keys():
                    if field_name not in visible_field_names:
                        result.add_warning(f"Field '{field_name}' in parameter '{param_name}' should not be visible")
    
    def migrate_profile(self, profile: UniversalProfile) -> UniversalProfile:
        """
        Migrate a profile to the current format and specifications.
        
        Args:
            profile: Profile to migrate
            
        Returns:
            UniversalProfile: Migrated profile
        """
        # For now, this is a simple migration that ensures all current parameters
        # have values (using defaults for missing ones)
        
        migrated_profile = profile.clone()
        
        # Add missing parameters with default values (do not mark as modified)
        # But avoid adding defaults if an equivalent parameter already exists with a section prefix
        all_param_names = self.spec_parser.get_all_parameters()
        existing_keys = set(migrated_profile.parameters.keys())
        for param_name in all_param_names:
            # Detect equivalent present keys (exact or with section prefixes, e.g. 'network_parameters.server_settings')
            def _has_equivalent(name: str) -> bool:
                if name in existing_keys:
                    return True
                for k in existing_keys:
                    if k == name or k.endswith("." + name):
                        return True
                return False

            if _has_equivalent(param_name):
                continue

            param_spec = self.spec_parser.get_parameter_spec(param_name)
            if param_spec:
                default_value = param_spec.get_default_value()
                migrated_profile.parameters[param_name] = ParameterValue(
                    value=default_value,
                    source="default"
                )

        # Clean up field visibility: remove fields that should not be visible
        for param_name, param_value in list(migrated_profile.parameters.items()):
            param_spec = self.spec_parser.get_parameter_spec(param_name)
            if not param_spec or not param_spec.fields:
                continue
            value = param_value.value
            if not isinstance(value, dict):
                continue
            # Determine which fields are visible based on current values
            visible_fields = {f.name for f in param_spec.get_visible_fields(value)}
            cleaned: Dict[str, Any] = {}
            for field in param_spec.fields:
                if field.visibility_condition is None:
                    # Always include non-conditional fields
                    if field.name in value:
                        cleaned[field.name] = value[field.name]
                else:
                    if field.name in visible_fields and field.name in value:
                        cleaned[field.name] = value[field.name]
            # Replace with cleaned value if different (do not mark as modified)
            if cleaned != value:
                pv = migrated_profile.parameters.get(param_name)
                if pv:
                    pv.value = cleaned
                else:
                    migrated_profile.parameters[param_name] = ParameterValue(
                        value=cleaned,
                        source="device"
                    )
        
        # Update metadata to reflect migration
        migrated_profile.metadata.parameter_spec_version = "1.0"  # Current version
        migrated_profile.modified_at = datetime.now()
        
        return migrated_profile

    def compare_profiles(self, profile1: UniversalProfile, profile2: UniversalProfile) -> Dict[str, Any]:
        """
        Compare two profiles and return differences.
        
        Args:
            profile1: First profile to compare
            profile2: Second profile to compare
            
        Returns:
            Dict[str, Any]: Comparison results
        """
        differences = {
            "added_parameters": [],
            "removed_parameters": [],
            "changed_parameters": [],
            "unchanged_parameters": []
        }
        
        params1 = set(profile1.parameters.keys())
        params2 = set(profile2.parameters.keys())
        
        # Find added and removed parameters
        differences["added_parameters"] = list(params2 - params1)
        differences["removed_parameters"] = list(params1 - params2)
        
        # Find changed and unchanged parameters
        common_params = params1 & params2
        
        for param_name in common_params:
            value1 = profile1.get_parameter_value(param_name)
            value2 = profile2.get_parameter_value(param_name)
            
            if value1 != value2:
                differences["changed_parameters"].append({
                    "parameter": param_name,
                    "old_value": value1,
                    "new_value": value2
                })
            else:
                differences["unchanged_parameters"].append(param_name)
        
        return differences

    def _init_apn_validation_patterns(self):
        """Initialize enhanced APN validation patterns."""
        # Domain name pattern - prevents consecutive dots and malformed domains
        self.apn_domain_pattern = r'^(?!.*\.\.+)(?!^\.)(?!.*\.$)[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?(?:\.[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?)*\.[a-zA-Z]{2,}$'
        
        # IP address pattern - validates IPv4 addresses
        self.apn_ip_pattern = r'^(?:(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)\.){3}(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)$'
        
        # URL pattern - validates HTTP/HTTPS URLs
        self.apn_url_pattern = r'^https?://(?:[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?\.)+[a-zA-Z]{2,}(?:/.*)?$'
        
        # Simple APN name pattern - for carrier-specific APNs without dots
        self.apn_simple_pattern = r'^[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?$'
        
        # Combined pattern for all APN formats
        self.apn_combined_pattern = f'({self.apn_domain_pattern[1:-1]}|{self.apn_ip_pattern[1:-1]}|{self.apn_url_pattern[1:-1]}|{self.apn_simple_pattern[1:-1]})'
        
        # APN credential pattern - allows alphanumeric, special chars commonly used in credentials
        self.apn_credential_pattern = r'^[a-zA-Z0-9@._-]*$'
        
        # Common carrier APN patterns for suggestions
        self.common_carrier_apns = {
            'vivo': ['zap.vivo.com.br', 'vivo.br'],
            'claro': ['claro.com.br', 'bandalarga.claro.com.br'],
            'tim': ['tim.br', 'timbrasil.br'],
            'oi': ['gprs.oi.com.br', 'oi.com.br'],
            'allcom': ['allcom.br']
        }
    
    def validate_parameter(self, param_name: str, value: Any, context: Optional[Dict[str, Any]] = None, 
                         source: str = 'unknown', security_context: Optional[SecurityContext] = None) -> ValidationResult:
        """
        Validate a parameter value according to its defined rules with security enhancements.
        
        Args:
            param_name: Name of the parameter to validate
            value: Value to validate
            context: Optional context with other parameter values for conditional validation
            source: Source of the validation request ('ui', 'command', 'api', 'profile')
            security_context: Security context for audit logging and rate limiting
            
        Returns:
            ValidationResult with validation status and error details
        """
        start_time = datetime.now()
        result = None
        param_type = 'unknown'

        # Short-circuit: if speed alert is disabled, skip validation for speed fields
        try:
            if param_name in ('speed_detection_time', 'speed_limit') and context:
                # Context may include either a flattened flag or a grouped dict
                enabled = None
                if isinstance(context, dict):
                    if 'speed_alert_enabled' in context:
                        enabled = context.get('speed_alert_enabled')
                    elif 'speed_alert' in context and isinstance(context.get('speed_alert'), dict):
                        enabled = context.get('speed_alert', {}).get('enabled')
                # Treat common false-ish values
                if enabled is False or (isinstance(enabled, str) and enabled.lower() in ('false', 'off', '0', 'no')):
                    return ValidationResult(is_valid=True)
        except Exception:
            # If any error occurs while checking context, fall back to normal validation
            pass
        
        # Create default security context if not provided
        if security_context is None:
            security_context = SecurityContext(
                user_id="system",
                session_id="default",
                ip_address="127.0.0.1",
                user_agent="ValidationService",
                timestamp=start_time,
                security_level=SecurityLevel.MEDIUM,
                permissions=[]
            )
        
        # Security enhancements
        if self._security_enabled:
            # 1. Rate limiting check
            rate_limit_id = f"{security_context.ip_address}:{security_context.user_id}"
            rate_allowed, rate_error = self.security_service.check_rate_limit(
                rate_limit_id, "validation", security_context
            )
            if not rate_allowed:
                return ValidationResult(
                    is_valid=False,
                    error_message=rate_error,
                    error_code='RATE_LIMITED'
                )
            
            # 2. Input sanitization for APN-related fields
            if param_name in ['apn', 'apn_user', 'apn_password'] and isinstance(value, str):
                field_type = 'apn' if param_name == 'apn' else 'username' if param_name == 'apn_user' else 'password'
                sanitized_value, sanitization_issues = self.security_service.sanitize_apn_input(value, field_type)
                
                if sanitization_issues:
                    # Log sanitization issues
                    self.security_service.audit_log(
                        AuditAction.SECURITY_VIOLATION,
                        security_context,
                        "input_sanitization",
                        param_name,
                        {"original_value_length": len(value), "issues": sanitization_issues},
                        True,
                        f"Input sanitization applied: {', '.join(sanitization_issues)}"
                    )
                
                # Use sanitized value for validation
                value = sanitized_value
            
            # 3. Validation integrity check
            integrity_ok, integrity_issues = self.security_service.verify_validation_integrity(self.validation_rules)
            if not integrity_ok:
                self.security_service.audit_log(
                    AuditAction.SECURITY_VIOLATION,
                    security_context,
                    "validation_integrity",
                    "validation_rules",
                    {"integrity_issues": integrity_issues},
                    False,
                    f"Validation integrity compromised: {', '.join(integrity_issues)}"
                )
                # Continue with validation but log the issue
        
        # Check cache first for performance optimization
        cache_key = self._generate_cache_key(param_name, value, context)
        cached_result = self._get_cached_validation_result(cache_key)
        if cached_result:
            # Update metrics for cached result
            execution_time = (datetime.now() - start_time).total_seconds() * 1000
            self.log_validation_attempt(param_name, str(value), 'cached', source, cached_result, execution_time, context)
            return cached_result
        
        try:
            if param_name not in self.validation_rules:
                result = ValidationResult(
                    is_valid=False,
                    error_message=self.i18n.get('validation_unknown_parameter', 'Unknown parameter: {param_name}').format(param_name=param_name),
                    error_code='UNKNOWN_PARAMETER'
                )
                param_type = 'unknown_parameter'
                return result
            
            rules = self.validation_rules[param_name]
            param_type = rules['type']
            
            # Get the translated display name for the parameter
            display_name = self.i18n.get(param_name, param_name)
            
            # For server_ip, use the server_url_ip translation
            if param_name == 'server_ip':
                display_name = self.i18n.get('server_url_ip', 'URL/IP')
            elif param_name == 'server_address':
                display_name = self.i18n.get('server_url_ip', 'URL/IP')
            
            # Handle empty values
            if value is None or (isinstance(value, str) and value.strip() == ''):
                if rules.get('required', False):
                    result = ValidationResult(
                        is_valid=False,
                        error_message=self.i18n.get('validation_required_field', '{param_name} is required and cannot be empty').format(param_name=display_name),
                        error_code='REQUIRED_FIELD'
                    )
                else:
                    result = ValidationResult(is_valid=True)
                return result
            
            # Check conditional validation rules
            conditional_result = self._validate_conditional_rules(param_name, value, context)
            if conditional_result is not None:
                result = conditional_result
                return result
            
            # Type-specific validation
            if param_type == 'numeric':
                result = self._validate_numeric(param_name, value, rules)
            elif param_type == 'text':
                result = self._validate_text(param_name, value, rules)
            elif param_type == 'apn':
                result = self._validate_apn(param_name, value, rules, context)
            elif param_type == 'apn_credential':
                result = self._validate_apn_credential(param_name, value, rules)
            elif param_type == 'ip_address':
                result = self._validate_ip_address(param_name, value)
            elif param_type == 'domain':
                result = self._validate_domain(param_name, value, rules)
            elif param_type == 'phone':
                result = self._validate_phone(param_name, value, rules)
            elif param_type == 'dropdown':
                result = self._validate_dropdown(param_name, value, rules)
            elif param_type == 'radio':
                result = self._validate_radio(param_name, value, rules)
            else:
                result = ValidationResult(is_valid=True)
            
            # Cache the result for future use
            if result is not None:
                self._cache_validation_result(cache_key, result)
            
            return result
            
        finally:
            # Always log the validation attempt, regardless of the path taken
            if result is not None:
                execution_time = (datetime.now() - start_time).total_seconds() * 1000
                self.log_validation_attempt(
                    param_name, str(value), param_type, source, result, execution_time, context
                )
                
                # Security audit logging
                if self._security_enabled and security_context:
                    self.security_service.audit_log(
                        AuditAction.VALIDATION_ATTEMPT,
                        security_context,
                        "parameter_validation",
                        param_name,
                        {
                            "param_type": param_type,
                            "validation_source": source,
                            "execution_time_ms": execution_time,
                            "has_context": context is not None
                        },
                        result.is_valid,
                        result.error_message if not result.is_valid else ""
                    )
    
    def validate_parameter_realtime(self, param_name: str, value: Any, context: Optional[Dict[str, Any]] = None, 
                                  source: str = 'ui', security_context: Optional[SecurityContext] = None) -> ValidationResult:
        """
        Validate a parameter value with real-time feedback optimizations and security enhancements.
        
        This method provides immediate validation feedback suitable for real-time UI updates.
        It includes progressive validation levels and user-friendly error messages.
        
        Args:
            param_name: Name of the parameter to validate
            value: Value to validate
            context: Optional context with other parameter values for conditional validation
            security_context: Security context for audit logging and rate limiting
            
        Returns:
            ValidationResult with validation status and detailed feedback
        """
        # Delegate to main validation method with security context
        return self.validate_parameter(param_name, value, context, source, security_context)
        # Use the standard validation but with enhanced feedback
        result = self.validate_parameter(param_name, value, context, source)
        
        # Add progressive validation feedback
        if not result.is_valid and result.error_code:
            result = self._enhance_validation_feedback(param_name, value, result)

    def _enhance_validation_feedback(self, param_name: str, value: Any, result: ValidationResult) -> ValidationResult:
        """
        Enhance validation result with more detailed feedback for real-time validation.
        
        Args:
            param_name: Name of the parameter
            value: Value being validated
            result: Original validation result
            
        Returns:
            Enhanced validation result
        """
        rules = self.validation_rules.get(param_name, {})
        param_type = rules.get('type', 'text')
        display_name = self.i18n.get(param_name, param_name)
        
        # Enhance numeric validation feedback
        if param_type == 'numeric' and result.error_code in ['RANGE_ERROR', 'MIN_VALUE', 'MAX_VALUE']:
            min_val = rules.get('min')
            max_val = rules.get('max')
            
            try:
                num_value = float(value) if value else 0
                
                if min_val is not None and max_val is not None:
                    # Provide contextual feedback based on how far off the value is
                    if num_value < min_val:
                        diff = min_val - num_value
                        if diff <= (max_val - min_val) * 0.1:  # Within 10% of range
                            result.validation_level = "warning"
                            result.warning_message = self.i18n.get(
                                'validation_close_to_range', 
                                '{param_name} is close to valid range. Minimum is {min_val}'
                            ).format(param_name=display_name, constraint=f"Minimum is {min_val}")
                    elif num_value > max_val:
                        diff = num_value - max_val
                        if diff <= (max_val - min_val) * 0.1:  # Within 10% of range
                            result.validation_level = "warning"
                            result.warning_message = self.i18n.get(
                                'validation_close_to_range', 
                                '{param_name} is close to valid range. Maximum is {max_val}'
                            ).format(param_name=display_name, constraint=f"Maximum is {max_val}")
                            
            except (ValueError, TypeError):
                pass
        
        # Enhance text validation feedback
        elif param_type == 'text' and result.error_code in ['MIN_LENGTH', 'MAX_LENGTH']:
            text_value = str(value) if value else ""
            min_length = rules.get('min_length')
            max_length = rules.get('max_length')
            
            if min_length and len(text_value) < min_length:
                remaining = min_length - len(text_value)
                if remaining <= 3:  # Close to minimum length
                    result.validation_level = "warning"
                    result.warning_message = self.i18n.get(
                        'validation_characters_remaining',
                        '{remaining} more characters needed for {param_name}'
                    ).format(remaining=remaining, param_name=display_name)
        
        return result
    
    def validate_parameter_group(self, parameters: Dict[str, Any]) -> Dict[str, ValidationResult]:
        """
        Validate a group of parameters with conditional validation support.
        
        Args:
            parameters: Dictionary of parameter names and values
            
        Returns:
            Dictionary of parameter names and their validation results
        """
        results = {}
        
        for param_name, value in parameters.items():
            results[param_name] = self.validate_parameter(param_name, value, parameters)
        
        return results

            if speed_enabled is False:  # Explicitly check for False
                # These fields should not be validated if speed alert is disabled
                return ValidationResult(is_valid=True)
        
        # Conditional validation for power alert parameters
        if param_name in ['power_detection_time', 'min_charge_time', 'battery_alert_type']:
            power_enabled = context.get('power_enabled')
            if power_enabled is False:  # Explicitly check for False

    def _validate_numeric(self, param_name: str, value: Any, rules: Dict) -> ValidationResult:
        # Get the translated display name for the parameter
        display_name = self.i18n.get(param_name, param_name)
        
        # For server_port, use the server_port translation
        if param_name == 'server_port':
            display_name = self.i18n.get('server_port', 'Porta do Servidor')
        """Validate numeric parameters with range checking."""
        try:
            if isinstance(value, str):
                if rules.get('integer', False):
                    num_value = int(value)
                else:
                    num_value = float(value)
            else:
                num_value = value
                
            # Check if integer is required
            if rules.get('integer', False) and not isinstance(num_value, int):
                if isinstance(num_value, float) and not num_value.is_integer():
                    return ValidationResult(
                        is_valid=False,
                        error_message=self.i18n.get('validation_integer_required', '{param_name} must be an integer value').format(param_name=display_name),
                        error_code='INTEGER_REQUIRED'
                    )
                num_value = int(num_value)
            
            # Check range
            min_val = rules.get('min')
            max_val = rules.get('max')
            
            if min_val is not None and num_value < min_val:
                # Use more specific range error message if both min and max are specified
                if max_val is not None:
                    return ValidationResult(
                        is_valid=False,
                        error_message=self.i18n.get('validation_range_error', '{param_name} must be between {min_val} and {max_val}').format(
                            param_name=display_name, min_val=min_val, max_val=max_val
                        ),
                        error_code='RANGE_ERROR',
                        suggested_value=str(min_val)
                    )
                else:
                    return ValidationResult(
                        is_valid=False,
                        error_message=self.i18n.get('validation_min_value', '{param_name} must be at least {min_val}').format(
                            param_name=display_name, min_val=min_val
                        ),
                        error_code='MIN_VALUE',
                        suggested_value=str(min_val)
                    )
            
            if max_val is not None and num_value > max_val:
                return ValidationResult(
                    is_valid=False,
                    error_message=self.i18n.get('validation_max_value', '{param_name} must be at most {max_val}').format(
                        param_name=display_name, max_val=max_val
                    ),
                    error_code='MAX_VALUE',
                    suggested_value=str(max_val)
                )
            
            # Check allowed values (for specific numeric constraints)
            allowed_values = rules.get('allowed_values')
            if allowed_values and num_value not in allowed_values:
                return ValidationResult(
                    is_valid=False,
                    error_message=self.i18n.get('validation_not_in_allowed_values', '{param_name} must be one of: {allowed_values}').format(
                        param_name=display_name, 
                        allowed_values=", ".join(map(str, allowed_values))
                    ),
                    error_code='NOT_IN_ALLOWED_VALUES'
                )
            
            return ValidationResult(is_valid=True)
            
        except (ValueError, TypeError):
            return ValidationResult(
                is_valid=False,
                error_message=self.i18n.get('validation_numeric_required', '{param_name} must be a numeric value').format(param_name=display_name),
                error_code='NUMERIC_REQUIRED'
            )
    
    def _validate_text(self, param_name: str, value: Any, rules: Dict) -> ValidationResult:
        # Get the translated display name for the parameter
        display_name = self.i18n.get(param_name, param_name)
        
        # For server_ip, use the server_url_ip translation
        if param_name == 'server_ip':
            display_name = self.i18n.get('server_url_ip', 'URL/IP')
        """Validate text parameters with length and pattern checking."""
        if not isinstance(value, str):
            value = str(value)
        
        # Check minimum length
        min_length = rules.get('min_length')
        if min_length and len(value) < min_length:
            return ValidationResult(
                is_valid=False,
                error_message=self.i18n.get('validation_min_length', '{param_name} must be at least {min_length} characters').format(
                    param_name=display_name, 
                    min_length=min_length
                ),
                error_code='MIN_LENGTH'
            )
        
        # Check maximum length
        max_length = rules.get('max_length')
        if max_length and len(value) > max_length:
            return ValidationResult(
                is_valid=False,
                error_message=self.i18n.get('validation_max_length', '{param_name} must be at most {max_length} characters').format(
                    param_name=param_name, 
                    max_length=max_length
                ),
                error_code='MAX_LENGTH'
            )
        
        # Check pattern
        pattern = rules.get('pattern')
        if pattern and not re.match(pattern, value):
            # Use specific error messages for known patterns
            if param_name == 'imei':
                return ValidationResult(
                    is_valid=False,
                    error_message=self.i18n.get('validation_invalid_imei', '{param_name} must be exactly 15 digits').format(param_name=display_name),
                    error_code='INVALID_IMEI'
                )
            elif param_name == 'iccid':
                return ValidationResult(
                    is_valid=False,
                    error_message=self.i18n.get('validation_invalid_iccid', '{param_name} must be 19-20 digits').format(param_name=display_name),
                    error_code='INVALID_ICCID'
                )
            elif param_name == 'imsi':
                return ValidationResult(
                    is_valid=False,
                    error_message=self.i18n.get('validation_invalid_imsi', '{param_name} must be exactly 15 digits').format(param_name=display_name),
                    error_code='INVALID_IMSI'
                )
            elif param_name in ['apn', 'apn_server']:
                # Use enhanced APN validation
                apn_result = self._validate_apn_format(value)
                if not apn_result.is_valid:
                    suggestions = self._get_apn_suggestions(value)
                    error_message = self.i18n.get('validation_invalid_apn', '{param_name} deve ser um nome de host ou endere├ºo IP v├ílido').format(param_name=display_name)
                    if suggestions:
                        error_message += f". Sugest├Áes: {', '.join(suggestions[:3])}"
                    
                    return ValidationResult(
                        is_valid=False,
                        error_message=error_message,
                        error_code='INVALID_APN'
                    )
            elif param_name in ['server_ip', 'server_address']:
                return ValidationResult(
                    is_valid=False,
                    error_message=self.i18n.get('validation_invalid_server_url_ip', '{param_name} deve ser uma URL ou endere├ºo IP v├ílido').format(param_name=display_name),
                    error_code='INVALID_SERVER_URL_IP'
                )
            else:
                return ValidationResult(
                    is_valid=False,
                    error_message=self.i18n.get('validation_invalid_format', '{param_name} has an invalid format').format(param_name=display_name),
                    error_code='INVALID_FORMAT'
                )
        
        return ValidationResult(is_valid=True)
    
    def _validate_ip_address(self, param_name: str, value: str) -> ValidationResult:
        """Validate IP address format using optimized compiled pattern."""
        ip_pattern = self._get_compiled_pattern('ip_address')
        
        # Use compiled pattern if available, otherwise fall back to regex
        if ip_pattern:
            is_valid = bool(ip_pattern.match(value))
        else:
            ip_pattern_str = r'^(?:(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)\.){3}(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)$'
            is_valid = bool(re.match(ip_pattern_str, value))
        
        if not is_valid:
            return ValidationResult(
                is_valid=False,
                error_message=self.i18n.get('validation_invalid_ip', '{param_name} must be a valid IP address (e.g. 192.168.1.1)').format(param_name=param_name),
                error_code='INVALID_IP'
            )
        
        return ValidationResult(is_valid=True)
    
    def _validate_domain(self, param_name: str, value: str, rules: Dict) -> ValidationResult:
        """Validate domain name format using optimized compiled pattern."""
        domain_pattern = self._get_compiled_pattern('domain')
        
        # Use compiled pattern if available, otherwise fall back to regex
        if domain_pattern:
            is_valid = bool(domain_pattern.match(value))
        else:
            domain_pattern_str = r'^[a-zA-Z0-9]([a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?(\.[a-zA-Z0-9]([a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?)*$'
            is_valid = bool(re.match(domain_pattern_str, value))
        
        if not is_valid:
            return ValidationResult(
                is_valid=False,
                error_message=self.i18n.get('validation_invalid_domain', '{param_name} must be a valid domain name (e.g. example.com)').format(param_name=param_name),
                error_code='INVALID_DOMAIN'
            )
        
        max_length = rules.get('max_length', 253)
        if len(value) > max_length:
            return ValidationResult(
                is_valid=False,
                error_message=self.i18n.get('validation_max_length', '{param_name} must be at most {max_length} characters').format(
                    param_name=param_name, 
                    max_length=max_length
                ),
                error_code='MAX_LENGTH'
            )
        
        return ValidationResult(is_valid=True)
    
    def _validate_phone(self, param_name: str, value: str, rules: Dict) -> ValidationResult:
        """Validate phone number format using optimized pattern matching."""
        # First check if the format looks like a phone number using compiled pattern
        phone_pattern = self._get_compiled_pattern('phone')
        if phone_pattern and not phone_pattern.match(value):
            return ValidationResult(
                is_valid=False,
                error_message=self.i18n.get('validation_invalid_phone', '{param_name} must contain only digits and common separators').format(param_name=param_name),
                error_code='INVALID_PHONE'
            )
        
        # Remove common phone number separators
        clean_phone = re.sub(r'[\s\-\(\)\+]', '', value)
        
        # Check if it contains only digits
        if not clean_phone.isdigit():
            return ValidationResult(
                is_valid=False,
                error_message=self.i18n.get('validation_invalid_phone', '{param_name} must contain only digits').format(param_name=param_name),
                error_code='INVALID_PHONE'
            )
        
        # Check length
        max_length = rules.get('max_length', 15)
        if len(clean_phone) > max_length:
            return ValidationResult(
                is_valid=False,
                error_message=self.i18n.get('validation_phone_too_long', '{param_name} must be at most {max_length} digits').format(
                    param_name=param_name,
                    max_length=max_length
                ),
                error_code='PHONE_TOO_LONG'
            )
        
        if len(clean_phone) < 7:  # Minimum reasonable phone number length
            return ValidationResult(
                is_valid=False,
                error_message=self.i18n.get('validation_phone_too_short', '{param_name} must be at least 7 digits').format(param_name=param_name),
                error_code='PHONE_TOO_SHORT'
            )
        
        return ValidationResult(is_valid=True)
