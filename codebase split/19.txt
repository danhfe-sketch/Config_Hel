"""
Parameter Factory - Factory pattern for creating different parameter input types.

This module provides a factory pattern for generating numeric inputs, dropdowns, 
and radio button groups with consistent validation and command generation interfaces.
"""

import customtkinter as ctk
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Dict, List, Optional, Callable, Any
from services.validation_service import ValidationService, ValidationResult
from services.device_command_mapper import DeviceCommandMapper


@dataclass
class ParameterDefinition:
    """Definition of a parameter with its properties and behavior."""
    name: str
    param_type: str  # 'numeric', 'dropdown', 'radio'
    display_label: str
    help_text: str
    at_command_template: str
    validation_rules: Dict
    group: str = "general"
    default_value: Any = None


class BaseParameter(ABC):
    """Base class for all parameter types with common validation and command generation."""
    
    def __init__(self, definition: ParameterDefinition, validation_service: ValidationService, 
                 i18n_service, send_command_callback: Callable[[str], None]):
        self.definition = definition
        self.validation_service = validation_service
        self.i18n = i18n_service
        self.send_command_callback = send_command_callback
        self.widget = None
        self.error_label = None
        self.send_button = None
        
    @abstractmethod
    def create_widget(self, parent) -> ctk.CTkFrame:
        """Create the UI widget for this parameter."""
        pass
    
    @abstractmethod
    def get_value(self) -> Any:
        """Get the current value from the widget."""
        pass
    
    @abstractmethod
    def set_value(self, value: Any) -> None:
        """Set the value in the widget."""
        pass
    
    def validate(self) -> ValidationResult:
        """Validate the current parameter value."""
        value = self.get_value()
        return self.validation_service.validate_parameter(self.definition.name, value)
    
    def send_command(self) -> None:
        """Validate and send the AT command for this parameter."""
        validation_result = self.validate()
        
        if not validation_result.is_valid:
            self._show_error(validation_result.error_message)
            return
        
        self._clear_error()
        value = self.get_value()
        
        if value is not None and str(value).strip():
            try:
                # Get the command mapper from the factory
                command_mapper = getattr(self, 'command_mapper', None)
                
                # If we have access to the command mapper, use it with validation
                if command_mapper:
                    command = command_mapper.get_command(self.definition.name, value, 
                                                       confirm_destructive=True, validate_command=True)
                else:
                    # Handle special cases for server parameters
                    if self.definition.name in ['server_ip', 'server_port']:
                        # Server parameters should not be sent individually
                        # They are handled by the ServerParameterGroup
                        self._show_error("Use the group 'Send' button to configure server settings")
                        return
                    else:
                        # Fall back to the template if no command mapper is available
                        command = self.definition.at_command_template.format(value=value)
                
                self.send_command_callback(command)
                
                # Show success message if command was sent successfully
                self._log_success(f"Command sent successfully: {self.definition.name}")
                
            except ValueError as e:
                error_msg = str(e)
                
                # Check if this is a validation error and provide better feedback
                if "Command validation failed" in error_msg:
                    # Extract the core error message and suggestions
                    if "Suggestions:" in error_msg:
                        parts = error_msg.split("Suggestions:")
                        core_error = parts[0].replace("Command validation failed for " + self.definition.name + ":", "").strip()
                        suggestions = parts[1].strip()
                        
                        # Show the core error in the UI
                        self._show_error(f"Validation failed: {core_error}")
                        
                        # Log suggestions for the user
                        if hasattr(self, '_log_message'):
                            self._log_message(f"Suggestions for {self.definition.name}: {suggestions}")
                    else:
                        core_error = error_msg.replace("Command validation failed for " + self.definition.name + ":", "").strip()
                        self._show_error(f"Validation failed: {core_error}")
                else:
                    self._show_error(str(e))
                return
                
            except KeyError as e:
                # Handle missing format keys
                missing_key = str(e).strip("'")
                self._show_error(f"Missing required parameter: {missing_key}")
                return
    
    def _show_error(self, message: str) -> None:
        """Show validation error message."""
        if self.error_label:
            self.error_label.configure(text=message, text_color="red")
    
    def _clear_error(self) -> None:
        """Clear validation error message."""
        if self.error_label:
            self.error_label.configure(text="", text_color="gray")
    
    def _log_success(self, message: str) -> None:
        """Log a success message."""
        # Try to access the app's log frame if available
        try:
            # This is a bit of a hack, but we need to access the app somehow
            import tkinter as tk
            root = self.widget
            while root and not isinstance(root, tk.Tk):
                root = root.master
            
            if root and hasattr(root, 'app') and hasattr(root.app, 'log_frame'):
                root.app.log_frame.log_message(f"SUCCESS: {message}\n")
        except:
            pass  # Ignore errors in logging
    
    def _log_message(self, message: str) -> None:
        """Log a general message."""
        # Try to access the app's log frame if available
        try:
            # This is a bit of a hack, but we need to access the app somehow
            import tkinter as tk
            root = self.widget
            while root and not isinstance(root, tk.Tk):
                root = root.master
            
            if root and hasattr(root, 'app') and hasattr(root.app, 'log_frame'):
                root.app.log_frame.log_message(f"INFO: {message}\n")
        except:
            pass  # Ignore errors in logging
    
    def update_language_display(self, i18n_service) -> None:
        """Update the parameter widget's language display."""
        # Update the i18n service reference
        self.i18n = i18n_service
        
        # Update send button text if it exists
        if self.send_button:
            self.send_button.configure(text=self.i18n.get("send", "Send"))
        
        # Update parameter label if it exists
        if hasattr(self, 'param_label'):
            self.param_label.configure(text=self.i18n.get(self.definition.display_label, self.definition.display_label))
    
    def add_update_indicator(self, value_changed: bool = True) -> None:
        """
        Add visual indicator to show this parameter was updated from device.
        
        Args:
            value_changed: True if the value changed, False if it was the same
        """
        try:
            if self.widget:
                if value_changed:
                    # Green border for changed values
                    original_border_color = self.widget.cget("border_color")
                    self.widget.configure(border_color="#00ff00", border_width=2)
                    
                    # Schedule removal after 3 seconds
                    if hasattr(self, '_master_widget'):
                        self._master_widget.after(3000, lambda: self._remove_update_indicator(original_border_color))
                else:
                    # Blue border for unchanged values
                    original_border_color = self.widget.cget("border_color")
                    self.widget.configure(border_color="#0066cc", border_width=1)
                    
                    # Schedule removal after 2 seconds
                    if hasattr(self, '_master_widget'):
                        self._master_widget.after(2000, lambda: self._remove_update_indicator(original_border_color))
                        
        except Exception as e:
            print(f"WARNING: Failed to add update indicator: {str(e)}")
    
    def _remove_update_indicator(self, original_border_color) -> None:
        """Remove the visual update indicator."""
        try:
            if self.widget:
                self.widget.configure(border_color=original_border_color, border_width=0)
        except Exception as e:
            print(f"WARNING: Failed to remove update indicator: {str(e)}")
    
    def _create_base_frame(self, parent) -> ctk.CTkFrame:
        """Create the base frame with label, input, and send button."""
        frame = ctk.CTkFrame(parent)
        frame.grid_columnconfigure(1, weight=1)  # Make input column expandable
        
        # Store reference to master widget for scheduling callbacks
        self._master_widget = parent
        
        # Parameter label - store reference for language updates
        self.param_label = ctk.CTkLabel(frame, text=self.i18n.get(self.definition.display_label, self.definition.display_label))
        self.param_label.grid(row=0, column=0, padx=10, pady=5, sticky="w")
        
        # Error label (initially empty)
        self.error_label = ctk.CTkLabel(frame, text="", text_color="red", font=ctk.CTkFont(size=10))
        self.error_label.grid(row=2, column=0, columnspan=3, padx=10, pady=(0, 5), sticky="w")
        
        return frame

            validation_result = self.validate()

class ParameterGroup:
    """Group of related parameters for logical organization."""
    
    def __init__(self, name: str, display_label: str, i18n_service):
        self.name = name
        self.display_label = display_label
        self.i18n = i18n_service
        self.parameters: List[BaseParameter] = []
        self.frame = None
    
    def add_parameter(self, parameter: BaseParameter) -> None:
        """Add a parameter to this group."""
        self.parameters.append(parameter)
    
    def create_widget(self, parent) -> ctk.CTkFrame:
        """Create the group frame with all parameters."""
        self.frame = ctk.CTkFrame(parent)
        self.frame.grid_columnconfigure(0, weight=1)
        
        # Group title
        title = ctk.CTkLabel(
            self.frame, 
            text=self.i18n.get(self.display_label, self.display_label), 
            font=ctk.CTkFont(weight="bold")
        )
        title.grid(row=0, column=0, padx=10, pady=(10, 5), sticky="w")
        
        # Add parameters
        for i, parameter in enumerate(self.parameters):
            param_widget = parameter.create_widget(self.frame)
            param_widget.grid(row=i+1, column=0, padx=10, pady=5, sticky="ew")
        
        return self.frame
    
    def get_values(self) -> Dict[str, Any]:
        """Get all parameter values in this group."""
        return {param.definition.name: param.get_value() for param in self.parameters}
    
    def set_values(self, values: Dict[str, Any]) -> None:
        """Set parameter values in this group."""
        for parameter in self.parameters:
            if parameter.definition.name in values:
                parameter.set_value(values[parameter.definition.name])
    
    def update_language_display(self, i18n_service) -> None:
        """Update the group's language display."""
        # Update the i18n service reference
        self.i18n = i18n_service
        
        # Update group title if frame exists
        if self.frame:
            # Find the title label (first child that's a CTkLabel with bold font)
            for child in self.frame.winfo_children():
                if isinstance(child, ctk.CTkLabel):
                    try:
                        if child.cget("font").cget("weight") == "bold":
                            child.configure(text=self.i18n.get(self.display_label, self.display_label))
                            break
                    except:
                        # If font checking fails, assume it's the title label if it's the first one
                        child.configure(text=self.i18n.get(self.display_label, self.display_label))
                        break
        
        # Update all parameters in this group
        for parameter in self.parameters:
            if hasattr(parameter, 'update_language_display'):
                parameter.update_language_display(i18n_service)

    def send_group_command(self) -> None:
        """Validate and send combined server command with IP and port."""
        all_valid = True
        server_ip = None
        server_port = None
        
        # Validate all parameters and collect values
        for parameter in self.parameters:
            validation_result = parameter.validate()
            if not validation_result.is_valid:
                all_valid = False
                parameter._show_error(validation_result.error_message)
            else:
                parameter._clear_error()
                # Collect parameter values
                if parameter.definition.name == 'server_ip':
                    server_ip = parameter.get_value()
                elif parameter.definition.name == 'server_port':
                    server_port = parameter.get_value()

        if all_valid and server_ip and server_port:
            # Send combined server command with both IP and port
            # Format: ATSMS:SERVER,0,IP,PORT,0#
            combined_command = f"ATSMS:SERVER,0,{server_ip},{server_port},0#"
            
            # Use the send command callback from the first parameter
            if self.parameters:
                self.parameters[0].send_command_callback(combined_command)
                
            # Log success message
            if hasattr(self.parameters[0], '_log_success'):
                self.parameters[0]._log_success(f"Server configuration sent: {server_ip}:{server_port}")
        elif all_valid:
            # Show error if missing values
            error_msg = "Both server IP and port are required"
            for parameter in self.parameters:
                parameter._show_error(error_msg)

    def _setup_enhanced_apn_validation_with_mixin(self):
        """Set up enhanced APN validation using ValidationMixin for APN-related fields.

        This method finds the APN parameter (usually 'apn_settings' under
        'network_parameters'), extracts the individual field UI elements, stores
        them in `self.apn_widgets` and registers them with
        `setup_real_time_validation` from the ValidationMixin so that real-time
        validation, tooltips and progressive feedback work.
        """
        try:
            # Default storage for APN widgets
            self.apn_widgets = {}

            # Find network parameters section and apn_settings parameter
            network_section = self.section_widgets.get('network_parameters') if hasattr(self, 'section_widgets') else None
            if not network_section:
                return

            apn_param = network_section.parameters.get('apn_settings') if hasattr(network_section, 'parameters') else None
            if not apn_param:
                return

            # apn_param.ui_elements contains BaseUIElement instances for each field
            for field_name, ui_element in apn_param.ui_elements.items():
                # Prefer the underlying input widget if available
                widget = None
                if hasattr(ui_element, 'widget') and ui_element.widget is not None:
                    widget = ui_element.widget
                elif hasattr(ui_element, 'container_frame') and ui_element.container_frame is not None:
                    widget = ui_element.container_frame
                else:
                    # Fallback to the ui_element object itself
                    widget = ui_element

                # Store reference for other APN-related logic
                self.apn_widgets[field_name] = widget

                # Register validation callbacks for each APN-related field
                if field_name == 'apn':
                    try:
                        # Use a named field key so the mixin can track state
                        self.setup_real_time_validation('apn', widget, lambda v, s=self: s._validate_apn_field_comprehensive(v), debounce_ms=500)
                    except Exception:
                        # Non-critical: continue if binding fails
                        pass
                elif field_name == 'user':
                    try:
                        self.setup_real_time_validation('apn_user', widget, lambda v, s=self: s._validate_apn_user_field_comprehensive(v), debounce_ms=400)
                    except Exception:
                        pass
                elif field_name == 'password':
                    try:
                        self.setup_real_time_validation('apn_password', widget, lambda v, s=self: s._validate_apn_password_field_comprehensive(v), debounce_ms=400)
                    except Exception:
                        pass

        except Exception as e:
            # Do not crash the application on validation setup failures
            print(f"WARNING: Failed to setup APN validation mixin: {e}")

                    legacy_param_widget['label'] = None  # UI element handles its own label
                elif param_widget.widget_type == 'complex':
                    legacy_param_widget['title'] = param_widget.title_widget

                    for field_name, ui_element in param_widget.ui_elements.items():
                        legacy_param_widget['fields'][field_name] = {
                            'config': param_widget.config.fields[field_name],
                            'element': ui_element,
                            'label': None  # UI element handles its own label
                        }
                
                self.parameter_widgets[legacy_param_key] = legacy_param_widget
                self.parameter_sections[section_name]['parameters'][param_name] = legacy_param_widget

    def highlight_updated_parameter(self, param_name: str, value_changed: bool = True):
        """
        Highlight a parameter that was updated from device configuration.
        
        Args:
            param_name: Name of the parameter to highlight
            value_changed: True if the value changed, False if it was the same
        """
        try:
            if param_name in self.parameter_widgets:
                parameter_widget = self.parameter_widgets[param_name]
                
                # Get the container frame to highlight
                container_frame = None
                if parameter_widget['type'] == 'simple':
                    ui_element = parameter_widget['element']
                    if hasattr(ui_element, 'container_frame'):
                        container_frame = ui_element.container_frame
                elif parameter_widget['type'] == 'complex':
                    container_frame = parameter_widget['frame']
                
                if container_frame and hasattr(container_frame, 'configure'):
                    if value_changed:
                        # Green border for changed values
                        original_border_color = container_frame.cget("border_color")
                        container_frame.configure(border_color="#00ff00", border_width=2)
                        
                        # Schedule removal after 3 seconds
                        self.after(3000, lambda: self._remove_parameter_highlight(container_frame, original_border_color))
                    else:
                        # Blue border for unchanged values
                        original_border_color = container_frame.cget("border_color")
                        container_frame.configure(border_color="#0066cc", border_width=1)
                        
                        # Schedule removal after 2 seconds
                        self.after(2000, lambda: self._remove_parameter_highlight(container_frame, original_border_color))
                        
        except Exception as e:
            print(f"WARNING: Failed to highlight parameter {param_name}: {str(e)}")

    def send_apn_command(self):
        """Validates and sends the complete APN configuration command with enhanced validation."""
        if not self.serial_service.is_connected:
            show_error(
                self,
                self.i18n.get("error"),
                self.i18n.get("not_connected_error")
            )
            return

        # Check if form submission should be prevented due to validation errors
        if self.prevent_form_submission_on_validation_errors():
            validation_errors = self.get_validation_errors()
            error_messages = []
            
            for field_name, error_message in validation_errors.items():
                field_display = field_name.replace('_', ' ').title()
                error_messages.append(f"{field_display}: {error_message}")
            
            show_error(
                self,
                self.i18n.get("validation_errors", "Validation Errors"),
                "Please fix the following validation errors before sending:\n\n" + "\n".join(error_messages)
            )
            return

        # Get APN values from the network parameters section
        apn_values = self._get_apn_values_from_section()
        if not apn_values:
            show_error(
                self,
                self.i18n.get("error"),
                self.i18n.get("apn_server_required")
            )
            return

        server = apn_values.get('apn', '').strip()
        user = apn_values.get('user', '').strip()
        password = apn_values.get('password', '').strip()

        # APN server is mandatory if any APN setting is being sent.
        if not server:
            show_error(
                self,
                self.i18n.get("error"),
                self.i18n.get("apn_server_required")
            )
            return

        # Final comprehensive validation using the APN validation component
        validation_result = self.apn_validation_component.validate_apn_configuration(server, user, password)
        if not validation_result.is_valid:
            show_error(
                self,
                self.i18n.get("invalid_apn_server_title", "Invalid APN Configuration"),
                validation_result.error_message + 
                (f"\n\nSuggestions:\nÔÇó " + "\nÔÇó ".join(validation_result.suggestions) if validation_result.suggestions else "")
            )
            return

        # Construct the command. User and password can be empty.
        command = f"ATSMS:APN,{server},{user},{password}#"
        # Direct UI send - bypass command validation to avoid blocking
        # well-formed APN commands initiated by the user.
        self.serial_service.send_command(command, command_type='apn', bypass_validation=True)

        # Show success feedback
        show_info(
            self,
            self.i18n.get("success", "Success"),
            f"APN configuration sent successfully:\nAPN: {server}\nUser: {user if user else '(none)'}"
        )

    def update_fields_from_dict(self, ui_field_values: Dict[str, Any]):
        """
        Update UI fields from a dictionary of profile-derived values.

        Keys can be simple parameter names (e.g., 'apn_settings') or complex
        field names (e.g., 'apn_settings.user'). Values are applied to the
        corresponding widgets using the specification factory.
        """
        try:
            factory = self.parameter_specification_factory
            section_widgets = self.section_widgets
            mapping = self._get_ui_parameter_mapping()

            for key, value in ui_field_values.items():
                try:
                    # Handle complex field keys like 'param.field'
                    if '.' in key:
                        base_param, field_name = key.split('.', 1)
                        section_info = mapping.get(base_param)
                        if not section_info:
                            print(f"WARNING: No UI mapping found for parameter {base_param}")
                            continue
                        section_name = section_info['section']
                        param_key = section_info['parameter']

                        # Get current complex value, update only the field, then set
                        current_value = {}
                        try:
                            current_value = factory.get_parameter_value(section_name, param_key, section_widgets) or {}
                        except Exception:
                            current_value = {}
                        if not isinstance(current_value, dict):
                            current_value = {}
                        current_value[field_name] = value
                        factory.set_parameter_value(section_name, param_key, current_value, section_widgets)
                        self.highlight_updated_parameter(f"{section_name}.{param_key}", True)
                    else:
                        # Simple parameter
                        section_info = mapping.get(key)
                        if not section_info:
                            print(f"WARNING: No UI mapping found for parameter {key}")
                            continue
                        section_name = section_info['section']
                        param_key = section_info['parameter']
                        factory.set_parameter_value(section_name, param_key, value, section_widgets)
                        self.highlight_updated_parameter(f"{section_name}.{param_key}", True)
                except Exception as e:
                    print(f"ERROR: Failed to apply UI value for '{key}': {e}")
        except Exception as e:
            print(f"ERROR: update_fields_from_dict failed: {e}")

                factory = self.parameter_specification_factory

                for param_name, parsed_values in parsed_parameters.items():
                    try:

                        if param_name.lower() in ['moving', 'moving_alert']:

                        if not section_info:
                            print(f"WARNING: No UI mapping found for parameter {param_name}")
                            continue
                        
                        section_name = section_info['section']
                        param_key = section_info['parameter']
                        
                        # Diagnostic logs for UI update flow
                        print(f"DEBUG: ui.update_parameter_fields - param='{param_name}', parsed_values_keys={list(parsed_values.keys()) if isinstance(parsed_values, dict) else parsed_values}")
                        try:
                            print(f"DEBUG: ui.update_parameter_fields - ui_selection present: {'ui_selection' in parsed_values if isinstance(parsed_values, dict) else False}")
                        except Exception as _e:
                            print(f"DEBUG: ui.update_parameter_fields - failed to check ui_selection presence: {_e}")
                        print(f"DEBUG: ui.update_parameter_fields - section_info resolved to: {section_info}")
                        print(f"DEBUG: Updating {section_name}.{param_key} with values: {parsed_values}")
                        
                        # No need to handle notification mode/text conversion here
                        # The response parser now returns the correct alert_type field
                        
                        # Update the parameter value using the factory
                        factory.set_parameter_value(section_name, param_key, parsed_values, section_widgets)

                        full_param_name = f"{section_name}.{param_key}"
                        self.highlight_updated_parameter(full_param_name, True)

                print(f"DEBUG: Completed updating {len(parsed_parameters)} parameters")

                print("WARNING: Parameter specification factory or section widgets not available")
                # Fallback to legacy update methods
                self._update_parameters_legacy(parsed_parameters)

            print(f"ERROR: Failed to update parameter fields: {str(e)}")

    def update_parameters_with_batch_processing(self, parameters: dict):
        """
        Update parameters using batch processing for optimal responsiveness.
        
        Args:
            parameters: Dictionary of parameter names to values
            
        Requirements: 7.1, 7.2, 7.4, 7.5
        """
        try:
            print(f"DEBUG: Using batch processing for {len(parameters)} parameters")
            
            # Import here to avoid circular imports
            from services.ui_responsiveness_service import UIResponsivenessService, UpdatePriority
            
            # Create UI responsiveness service if not available
            if not hasattr(self, '_ui_responsiveness_service'):
                self._ui_responsiveness_service = UIResponsivenessService(self)
            
            # Process parameters in batches with visual feedback
            def progress_callback(current, total):
                print(f"DEBUG: Batch update progress: {current}/{total}")
            
            # Queue parameters for batch processing
            self._ui_responsiveness_service.queue_multiple_parameter_updates(
                parameters, UpdatePriority.NORMAL, True
            )
            
            # Process the queue
            result = self._ui_responsiveness_service.process_update_queue()
            
            # Log results
            print(f"DEBUG: Batch processing completed: {result.successful_updates}/{result.total_updates} successful in {result.update_duration_ms:.1f}ms")
            
            if result.update_duration_ms > 100:
                print(f"WARNING: Batch processing exceeded 100ms target: {result.update_duration_ms:.1f}ms")
            
            if result.errors:
                print(f"WARNING: Batch processing had {len(result.errors)} errors")
                for error in result.errors[:3]:  # Show first 3 errors
                    print(f"  - {error}")
                    
        except Exception as e:
            print(f"ERROR: Batch processing failed: {str(e)}")
            # Fallback to individual updates
            print("DEBUG: Falling back to individual parameter updates")

        """
        Enable UI responsiveness monitoring and optimization.
        
        This method sets up performance monitoring for UI updates to ensure
        they meet the 100ms responsiveness requirement.
        
        Requirements: 7.1, 7.5
        """
        try:
            from services.ui_responsiveness_service import UIResponsivenessService
            
            if not hasattr(self, '_ui_responsiveness_service'):

    def _load_error_data(self):
        """Load and display error data."""
        # Combine errors and warnings for navigation
        self.all_issues = []
        
        # Add critical errors first
        for error in self.validation_result.get_errors_by_severity('critical'):
            self.all_issues.append(('critical', error))
        
        # Add regular errors
        for error in self.validation_result.get_errors_by_severity('error'):
            self.all_issues.append(('error', error))
        
        # Add warnings
        for warning in self.validation_result.warnings:
            self.all_issues.append(('warning', warning))
        
        # Update navigation
        self._update_navigation()
        
        # Display first error
        if self.all_issues:
            self._display_current_error()
    
    def _update_navigation(self):
        """Update navigation controls."""
        total_issues = len(self.all_issues)
        
        if total_issues == 0:
            self.error_counter_label.configure(text="No issues")
            self.prev_button.configure(state="disabled")
            self.next_button.configure(state="disabled")
            return
        
        # Update counter
        self.error_counter_label.configure(text=f"{self.current_error_index + 1} / {total_issues}")
        
        # Update button states
        self.prev_button.configure(state="normal" if self.current_error_index > 0 else "disabled")
        self.next_button.configure(state="normal" if self.current_error_index < total_issues - 1 else "disabled")

class ColumnDivider(ctk.CTkFrame):
    """A draggable divider for resizing columns."""
    
    def __init__(self, master, left_column, right_column, min_left=100, min_right=100, **kwargs):
        """
        Initialize a column divider.
        
        Args:
            master: Parent widget
            left_column: Index of the column to the left of the divider
            right_column: Index of the column to the right of the divider
            min_left: Minimum width for the left column
            min_right: Minimum width for the right column
        """
        # Set a narrow width for the divider
        kwargs["width"] = kwargs.get("width", 5)
        kwargs["fg_color"] = kwargs.get("fg_color", "#888888")
        kwargs["cursor"] = "sb_h_double_arrow"  # Horizontal resize cursor
        
        super().__init__(master, **kwargs)
        
        self.master = master
        self.left_column = left_column
        self.right_column = right_column
        self.min_left = min_left
        self.min_right = min_right
        
        # Track drag state
        self.dragging = False
        self.start_x = 0
        self.start_width_left = 0
        self.start_width_right = 0
        
        # Bind mouse events
        self.bind("<Enter>", self._on_enter)
        self.bind("<Leave>", self._on_leave)
        self.bind("<ButtonPress-1>", self._on_drag_start)
        self.bind("<ButtonRelease-1>", self._on_drag_end)
        self.bind("<B1-Motion>", self._on_drag_motion)
    
    def _on_enter(self, event):
        """Handle mouse enter event."""
        self.configure(fg_color="#AAAAAA")  # Highlight on hover
    
    def _on_leave(self, event):
        """Handle mouse leave event."""
        if not self.dragging:
            self.configure(fg_color="#888888")  # Reset color when not dragging
    
    def _on_drag_start(self, event):
        """Handle drag start event."""
        self.dragging = True
        self.start_x = event.x_root
        
        # Get current column widths
        self.start_width_left = self.master.grid_bbox(column=self.left_column)[2]  # Width of left column
        self.start_width_right = self.master.grid_bbox(column=self.right_column)[2]  # Width of right column
        
        # Change color to indicate dragging
        self.configure(fg_color="#5555FF")
    
    def _on_drag_end(self, event):
        """Handle drag end event."""
        self.dragging = False
        self.configure(fg_color="#888888")
    
    def _on_drag_motion(self, event):
        """Handle drag motion event."""
        if not self.dragging:
            return
        
        # Calculate the drag distance
        delta_x = event.x_root - self.start_x
        
        # Calculate new widths
        new_width_left = max(self.min_left, self.start_width_left + delta_x)
        new_width_right = max(self.min_right, self.start_width_right - delta_x)
        
        # Update column widths
        self.master.grid_columnconfigure(self.left_column, minsize=new_width_left)
        self.master.grid_columnconfigure(self.right_column, minsize=new_width_right)
        
        # Update the UI
        self.master.update_idletasks()

        Args:

        """
        super().__init__(master, **kwargs)
        
        self.column_count = column_count
        self.dividers = []

        for i in range(column_count):
            self.grid_columnconfigure(i*2, weight=1)

                self,

                right_column=(i+1)*2
            )

            font=ctk.CTkFont(size=16, weight="bold")
        )
        self.title_label.pack(pady=(0, 15))
        
        # Status message
        self.status_label = ctk.CTkLabel(
            main_frame,
            text=self.i18n.get('initializing', 'Inicializando...'),
            font=ctk.CTkFont(size=12)
        )
        self.status_label.pack(pady=(0, 10))
        
        # Progress bar
        self.progress_bar = ctk.CTkProgressBar(main_frame, width=350)
        self.progress_bar.pack(pady=(0, 10))
        self.progress_bar.set(0)
        
        # Progress percentage label
        self.percentage_label = ctk.CTkLabel(
            main_frame,
            text="0%",
            font=ctk.CTkFont(size=11)
        )
        self.percentage_label.pack(pady=(0, 15))
        
        # Cancel button
        self.cancel_button = ctk.CTkButton(
            main_frame,
            text=self.i18n.get('cancel', 'Cancelar'),

            width=100,
            fg_color="gray",
            hover_color="darkgray"
        )
        self.cancel_button.pack()
        
    def _get_operation_title(self) -> str:
        """Get localized operation title."""
        titles = {
            'device_to_profile': self.i18n.get('saving_device_config', 'Salvando Configura├º├úo do Dispositivo'),
            'profile_to_device': self.i18n.get('loading_config_to_device', 'Carregando Configura├º├úo para Dispositivo'),
            'device_operation': self.i18n.get('device_operation', 'Opera├º├úo do Dispositivo')
        }
        return titles.get(self.operation_type, titles['device_operation'])
        
    def update_progress(self, progress_info: SimpleProgressInfo):
        """Update the progress dialog with new information."""
        try:
            # Check if dialog is being destroyed
            if self.is_destroying:
                return
                
            # Update last activity time
            self.last_update_time = time.time()
                
            # Multiple safety checks to prevent widget access after destruction
            if not hasattr(self, 'dialog') or self.dialog is None:
                return
                
            if not self.dialog.winfo_exists():
                return
                
            # Check if widgets still exist before updating
            if not hasattr(self, 'status_label') or not hasattr(self, 'progress_bar') or not hasattr(self, 'percentage_label'):
                return
                
            # Create enhanced status message with parameter info
            status_message = progress_info.status_message
            if progress_info.total_parameters > 0 and progress_info.current_parameter > 0:
                if progress_info.parameter_name:
                    status_message = f"{status_message}\n{progress_info.parameter_name} ({progress_info.current_parameter}/{progress_info.total_parameters})"
                else:
                    status_message = f"{status_message}\nPar├ómetro {progress_info.current_parameter}/{progress_info.total_parameters}"
            
            # Update status message with safety check
            try:
                if hasattr(self.status_label, 'configure'):
                    self.status_label.configure(text=status_message)
            except Exception:
                pass
            
            # Update progress bar with safety check
            try:
                if hasattr(self.progress_bar, 'set'):
                    progress_value = progress_info.progress_percentage / 100.0
                    self.progress_bar.set(progress_value)
            except Exception:
                pass
            
            # Update percentage label with parameter info and safety check
            try:
                if hasattr(self.percentage_label, 'configure'):
                    percentage_text = f"{int(progress_info.progress_percentage)}%"
                    if progress_info.total_parameters > 0:
                        percentage_text += f" ({progress_info.current_parameter}/{progress_info.total_parameters})"
                    self.percentage_label.configure(text=percentage_text)
            except Exception:
                pass
            
            # Handle completion
            if progress_info.is_complete:
                if progress_info.has_error:
                    self._show_error(progress_info.error_message)
                else:
                    self._show_success()
                    
        except Exception as e:
            # Dialog may have been closed, ignore all errors to prevent crashes
            pass
            
    def _show_success(self):
        """Show success state."""
        try:
            if hasattr(self, 'status_label') and hasattr(self.status_label, 'configure'):
                self.status_label.configure(text=self.i18n.get('operation_completed', 'Opera├º├úo conclu├¡da com sucesso!'))
        except Exception:
            pass
            
        try:
            if hasattr(self, 'cancel_button') and hasattr(self.cancel_button, 'configure'):
                self.cancel_button.configure(
                    text=self.i18n.get('close', 'Fechar'),
                    fg_color="green",
                    hover_color="darkgreen"
                )
        except Exception:
            pass
        
    def _show_error(self, error_message: str):
        """Show error state."""
        try:
            if hasattr(self, 'status_label') and hasattr(self.status_label, 'configure'):
                self.status_label.configure(text=f"{self.i18n.get('operation_failed', 'Opera├º├úo falhou')}: {error_message}")
        except Exception:
            pass
            
        try:
            if hasattr(self, 'cancel_button') and hasattr(self.cancel_button, 'configure'):
                self.cancel_button.configure(
                    text=self.i18n.get('close', 'Fechar'),
                    fg_color="red",
                    hover_color="darkred"
                )
        except Exception:
            pass
        
    def set_cancel_callback(self, callback: Callable):
        """Set callback function for cancel operation."""
        self.cancel_callback = callback
        
    def _on_cancel(self):
        """Handle cancel button click."""
        if self.cancel_callback and not self.is_cancelled:
            self.is_cancelled = True
            self.cancel_callback()
            
        self._close_dialog()

        """Handle window close event."""
