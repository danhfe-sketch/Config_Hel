    def _perform_migration(self, source_file: str, target_file: str, 
                          config: MigrationConfig) -> MigrationResult:
        """Perform the actual migration operation."""
        try:
            # Read source file
            config, error_msg = self.file_handler.read_ini_file(source_file)
            
            # Create file_result-like object for compatibility
            class FileResult:
                def __init__(self, success, config=None, error_message=None, profile_data=None):
                    self.success = success
                    self.config = config
                    self.error_message = error_message
                    self.profile_data = profile_data
            
            if config is not None:
                # Create basic profile data from config
                profile_data = {'config': config}
                file_result = FileResult(True, config, profile_data=profile_data)
            else:
                file_result = FileResult(False, error_message=error_msg)
            
            if not file_result.success:
                return MigrationResult(
                    success=False,
                    original_version="unknown",
                    target_version=config.target_format_version,
                    errors=[]
                )
            
            # Convert to current format
            conversion_result = self.converter.convert_to_standard_format(
                self._profile_data_to_dict(file_result.profile_data)
            )
            
            if not conversion_result.success:
                return MigrationResult(
                    success=False,
                    original_version="unknown",
                    target_version=config.target_format_version,
                    errors=conversion_result.errors
                )
            
            # Write migrated file
            write_result = self.file_handler.write_profile_file(
                target_file, conversion_result.profile_data
            )
            
            if not write_result.success:
                return MigrationResult(
                    success=False,
                    original_version="unknown",
                    target_version=config.target_format_version,
                    errors=[]
                )
            
            return MigrationResult(
                success=True,
                original_version="unknown",
                target_version=config.target_format_version,
                migrated_data=conversion_result.profile_data,
                migration_log=["Migration completed successfully"]
            )
            
        except Exception as e:
            return MigrationResult(
                success=False,
                original_version="unknown",
                target_version=config.target_format_version,
                errors=[],
                migration_log=[f"Migration error: {str(e)}"]
            )

    def _profile_data_to_dict(self, profile_data: ProfileData) -> Dict[str, Any]:
        """Convert ProfileData to dictionary format for converter."""
        result = {
            'Profile_Info': {
                'profile_name': profile_data.metadata.profile_name,
                'created_date': profile_data.metadata.created_date,
                'format_version': profile_data.format_version,
                'app_version': profile_data.metadata.app_version,
                'parameter_count': str(profile_data.metadata.parameter_count)
            },
            'Device_Parameters': profile_data.device_parameters.copy()
        }
        
        if profile_data.metadata.device_imei:
            result['Profile_Info']['device_imei'] = profile_data.metadata.device_imei
        if profile_data.metadata.device_version:
            result['Profile_Info']['device_version'] = profile_data.metadata.device_version
        if profile_data.metadata.checksum:
            result['Profile_Info']['checksum'] = profile_data.metadata.checksum
        
        return result

    def _create_migration_log_file(self, operation: MigrationOperation):

        try:
            log_file = f"{operation.target_file}.migration.log"

                f.write(f"Status: {operation.status.value}\n")
                f.write(f"Start Time: {operation.start_time}\n")
                f.write(f"End Time: {operation.end_time}\n")
                
                if operation.backup_file:
                    f.write(f"Backup File: {operation.backup_file}\n")
                
                if operation.error_message:
                    f.write(f"Error: {operation.error_message}\n")
                
                f.write("\nMigration Log:\n")
                f.write("-" * 20 + "\n")
                for log_entry in operation.migration_log:

                    f.write(f"Errors: {len(operation.validation_results.errors)}\n")
                    f.write(f"Warnings: {len(operation.validation_results.warnings)}\n")
            
            self.logger.debug(f"Created migration log: {log_file}")
            
        except Exception as e:

    def _generate_batch_summary_report(self, result: BatchMigrationResult) -> str:

"""
Profile Operation Logger Service

This service provides comprehensive logging and diagnostics for all profile operations
including detailed timestamps, performance monitoring, and diagnostic information collection.
"""

import logging
import json
import time
import traceback
import threading
from datetime import datetime, timedelta
from typing import Dict, Any, List, Optional, Union
from dataclasses import dataclass, asdict
from pathlib import Path
import os
import platform
from contextlib import contextmanager

# Optional dependency for system monitoring
try:
    import psutil
    PSUTIL_AVAILABLE = True
except ImportError:
    PSUTIL_AVAILABLE = False
    psutil = None


@dataclass
class ProfileOperationMetrics:
    """Metrics for profile operations."""
    operation_id: str
    operation_type: str  # 'load', 'save', 'validate', 'convert', 'migrate'
    start_time: datetime
    end_time: Optional[datetime] = None
    duration_ms: Optional[float] = None
    success: Optional[bool] = None
    error_message: Optional[str] = None
    parameters_processed: int = 0
    file_size_bytes: Optional[int] = None
    memory_usage_mb: Optional[float] = None
    cpu_usage_percent: Optional[float] = None


@dataclass
class DiagnosticInfo:
    """Diagnostic information for troubleshooting."""
    timestamp: datetime
    operation_id: str
    system_info: Dict[str, Any]
    application_state: Dict[str, Any]
    error_context: Optional[Dict[str, Any]] = None
    stack_trace: Optional[str] = None


class ProfileOperationLogger:
    """
    Comprehensive logging service for profile operations with performance monitoring
    and diagnostic information collection.
    """
    
    def __init__(self, log_directory: str = "logs"):
        """
        Initialize the profile operation logger.
        
        Args:
            log_directory: Directory to store log files
        """
        self.log_directory = Path(log_directory)
        self.log_directory.mkdir(exist_ok=True)
        
        # Setup loggers
        self._setup_loggers()
        
        # Operation tracking
        self._active_operations: Dict[str, ProfileOperationMetrics] = {}
        self._operation_history: List[ProfileOperationMetrics] = []
        self._lock = threading.Lock()
        
        # Performance monitoring
        self._performance_samples: List[Dict[str, Any]] = []
        self._diagnostic_data: List[DiagnosticInfo] = []
        
        # Initialize system monitoring
        self._init_system_monitoring()
    
    def _setup_loggers(self):
        """Setup specialized loggers for different aspects of profile operations."""
        
        # Main profile operations logger
        self.profile_logger = logging.getLogger('profile_operations')
        self.profile_logger.setLevel(logging.DEBUG)
        
        # Performance logger
        self.performance_logger = logging.getLogger('profile_performance')
        self.performance_logger.setLevel(logging.INFO)
        
        # Diagnostics logger
        self.diagnostics_logger = logging.getLogger('profile_diagnostics')
        self.diagnostics_logger.setLevel(logging.DEBUG)
        
        # Error logger
        self.error_logger = logging.getLogger('profile_errors')
        self.error_logger.setLevel(logging.ERROR)
        
        # Setup file handlers
        self._setup_file_handlers()
    
    def _setup_file_handlers(self):
        """Setup file handlers for different log types."""
        
        # Create formatters
        detailed_formatter = logging.Formatter(
            '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        )
        
        json_formatter = logging.Formatter('%(message)s')
        
        # Profile operations log
        profile_handler = logging.FileHandler(
            self.log_directory / 'profile_operations.log',
            encoding='utf-8'
        )
        profile_handler.setFormatter(detailed_formatter)
        self.profile_logger.addHandler(profile_handler)
        
        # Performance log (JSON format)
        performance_handler = logging.FileHandler(
            self.log_directory / 'profile_performance.log',
            encoding='utf-8'
        )
        performance_handler.setFormatter(json_formatter)
        self.performance_logger.addHandler(performance_handler)
        
        # Diagnostics log
        diagnostics_handler = logging.FileHandler(
            self.log_directory / 'profile_diagnostics.log',
            encoding='utf-8'
        )
        diagnostics_handler.setFormatter(detailed_formatter)
        self.diagnostics_logger.addHandler(diagnostics_handler)
        
        # Error log
        error_handler = logging.FileHandler(
            self.log_directory / 'profile_errors.log',
            encoding='utf-8'
        )
        error_handler.setFormatter(detailed_formatter)
        self.error_logger.addHandler(error_handler)
    
    def _init_system_monitoring(self):
        """Initialize system monitoring capabilities."""
        try:
            # Get initial system info
            self.system_info = {
                'platform': platform.platform(),
                'python_version': platform.python_version(),
            }
            
            if PSUTIL_AVAILABLE:
                self.system_info.update({
                    'cpu_count': psutil.cpu_count(),
                    'memory_total_gb': round(psutil.virtual_memory().total / (1024**3), 2),
                    'disk_free_gb': round(psutil.disk_usage('.').free / (1024**3), 2)
                })
            else:
                self.system_info.update({
                    'cpu_count': 'N/A (psutil not available)',
                    'memory_total_gb': 'N/A (psutil not available)',
                    'disk_free_gb': 'N/A (psutil not available)'

        except Exception as e:

    def start_operation(self, operation_type: str, context: Dict[str, Any] = None) -> str:
        """
        Start tracking a profile operation.
        
        Args:
            operation_type: Type of operation (load, save, validate, etc.)
            context: Additional context information
            
        Returns:
            Operation ID for tracking
        """
        operation_id = f"{operation_type}_{int(time.time() * 1000)}"
        
        metrics = ProfileOperationMetrics(
            operation_id=operation_id,
            operation_type=operation_type,
            start_time=datetime.now()
        )
        
        with self._lock:
            self._active_operations[operation_id] = metrics
        
        # Log operation start
        self.profile_logger.info(
            f"OPERATION_START: {operation_id} - {operation_type}",
            extra={'operation_id': operation_id, 'context': context or {}}
        )
        
        # Collect initial diagnostic info
        self._collect_diagnostic_info(operation_id, "operation_start", context)
        
        return operation_id
    
    def end_operation(self, operation_id: str, success: bool = True, 
                     error_message: str = None, additional_metrics: Dict[str, Any] = None):
        """
        End tracking a profile operation.
        
        Args:
            operation_id: Operation ID from start_operation
            success: Whether the operation succeeded
            error_message: Error message if operation failed
            additional_metrics: Additional metrics to record
        """
        with self._lock:
            if operation_id not in self._active_operations:
                self.profile_logger.warning(f"Unknown operation ID: {operation_id}")
                return
            
            metrics = self._active_operations[operation_id]
            metrics.end_time = datetime.now()
            metrics.duration_ms = (metrics.end_time - metrics.start_time).total_seconds() * 1000
            metrics.success = success
            metrics.error_message = error_message
            
            # Add additional metrics
            if additional_metrics:
                if 'parameters_processed' in additional_metrics:
                    metrics.parameters_processed = additional_metrics['parameters_processed']
                if 'file_size_bytes' in additional_metrics:
                    metrics.file_size_bytes = additional_metrics['file_size_bytes']
            
            # Collect system metrics
            try:
                if PSUTIL_AVAILABLE:
                    metrics.memory_usage_mb = round(psutil.Process().memory_info().rss / (1024**2), 2)
                    metrics.cpu_usage_percent = round(psutil.Process().cpu_percent(), 2)
            except Exception:
                pass
            
            # Move to history
            self._operation_history.append(metrics)
            del self._active_operations[operation_id]
        
        # Log operation end
        log_level = logging.INFO if success else logging.ERROR
        status = "SUCCESS" if success else "FAILURE"
        
        self.profile_logger.log(
            log_level,
            f"OPERATION_END: {operation_id} - {status} - {metrics.duration_ms:.2f}ms"
        )
        
        # Log performance metrics
        self._log_performance_metrics(metrics)
        
        # Log error if failed
        if not success and error_message:
            self.error_logger.error(
                f"Operation {operation_id} failed: {error_message}",
                extra={'operation_id': operation_id, 'operation_type': metrics.operation_type}
            )
            
            # Collect error diagnostic info
            self._collect_diagnostic_info(operation_id, "operation_error", {
                'error_message': error_message,
                'stack_trace': traceback.format_exc()
            })
    
    @contextmanager
    def operation_context(self, operation_type: str, context: Dict[str, Any] = None):
        """
        Context manager for tracking profile operations.
        
        Args:
            operation_type: Type of operation
            context: Additional context information
        """
        operation_id = self.start_operation(operation_type, context)
        try:
            yield operation_id
            self.end_operation(operation_id, success=True)
        except Exception as e:
            self.end_operation(operation_id, success=False, error_message=str(e))
            raise
    
    def log_operation_step(self, operation_id: str, step: str, details: Dict[str, Any] = None):

            operation_id: Operation ID
            validation_result: Validation result details

        is_valid = validation_result.get('is_valid', False)
        log_level = logging.INFO if is_valid else logging.WARNING

    def _log_performance_metrics(self, metrics: ProfileOperationMetrics):

        performance_data = {

        with self._lock:

            # Keep only recent samples (last 1000)

    def _collect_diagnostic_info(self, operation_id: str, event_type: str, 
                                context: Dict[str, Any] = None):
        """
        Collect diagnostic information for troubleshooting.
        
        Args:
            operation_id: Operation ID
            event_type: Type of diagnostic event
            context: Additional context information

        try:
            # Collect system state
            system_state = {
                'active_operations': len(self._active_operations),
                'operation_history_count': len(self._operation_history)

            if PSUTIL_AVAILABLE:
                system_state.update({
                    'memory_usage': psutil.virtual_memory()._asdict(),

                system_state.update({
                    'memory_usage': 'N/A (psutil not available)',
                    'cpu_usage': 'N/A (psutil not available)',
                    'disk_usage': 'N/A (psutil not available)'
                })
            
            # Collect application state
            app_state = {
                'event_type': event_type,
                'thread_count': threading.active_count(),
                'log_directory_size': self._get_log_directory_size()
            }
            
            diagnostic_info = DiagnosticInfo(

                system_info=system_state,
                application_state=app_state,

            )
            
            if context and 'stack_trace' in context:

            with self._lock:

                # Keep only recent diagnostic data (last 500 entries)

            # Log diagnostic info

    def get_operation_metrics(self, operation_type: str = None,

        """
        Get operation metrics for analysis.
        
        Args:
            operation_type: Filter by operation type (optional)
            hours: Number of hours to look back
            
        Returns:
            List of operation metrics

        with self._lock:
            filtered_metrics = [

    def get_performance_summary(self, hours: int = 24) -> Dict[str, Any]:
        """
        Get performance summary for the specified time period.
        
        Args:
            hours: Number of hours to analyze
            
        Returns:
            Performance summary statistics
        """
        metrics = self.get_operation_metrics(hours=hours)
        
        if not metrics:
            return {'message': 'No operations found in the specified time period'}
        
        # Calculate statistics
        durations = [m.duration_ms for m in metrics if m.duration_ms is not None]
        success_count = sum(1 for m in metrics if m.success)
        total_count = len(metrics)
        
        summary = {
            'time_period_hours': hours,
            'total_operations': total_count,
            'successful_operations': success_count,
            'success_rate_percent': round((success_count / total_count) * 100, 2) if total_count > 0 else 0,
            'average_duration_ms': round(sum(durations) / len(durations), 2) if durations else 0,
            'min_duration_ms': min(durations) if durations else 0,
            'max_duration_ms': max(durations) if durations else 0,
            'operations_by_type': {}
        }
        
        # Group by operation type
        for metric in metrics:
            op_type = metric.operation_type
            if op_type not in summary['operations_by_type']:
                summary['operations_by_type'][op_type] = {
                    'count': 0,
                    'success_count': 0,
                    'average_duration_ms': 0
                }
            
            summary['operations_by_type'][op_type]['count'] += 1
            if metric.success:
                summary['operations_by_type'][op_type]['success_count'] += 1
            
            if metric.duration_ms:
                current_avg = summary['operations_by_type'][op_type]['average_duration_ms']
                count = summary['operations_by_type'][op_type]['count']
                summary['operations_by_type'][op_type]['average_duration_ms'] = round(
                    (current_avg * (count - 1) + metric.duration_ms) / count, 2
                )
        
        return summary

"""
Profile Operation Manager for LT32Pro Devices

This module provides a high-level interface for orchestrating complete profile operations
including loading profiles from devices and sending profiles to devices with real-time
progress tracking and comprehensive error handling.
"""

import time
import threading
from typing import Dict, Optional, Callable, Any, List, Tuple
from dataclasses import dataclass
from enum import Enum
import logging

from services.configuration_reader_service import ConfigurationReaderService
from services.unified_device_communicator import UnifiedDeviceCommunicator
from services.lt32pro_device_writer import LT32ProDeviceWriter
from services.enhanced_parameter_extractor import EnhancedParameterExtractor
from services.parameter_processor import ParameterProcessor
from services.parameter_error_feedback_service import ParameterErrorFeedbackService
from services.lt32pro_error_handler import LT32ProErrorHandler
from services.ui_responsiveness_service import UIResponsivenessService, UpdatePriority, BatchUpdateResult

logger = logging.getLogger(__name__)


class OperationType(Enum):
    """Types of profile operations."""
    LOAD_FROM_DEVICE = "load_from_device"
    SEND_TO_DEVICE = "send_to_device"


class OperationStatus(Enum):
    """Status of profile operations."""
    NOT_STARTED = "not_started"
    IN_PROGRESS = "in_progress"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELLED = "cancelled"


@dataclass
class OperationResult:
    """Result of a profile operation."""
    operation_type: OperationType
    status: OperationStatus
    success_count: int
    failure_count: int
    total_count: int
    duration_seconds: float
    parameters: Optional[Dict[str, str]] = None
    write_results: Optional[Dict[str, bool]] = None
    error_message: Optional[str] = None
    warnings: Optional[List[str]] = None


@dataclass
class ProgressUpdate:
    """Progress update information."""
    current_step: int
    total_steps: int
    step_description: str
    success: Optional[bool] = None
    operation_type: Optional[OperationType] = None
    elapsed_time: float = 0.0


class ProfileOperationManager:
    """
    Orchestrates complete profile operations with progress tracking.
    
    This class provides high-level methods for loading profiles from devices
    and sending profiles to devices, with real-time progress tracking,
    comprehensive error handling, and detailed operation reporting.
    """
    
    def __init__(self, serial_service, progress_callback: Optional[Callable[[ProgressUpdate], None]] = None):
        """
        Initialize the profile operation manager.
        
        Args:
            serial_service: The serial communication service instance
            progress_callback: Optional callback for progress updates
        """
        self.serial_service = serial_service
        self.progress_callback = progress_callback
        
        # Initialize components - using ConfigurationReaderService instead of LT32ProDeviceReader
        self.config_reader = ConfigurationReaderService(serial_service.app, serial_service.app.i18n)
        self.device_communicator = UnifiedDeviceCommunicator(self.config_reader)
        self.device_writer = LT32ProDeviceWriter(serial_service)
        self.parameter_extractor = EnhancedParameterExtractor()
        self.parameter_processor = ParameterProcessor()
        self.feedback_service = ParameterErrorFeedbackService()
        self.error_handler = LT32ProErrorHandler()
        
        # UI responsiveness service for batch updates and visual feedback
        self.ui_responsiveness = UIResponsivenessService()
        
        # Operation state
        self._current_operation = None
        self._operation_cancelled = False
        self._operation_start_time = None
        self._operation_lock = threading.Lock()
        
        # Results tracking
        self._last_result = None
        self._operation_warnings = []
        
    def load_profile_from_device(self, progress_callback: Optional[Callable[[ProgressUpdate], None]] = None) -> OperationResult:
        """
        Load a profile from the connected LT32Pro device.
        
        This method orchestrates the complete process of reading device configuration,
        extracting individual parameters, and creating a profile dictionary.
        
        Args:
            progress_callback: Optional callback for progress updates
            
        Returns:
            OperationResult with extracted parameters and operation details
            
        Raises:
            ConnectionError: If device is not connected
            RuntimeError: If another operation is in progress
        """
        with self._operation_lock:
            if self._current_operation is not None:
                raise RuntimeError(f"Another operation is in progress: {self._current_operation}")
            
            self._current_operation = OperationType.LOAD_FROM_DEVICE
            
        # Use provided callback or instance callback
        callback = progress_callback or self.progress_callback
        
        # Initialize operation state
        self._operation_cancelled = False
        self._operation_start_time = time.time()
        self._operation_warnings = []
        
        # Total steps: device reading + parameter extraction + parameter processing + validation
        total_steps = len(self.device_communicator.READ_COMMAND_SEQUENCE) + 3
        current_step = 0
        
        try:
            # Update progress - starting operation
            if callback:
                progress = ProgressUpdate(
                    current_step=0,
                    total_steps=total_steps,
                    step_description="Starting profile load operation",
                    operation_type=OperationType.LOAD_FROM_DEVICE,
                    elapsed_time=0.0
                )
                callback(progress)
            
            # Step 1: Read device configuration using ConfigurationReaderService
            device_responses = {}
            
            def device_read_progress(progress_info):
                nonlocal current_step
                if callback:
                    # Extract progress information from ConfigurationReaderService format
                    current_step = progress_info.get('current_step', 0)
                    description = progress_info.get('status_text', 'Reading device configuration...')
                    success = progress_info.get('successful_count', 0) > 0
                    
                    progress = ProgressUpdate(
                        current_step=current_step,
                        total_steps=total_steps,
                        step_description=f"Device reading: {description}",
                        success=success,
                        operation_type=OperationType.LOAD_FROM_DEVICE,
                        elapsed_time=time.time() - self._operation_start_time
                    )
                    callback(progress)
            
            try:
                device_responses = self.device_communicator.read_device_configuration(device_read_progress)
                
                if not device_responses:
                    raise RuntimeError("No device responses received")
                    
            except Exception as e:
                error_msg = f"Failed to read device configuration: {str(e)}"
                self._operation_warnings.append(error_msg)
                
                # Try to handle the error and continue if possible
                try:
                    # Use the available error handling methods
                    if "timeout" in str(e).lower():
                        can_continue = self.error_handler.handle_communication_timeout("READ_CONFIG", 1)
                    else:
                        can_continue = False
                    
                    if not can_continue:
                        raise RuntimeError(error_msg)
                except Exception:
                    # If error handling fails, raise the original error
                    raise RuntimeError(error_msg)
            
            # Step 2: Extract parameters from responses
            current_step += 1
            if callback:
                progress = ProgressUpdate(
                    current_step=current_step,
                    total_steps=total_steps,
                    step_description="Extracting parameters from device responses",
                    operation_type=OperationType.LOAD_FROM_DEVICE,
                    elapsed_time=time.time() - self._operation_start_time
                )
                callback(progress)
            
            try:
                extracted_parameters = self.parameter_extractor.extract_parameters_from_responses(device_responses)
                
                if not extracted_parameters:
                    self._operation_warnings.append("No parameters could be extracted from device responses")
                    
            except Exception as e:
                error_msg = f"Failed to extract parameters: {str(e)}"
                self._operation_warnings.append(error_msg)
                extracted_parameters = {}
            
            # Step 3: Process parameters for clean display
            current_step += 1
            if callback:
                progress = ProgressUpdate(
                    current_step=current_step,
                    total_steps=total_steps,
                    step_description="Processing parameters for clean display",
                    operation_type=OperationType.LOAD_FROM_DEVICE,
                    elapsed_time=time.time() - self._operation_start_time
                )
                callback(progress)
            
            try:
                processed_parameters = self._process_parameters_for_display(extracted_parameters, device_responses)
                
                # Count processing results
                original_count = len(extracted_parameters)
                processed_count = len(processed_parameters)
                
                if processed_count != original_count:
                    warning_msg = f"Parameter processing changed {abs(original_count - processed_count)} parameters"
                    self._operation_warnings.append(warning_msg)
                    
            except Exception as e:
                error_msg = f"Parameter processing failed: {str(e)}"
                self._operation_warnings.append(error_msg)
                processed_parameters = extracted_parameters  # Use unprocessed parameters
            
            # Step 4: Validate processed parameters
            current_step += 1
            if callback:
                progress = ProgressUpdate(
                    current_step=current_step,
                    total_steps=total_steps,
                    step_description="Validating processed parameters",
                    operation_type=OperationType.LOAD_FROM_DEVICE,
                    elapsed_time=time.time() - self._operation_start_time
                )
                callback(progress)
            
            try:
                validated_parameters = self.parameter_extractor.validate_extracted_parameters(processed_parameters)
                
                # Count validation results
                original_count = len(processed_parameters)
                validated_count = len(validated_parameters)
                
                if validated_count < original_count:
                    warning_msg = f"Parameter validation removed {original_count - validated_count} invalid parameters"
                    self._operation_warnings.append(warning_msg)
                    
            except Exception as e:
                error_msg = f"Parameter validation failed: {str(e)}"
                self._operation_warnings.append(error_msg)
                validated_parameters = processed_parameters  # Use unvalidated parameters
            
            # Calculate operation results
            duration = time.time() - self._operation_start_time
            success_count = len(device_responses)
            total_count = len(self.device_communicator.READ_COMMAND_SEQUENCE)
            failure_count = total_count - success_count
            
            # Determine overall status
            if success_count == 0:
                status = OperationStatus.FAILED
            elif failure_count > 0:
                status = OperationStatus.COMPLETED  # Partial success
            else:
                status = OperationStatus.COMPLETED
            
            # Create result
            result = OperationResult(
                operation_type=OperationType.LOAD_FROM_DEVICE,
                status=status,
                success_count=success_count,
                failure_count=failure_count,
                total_count=total_count,
                duration_seconds=duration,
                parameters=validated_parameters,
                warnings=self._operation_warnings.copy() if self._operation_warnings else None
            )
            
            # Final progress update
            if callback:
                progress = ProgressUpdate(
                    current_step=total_steps,
                    total_steps=total_steps,
                    step_description=f"Load completed: {success_count}/{total_count} parameters extracted",
                    success=status == OperationStatus.COMPLETED,
                    operation_type=OperationType.LOAD_FROM_DEVICE,
                    elapsed_time=duration
                )
                callback(progress)
            
            self._last_result = result
            return result
            
        except Exception as e:
            # Handle operation failure
            duration = time.time() - self._operation_start_time
            error_msg = str(e)
            
            result = OperationResult(
                operation_type=OperationType.LOAD_FROM_DEVICE,
                status=OperationStatus.FAILED,
                success_count=0,
                failure_count=1,
                total_count=1,
                duration_seconds=duration,
                error_message=error_msg,
                warnings=self._operation_warnings.copy() if self._operation_warnings else None
            )
            
            # Final progress update with error
            if callback:
                progress = ProgressUpdate(
                    current_step=total_steps,
                    total_steps=total_steps,
                    step_description=f"Load failed: {error_msg}",
                    success=False,
                    operation_type=OperationType.LOAD_FROM_DEVICE,
                    elapsed_time=duration
                )
                callback(progress)
            
            self._last_result = result
            raise
            
        finally:
            with self._operation_lock:
                self._current_operation = None
    
    def send_profile_to_device(self, parameters: Dict[str, str], 
                             progress_callback: Optional[Callable[[ProgressUpdate], None]] = None) -> OperationResult:
        """
        Send a profile to the connected LT32Pro device.
        
        This method orchestrates the complete process of validating profile parameters,
        sending configuration commands to the device, and tracking the results.
        
        Args:
            parameters: Dictionary of parameters to send to device
            progress_callback: Optional callback for progress updates
            
        Returns:
            OperationResult with write results and operation details
            
        Raises:
            ConnectionError: If device is not connected
            ValueError: If parameters are invalid
            RuntimeError: If another operation is in progress
        """
        with self._operation_lock:
            if self._current_operation is not None:
                raise RuntimeError(f"Another operation is in progress: {self._current_operation}")
            
            self._current_operation = OperationType.SEND_TO_DEVICE
            
        # Use provided callback or instance callback
        callback = progress_callback or self.progress_callback
        
        # Initialize operation state
        self._operation_cancelled = False
        self._operation_start_time = time.time()
        self._operation_warnings = []
        
        if not parameters:
            raise ValueError("No parameters provided for sending to device")
        
        # Group parameters by command for progress calculation
        command_groups = self.device_writer._group_parameters_by_command(parameters)
        total_steps = len(command_groups) + 2  # +2 for validation and completion
        current_step = 0
        
        try:
            # Update progress - starting operation
            if callback:
                progress = ProgressUpdate(
                    current_step=0,
                    total_steps=total_steps,
                    step_description="Starting profile send operation",
                    operation_type=OperationType.SEND_TO_DEVICE,
                    elapsed_time=0.0
                )
                callback(progress)
            
            # Step 1: Validate parameters
            current_step += 1
            if callback:
                progress = ProgressUpdate(
                    current_step=current_step,
                    total_steps=total_steps,
                    step_description=f"Validating {len(parameters)} parameters",
                    operation_type=OperationType.SEND_TO_DEVICE,
                    elapsed_time=time.time() - self._operation_start_time
                )
                callback(progress)
            
            # Validate parameters before sending
            validated_parameters = self._validate_parameters_for_writing(parameters)
            
            if not validated_parameters:
                raise ValueError("No valid parameters found for device configuration")
            
            validation_warnings = len(parameters) - len(validated_parameters)
            if validation_warnings > 0:
                warning_msg = f"Parameter validation removed {validation_warnings} invalid parameters"
                self._operation_warnings.append(warning_msg)
            
            # Step 2: Send parameters to device
            write_results = {}
            
            def device_write_progress(step, total, description, success):
                nonlocal current_step
                if callback:
                    progress = ProgressUpdate(
                        current_step=current_step + step,
                        total_steps=total_steps,
                        step_description=f"Device writing: {description}",
                        success=success,
                        operation_type=OperationType.SEND_TO_DEVICE,
                        elapsed_time=time.time() - self._operation_start_time
                    )
                    callback(progress)
            
            try:
                write_results = self.device_writer.write_device_configuration(
                    validated_parameters, 
                    device_write_progress
                )
                
                if not write_results:
                    raise RuntimeError("No write results received from device")
                    
            except Exception as e:
                error_msg = f"Failed to write device configuration: {str(e)}"
                self._operation_warnings.append(error_msg)
                
                # Try to handle the error and continue if possible
                try:
                    # Use the available error handling methods
                    if "timeout" in str(e).lower():
                        can_continue = self.error_handler.handle_communication_timeout("WRITE_CONFIG", 1)
                    else:
                        can_continue = False
                    
                    if not can_continue:
                        raise RuntimeError(error_msg)
                except Exception:
                    # If error handling fails, raise the original error
                    raise RuntimeError(error_msg)
                
                # Use partial results if available
                write_results = self.device_writer.get_write_results()
            
            # Step 3: Final validation and completion
            current_step = total_steps - 1
            if callback:
                progress = ProgressUpdate(
                    current_step=current_step,
                    total_steps=total_steps,
                    step_description="Finalizing operation results",
                    operation_type=OperationType.SEND_TO_DEVICE,
                    elapsed_time=time.time() - self._operation_start_time
                )
                callback(progress)
            
            # Calculate operation results
            duration = time.time() - self._operation_start_time
            success_count = sum(1 for success in write_results.values() if success)
            total_count = len(write_results)
            failure_count = total_count - success_count
            
            # Determine overall status
            if success_count == 0:
                status = OperationStatus.FAILED
            elif failure_count > 0:
                status = OperationStatus.COMPLETED  # Partial success
            else:
                status = OperationStatus.COMPLETED
            
            # Create result
            result = OperationResult(
                operation_type=OperationType.SEND_TO_DEVICE,
                status=status,
                success_count=success_count,
                failure_count=failure_count,
                total_count=total_count,
                duration_seconds=duration,
                write_results=write_results,
                warnings=self._operation_warnings.copy() if self._operation_warnings else None
            )
            
            # Final progress update
            if callback:
                progress = ProgressUpdate(
                    current_step=total_steps,
                    total_steps=total_steps,
                    step_description=f"Send completed: {success_count}/{total_count} parameters written",
                    success=status == OperationStatus.COMPLETED,
                    operation_type=OperationType.SEND_TO_DEVICE,
                    elapsed_time=duration
                )
                callback(progress)
            
            self._last_result = result
            return result
            
        except Exception as e:
            # Handle operation failure
            duration = time.time() - self._operation_start_time
            error_msg = str(e)
            
            result = OperationResult(
                operation_type=OperationType.SEND_TO_DEVICE,
                status=OperationStatus.FAILED,
                success_count=0,
                failure_count=1,
                total_count=1,
                duration_seconds=duration,
                error_message=error_msg,
                warnings=self._operation_warnings.copy() if self._operation_warnings else None
            )
            
            # Final progress update with error
            if callback:
                progress = ProgressUpdate(
                    current_step=total_steps,
                    total_steps=total_steps,
                    step_description=f"Send failed: {error_msg}",
                    success=False,
                    operation_type=OperationType.SEND_TO_DEVICE,
                    elapsed_time=duration
                )
                callback(progress)
            
            self._last_result = result
            raise
            
        finally:
            with self._operation_lock:
                self._current_operation = None
    
    def _process_parameters_for_display(self, extracted_parameters: Dict[str, str], device_responses: Dict[str, str]) -> Dict[str, str]:
        """
        Process extracted parameters for clean UI display with comprehensive error handling.
        
        This method applies parameter processing to clean up time parameters,
        odometer values, and hourmeter values according to the requirements.
        Uses safe parameter processing with fallbacks for malformed data.
        
        Args:
            extracted_parameters: Parameters extracted from device responses
            device_responses: Raw device responses for additional processing
            
        Returns:
            Dictionary of processed parameters with clean values for UI display
            
        Requirements: 1.4, 1.5, 2.1, 2.2, 5.1, 5.2, 5.3, 5.4, 5.5
        """
        try:
            # Use the safe parameter processing method
            processing_summary = self.parameter_processor.safe_process_parameters(
                extracted_parameters, 
                device_responses
            )
            
            # Generate user feedback for processing results
            feedback_messages = self.feedback_service.generate_processing_feedback(processing_summary)
            
            # Add feedback messages to operation warnings for user visibility
            for feedback in feedback_messages:
                if feedback.level.value in ['warning', 'error']:
                    self._operation_warnings.append(f"{feedback.title}: {feedback.message}")
            
            # Extract processed parameters from results
            processed_parameters = {}
            for param_name, result in processing_summary.processing_results.items():
                processed_parameters[param_name] = result.processed_value
            
            # Use batch UI updates for processed parameters (Requirements 7.1, 7.2)
            if processed_parameters:
                try:
                    batch_result = self.ui_responsiveness.process_large_parameter_set(
                        processed_parameters,
                        progress_callback=lambda current, total: logger.debug(f"UI update progress: {current}/{total}")
                    )
                    
                    if batch_result.update_duration_ms > 100:  # Requirement 7.1
                        warning_msg = f"UI updates took {batch_result.update_duration_ms:.1f}ms, exceeding 100ms target"
                        self._operation_warnings.append(warning_msg)
                        logger.warning(warning_msg)
                    else:
                        logger.debug(f"UI updates completed in {batch_result.update_duration_ms:.1f}ms")
                        
                except Exception as e:
                    warning_msg = f"UI batch update failed: {str(e)}"
                    self._operation_warnings.append(warning_msg)
                    logger.warning(warning_msg)
            
            # Log comprehensive processing summary
            logger.info(f"Parameter processing completed: {processing_summary.successful_parameters}/{processing_summary.total_parameters} parameters processed successfully")
            
            if processing_summary.failed_parameters > 0:
                logger.warning(f"Parameter processing had {processing_summary.failed_parameters} failures: {processing_summary.errors[:3]}")
            
            if processing_summary.warnings:
                logger.info(f"Parameter processing had {len(processing_summary.warnings)} warnings")
            
            return processed_parameters
            
        except Exception as e:
            # Handle catastrophic failure in parameter processing
            error_msg = f"Critical error in parameter processing: {str(e)}"
            logger.error(error_msg)
            self._operation_warnings.append(error_msg)
            
            # Generate error feedback for user
            try:
                error_feedback = self.feedback_service.generate_malformed_data_feedback(
                    "parameter_processing", 
                    str(extracted_parameters)[:200], 
                    error_msg
                )
                self._operation_warnings.append(f"{error_feedback.title}: {error_feedback.message}")
            except Exception:
                # If even feedback generation fails, just log it
                logger.error("Failed to generate error feedback for parameter processing failure")
            
            # Return original parameters as fallback (Requirement 5.5)
            logger.info("Falling back to original extracted parameters due to processing failure")
            return extracted_parameters
    
    def _find_specific_odometer_response(self, param_name: str, device_responses: Dict[str, str]) -> Optional[str]:
        """
        Find the most appropriate raw device response for a specific odometer parameter.
        
        Args:
            param_name: Name of the parameter being processed
            device_responses: Raw device responses
            
        Returns:
            Raw response most appropriate for this parameter, or None if not found
        """
        param_name_lower = param_name.lower()
        
        # Priority order for finding responses based on parameter name
        if 'current' in param_name_lower or 'cur' in param_name_lower:
            # For current_mileage, prefer status response with "Cur Mileage Value"
            for command, response in device_responses.items():
                if response and 'Cur Mileage Value' in response:
                    return response
        
        # For general odometer/mileage parameters, prefer direct odometer command
        for command, response in device_responses.items():
            if not response:
                continue
                
            command_lower = command.lower()
            response_upper = response.upper()
            
            # Prefer direct odometer/mileage commands
            if 'odometer' in command_lower or 'mileage' in command_lower:
                if 'MILEAGE=' in response_upper:
                    return response
        
        # Fallback to any response containing odometer data
        return self._find_odometer_response(device_responses)
    
    def _find_specific_hourmeter_response(self, param_name: str, device_responses: Dict[str, str]) -> Optional[str]:
        """
        Find the most appropriate raw device response for a specific hourmeter parameter.
        
        Args:
            param_name: Name of the parameter being processed
            device_responses: Raw device responses
            
        Returns:
            Raw response most appropriate for this parameter, or None if not found
        """
        param_name_lower = param_name.lower()
        
        # First, try to find direct command match for the parameter
        for command, response in device_responses.items():
            if not response:
                continue
                
            command_lower = command.lower()
            response_upper = response.upper()
            
            # Direct command match - prefer exact parameter name matches
            if param_name_lower in command_lower:
                if 'ACCONT=' in response_upper:
                    return response
        
        # Priority order for finding responses based on parameter name
        if 'acc_on' in param_name_lower and 'accont' != param_name_lower:
            # For acc_on_time parameters (but not 'accont' itself), prefer status response with "Get ACC ON Time"
            for command, response in device_responses.items():
                if response and 'Get ACC ON Time' in response:
                    return response
        
        # For general hourmeter parameters, prefer direct hourmeter/accont commands
        for command, response in device_responses.items():
            if not response:
                continue
                
            command_lower = command.lower()
            response_upper = response.upper()
            
            # Prefer direct hourmeter/accont commands
            if 'hourmeter' in command_lower or 'accont' in command_lower:
                if 'ACCONT=' in response_upper:
                    return response
        
        # Fallback to any response containing hourmeter data
        return self._find_hourmeter_response(device_responses)
    
    def _find_odometer_response(self, device_responses: Dict[str, str]) -> Optional[str]:
        """
        Find the raw device response containing odometer data.
        
        Args:
            device_responses: Raw device responses
            
        Returns:
            Raw response containing odometer data, or None if not found
        """
        # Look for responses that might contain odometer data
        odometer_commands = ['odometer', 'mileage', 'status']
        
        for command, response in device_responses.items():
            if not response:
                continue
                
            command_lower = command.lower()
            response_upper = response.upper()
            
            # Check if this command might contain odometer data
            if any(cmd in command_lower for cmd in odometer_commands):
                # Check if response contains odometer-related keywords
                if any(keyword in response_upper for keyword in ['MILEAGE', 'ODOMETER', 'KM']):
                    return response
        
        return None
    
    def _find_hourmeter_response(self, device_responses: Dict[str, str]) -> Optional[str]:
        """
        Find the raw device response containing hourmeter data.
        
        Args:
            device_responses: Raw device responses
            
        Returns:
            Raw response containing hourmeter data, or None if not found
        """
        # Look for responses that might contain hourmeter data
        hourmeter_commands = ['hourmeter', 'accont', 'acc_on', 'status']
        
        for command, response in device_responses.items():
            if not response:
                continue
                
            command_lower = command.lower()
            response_upper = response.upper()
            
            # Check if this command might contain hourmeter data
            if any(cmd in command_lower for cmd in hourmeter_commands):
                # Check if response contains hourmeter-related keywords
                if any(keyword in response_upper for keyword in ['ACCONT', 'ACC ON', 'HOURMETER', 'MINUTE', 'HOUR']):
                    return response
        
        return None

    def _validate_parameters_for_writing(self, parameters: Dict[str, str]) -> Dict[str, str]:
        """
        Validate parameters before writing to device.
        
        Args:
            parameters: Parameters to validate
            
        Returns:
            Dictionary of validated parameters
        """
        validated = {}
        supported_params = self.device_writer.get_supported_parameters()
        
        for param_name, value in parameters.items():
            # Check if parameter is supported
            if param_name.lower() not in [p.lower() for p in supported_params]:
                warning_msg = f"Unsupported parameter ignored: {param_name}"
                self._operation_warnings.append(warning_msg)
                continue
            
            # Check if value is valid
            if not value or not str(value).strip():
                warning_msg = f"Empty parameter ignored: {param_name}"
                self._operation_warnings.append(warning_msg)
                continue
            
            # Basic validation for specific parameter types
            if self._validate_parameter_value(param_name, value):
                validated[param_name] = str(value).strip()
            else:
                warning_msg = f"Invalid parameter value ignored: {param_name}={value}"
                self._operation_warnings.append(warning_msg)
        
        return validated
    
    def _validate_parameter_value(self, param_name: str, value: str) -> bool:
        """
        Validate individual parameter values.
        
        Args:
            param_name: Name of the parameter
            value: Value to validate
            
        Returns:
            True if value is valid for the parameter
        """
        param_lower = param_name.lower()
        value_str = str(value).strip()
        
        # Server IP validation
        if 'server_ip' in param_lower or param_lower == 'server':
            # Basic IP address or domain name validation
            if not value_str or len(value_str) < 3:
                return False
            # Allow IP addresses and domain names
            return True
        
        # Port validation
        if 'port' in param_lower:
            try:
                port = int(value_str)
                return 1 <= port <= 65535
            except ValueError:
                return False
        
        # Protocol validation
        if param_lower in ['protocol', 'gt06sel']:
            return value_str in ['0', '1', '2', '3', '4']
        
        # TCP ACK validation
        if param_lower in ['tcp_ack', 'tcp_acknowledgment', 'posbyack']:
            return value_str in ['0', '1']
        
        # APN validation
        if 'apn' in param_lower:
            # APN values can be empty for user/password but not for server
            if param_lower == 'apn_server' or param_lower == 'apn':
                return len(value_str) > 0
            return True  # User and password can be empty
        
        # Default: accept non-empty values
        return len(value_str) > 0

"""
Profile Parameter Filter for LT32Pro Devices

This module provides filtering functionality to identify which parameters
can be written back to LT32Pro devices, separating read-only parameters
from writable configuration parameters.
"""

from typing import Dict, Set, List, Tuple
import logging

logger = logging.getLogger(__name__)


class ProfileParameterFilter:
    """
    Filters profile parameters to identify writable vs read-only parameters.
    
    This class helps ensure that only appropriate parameters are sent back
    to the device during profile operations, improving success rates and
    avoiding unnecessary errors.
    """
    
    # Parameters that are typically writable to LT32Pro devices
    WRITABLE_PARAMETERS = {
        'apn_server',
        'apn_user', 
        'apn_password',
        'server_ip',
        'server_port',
        'server_type',
        'protocol',
        'tcp_ack',
        'tcp_acknowledgment'
    }
    
    # Parameters that are read-only (device status/info)
    READ_ONLY_PARAMETERS = {
        'battery_voltage',
        'gprs_status',
        'gsm_signal_level',
        'acc_status',
        'gps_status',
        'device_imei',
        'device_id',
        'imei',
        'device_iccid',
        'iccid',
        'device_version',
        'firmware_date'
    }
    
    # Parameters that may not be supported by all device versions
    UNSUPPORTED_PARAMETERS = {
        'heartbeat_interval',  # Use different command format
        'report_interval',     # Use different command format
        'timezone',           # May not be supported
        'device_name',        # May not be writable
        'upload_interval',    # May not be supported
        'sleep_mode',         # May not be supported
        'power_save_mode',    # May not be supported
        'tracking_mode',      # May not be supported
        'server_protocol',    # Derived parameter
        'protocol_name'       # Derived parameter
    }
    
    def __init__(self):
        """Initialize the parameter filter."""
        self.logger = logging.getLogger(__name__)
    
    def filter_writable_parameters(self, parameters: Dict[str, str]) -> Dict[str, str]:
        """
        Filter parameters to only include those that can be written to the device.
        
        Args:
            parameters: Dictionary of all parameters
            
        Returns:
            Dictionary containing only writable parameters
        """
        if not parameters:
            return {}
        
        writable_params = {}
        filtered_count = 0
        
        for param_name, param_value in parameters.items():
            if self._is_writable_parameter(param_name):
                writable_params[param_name] = param_value
            else:
                filtered_count += 1
                self.logger.debug(f"Filtered out parameter: {param_name} (not writable)")
        
        self.logger.info(f"Parameter filtering: {len(writable_params)} writable, {filtered_count} filtered out")
        
        return writable_params
    
    def _is_writable_parameter(self, param_name: str) -> bool:
        """
        Check if a parameter is writable to the device.
        
        Args:
            param_name: Name of the parameter
            
        Returns:
            True if parameter is writable, False otherwise
        """
        param_lower = param_name.lower()
        
        # Check if it's explicitly writable
        if any(writable in param_lower for writable in self.WRITABLE_PARAMETERS):
            return True
        
        # Check if it's explicitly read-only
        if any(readonly in param_lower for readonly in self.READ_ONLY_PARAMETERS):
            return False
        
        # Check if it's unsupported
        if any(unsupported in param_lower for unsupported in self.UNSUPPORTED_PARAMETERS):
            return False
        
        # Default to not writable for unknown parameters (conservative approach)
        return False
    
    def categorize_parameters(self, parameters: Dict[str, str]) -> Dict[str, Dict[str, str]]:
        """
        Categorize parameters into writable, read-only, and unsupported.
        
        Args:
            parameters: Dictionary of all parameters
            
        Returns:
            Dictionary with categories: 'writable', 'read_only', 'unsupported'
        """
        categories = {
            'writable': {},
            'read_only': {},
            'unsupported': {}
        }
        
        for param_name, param_value in parameters.items():
            param_lower = param_name.lower()
            
            if any(writable in param_lower for writable in self.WRITABLE_PARAMETERS):
                categories['writable'][param_name] = param_value
            elif any(readonly in param_lower for readonly in self.READ_ONLY_PARAMETERS):
                categories['read_only'][param_name] = param_value
            elif any(unsupported in param_lower for unsupported in self.UNSUPPORTED_PARAMETERS):
                categories['unsupported'][param_name] = param_value
            else:
                # Unknown parameters go to unsupported for safety
                categories['unsupported'][param_name] = param_value
        
        return categories
    
    def get_filtering_report(self, parameters: Dict[str, str]) -> Dict[str, any]:
        """
        Generate a detailed filtering report.
        
        Args:
            parameters: Dictionary of all parameters
            
        Returns:
            Dictionary with filtering statistics and recommendations
        """
        categories = self.categorize_parameters(parameters)
        
        total_params = len(parameters)
        writable_count = len(categories['writable'])
        readonly_count = len(categories['read_only'])
        unsupported_count = len(categories['unsupported'])
        
        report = {
            'total_parameters': total_params,
            'writable_parameters': writable_count,
            'read_only_parameters': readonly_count,
            'unsupported_parameters': unsupported_count,
            'writable_percentage': (writable_count / total_params * 100) if total_params > 0 else 0,
            'categories': categories,
            'recommendations': self._generate_recommendations(categories)
        }
        
        return report
    
    def _generate_recommendations(self, categories: Dict[str, Dict[str, str]]) -> List[str]:
        """Generate recommendations based on parameter categorization."""
        recommendations = []
        
        writable_count = len(categories['writable'])
        readonly_count = len(categories['read_only'])
        unsupported_count = len(categories['unsupported'])
        
        if writable_count == 0:
            recommendations.append("No writable parameters found - check parameter extraction")
        elif writable_count < 5:
            recommendations.append("Few writable parameters found - verify device compatibility")
        
        if readonly_count > writable_count:
            recommendations.append("Many read-only parameters detected - consider filtering before send operations")
        
        if unsupported_count > 0:
            recommendations.append(f"{unsupported_count} unsupported parameters found - update parameter mapping if needed")
        
        if writable_count > 0:
            recommendations.append(f"Recommend sending only {writable_count} writable parameters to device")
        
        return recommendations

            # Record metric
            self._record_metric(
                PerformanceMetricType.UI_RESPONSIVENESS,
                operation_id,
                profile.operation_type,
                duration,
                "seconds",
                {"freeze_duration": duration}
            )
    
    def finish_operation_monitoring(self, operation_id: str, success: bool = True, 
                                  error_message: Optional[str] = None) -> OperationPerformanceProfile:
        """
        Finish monitoring an operation and generate performance profile.
        
        Args:
            operation_id: Operation identifier
            success: Whether operation completed successfully
            error_message: Error message if operation failed
            
        Returns:
            Complete performance profile for the operation
        """
        if operation_id not in self.operation_profiles:
            raise ValueError(f"Operation {operation_id} not found in monitoring")
            
        profile = self.operation_profiles[operation_id]
        op_data = self.active_operations.get(operation_id, {})
        
        # Finalize profile
        profile.end_time = datetime.now()
        profile.total_duration = (profile.end_time - profile.start_time).total_seconds()
        profile.success = success
        profile.error_message = error_message
        
        # Calculate final metrics
        if op_data:
            profile.parameters_processed = op_data['parameters_processed']
            if profile.total_duration > 0:
                profile.parameters_per_second = profile.parameters_processed / profile.total_duration
            
            profile.progress_updates_sent = op_data['progress_update_count']
            if profile.total_duration > 0:
                profile.progress_update_frequency = profile.progress_updates_sent / profile.total_duration
            
            # Memory usage statistics
            memory_samples = op_data.get('memory_samples', [])
            if memory_samples:
                profile.peak_memory_usage = max(memory_samples)
                profile.average_memory_usage = sum(memory_samples) / len(memory_samples)
            
            # Thread count statistics
            thread_samples = op_data.get('thread_count_samples', [])
            if thread_samples:
                profile.thread_count_peak = max(thread_samples)
        
        # Generate optimization recommendations
        profile.optimization_recommendations = self._generate_optimization_recommendations(profile)
        
        # Clean up active operation tracking
        if operation_id in self.active_operations:
            del self.active_operations[operation_id]
        
        self.logger.info(f"Finished monitoring operation {operation_id}: "
                        f"duration={profile.total_duration:.2f}s, "
                        f"params={profile.parameters_processed}, "
                        f"rate={profile.parameters_per_second:.2f} params/sec")
        
        return profile
    
    def get_performance_summary(self, operation_type: Optional[str] = None, 
                              time_window: Optional[timedelta] = None) -> Dict[str, Any]:
        """
        Get performance summary statistics.
        
        Args:
            operation_type: Filter by operation type (optional)
            time_window: Time window for analysis (optional)
            
        Returns:
            Dictionary containing performance summary
        """
        # Filter profiles based on criteria
        profiles = list(self.operation_profiles.values())
        
        if operation_type:
            profiles = [p for p in profiles if p.operation_type == operation_type]
            
        if time_window:
            cutoff_time = datetime.now() - time_window
            profiles = [p for p in profiles if p.start_time >= cutoff_time]
        
        if not profiles:
            return {"message": "No operations found matching criteria"}
        
        # Calculate summary statistics
        total_operations = len(profiles)
        successful_operations = len([p for p in profiles if p.success])
        
        durations = [p.total_duration for p in profiles if p.total_duration > 0]
        memory_peaks = [p.peak_memory_usage for p in profiles if p.peak_memory_usage > 0]
        processing_rates = [p.parameters_per_second for p in profiles if p.parameters_per_second > 0]
        
        summary = {
            "total_operations": total_operations,
            "successful_operations": successful_operations,
            "success_rate": successful_operations / total_operations if total_operations > 0 else 0,
            "performance_metrics": {}
        }
        
        if durations:
            summary["performance_metrics"]["execution_time"] = {
                "average": sum(durations) / len(durations),
                "min": min(durations),
                "max": max(durations),
                "median": sorted(durations)[len(durations) // 2]
            }
        
        if memory_peaks:
            summary["performance_metrics"]["memory_usage"] = {
                "average_peak": sum(memory_peaks) / len(memory_peaks),
                "min_peak": min(memory_peaks),
                "max_peak": max(memory_peaks)
            }
        
        if processing_rates:
            summary["performance_metrics"]["processing_rate"] = {
                "average": sum(processing_rates) / len(processing_rates),
                "min": min(processing_rates),
                "max": max(processing_rates)
            }
        
        # Generate overall recommendations
        summary["optimization_recommendations"] = self._generate_overall_recommendations(profiles)
        
        return summary
    
    def optimize_progress_update_frequency(self, operation_id: str) -> float:
        """
        Calculate optimal progress update frequency for an operation.
        
        Args:
            operation_id: Operation identifier
            
        Returns:
            Recommended update interval in seconds
        """
        if operation_id not in self.active_operations:
            return self.optimal_progress_update_interval
            
        op_data = self.active_operations[operation_id]
        
        # Analyze current update frequency
        current_frequency = op_data['progress_update_count']
        elapsed_time = time.time() - op_data['start_time']
        
        if elapsed_time > 0:
            updates_per_second = current_frequency / elapsed_time
            
            # Adjust based on performance thresholds
            if updates_per_second > self.performance_thresholds[PerformanceMetricType.PROGRESS_UPDATE_FREQUENCY]:
                # Too frequent, increase interval
                return min(self.optimal_progress_update_interval * 1.5, 1.0)
            elif updates_per_second < 1.0:
                # Too infrequent, decrease interval
                return max(self.optimal_progress_update_interval * 0.5, 0.05)
        
        return self.optimal_progress_update_interval
    
    def should_trigger_memory_cleanup(self) -> bool:
        """
        Determine if memory cleanup should be triggered.
        
        Returns:
            True if memory cleanup is recommended
        """
        if not self.psutil_available or not self.process:
            return False
            
        try:
            current_memory = self.process.memory_info().rss / 1024 / 1024  # MB
            memory_increase = current_memory - self.baseline_memory
            
            return memory_increase > self.memory_cleanup_threshold
        except Exception:
            return False
    
    def trigger_memory_cleanup(self) -> Tuple[float, float]:
        """
        Trigger memory cleanup and return before/after memory usage.
        
        Returns:
            Tuple of (memory_before_mb, memory_after_mb)
        """
        if not self.psutil_available or not self.process:
            # Just run garbage collection without memory monitoring
            gc.collect()
            return 0.0, 0.0
            
        try:
            memory_before = self.process.memory_info().rss / 1024 / 1024
            
            # Force garbage collection
            gc.collect()
            
            memory_after = self.process.memory_info().rss / 1024 / 1024
            
            self.logger.info(f"Memory cleanup: {memory_before:.1f}MB -> {memory_after:.1f}MB "
                           f"(freed {memory_before - memory_after:.1f}MB)")
            
            return memory_before, memory_after
            
        except Exception as e:
            self.logger.error(f"Memory cleanup failed: {e}")
            return 0.0, 0.0
    
    def get_optimization_recommendations(self, operation_id: str) -> List[str]:

        # Check memory usage
        if self.should_trigger_memory_cleanup():

        # Check processing rate
        if op_data['parameters_processed'] > 0 and elapsed_time > 5:
            rate = op_data['parameters_processed'] / elapsed_time
            if rate < 1.0:
                recommendations.append("Parameter processing rate is slow, consider optimization")
        
        return recommendations
    
    def _monitoring_loop(self) -> None:
        """Main monitoring loop running in background thread."""
        while self.monitoring_active:
            try:
                # Sample system metrics for active operations
                if self.psutil_available and self.process:
                    current_memory = self.process.memory_info().rss / 1024 / 1024  # MB
                else:
                    current_memory = 0.0
                current_threads = threading.active_count()
                
                # Update active operations with current samples
                for operation_id, op_data in self.active_operations.items():
                    op_data['memory_samples'].append(current_memory)
                    op_data['thread_count_samples'].append(current_threads)
                    
                    # Keep only recent samples (last 100)
                    if len(op_data['memory_samples']) > 100:
                        op_data['memory_samples'] = op_data['memory_samples'][-100:]
                    if len(op_data['thread_count_samples']) > 100:
                        op_data['thread_count_samples'] = op_data['thread_count_samples'][-100:]
                
                # Record system-wide metrics
                if self.active_operations:
                    for operation_id in list(self.active_operations.keys()):
                        if operation_id in self.operation_profiles:
                            self._record_metric(
                                PerformanceMetricType.MEMORY_USAGE,
                                operation_id,
                                self.operation_profiles[operation_id].operation_type,
                                current_memory,
                                "MB"
                            )
                            
                            self._record_metric(
                                PerformanceMetricType.THREAD_COUNT,
                                operation_id,
                                self.operation_profiles[operation_id].operation_type,
                                current_threads,
                                "threads"
                            )
                
                # Sleep for monitoring interval
                time.sleep(1.0)
                
            except Exception as e:
                self.logger.error(f"Monitoring loop error: {e}")
                time.sleep(5.0)  # Longer sleep on error

"""
Profile Performance Optimizer

This module provides performance optimization utilities for LT32Pro profile operations
including batch processing, memory management, and operation caching.
"""

import time
import threading
import gc
from typing import Dict, List, Optional, Any, Callable, Tuple
from dataclasses import dataclass
from concurrent.futures import ThreadPoolExecutor, as_completed
import logging

logger = logging.getLogger(__name__)


@dataclass
class OptimizationConfig:
    """Configuration for performance optimization."""
    enable_batch_processing: bool = True
    batch_size: int = 10
    enable_caching: bool = True
    cache_size: int = 100
    enable_parallel_processing: bool = True
    max_workers: int = 3
    enable_memory_optimization: bool = True
    gc_threshold: int = 50  # Number of operations before forcing garbage collection


class ProfilePerformanceOptimizer:
    """
    Performance optimizer for LT32Pro profile operations.
    
    Provides optimization strategies including:
    - Batch processing for large parameter sets
    - Memory management and garbage collection
    - Operation result caching
    - Parallel processing for independent operations
    """
    
    def __init__(self, config: OptimizationConfig = None):
        """Initialize the performance optimizer."""
        self.config = config or OptimizationConfig()
        self.operation_cache: Dict[str, Any] = {}
        self.operation_count = 0
        self.cache_hits = 0
        self.cache_misses = 0
        self._lock = threading.Lock()
        
        logger.info(f"ProfilePerformanceOptimizer initialized with config: {self.config}")
    
    def optimize_parameter_processing(self, parameters: Dict[str, str], 
                                    processor_func: Callable[[Dict[str, str]], Any]) -> Any:
        """
        Optimize parameter processing using batch processing and caching.
        
        Args:
            parameters: Dictionary of parameters to process
            processor_func: Function to process parameters
            
        Returns:
            Processing result
        """
        if not self.config.enable_batch_processing or len(parameters) <= self.config.batch_size:
            return self._process_with_caching(parameters, processor_func)
        
        # Process in batches
        logger.info(f"Processing {len(parameters)} parameters in batches of {self.config.batch_size}")
        
        parameter_items = list(parameters.items())
        batches = [
            dict(parameter_items[i:i + self.config.batch_size])
            for i in range(0, len(parameter_items), self.config.batch_size)
        ]
        
        batch_results = []
        
        if self.config.enable_parallel_processing and len(batches) > 1:
            # Process batches in parallel
            with ThreadPoolExecutor(max_workers=self.config.max_workers) as executor:
                future_to_batch = {
                    executor.submit(self._process_with_caching, batch, processor_func): batch
                    for batch in batches
                }
                
                for future in as_completed(future_to_batch):
                    try:
                        result = future.result()
                        batch_results.append(result)
                    except Exception as e:
                        logger.error(f"Batch processing failed: {e}")
                        batch_results.append(None)
        else:
            # Process batches sequentially
            for batch in batches:
                try:
                    result = self._process_with_caching(batch, processor_func)
                    batch_results.append(result)
                except Exception as e:
                    logger.error(f"Batch processing failed: {e}")
                    batch_results.append(None)
        
        # Combine batch results
        return self._combine_batch_results(batch_results)
    
    def _process_with_caching(self, parameters: Dict[str, str], 
                            processor_func: Callable[[Dict[str, str]], Any]) -> Any:
        """Process parameters with caching support."""
        if not self.config.enable_caching:
            return processor_func(parameters)
        
        # Generate cache key
        cache_key = self._generate_cache_key(parameters)
        
        with self._lock:
            # Check cache
            if cache_key in self.operation_cache:
                self.cache_hits += 1
                logger.debug(f"Cache hit for key: {cache_key[:20]}...")
                return self.operation_cache[cache_key]
            
            self.cache_misses += 1
        
        # Process parameters
        result = processor_func(parameters)
        
        with self._lock:
            # Store in cache
            if len(self.operation_cache) >= self.config.cache_size:
                # Remove oldest entry (simple FIFO)
                oldest_key = next(iter(self.operation_cache))
                del self.operation_cache[oldest_key]
            
            self.operation_cache[cache_key] = result
        
        logger.debug(f"Cache miss for key: {cache_key[:20]}...")
        return result
    
    def _generate_cache_key(self, parameters: Dict[str, str]) -> str:
        """Generate a cache key for parameters."""
        # Sort parameters for consistent key generation
        sorted_params = sorted(parameters.items())
        return str(hash(tuple(sorted_params)))
    
    def _combine_batch_results(self, batch_results: List[Any]) -> Any:
        """Combine results from batch processing."""
        # This is a generic implementation - specific processors may need custom logic
        if not batch_results:
            return None
        
        # Filter out None results
        valid_results = [r for r in batch_results if r is not None]
        
        if not valid_results:
            return None
        
        # If results are dictionaries, merge them
        if all(isinstance(r, dict) for r in valid_results):
            combined = {}
            for result in valid_results:
                combined.update(result)
            return combined
        
        # If results are lists, concatenate them
        if all(isinstance(r, list) for r in valid_results):
            combined = []
            for result in valid_results:
                combined.extend(result)
            return combined
        
        # For other types, return the first valid result
        return valid_results[0]
    
    def optimize_memory_usage(self):
        """Optimize memory usage by forcing garbage collection."""
        if not self.config.enable_memory_optimization:
            return
        
        self.operation_count += 1
        
        if self.operation_count >= self.config.gc_threshold:
            logger.debug("Forcing garbage collection for memory optimization")
            gc.collect()
            self.operation_count = 0
    
    def optimize_device_communication(self, commands: List[str], 
                                    communication_func: Callable[[str], str]) -> Dict[str, str]:
        """
        Optimize device communication by batching commands and parallel execution.
        
        Args:
            commands: List of commands to send to device
            communication_func: Function to send individual commands
            
        Returns:
            Dictionary of command -> response mappings
        """
        if not commands:
            return {}
        
        logger.info(f"Optimizing communication for {len(commands)} commands")
        
        if not self.config.enable_parallel_processing or len(commands) <= 2:
            # Sequential processing for small command sets
            results = {}
            for command in commands:
                try:
                    response = communication_func(command)
                    results[command] = response
                except Exception as e:
                    logger.error(f"Command {command} failed: {e}")
                    results[command] = f"ERROR: {str(e)}"
            return results
        
        # Parallel processing for larger command sets
        results = {}
        
        with ThreadPoolExecutor(max_workers=min(self.config.max_workers, len(commands))) as executor:
            future_to_command = {
                executor.submit(communication_func, command): command
                for command in commands
            }
            
            for future in as_completed(future_to_command):
                command = future_to_command[future]
                try:
                    response = future.result()
                    results[command] = response
                except Exception as e:
                    logger.error(f"Command {command} failed: {e}")
                    results[command] = f"ERROR: {str(e)}"
        
        return results
    
    def get_optimization_stats(self) -> Dict[str, Any]:
        """Get optimization statistics."""
        with self._lock:
            total_cache_requests = self.cache_hits + self.cache_misses
            cache_hit_rate = (self.cache_hits / total_cache_requests * 100) if total_cache_requests > 0 else 0
            
            return {
                'cache_hits': self.cache_hits,
                'cache_misses': self.cache_misses,
                'cache_hit_rate': cache_hit_rate,
                'cache_size': len(self.operation_cache),
                'operation_count': self.operation_count,
                'config': {
                    'batch_processing': self.config.enable_batch_processing,
                    'batch_size': self.config.batch_size,
                    'caching': self.config.enable_caching,
                    'parallel_processing': self.config.enable_parallel_processing,
                    'max_workers': self.config.max_workers
                }
            }
    
    def clear_cache(self):
        """Clear the operation cache."""
        with self._lock:
            self.operation_cache.clear()
            logger.info("Operation cache cleared")
    
    def reset_stats(self):
        """Reset optimization statistics."""
        with self._lock:
            self.cache_hits = 0
            self.cache_misses = 0
            self.operation_count = 0
            logger.info("Optimization statistics reset")


class MemoryOptimizer:
    """Memory optimization utilities for profile operations."""
    
    @staticmethod
    def optimize_parameter_dict(parameters: Dict[str, str]) -> Dict[str, str]:
        """Optimize parameter dictionary for memory usage."""
        if not parameters:
            return parameters
        
        # Remove empty values
        optimized = {k: v for k, v in parameters.items() if v and v.strip()}
        
        # Intern string values for memory efficiency
        optimized = {k: str(v).strip() for k, v in optimized.items()}
        
        return optimized
    
    @staticmethod
    def estimate_memory_usage(data: Any) -> int:
        """Estimate memory usage of data in bytes."""
        import sys
        
        if isinstance(data, dict):
            return sum(sys.getsizeof(k) + sys.getsizeof(v) for k, v in data.items())
        elif isinstance(data, (list, tuple)):
            return sum(sys.getsizeof(item) for item in data)
        else:
            return sys.getsizeof(data)
    
    @staticmethod
    def optimize_for_large_datasets(data: Any, chunk_size: int = 1000) -> List[Any]:
        """Optimize large datasets by chunking."""
        if isinstance(data, dict):
            items = list(data.items())
            return [dict(items[i:i + chunk_size]) for i in range(0, len(items), chunk_size)]
        elif isinstance(data, list):
            return [data[i:i + chunk_size] for i in range(0, len(data), chunk_size)]
        else:
            return [data]


class ConcurrencyOptimizer:
    """Concurrency optimization utilities for profile operations."""
    
    def __init__(self, max_workers: int = 3):
        """Initialize concurrency optimizer."""
        self.max_workers = max_workers
        self.active_operations = 0
        self._lock = threading.Lock()
    
    def execute_concurrent_operations(self, operations: List[Callable], 
                                    timeout: float = 30.0) -> List[Tuple[bool, Any]]:
        """
        Execute multiple operations concurrently.
        
        Args:
            operations: List of callable operations
            timeout: Timeout for each operation
            
        Returns:
            List of (success, result) tuples
        """
        if not operations:
            return []
        
        results = []
        
        with ThreadPoolExecutor(max_workers=min(self.max_workers, len(operations))) as executor:
            future_to_index = {
                executor.submit(self._execute_with_timeout, op, timeout): i
                for i, op in enumerate(operations)
            }
            
            # Initialize results list with None values
            results = [None] * len(operations)
            
            for future in as_completed(future_to_index):
                index = future_to_index[future]
                try:
                    success, result = future.result()
                    results[index] = (success, result)
                except Exception as e:
                    results[index] = (False, str(e))
        
        return results
    
    def _execute_with_timeout(self, operation: Callable, timeout: float) -> Tuple[bool, Any]:
        """Execute operation with timeout."""
        try:
            with self._lock:
                self.active_operations += 1
            
            start_time = time.time()
            result = operation()
            duration = time.time() - start_time
            
            if duration > timeout:
                return False, f"Operation timed out after {duration:.2f}s"
            
            return True, result
            
        except Exception as e:
            return False, str(e)
        finally:
            with self._lock:
                self.active_operations -= 1
    
    def get_active_operations(self) -> int:
        """Get number of active operations."""
        with self._lock:
            return self.active_operations

        param_specs = self.spec_parser.parse_specifications()
        
        # For each parameter in the profile, populate the corresponding UI element
        for param_name, param_value in profile.parameters.items():
            if param_name in ui_elements:
                ui_element = ui_elements[param_name]
                spec = param_specs.get(param_name)
                
                if spec and ui_element:
                    # Set the value in the UI element
                    try:
                        ui_element.set_value(param_value.value)
                    except Exception as e:
                        result.add_error(f"Failed to set value for {param_name}: {str(e)}")
                        
                    # Validate the value
                    if not ui_element.validate():
                        error_msg = ui_element.get_validation_error()
                        if error_msg:
                            result.add_error(f"Validation failed for {param_name}: {error_msg}")
                            
        return result

                    if not ui_element.validate():

                        if error_msg:
                            result.add_error(f"Validation failed for {param_name}: {error_msg}")
                else:
                    # For empty values, don't validate and don't include in profile

                    pass

    def apply_visibility_conditions(self, ui_elements: Dict[str, Any], context: Dict[str, Any]) -> None:
        """
        Apply visibility conditions to UI elements based on context values.

    def _evaluate_visibility_condition(self, condition: str, context: Dict[str, Any]) -> bool:
        """
        Evaluate a visibility condition against context values.
        
        Args:
            condition: Visibility condition string (e.g., "field_name == 'value'")
            context: Current parameter values
            
        Returns:
            bool: True if condition is met, False otherwise
        """
        try:
            # Simple condition parsing: "field_name == 'value'" or "field_name == true"
            if "==" in condition:
                field_name, expected_value = condition.split("==")
                field_name = field_name.strip()
                expected_value = expected_value.strip().strip("'\"")
                
                # Convert expected value to appropriate type
                if expected_value.lower() == "true":
                    expected_value = True
                elif expected_value.lower() == "false":
                    expected_value = False
                elif expected_value.isdigit():
                    expected_value = int(expected_value)
                    
                # Get actual value from context
                actual_value = context.get(field_name)
                
                # Compare values
                return actual_value == expected_value
                
        except Exception:
            # If parsing fails, default to visible
            pass
            
        return True
    
    def get_field_dependencies(self, param_name: str) -> List[str]:

        dependencies = []

        param_spec = self.spec_parser.get_parameter_spec(param_name)

            return dependencies
            
        # Check main parameter visibility condition
        if param_spec.visibility_condition:
            dependency = self._extract_dependency_from_condition(param_spec.visibility_condition)
            if dependency:
                dependencies.append(dependency)
                
        # Check field-level visibility conditions for complex parameters
        if param_spec.fields:
            for field in param_spec.fields:
                if field.visibility_condition:
                    dependency = self._extract_dependency_from_condition(field.visibility_condition)
                    if dependency and dependency not in dependencies:
                        dependencies.append(dependency)
                        
        return dependencies

                # Initialize the grouped parameter if it doesn't exist
                if grouped_key not in grouped_params:

                # Add the field value to the grouped parameter
                grouped_params[grouped_key].value[field_name] = processed_value
            else:
                # For simple parameters, if processed_value is a dict containing the field
                # matching the parameter name (common when a single-field dict was serialized),
                # extract the inner value so UI receives the raw value (e.g., '59') instead of {'odometer': '59'}.
                from models.universal_profile import ParameterValue
                # Determine the simple parameter name (last segment)
                simple_name = param_key.split('.')[-1] if param_key else None
                if isinstance(processed_value, dict) and simple_name and simple_name in processed_value and len(processed_value) == 1:
                    extracted_value = processed_value[simple_name]
                    grouped_params[param_key] = ParameterValue(value=extracted_value, source=param_value.source)
                else:
                    grouped_params[param_key] = ParameterValue(value=processed_value, source=param_value.source)
        
        # Update each section's parameters
        for section_name, section_widget in section_widgets.items():
            if hasattr(section_widget, 'parameters'):
                for param_name, param_widget in section_widget.parameters.items():

                    if profile_param and param_widget:

                            if param_widget.widget_type == 'simple':
                                if 'main' in param_widget.ui_elements:
                                    param_widget.ui_elements['main'].set_value(profile_param.value)
                                    updated_count += 1

                            elif param_widget.widget_type == 'complex' and isinstance(profile_param.value, dict):
                                # Set values for each field
                                for field_name, field_value in profile_param.value.items():
                                    if field_name in param_widget.ui_elements:
                                        param_widget.ui_elements[field_name].set_value(field_value)
                                        updated_count += 1

                            error_msg = f"Failed to update {full_param_name}: {str(e)}"
                            result.add_error(error_msg)
                            print(f"ERROR: {error_msg}")

        if updated_count > 0:

    def _parse_parameter_value(self, value):
        """
        Parse parameter values that might be stored as string representations of dictionaries.
        
        Args:
            value: The parameter value to parse
            
        Returns:
            The parsed value (dict if it was a string representation, otherwise original value)
        """
        if isinstance(value, str):
            # Check if this looks like a string representation of a dictionary
            if value.startswith('{') and value.endswith('}'):
                try:
                    # Try to safely evaluate the string as a Python literal
                    import ast
                    parsed_value = ast.literal_eval(value)
                    return parsed_value
                except (ValueError, SyntaxError):
                    # If parsing fails, return the original string
                    return value
            else:
                # Regular string value
                return value
        else:
            # Non-string value, return as-is
            return value

        # Validate required fields
        for field_name, field_config in self.required_metadata_fields.items():
            field_value = getattr(metadata, field_name, None)
            
            if field_value is None or (isinstance(field_value, str) and not field_value.strip()):
                errors.append(ValidationError(
                    error_type=ValidationErrorType.MISSING_SECTION,
                    location=f"Profile_Info.{field_name}",
                    message=f"Required metadata field '{field_name}' is missing or empty",
                    severity=ValidationSeverity.ERROR,
                    suggestion=f"Add {field_config['description'].lower()}",
                    parameter_name=field_name,
                    section_name="Profile_Info"
                ))
        
        # Validate profile name format
        if metadata.profile_name:
            if len(metadata.profile_name) > 100:
                errors.append(ValidationError(
                    error_type=ValidationErrorType.FORMAT_ERROR,
                    location="Profile_Info.profile_name",
                    message="Profile name is too long (maximum 100 characters)",
                    severity=ValidationSeverity.ERROR,
                    suggestion="Shorten the profile name",
                    parameter_name="profile_name",
                    current_value=metadata.profile_name
                ))
            
            # Check for invalid characters
            invalid_chars = re.findall(r'[<>:"/\\|?*]', metadata.profile_name)
            if invalid_chars:
                errors.append(ValidationError(
                    error_type=ValidationErrorType.FORMAT_ERROR,
                    location="Profile_Info.profile_name",
                    message=f"Profile name contains invalid characters: {', '.join(set(invalid_chars))}",
                    severity=ValidationSeverity.ERROR,
                    suggestion="Remove invalid characters from profile name",
                    parameter_name="profile_name",
                    current_value=metadata.profile_name
                ))
        
        # Validate created_date format
        if metadata.created_date:
            try:
                # Try to parse the date
                datetime.fromisoformat(metadata.created_date.replace('Z', '+00:00'))
            except ValueError:
                errors.append(ValidationError(
                    error_type=ValidationErrorType.FORMAT_ERROR,
                    location="Profile_Info.created_date",
                    message="Invalid date format for created_date",
                    severity=ValidationSeverity.WARNING,
                    suggestion="Use ISO format (YYYY-MM-DD HH:MM:SS)",
                    parameter_name="created_date",
                    current_value=metadata.created_date,
                    expected_format="YYYY-MM-DD HH:MM:SS"
                ))
        
        # Validate IMEI format if present
        if metadata.device_imei:
            if not re.match(r'^\d{15}$', metadata.device_imei):
                errors.append(ValidationError(
                    error_type=ValidationErrorType.FORMAT_ERROR,
                    location="Profile_Info.device_imei",
                    message="Invalid IMEI format (must be 15 digits)",
                    severity=ValidationSeverity.WARNING,
                    suggestion="Ensure IMEI is exactly 15 digits",
                    parameter_name="device_imei",
                    current_value=metadata.device_imei,
                    expected_format="15 digits"
                ))
        
        # Validate parameter count consistency
        if metadata.parameter_count is not None and metadata.parameter_count < 0:
            errors.append(ValidationError(
                error_type=ValidationErrorType.RANGE_ERROR,
                location="Profile_Info.parameter_count",
                message="Parameter count cannot be negative",
                severity=ValidationSeverity.ERROR,
                suggestion="Set parameter count to actual number of parameters",
                parameter_name="parameter_count",
                current_value=str(metadata.parameter_count)
            ))
        
        is_valid = len([e for e in errors if e.severity in [ValidationSeverity.CRITICAL, ValidationSeverity.ERROR]]) == 0
        
        return MetadataValidationResult(
            is_valid=is_valid,
            errors=errors,
            warnings=warnings
        )

    def check_compatibility(self, profile_version: str) -> CompatibilityResult:
        """
        Check compatibility of a profile version with the current application.
        
        Args:
            profile_version: Version string from the profile
            
        Returns:
            CompatibilityResult with compatibility information
        """
        self.logger.debug(f"Checking compatibility for profile version: {profile_version}")
        
        # Parse version strings
        try:
            profile_ver = self._parse_version(profile_version)
            current_ver = self._parse_version(self.current_app_version)
        except ValueError as e:
            return CompatibilityResult(
                is_compatible=False,
                profile_version=profile_version,
                current_app_version=self.current_app_version,
                compatibility_level='incompatible',
                migration_required=False,
                migration_suggestions=[f"Invalid version format: {str(e)}"]
            )
        
        # Determine compatibility level
        compatibility_level = 'full'
        migration_required = False
        migration_suggestions = []
        unsupported_features = []
        
        # Major version compatibility
        if profile_ver['major'] > current_ver['major']:
            compatibility_level = 'incompatible'
            migration_suggestions.append(
                f"Profile was created with a newer version ({profile_version}). "
                f"Update application to version {profile_version} or higher."
            )
        elif profile_ver['major'] < current_ver['major']:
            compatibility_level = 'partial'
            migration_required = True
            migration_suggestions.append(
                f"Profile format is outdated (v{profile_version}). "
                f"Automatic migration to v{self.current_app_version} is recommended."
            )
        
        # Minor version compatibility
        elif profile_ver['minor'] > current_ver['minor']:
            compatibility_level = 'partial'
            migration_suggestions.append(
                f"Profile uses newer features from v{profile_version}. "
                f"Some features may not be available."
            )
        elif profile_ver['minor'] < current_ver['minor']:
            migration_suggestions.append(
                f"Profile can be upgraded to use new features from v{self.current_app_version}."
            )
        
        # Check for known compatibility issues
        compatibility_issues = self._check_version_specific_issues(profile_version)
        if compatibility_issues:
            unsupported_features.extend(compatibility_issues)
            if compatibility_level == 'full':
                compatibility_level = 'partial'
        
        is_compatible = compatibility_level != 'incompatible'
        
        return CompatibilityResult(
            is_compatible=is_compatible,
            profile_version=profile_version,
            current_app_version=self.current_app_version,
            compatibility_level=compatibility_level,
            migration_required=migration_required,
            migration_suggestions=migration_suggestions,
            unsupported_features=unsupported_features
        )

    def validate_config_parser_structure(self, config: configparser.ConfigParser) -> ValidationResult:
        """
        Validate the structure of a ConfigParser object representing a profile.
        
        Args:
            config: ConfigParser object to validate
            
        Returns:
            ValidationResult with structure validation details
        """
        self.logger.debug("Validating ConfigParser structure")
        
        errors = []
        warnings = []
        suggestions = []
        
        # Check for required sections
        for section_name, section_config in self.required_sections.items():
            if not config.has_section(section_name):
                errors.append(ValidationError(
                    error_type=ValidationErrorType.MISSING_SECTION,
                    location=section_name,
                    message=f"Required section '{section_name}' is missing",
                    severity=ValidationSeverity.CRITICAL,
                    suggestion=f"Add [{section_name}] section with {section_config['description'].lower()}",
                    section_name=section_name
                ))
        
        # Check for unknown sections (warnings only)
        known_sections = set(self.required_sections.keys()) | set(self.optional_sections.keys())
        for section_name in config.sections():
            if section_name not in known_sections:
                warnings.append(ValidationWarning(
                    location=section_name,
                    message=f"Unknown section '{section_name}' found",
                    suggestion=f"Verify that section '{section_name}' is needed"
                ))
        
        # Validate Profile_Info section structure
        if config.has_section('Profile_Info'):
            self._validate_profile_info_section(config['Profile_Info'], errors, warnings)
        
        # Count parameters in Device_Parameters section
        parameter_count = 0
        if config.has_section('Device_Parameters'):
            parameter_count = len(config['Device_Parameters'])
            if parameter_count == 0:
                warnings.append(ValidationWarning(
                    location="Device_Parameters",
                    message="Device_Parameters section is empty",
                    suggestion="Add device configuration parameters"
                ))
        
        # Create validation summary
        summary = ValidationSummary(
            total_parameters=parameter_count,
            valid_parameters=0,  # Will be updated by parameter validation
            invalid_parameters=0,  # Will be updated by parameter validation
            missing_required_sections=len([e for e in errors if e.error_type == ValidationErrorType.MISSING_SECTION]),
            compatibility_issues=0,
            format_version="unknown"
        )
        
        is_valid = len([e for e in errors if e.severity in [ValidationSeverity.CRITICAL, ValidationSeverity.ERROR]]) == 0
        
        return ValidationResult(
            is_valid=is_valid,
            errors=errors,
            warnings=warnings,
            suggestions=suggestions,
            validation_summary=summary
        )

    def _validate_single_parameter(self, param_name: str, param_value: Any) -> ParameterValidationResult:
        """
        Validate a single parameter value.
        
        Args:
            param_name: Name of the parameter
            param_value: Value to validate
            
        Returns:
            ParameterValidationResult with validation details
        """
        errors = []
        warnings = []
        suggested_value = None
        
        # Get validation rules for this parameter
        if param_name not in self.parameter_validation_rules:
            # Unknown parameter - create warning
            warnings.append(ValidationWarning(
                location=f"Device_Parameters.{param_name}",
                message=f"Unknown parameter '{param_name}'",
                suggestion="Verify parameter name is correct"
            ))
            return ParameterValidationResult(
                is_valid=True,  # Unknown parameters don't fail validation
                parameter_name=param_name,
                current_value=param_value,
                warnings=warnings
            )
        
        rules = self.parameter_validation_rules[param_name]
        param_type = rules['type']
        
        # Convert value to string for validation
        str_value = str(param_value) if param_value is not None else ""
        
        # Check if required parameter is empty
        if rules.get('required', False) and not str_value.strip():
            errors.append(ValidationError(
                error_type=ValidationErrorType.INVALID_PARAMETER,
                location=f"Device_Parameters.{param_name}",
                message=f"Required parameter '{param_name}' is empty",
                severity=ValidationSeverity.ERROR,
                suggestion="Provide a value for this required parameter",
                parameter_name=param_name,
                current_value=str_value
            ))
            return ParameterValidationResult(
                is_valid=False,
                parameter_name=param_name,
                current_value=param_value,
                errors=errors
            )
        
        # Skip validation for empty optional parameters
        if not str_value.strip() and not rules.get('required', False):
            return ParameterValidationResult(
                is_valid=True,
                parameter_name=param_name,
                current_value=param_value
            )
        
        # Type-specific validation
        if param_type == 'numeric':
            self._validate_numeric_parameter(param_name, str_value, rules, errors, warnings)
            if errors and 'min' in rules and 'max' in rules:
                suggested_value = str(max(rules['min'], min(rules['max'], rules.get('default', rules['min']))))
        
        elif param_type == 'text':
            self._validate_text_parameter(param_name, str_value, rules, errors, warnings)
        
        elif param_type == 'dropdown':
            self._validate_dropdown_parameter(param_name, str_value, rules, errors, warnings)
            if errors and 'allowed_values' in rules:
                suggested_value = rules['allowed_values'][0]
        
        elif param_type == 'radio':
            self._validate_radio_parameter(param_name, str_value, rules, errors, warnings)
            if errors and 'allowed_values' in rules:
                suggested_value = rules['allowed_values'][0]
        
        elif param_type in ['apn', 'apn_credential']:
            self._validate_apn_parameter(param_name, str_value, rules, errors, warnings)
        
        elif param_type == 'phone':
            self._validate_phone_parameter(param_name, str_value, rules, errors, warnings)
        
        elif param_type == 'domain':
            self._validate_domain_parameter(param_name, str_value, rules, errors, warnings)
        
        is_valid = len([e for e in errors if e.severity in [ValidationSeverity.CRITICAL, ValidationSeverity.ERROR]]) == 0
