    def _generate_server_command(self, server_params: Dict[str, Any], validate_commands: bool) -> Optional[str]:
        """Generate server command from flattened parameters.
        
        Be tolerant of different key names produced by various code paths:
        accept 'server_address'|'address'|'server_ip'|'server_domain' and
        'server_port'|'port'. Coerce port to integer and return None when required
        pieces are missing.
        """
        try:
            # Try multiple possible keys for address and port
            server_address = (server_params.get('server_address') or
                              server_params.get('address') or
                              server_params.get('server_ip') or
                              server_params.get('server_domain') or
                              '')
    
            server_port = (server_params.get('server_port') or
                           server_params.get('port') or
                           '')
    
            # Normalize and validate
            server_address = '' if server_address is None else str(server_address)
    
            if server_port is None or server_port == '':
                self.logger.warning("Both server address and port are required for server command")
                return None
    
            # Coerce port to integer where possible, fall back to 0
            try:
                server_port_int = int(float(server_port))
            except (ValueError, TypeError):
                self.logger.warning(f"Invalid server port value '{server_port}', defaulting to 0")
                server_port_int = 0
    
            if not server_address.strip():
                self.logger.warning("Server address is required for server command")
                return None
    
            return self.format_server_command(str(server_address), str(server_port_int))
    
        except Exception as e:
            self.logger.error(f"Error generating server command: {str(e)}")
            return None

"""
Device Communication Manager for the LT32 PRO Universal Profile System.

This module provides a unified interface for device communication that integrates
with existing device reader/writer services and uses parameter specifications
for command formatting and response parsing.

import asyncio
import time
from typing import Dict, List, Optional, Callable, Any, Tuple
from datetime import datetime

from models.parameter_spec import ParameterSpec, ParameterCommands

class DeviceCommunicationManager:
    """
    Manages device communication for reading and writing parameters using
    the Parameter UI List specifications.
    
    This class provides a unified interface that integrates with existing
    device services while using parameter specifications for proper command
    formatting and response parsing.

    def __init__(self, serial_service, spec_parser: ParameterSpecificationParser):
        """
        Initialize the device communication manager.
        
        Args:
            serial_service: The serial communication service instance
            spec_parser: Parameter specification parser instance
        """
        self.serial_service = serial_service
        self.spec_parser = spec_parser
        
        # Initialize device reader and writer
        self.device_reader = LT32ProDeviceReader(serial_service)
        self.device_writer = LT32ProDeviceWriter(serial_service)
        
        # Operation state
        self._operation_cancelled = False
        self._current_operation: Optional[str] = None
    
    async def read_all_parameters(self, progress_callback: Optional[Callable] = None) -> Dict[str, str]:
        """
        Read all parameters from the device based on parameter specifications.
        
        Args:
            progress_callback: Optional callback for progress updates
                              Called with (current_step, total_steps, step_description, success)
        
        Returns:
            Dict[str, str]: Dictionary mapping parameter names to raw device responses
            
        Raises:
            ConnectionError: If device is not connected
            RuntimeError: If operation fails
        """
        if not self.serial_service.is_connected:
            raise ConnectionError("Device is not connected")
        
        self._operation_cancelled = False
        self._current_operation = "read_all_parameters"
        
        try:
            # Get all parameter specifications
            all_specs = self.spec_parser.parse_specifications()
            
            # Filter parameters that have read commands
            readable_params = []
            for param_name, spec in all_specs.items():
                if self._has_read_command(spec):
                    readable_params.append((param_name, spec))
            
            if not readable_params:
                return {}
            
            total_params = len(readable_params)
            device_responses = {}
            
            # Read each parameter
            for i, (param_name, spec) in enumerate(readable_params):
                if self._operation_cancelled:
                    break
                
                # Update progress
                if progress_callback:
                    progress_callback(i, total_params, f"Reading {param_name}", None)
                
                try:
                    # Get the read command for this parameter
                    read_command = self._get_read_command(spec)
                    if read_command:
                        # Execute the read command
                        response = await self._execute_read_command(read_command, param_name)
                        if response:
                            device_responses[param_name] = response
                            
                            # Update progress with success
                            if progress_callback:
                                progress_callback(i + 1, total_params, 
                                                f"Read {param_name}: {response[:50]}...", True)
                        else:
                            # Update progress with failure
                            if progress_callback:
                                progress_callback(i + 1, total_params, 
                                                f"Failed to read {param_name}", False)
                    
                except Exception as e:
                    # Log error but continue with other parameters
                    if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                        self.serial_service.app.log_frame.log_message(
                            f"ERROR: Failed to read {param_name}: {str(e)}\n"
                        )
                    
                    # Update progress with failure
                    if progress_callback:
                        progress_callback(i + 1, total_params, 
                                        f"Error reading {param_name}: {str(e)}", False)
                
                # Small delay between commands
                if i < total_params - 1:
                    await asyncio.sleep(0.3)
            
            # Final progress update
            if progress_callback:
                success_count = len(device_responses)
                progress_callback(total_params, total_params, 
                                f"Completed: {success_count}/{total_params} parameters read", 
                                success_count > 0)
            
            return device_responses
            
        finally:
            self._current_operation = None
    
    async def write_all_parameters(self, parameters: Dict[str, Any], 
                                 progress_callback: Optional[Callable] = None) -> Dict[str, bool]:
        """
        Write all parameters to the device based on parameter specifications.
        
        Args:
            parameters: Dictionary mapping parameter names to values
            progress_callback: Optional callback for progress updates
                              Called with (current_step, total_steps, step_description, success)
        
        Returns:
            Dict[str, bool]: Dictionary mapping parameter names to success status
            
        Raises:
            ConnectionError: If device is not connected
            ValueError: If parameters contain invalid values
        """
        if not self.serial_service.is_connected:
            raise ConnectionError("Device is not connected")
        
        if not parameters:
            raise ValueError("No parameters provided for writing")
        
        self._operation_cancelled = False
        self._current_operation = "write_all_parameters"
        
        try:
            all_specs = self.spec_parser.parse_specifications()
            
            writable_params = []
            for param_name, value in parameters.items():
                spec = all_specs.get(param_name)
                if spec and self._has_write_command(spec):
                    validation_result = spec.validate_value(value)
                    if validation_result.is_valid:
                        writable_params.append((param_name, value, spec))
                    else:
                        if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                            error_msg = f"Validation failed for {param_name}: {', '.join(validation_result.errors)}"
                            self.serial_service.app.log_frame.log_message(f"ERROR: {error_msg}\n")

            if not writable_params:
                return {}

            total_params = len(writable_params)
            write_results = {}

            for i, (param_name, value, spec) in enumerate(writable_params):
                if self._operation_cancelled:
                    break

                if progress_callback:
                    progress_callback(i, total_params, f"Writing {param_name}", None)

                try:
                    if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                        try:
                            self.serial_service.app.log_frame.log_message(
                                f"DEBUG: Preparing to write parameter '{param_name}' with value: {repr(value)} and spec: {getattr(spec, 'name', 'unknown')}\n"
                            )
                        except Exception:
                            pass
                    else:
                        print(f"DEBUG: Preparing to write parameter '{param_name}' with value: {repr(value)} and spec: {getattr(spec, 'name', 'unknown')}")

                    success = await self._execute_write_command(spec, param_name, value)
                    write_results[param_name] = success

                    if progress_callback:
                        status_msg = "Success" if success else "Failed"
                        progress_callback(i + 1, total_params,
                                        f"{param_name}: {status_msg}", success)

                except Exception as e:
                    write_results[param_name] = False

                    if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                        self.serial_service.app.log_frame.log_message(
                            f"ERROR: Failed to write {param_name}: {str(e)}\n"
                        )

                    if progress_callback:
                        progress_callback(i + 1, total_params,
                                        f"Error writing {param_name}: {str(e)}", False)
                
                if i < total_params - 1:
                    await asyncio.sleep(0.5)

            if progress_callback:
                success_count = sum(1 for success in write_results.values() if success)
                progress_callback(total_params, total_params,
                                f"Completed: {success_count}/{total_params} parameters written",
                                success_count > 0)

            return write_results

        finally:
            self._current_operation = None
    
    def parse_device_response(self, command: str, response: str, param_spec: ParameterSpec) -> Optional[Dict[str, Any]]:
        """

    def format_parameter_for_device(self, param_spec: ParameterSpec, value: Any) -> Optional[str]:
        """

        Args:
            param_spec: Parameter specification
            value: Parameter value to format
            
        Returns:
            Optional[str]: Formatted device command or None if formatting fails
        """
        try:
            # Use DeviceCommandMapper for proper value mapping and command generation
            from services.device_command_mapper import DeviceCommandMapper
            mapper = DeviceCommandMapper()
            
            # Prepare the value in the format expected by DeviceCommandMapper
            if param_spec.fields and isinstance(value, dict):
                # Complex parameter with multiple fields
                # Check if value has nested structure with 'value' key (from profile files)
                if 'value' in value and isinstance(value['value'], dict):
                    # Extract the actual field values from the nested structure
                    prepared_value = value['value']
                else:
                    # Value is already in the correct format
                    prepared_value = value
            else:
                # Simple parameter - extract value if it's nested
                if isinstance(value, dict) and 'value' in value:
                    prepared_value = value['value']

                    prepared_value = value

            try:
                if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                    self.serial_service.app.log_frame.log_message(
                        f"DEBUG: Prepared value for {param_spec.name}: {repr(prepared_value)}\n"
                    )
                else:
                    print(f"DEBUG: Prepared value for {param_spec.name}: {repr(prepared_value)}")
            except Exception:
                pass
            
            # Generate command using DeviceCommandMapper
            # Keep the full command (including ATSMS: prefix) so the serial layer sends exactly what the mapper produced.
            full_command = mapper.get_command(param_spec.name, prepared_value, validate_command=False)
            if not full_command:

    def _parse_response_basic(self, response: str, param_spec: ParameterSpec) -> Dict[str, Any]:
        """
        Basic response parsing fallback.
        
        Args:
            response: Raw device response
            param_spec: Parameter specification
            
        Returns:
            Dict[str, Any]: Parsed values
        """
        # Remove common prefixes and suffixes
        clean_response = response.strip().rstrip('#')
        
        # Handle different response formats based on parameter type
        if param_spec.name == "apn_settings":
            # APN format: "APN:server,user,password"
            if clean_response.startswith("APN:"):
                parts = clean_response[4:].split(',')
                return {
                    "apn": parts[0] if len(parts) > 0 else "",
                    "user": parts[1] if len(parts) > 1 else "",
                    "password": parts[2] if len(parts) > 2 else ""
                }
        elif param_spec.name == "server_settings":
            # SERVER format: "SERVER:type,address,port,protocol"
            if clean_response.startswith("SERVER:"):
                parts = clean_response[7:].split(',')
                return {
                    "address": parts[1] if len(parts) > 1 else "",
                    "port": parts[2] if len(parts) > 2 else ""
                }
        elif param_spec.name == "protocol":
            # Protocol format: "GT06SEL=X"
            if "GT06SEL=" in clean_response:
                value = clean_response.split("=")[1]
                return {"protocol": value}
        elif param_spec.name == "tcp_ack":
            # TCP ACK format: "POSBYACK=X"
            if "POSBYACK=" in clean_response:
                value = clean_response.split("=")[1]
                return {"tcp_ack": value}
        elif param_spec.name == "odometer":
            # Odometer formats:
            # - "MILEAGE=XXXXX" (direct format)
            # - "Cur Mileage Value:XXXXKM" (verbose format)
            if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                self.serial_service.app.log_frame.log_message(
                    f"DEBUG: Parsing odometer response: '{clean_response}'\n"
                )
            
            if clean_response.startswith("MILEAGE="):
                value = clean_response[8:]  # Remove "MILEAGE=" prefix
                # Validate that it's numeric
                try:
                    int(value)
                    result = {"odometer": value}
                    if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                        self.serial_service.app.log_frame.log_message(
                            f"DEBUG: Parsed odometer from MILEAGE format: {result}\n"
                        )
                    return result
                except ValueError:
                    if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                        self.serial_service.app.log_frame.log_message(
                            f"DEBUG: Invalid odometer value in MILEAGE format: '{value}'\n"
                        )
                    return {"odometer": "0"}  # Default to 0 for invalid values
            elif "Cur Mileage Value:" in clean_response:
                # Extract value from "Cur Mileage Value:XXXXKM"
                parts = clean_response.split(":")
                if len(parts) >= 2:
                    value_part = parts[1]
                    # Remove "KM" suffix if present
                    value = value_part.replace("KM", "").strip()
                    # Validate that it's numeric
                    try:
                        int(value)
                        result = {"odometer": value}
                        if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                            self.serial_service.app.log_frame.log_message(
                                f"DEBUG: Parsed odometer from verbose format: {result}\n"
                            )
                        return result
                    except ValueError:
                        if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                            self.serial_service.app.log_frame.log_message(
                                f"DEBUG: Invalid odometer value in verbose format: '{value}'\n"
                            )
                        return {"odometer": "0"}  # Default to 0 for invalid values
                else:
                    if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                        self.serial_service.app.log_frame.log_message(
                            f"DEBUG: Malformed odometer response in verbose format: '{clean_response}'\n"
                        )
                    return {"odometer": "0"}  # Default to 0
            else:
                # Try to extract any numeric value
                import re
                numbers = re.findall(r'\d+', clean_response)
                if numbers:
                    result = {"odometer": numbers[-1]}  # Use last number found
                    if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                        self.serial_service.app.log_frame.log_message(
                            f"DEBUG: Parsed odometer using fallback extraction: {result}\n"
                        )
                    return result
                else:
                    if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                        self.serial_service.app.log_frame.log_message(
                            f"DEBUG: No numeric values found in odometer response: '{clean_response}'\n"
                        )
                    return {"odometer": "0"}  # Default to 0
        elif param_spec.name == "hourmeter":
            # Hourmeter formats:
            # - "ACCONT=XXXXX" (direct format)
            # - "Get ACC ON Time:XXXX Minute" (verbose format)
            if hasattr(self.serial_service, 'app') and hasattr(self.serial_service.app, 'log_frame'):
                self.serial_service.app.log_frame.log_message(
                    f"DEBUG: Parsing hourmeter response: '{clean_response}'\n"
                )
            
            if clean_response.startswith("ACCONT="):
                value = clean_response[7:]  # Remove "ACCONT=" prefix
                # Validate that it's numeric
                try:

"""
Enhanced Parameter Extractor for LT32Pro Device Responses

This module provides functionality to parse LT32Pro device responses and extract
individual parameters for profile file creation and validation.
"""

import re
from typing import Dict, Optional
import logging

logger = logging.getLogger(__name__)


class EnhancedParameterExtractor:
    """
    Extracts individual parameters from LT32Pro device responses.
    
    Converts device-specific response formats into standardized parameter
    dictionaries for profile file creation and validation.
    """
    
    # Protocol mapping for GT06SEL values
    PROTOCOL_MAPPING = {
        "0": "GT06",
        "1": "GT06N", 
        "2": "CRX",
        "3": "CONCOX",
        "4": "H02"
    }
    
    def __init__(self):
        """Initialize the parameter extractor."""
        self.logger = logging.getLogger(__name__)
    
    def extract_parameters_from_responses(self, device_responses: Dict[str, str]) -> Dict[str, str]:
        """
        Convert device responses to individual parameters with comprehensive error handling.
        
        Args:
            device_responses: Dictionary of command -> response mappings
            
        Returns:
            Dictionary of extracted parameters
            
        Requirements: 5.1, 5.2 - Log errors and continue processing other parameters
        """
        parameters = {}
        extraction_errors = []
        successful_extractions = 0
        
        try:
            # Process each response type with individual error handling
            for command, response in device_responses.items():
                try:
                    if not response:
                        self.logger.debug(f"Skipping empty response for command: {command}")
                        continue
                    
                    extracted_params = {}
                    
                    # Parse APN responses
                    if command.upper() in ['APN', 'ATSMS:APN#']:
                        extracted_params = self._safe_parse_apn_response(response, command)
                    
                    # Parse SERVER responses
                    elif command.upper() in ['SERVER', 'ATSMS:SERVER#']:
                        extracted_params = self._safe_parse_server_response(response, command)
                    
                    # Parse READOK responses (IMEI, ICCID, GT06SEL, POSBYACK)
                    elif 'READOK' in response.upper():
                        extracted_params = self._safe_parse_readok_response(response, command)
                    
                    # Parse STATUS responses
                    elif command.upper() in ['STATUS', 'ATSMS:STATUS#']:
                        extracted_params = self._safe_parse_status_response(response, command)
                    
                    # Parse VERSION responses
                    elif command.upper() in ['VERSION', 'ATSMS:VERSION#']:
                        extracted_params = self._safe_parse_version_response(response, command)
                    
                    # Parse HOURMETER/ACCONT responses
                    elif command.upper() in ['HOURMETER', 'ACCONT', 'ATSMS:ACCONT#']:
                        extracted_params = self._safe_parse_hourmeter_response(response, command)
                    
                    # Parse ODOMETER/MILEAGE responses
                    elif command.upper() in ['ODOMETER', 'MILEAGE', 'ATSMS:MILEAGE#']:
                        extracted_params = self._safe_parse_odometer_response(response, command)
                    
                    else:
                        self.logger.debug(f"No parser available for command: {command}")
                        continue
                    
                    # Update parameters with extracted values
                    if extracted_params:
                        parameters.update(extracted_params)
                        successful_extractions += 1
                        self.logger.debug(f"Successfully extracted {len(extracted_params)} parameters from {command}")
                    else:
                        self.logger.warning(f"No parameters extracted from {command} response")
                        
                except Exception as e:
                    error_msg = f"Failed to process {command} response: {str(e)}"
                    extraction_errors.append(error_msg)
                    self.logger.error(error_msg)
                    # Continue processing other responses (Requirement 5.2)
                    continue
            
            # Add default parameters with error handling
            try:
                parameters = self.add_default_parameters(parameters)
            except Exception as e:
                error_msg = f"Failed to add default parameters: {str(e)}"
                extraction_errors.append(error_msg)
                self.logger.error(error_msg)
            
            # Log extraction summary
            total_responses = len(device_responses)
            if extraction_errors:
                self.logger.warning(f"Parameter extraction completed with {len(extraction_errors)} errors. "
                                  f"Successfully processed {successful_extractions}/{total_responses} responses. "
                                  f"Extracted {len(parameters)} total parameters.")
            else:
                self.logger.info(f"Successfully extracted {len(parameters)} parameters from {total_responses} device responses")
            
            return parameters
            
        except Exception as e:
            # Handle catastrophic failure
            error_msg = f"Critical error during parameter extraction: {str(e)}"
            self.logger.error(error_msg)
            extraction_errors.append(error_msg)
            
            # Return whatever parameters we managed to extract
            return parameters
    
    def _parse_apn_response(self, response: str) -> Dict[str, str]:
        """
        Parse APN response format: APN:server,user,password
        
        Args:
            response: APN response string
            
        Returns:
            Dictionary with apn_server, apn_user, apn_password
        """
        parameters = {}
        
        try:
            # Match APN:server,user,password format
            apn_match = re.search(r'APN:([^,]*),([^,]*),([^,\r\n]*)', response, re.IGNORECASE)
            
            if apn_match:
                parameters['apn_server'] = apn_match.group(1).strip()
                parameters['apn_user'] = apn_match.group(2).strip()
                parameters['apn_password'] = apn_match.group(3).strip()
                
                self.logger.debug(f"Parsed APN: server={parameters['apn_server']}, "
                                f"user={parameters['apn_user']}, password=***")
            else:
                self.logger.warning(f"Could not parse APN response: {response}")
                
        except Exception as e:
            self.logger.error(f"Error parsing APN response '{response}': {e}")
            
        return parameters
    
    def _parse_server_response(self, response: str) -> Dict[str, str]:
        """
        Parse SERVER response format: SERVER:0,IP,PORT,0
        
        Args:
            response: SERVER response string
            
        Returns:
            Dictionary with server_ip, server_port, server_type
        """
        parameters = {}
        
        try:
            # Match SERVER:0,IP,PORT,0 format
            server_match = re.search(r'SERVER:(\d+),([^,]+),(\d+),(\d+)', response, re.IGNORECASE)
            
            if server_match:
                parameters['server_type'] = server_match.group(1).strip()
                parameters['server_ip'] = server_match.group(2).strip()
                parameters['server_port'] = server_match.group(3).strip()
                parameters['server_protocol'] = server_match.group(4).strip()
                
                self.logger.debug(f"Parsed SERVER: ip={parameters['server_ip']}, "
                                f"port={parameters['server_port']}, type={parameters['server_type']}")
            else:
                self.logger.warning(f"Could not parse SERVER response: {response}")
                
        except Exception as e:
            self.logger.error(f"Error parsing SERVER response '{response}': {e}")
            
        return parameters
    
    def _parse_readok_response(self, response: str) -> Dict[str, str]:
        """
        Parse READOK response formats:
        - READOK: IMEI=value
        - READOK: ICCID=value  
        - READOK: GT06SEL=value
        - READOK: POSBYACK=value
        
        Args:
            response: READOK response string
            
        Returns:
            Dictionary with extracted parameters
        """
        parameters = {}
        
        try:
            # Parse IMEI
            imei_match = re.search(r'READOK:\s*IMEI=([^\r\n]+)', response, re.IGNORECASE)
            if imei_match:
                imei_value = imei_match.group(1).strip()
                parameters['device_imei'] = imei_value
                parameters['device_id'] = imei_value
                parameters['imei'] = imei_value
                self.logger.debug(f"Parsed IMEI: {imei_value}")
            
            # Parse ICCID
            iccid_match = re.search(r'READOK:\s*ICCID=([^\r\n]+)', response, re.IGNORECASE)
            if iccid_match:
                iccid_value = iccid_match.group(1).strip()
                parameters['device_iccid'] = iccid_value
                parameters['iccid'] = iccid_value
                self.logger.debug(f"Parsed ICCID: {iccid_value}")
            
            # Parse GT06SEL (protocol)
            protocol_match = re.search(r'READOK:\s*GT06SEL=([^\r\n]+)', response, re.IGNORECASE)
            if protocol_match:
                protocol_value = protocol_match.group(1).strip()
                parameters['protocol'] = protocol_value
                parameters['protocol_name'] = self.PROTOCOL_MAPPING.get(protocol_value, f"Unknown({protocol_value})")
                self.logger.debug(f"Parsed Protocol: {protocol_value} -> {parameters['protocol_name']}")
            
            # Parse POSBYACK (TCP acknowledgment)
            posbyack_match = re.search(r'READOK:\s*POSBYACK=([^\r\n]+)', response, re.IGNORECASE)
            if posbyack_match:
                posbyack_value = posbyack_match.group(1).strip()
                parameters['tcp_ack'] = posbyack_value
                parameters['tcp_acknowledgment'] = posbyack_value
                self.logger.debug(f"Parsed POSBYACK: {posbyack_value}")
                
        except Exception as e:
            self.logger.error(f"Error parsing READOK response '{response}': {e}")
            
        return parameters
    
    def _parse_status_response(self, response: str) -> Dict[str, str]:

            gprs_match = re.search(r'GPRS:([^;]+)', response, re.IGNORECASE)

            # Parse GPS status
            gps_match = re.search(r'GPS:([^;]+)', response, re.IGNORECASE)

            # Parse hourmeter from "Get ACC ON Time:value Minute" format in status

            if cur_mileage_match:
                parameters['current_mileage'] = cur_mileage_match.group(1)

        except Exception as e:

    def _parse_hourmeter_response(self, response: str, command: str = '') -> Dict[str, str]:
        """
        Parse HOURMETER/ACCONT response formats.
        
        Handles various formats:
        - "ACCONT=5678"
        - "Get ACC ON Time:5678 Minute"
        - "Engine Hours: 1234 hours"
        
        Args:
            response: Hourmeter response string
            command: Command that generated this response
            
        Returns:
            Dictionary with hourmeter parameters
        """
        parameters = {}
        
        try:
            # Pattern to match ACCONT=value format
            accont_match = re.search(r'ACCONT[=:](\d+)', response, re.IGNORECASE)
            if accont_match:
                # Use different parameter names based on the command to avoid conflicts
                if command.upper() == 'ACCONT':
                    param_name = 'accont'
                else:
                    param_name = 'hourmeter'
                
                parameters[param_name] = accont_match.group(1)
                self.logger.debug(f"Extracted {param_name} from ACCONT format: {accont_match.group(1)}")
                return parameters
            
            # Pattern to match "Get ACC ON Time:value Minute" format
            acc_on_time_match = re.search(r'Get\s+ACC\s+ON\s+Time[=:](\d+)', response, re.IGNORECASE)
            if acc_on_time_match:
                parameters['acc_on_time'] = acc_on_time_match.group(1)

                return parameters

                self.logger.debug(f"Extracted hourmeter (fallback): {numeric_match.group(1)}")

    def handle_error(self, error_type: str, severity: str, context: Dict[str, Any] = None) -> None:
        """
        Handle an error with appropriate user feedback.
        
        Args:
            error_type: The type of error (from class constants)
            severity: The severity level (from class constants)
            context: Optional context dictionary with error details
        """
        if context is None:
            context = {}
            
        # Update error tracking
        current_time = datetime.datetime.now()
        self.last_error_time[error_type] = current_time
        self.error_count[error_type] = self.error_count.get(error_type, 0) + 1
        
        # Get the error message template
        message_template = self.error_messages.get(error_type, "Unknown error: {error_message}")
        
        # Format the error message with context
        try:
            error_message = message_template.format(**context)

            error_message = message_template.format(error_message=str(context))
        
        # Log the error with appropriate prefix based on severity
        if severity == self.SEVERITY_ERROR or severity == self.SEVERITY_CRITICAL:
            self.app.log_frame.log_message(f"ERROR: {error_message}\n")
            
            # For critical errors, show a dialog
            if severity == self.SEVERITY_CRITICAL:
                self.app.after(0, lambda: messagebox.showerror("Critical Error", error_message))
        elif severity == self.SEVERITY_WARNING:
            self.app.log_frame.log_message(f"WARNING: {error_message}\n")
        else:
            self.app.log_frame.log_message(f"INFO: {error_message}\n")
        
        # Handle connection errors specially
        if error_type.startswith("conn_error_"):
            self._handle_connection_error(error_type, context)
            
            # Add resolution suggestions for connection errors
            if error_type in self.error_resolutions:
                suggestions = self.error_resolutions[error_type]
                self.app.log_frame.log_message(f"INFO: Suggested solutions:\n")

        # For unexpected disconnections or timeouts, attempt to reconnect if enabled

            self.CONN_ERROR_DEVICE_DISCONNECTED,

    def _attempt_reconnection(self, port: Optional[str]) -> None:
        """
        Attempt to reconnect to the device.
        
        Args:
            port: The COM port to reconnect to
        """
        # Don't start a new reconnection thread if one is already running
        if self.reconnect_thread and self.reconnect_thread.is_alive():
            return
            
        # Reset reconnection state
        self.stop_reconnect = False
        self.reconnect_attempts = 0
        
        # Log the start of reconnection attempts
        self.app.log_frame.log_message(
            f"INFO: Starting automatic reconnection to {port}...\n"
        )
        
        # Start reconnection in a separate thread
        self.reconnect_thread = threading.Thread(
            target=self._reconnection_worker,
            args=(port,),
            daemon=True
        )
        self.reconnect_thread.start()
        
        # Update UI to show reconnection in progress
        if hasattr(self.app, 'connection_frame'):
            self.app.connection_frame.show_reconnecting_status()
    
    def _reconnection_worker(self, port: Optional[str]) -> None:
        """
        Worker thread for reconnection attempts.
        
        Args:
            port: The COM port to reconnect to
        """
        if not port or not hasattr(self.app, 'serial_service'):
            return
            
        while (not self.stop_reconnect and 
               self.reconnect_attempts < self.max_reconnect_attempts):
            
            self.reconnect_attempts += 1
            
            # Calculate backoff delay (increases with each attempt)
            current_delay = self.reconnect_delay * self.reconnect_attempts
            
            # Log reconnection attempt with progress bar
            self.app.log_frame.log_message(
                f"INFO: Reconnection attempt {self.reconnect_attempts}/{self.max_reconnect_attempts} "
                f"[{'=' * self.reconnect_attempts}{' ' * (self.max_reconnect_attempts - self.reconnect_attempts)}]\n"
            )
            
            # Update UI to show current reconnection attempt
            if hasattr(self.app, 'connection_frame'):
                self.app.after(0, self.app.connection_frame.update_reconnection_status, 
                              self.reconnect_attempts, self.max_reconnect_attempts)
            
            # Attempt to reconnect
            try:
                self.app.serial_service.connect(port)
                
                # Check if connection was successful
                if self.app.serial_service.is_connected:
                    self.app.log_frame.log_message(f"INFO: Ô£ô Reconnection successful\n")
                    
                    # Update connection quality
                    self._update_connection_quality(20)  # Successful reconnection improves quality
                    
                    # Reset reconnection UI
                    if hasattr(self.app, 'connection_frame'):
                        self.app.after(0, self.app.connection_frame.hide_reconnecting_status)
                    
                    return
            except Exception as e:
                # Log reconnection failure with details
                error_type, context = self.parse_error_from_exception(e)
                error_message = self.error_messages.get(error_type, str(e))
                
                try:
                    formatted_error = error_message.format(**context)
                except (KeyError, ValueError):
                    formatted_error = str(e)
                
                self.app.log_frame.log_message(f"ERROR: Reconnection failed: {formatted_error}\n")
            
            # Wait before next attempt with backoff
            for i in range(int(current_delay)):
                if self.stop_reconnect:
                    break
                time.sleep(1)
                
                # Update countdown in UI
                remaining = current_delay - i - 1
                if remaining > 0 and hasattr(self.app, 'connection_frame'):
                    self.app.after(0, self.app.connection_frame.update_reconnection_countdown, remaining)
        
        # Log if all reconnection attempts failed
        if self.reconnect_attempts >= self.max_reconnect_attempts:
            self.app.log_frame.log_message(
                f"ERROR: Failed to reconnect after {self.max_reconnect_attempts} attempts\n"
            )
            
            # Show reconnection failure in UI
            if hasattr(self.app, 'connection_frame'):
                self.app.after(0, self.app.connection_frame.show_reconnection_failed)
                
            # Suggest manual reconnection
            self.app.log_frame.log_message(
                f"INFO: Please check your device connection and try reconnecting manually\n"
            )
    
    def stop_reconnection(self) -> None:
        """Stop any ongoing reconnection attempts."""
        self.stop_reconnect = True
        if self.reconnect_thread and self.reconnect_thread.is_alive():
            self.reconnect_thread.join(timeout=1)
            
        # Reset reconnection UI
        if hasattr(self.app, 'connection_frame'):
            self.app.connection_frame.hide_reconnecting_status()
            
        self.app.log_frame.log_message("INFO: Automatic reconnection stopped\n")

import json
import logging
import os
from utils.helpers import resource_path
logger = logging.getLogger(__name__)
from services.app_config_service import AppConfigService

class I18nService:
    """Handles internationalization and language switching."""

    DEFAULT_LANGUAGE = "pt_BR"
    
    def __init__(self, language=None):
        self.language_map = {}
        self.current_language = None
        
        # Use the centralized app config service
        self.app_config = AppConfigService()
        
        # Try to load saved language preference
        if language is None:
            try:
                saved_language = self.app_config.get_language()
                print(f"Loaded saved language preference: {saved_language}")
                language = saved_language if saved_language else self.DEFAULT_LANGUAGE
            except Exception as e:
                print(f"Error loading language preference: {e}")
                language = self.DEFAULT_LANGUAGE
                
        print(f"Initializing I18nService with language: {language}")
        self.set_language(language)

    def set_language(self, language_code):
        """Loads a new language file from the assets/locales directory."""
        locale_path = resource_path(os.path.join('assets', 'locales', f'{language_code}.json'))
        try:
            with open(locale_path, 'r', encoding='utf-8') as f:
                self.language_map = json.load(f)
                self.current_language = language_code
                print(f"Loaded language: {language_code}")
                
                # Save the language preference
                self.app_config.set_language(language_code)
        except FileNotFoundError:
            print(f"Error: Language file not found for code: {language_code}")
            # Fallback to a default or handle the error appropriately
            if not self.language_map: # if no language is loaded at all
                self.language_map = {"error": "Language file missing"}

    def get(self, key, default_value=''):
        """Returns the translated string for a given key."""
        if key in self.language_map:
            value = self.language_map[key]
        else:
            fallback = default_value or f"_{key}_"
            logger.warning(
                "Missing translation key '%s' for language '%s'. Falling back to '%s'.",
                key,
                self.current_language,
                fallback
            )
            value = fallback
        # If the locale file contains escaped newline sequences (e.g. "\\n")
        # convert them to real newlines so dialogs and labels render correctly.
        try:
            if isinstance(value, str):
                value = value.replace('\\n', '\n').replace('\\r', '\r')
        except Exception:
            # If any unexpected type or issue occurs, fall back to the raw value
            pass
        return value
    
    def get_current_language(self):
        """Returns the current language code."""
        return self.current_language

        formatted_command = f"ATSMS:{base_command}"
        
        return formatted_command
    
    def format_write_command(self, command: str, value: str) -> str:
        """
        Format write commands with ATSMS prefix for USB communication.
        """
        if not command:
            raise ValueError("Command cannot be empty")
        if not value:
            raise ValueError("Value cannot be empty")
            
        command = command.upper().strip()
        value = value.strip()
        
        if command not in self.WRITE_COMMAND_MAPPINGS:
            raise ValueError(f"Unsupported write command: {command}")

        # Special handling for JAMMER: when state is OFF, do not include the extra mode/field
        if command == 'JAMMER':
            # Accept values like 'OFF', 'OFF,0', 'ON,1' etc.
            parts = [p.strip() for p in value.split(',') if p is not None]
            if len(parts) == 0:
                raise ValueError("JAMMER command requires at least a state value")
            state = parts[0].upper()
            if state == 'OFF':
                return f"ATSMS:JAMMER,OFF#"
            # For non-OFF states, preserve original formatting (include remaining parts if any)
            rest = ','.join(parts[1:]) if len(parts) > 1 else ''
            base = f"JAMMER,{state}"
            if rest:
                base = f"{base},{rest}"
            base = f"{base}#"
            return f"ATSMS:{base}"

        # Special handling for SPEED: when state is OFF, do not include trailing numeric fields
        if command == 'SPEED':
            # Accept values like 'OFF', 'OFF,15,0,0', 'ON,80,1,0' etc.
            parts = [p.strip() for p in value.split(',') if p is not None]
            if len(parts) == 0:
                raise ValueError("SPEED command requires at least a state value")
            state = parts[0].upper()
            if state == 'OFF':
                return f"ATSMS:SPEED,OFF#"
            # For non-OFF states, preserve original formatting and append '#'
            rest = ','.join(parts[1:]) if len(parts) > 1 else ''
            base = f"SPEED,{state}"
            if rest:
                base = f"{base},{rest}"
            base = f"{base}#"
            return f"ATSMS:{base}"

        self._validate_write_command_value(command, value)

        # For specific commands, if the value contains spaces, enclose it in quotes.
        if command in self.COMMANDS_REQUIRING_QUOTES and ' ' in value:
            value = f'"{value}"'

        base_command = self.WRITE_COMMAND_MAPPINGS[command].format(value=value)
        return f"ATSMS:{base_command}"
    
    def _validate_write_command_value(self, command: str, value: str) -> None:
        """
        Validate command values according to LT32Pro specifications.
        """
        if command == 'GT06SEL':
            if value not in self.VALID_PROTOCOLS:
                raise ValueError(f"Invalid protocol value: {value}. Must be one of {self.VALID_PROTOCOLS}")
                
        elif command == 'POSBYACK':
            if value not in self.VALID_TCP_ACK:
                raise ValueError(f"Invalid TCP ACK value: {value}. Must be one of {self.VALID_TCP_ACK}")
                
        elif command == 'APN':
            parts = value.split(',')
            if len(parts) != 3:
                raise ValueError("APN value must be in format: server,user,password")
            if not parts[0]:
                raise ValueError("APN server cannot be empty")
                
        elif command == 'SERVER':
            parts = value.split(',')
            if len(parts) != 4:
                raise ValueError("SERVER value must be in format: type,ip,port,protocol")
            
            try:
                server_type = int(parts[0])
                if server_type not in [0, 1]:
                    raise ValueError("Server type must be 0 (IP) or 1 (domain)")
                
                port = int(parts[2])
                if not (1 <= port <= 65535):
                    raise ValueError("Port must be between 1 and 65535")
                
                protocol = int(parts[3])
                if protocol != 0:
                    raise ValueError("Protocol must be 0 (TCP)")
                    
            except ValueError as e:
                if "invalid literal" in str(e):
                    raise ValueError("SERVER numeric values must be valid integers")
                raise
    
    def validate_command_syntax(self, formatted_command: str) -> bool:
        """
        Validate that a formatted command follows proper LT32Pro syntax.
        """
        if not formatted_command:
            raise ValueError("Command cannot be empty")
            
        if not formatted_command.startswith("ATSMS:"):
            raise ValueError("USB commands must start with 'ATSMS:' prefix")
            
        command_body = formatted_command[6:]
        
        valid_patterns = [
            r'^[A-Z]+#$',                    # Simple commands like APN#, STATUS#
            r'^[A-Z]+,.+#$',                 # Commands with values like APN,value#
            r'^SZCS#[A-Z0-9]+$',            # Read commands like SZCS#IMEI, SZCS#GT06SEL
            r'^SZCS#[A-Z0-9]+=.+$',         # Set commands like SZCS#GT06SEL=2
            r'^CXCS#[A-Z0-9_]+$',           # Legacy read commands like CXCS#IMEI, CXCS#GT06SEL, CXCS#SPD_LED_SHOW
            r'^CXCS#[A-Z0-9_]+#[A-Z0-9_]+$' # Multi-part legacy commands like CXCS#ACCLINE#ACC_CFG
        ]
        
        for pattern in valid_patterns:
            if re.match(pattern, command_body):
                return True
                
        raise ValueError(f"Invalid command syntax: {formatted_command}")
    
    def get_supported_read_commands(self) -> List[str]:
        """
        Get list of supported read commands.
        """
        return list(self.READ_COMMAND_MAPPINGS.keys())
    
    def get_supported_write_commands(self) -> List[str]:
        """
        Get list of supported write commands.
        """
        return list(self.WRITE_COMMAND_MAPPINGS.keys())
    
    def is_read_command_supported(self, command: str) -> bool:
        """
        Check if a read command is supported.
        """
        return command.upper().strip() in self.READ_COMMAND_MAPPINGS
    
    def is_write_command_supported(self, command: str) -> bool:
        """
        Check if a write command is supported.
        """
        return command.upper().strip() in self.WRITE_COMMAND_MAPPINGS
    
    def get_alternative_read_commands(self, command: str) -> List[str]:
        """
        Get alternative read commands for devices that don't support CXCS# commands.
        """
        command = command.upper().strip()
        return self.ALTERNATIVE_READ_COMMANDS.get(command, [])
    
    def has_alternative_commands(self, command: str) -> bool:
        """
        Check if a command has alternative formats available.
        """
        command = command.upper().strip()
        return command in self.ALTERNATIVE_READ_COMMANDS

"""
LT32Pro Error Handler

Provides device-specific error handling with recovery strategies for LT32Pro communication.
Handles connection errors, communication timeouts, protocol errors, and validation issues.
"""

import time
import logging
from typing import Dict, Optional, Tuple, List
from enum import Enum


class ErrorCategory(Enum):
    """Categories of errors that can occur during LT32Pro operations."""
    CONNECTION = "connection"
    COMMUNICATION = "communication"
    PROTOCOL = "protocol"
    VALIDATION = "validation"
    FILE = "file"


class LT32ProError(Exception):
    """Base exception for LT32Pro specific errors."""
    
    def __init__(self, message: str, category: ErrorCategory, details: Optional[Dict] = None):
        super().__init__(message)
        self.category = category
        self.details = details or {}


class ConnectionError(LT32ProError):
    """Error related to device connection issues."""
    
    def __init__(self, message: str, details: Optional[Dict] = None):
        super().__init__(message, ErrorCategory.CONNECTION, details)


class CommunicationError(LT32ProError):
    """Error related to device communication issues."""
    
    def __init__(self, message: str, details: Optional[Dict] = None):
        super().__init__(message, ErrorCategory.COMMUNICATION, details)


class ProtocolError(LT32ProError):
    """Error related to protocol or command format issues."""
    
    def __init__(self, message: str, details: Optional[Dict] = None):
        super().__init__(message, ErrorCategory.PROTOCOL, details)


class ValidationError(LT32ProError):
    """Error related to parameter validation issues."""
    
    def __init__(self, message: str, details: Optional[Dict] = None):
        super().__init__(message, ErrorCategory.VALIDATION, details)


class FileError(LT32ProError):
    """Error related to profile file operations."""
    
    def __init__(self, message: str, details: Optional[Dict] = None):
        super().__init__(message, ErrorCategory.FILE, details)


class LT32ProErrorHandler:
    """
    Handles LT32Pro specific errors with recovery strategies.
    
    Provides error categorization, retry logic with exponential backoff,
    partial data recovery, and user-friendly error messages.
    """
    
    def __init__(self, max_retries: int = 3, base_delay: float = 1.0):
        """
        Initialize error handler.
        
        Args:
            max_retries: Maximum number of retry attempts
            base_delay: Base delay for exponential backoff (seconds)
        """
        self.max_retries = max_retries
        self.base_delay = base_delay
        self.logger = logging.getLogger(__name__)
        
        # Device error codes and their meanings
        self.device_error_codes = {
            "ERROR": "General command error",
            "TIMEOUT": "Command timeout",
            "INVALID": "Invalid command format",
            "BUSY": "Device is busy processing another command",
            "NODATA": "No data available",
            "FAIL": "Command execution failed"
        }
    
    def handle_communication_timeout(self, command: str, attempt: int) -> bool:
        """
        Handle command timeout with exponential backoff retry.
        
        Args:
            command: The command that timed out
            attempt: Current attempt number (0-based)
            
        Returns:
            True if should retry, False if max retries exceeded
        """
        if attempt >= self.max_retries:
            self.logger.error(f"Command '{command}' failed after {self.max_retries} attempts")
            return False
        
        # Calculate exponential backoff delay
        delay = self.base_delay * (2 ** attempt)
        
        self.logger.warning(
            f"Command '{command}' timed out (attempt {attempt + 1}/{self.max_retries + 1}). "
            f"Retrying in {delay:.1f} seconds..."
        )
        
        time.sleep(delay)
        return True
    
    def handle_invalid_response(self, command: str, response: str) -> Tuple[str, Dict[str, str]]:
        """
        Handle invalid device responses with partial data recovery.
        
        Args:
            command: The command that generated the response
            response: The invalid response received
            
        Returns:
            Tuple of (cleaned_response, recovered_data)
        """
        self.logger.warning(f"Invalid response for command '{command}': {response}")
        
        recovered_data = {}
        cleaned_response = response.strip()
        
        # Try to extract partial data from malformed responses
        if ":" in cleaned_response:
            try:
                # Handle responses like "APN:server,user,pass" even if malformed
                if cleaned_response.startswith("APN:"):
                    parts = cleaned_response.split(":", 1)[1].split(",")
                    if len(parts) >= 1:
                        recovered_data["apn_server"] = parts[0].strip()
                    if len(parts) >= 2:
                        recovered_data["apn_user"] = parts[1].strip()
                    if len(parts) >= 3:
                        recovered_data["apn_password"] = parts[2].strip()
                
                # Handle SERVER responses
                elif cleaned_response.startswith("SERVER:"):
                    parts = cleaned_response.split(":", 1)[1].split(",")
                    if len(parts) >= 2:
                        recovered_data["server_ip"] = parts[1].strip()
                    if len(parts) >= 3:
                        recovered_data["server_port"] = parts[2].strip()
                
                # Handle READOK responses
                elif "READOK:" in cleaned_response and "=" in cleaned_response:
                    # Extract key=value pairs
                    value_part = cleaned_response.split("READOK:")[-1].strip()
                    if "=" in value_part:
                        key, value = value_part.split("=", 1)
                        recovered_data[key.strip().lower()] = value.strip()
                
            except Exception as e:
                self.logger.error(f"Failed to recover partial data from response: {e}")
        
        return cleaned_response, recovered_data
    
    def handle_device_error_response(self, command: str, error_response: str) -> Tuple[bool, str]:
        """
        Handle device error responses with user-friendly messages.
        
        Args:
            command: The command that generated the error
            error_response: The error response from device
            
        Returns:
            Tuple of (should_retry, user_friendly_message)
        """
        error_response = error_response.strip().upper()
        
        # Check for known error codes
        for error_code, description in self.device_error_codes.items():
            if error_code in error_response:
                user_message = f"Device error for command '{command}': {description}"
                
                # Determine if we should retry based on error type
                should_retry = error_code in ["TIMEOUT", "BUSY"]
                
                self.logger.error(f"Device error: {error_response} - {description}")
                return should_retry, user_message
        
        # Handle unknown error responses
        user_message = f"Unknown device error for command '{command}': {error_response}"
        self.logger.error(f"Unknown device error: {error_response}")
        
        # Default to not retrying for unknown errors
        return False, user_message
    
    def handle_connection_error(self, port: str, error: Exception) -> str:
        """
        Handle connection errors with specific troubleshooting guidance.
        
        Args:
            port: The COM port that failed to connect
            error: The connection error exception
            
        Returns:
            User-friendly troubleshooting message
        """
        error_str = str(error).lower()
        
        if "access is denied" in error_str or "permission denied" in error_str:
            return (
                f"Cannot access COM port {port}. "
                f"Troubleshooting steps:\n"
                f"1. Close any other applications using the port\n"
                f"2. Check if device is properly connected\n"
                f"3. Try unplugging and reconnecting the device\n"
                f"4. Restart the application with administrator privileges"
            )
        
        elif "file not found" in error_str or "no such file" in error_str:
            return (
                f"COM port {port} not found. "
                f"Troubleshooting steps:\n"
                f"1. Check Device Manager for available COM ports\n"
                f"2. Verify the LT32Pro device is connected via USB\n"
                f"3. Install or update the device drivers\n"
                f"4. Try a different USB cable or port"
            )
        
        elif "timeout" in error_str:
            return (
                f"Connection to {port} timed out. "
                f"Troubleshooting steps:\n"
                f"1. Check if the device is powered on\n"
                f"2. Verify the correct COM port is selected\n"
                f"3. Try reducing the baud rate\n"
                f"4. Check USB cable connection"
            )
        
        else:
            return (
                f"Failed to connect to {port}: {error}. "
                f"General troubleshooting steps:\n"
                f"1. Check device connection and power\n"
                f"2. Verify COM port settings\n"
                f"3. Close other applications using the port\n"
                f"4. Restart the device and try again"
            )
    
    def categorize_error(self, error: Exception) -> ErrorCategory:
        """
        Categorize an error based on its type and message.
        
        Args:
            error: The exception to categorize
            
        Returns:
            The error category
        """
        if isinstance(error, LT32ProError):
            return error.category
        
        error_str = str(error).lower()
        
        # Connection-related errors
        if any(keyword in error_str for keyword in [
            "connection", "port", "access denied", "permission denied", 
            "file not found", "no such file", "device not found"
        ]):
            return ErrorCategory.CONNECTION
        
        # Communication-related errors
        elif any(keyword in error_str for keyword in [
            "timeout", "read", "write", "serial", "communication"
        ]):
            return ErrorCategory.COMMUNICATION
        
        # Protocol-related errors
        elif any(keyword in error_str for keyword in [
            "protocol", "command", "format", "syntax", "invalid response"
        ]):
            return ErrorCategory.PROTOCOL
        
        # Validation-related errors
        elif any(keyword in error_str for keyword in [
            "validation", "invalid", "missing", "required", "format"
        ]):
            return ErrorCategory.VALIDATION
        
        # File-related errors
        elif any(keyword in error_str for keyword in [
            "file", "directory", "path"
        ]) and not any(keyword in error_str for keyword in [
            "access denied", "permission denied", "port", "connection"
        ]):
            return ErrorCategory.FILE
        
        # Default to communication error
        return ErrorCategory.COMMUNICATION
    
    def get_recovery_suggestions(self, category: ErrorCategory, error: Exception) -> List[str]:
        """
        Get recovery suggestions based on error category.
        
        Args:
            category: The error category
            error: The original exception
            
        Returns:
            List of recovery suggestions
        """
        suggestions = []
        
        if category == ErrorCategory.CONNECTION:
            suggestions.extend([
                "Check device connection and power",
                "Verify correct COM port selection",
                "Close other applications using the port",
                "Try unplugging and reconnecting the device",
                "Check Device Manager for port availability"
            ])
        
        elif category == ErrorCategory.COMMUNICATION:
            suggestions.extend([
                "Check device power and connection stability",
                "Verify baud rate and communication settings",
                "Try reducing communication speed",
                "Check for electromagnetic interference",
                "Restart the device and try again"
            ])
        
        elif category == ErrorCategory.PROTOCOL:
            suggestions.extend([
                "Verify device firmware version compatibility",
                "Check command format and syntax",
                "Ensure proper ATSMS prefix for USB communication",
                "Try basic commands first (STATUS, VERSION)",
                "Consult device manual for command reference"
            ])
        
        elif category == ErrorCategory.VALIDATION:
            suggestions.extend([
                "Check parameter values and formats",
                "Verify required parameters are present",
                "Validate IP addresses and port numbers",
                "Check profile file format and structure",
                "Use default values for optional parameters"
            ])
        
        elif category == ErrorCategory.FILE:
            suggestions.extend([
                "Check file permissions and access rights",
                "Verify file path exists and is accessible",
                "Ensure sufficient disk space",
                "Check file format and encoding",
                "Try saving to a different location"
            ])
        
        return suggestions
    
    def create_error_report(self, error: Exception, context: Dict[str, str] = None) -> Dict[str, any]:
        """
        Create a comprehensive error report.
        
        Args:
            error: The exception that occurred
            context: Additional context information
            
        Returns:
            Dictionary containing error report details
        """
        category = self.categorize_error(error)
        suggestions = self.get_recovery_suggestions(category, error)
        
        report = {
            "error_type": type(error).__name__,
            "error_message": str(error),
            "category": category.value,
            "timestamp": time.strftime("%Y-%m-%d %H:%M:%S"),
            "suggestions": suggestions,
            "context": context or {}
        }
        
        if isinstance(error, LT32ProError):
            report["details"] = error.details
        
        return report
