            # Parse bullet points (handle both * and indented content)
            if line.startswith('*') or (in_fields_section and line.startswith('    *')) or (in_commands_section and line.startswith('    *')):
                # Check for main sections
                if '**fields:**' in line:
                    in_fields_section = True
                    in_commands_section = False
                    config_data['fields'] = []
                    current_field = None
                    current_field_data = {}
                elif '**commands:**' in line:
                    in_commands_section = True
                    in_fields_section = False
                    config_data['commands'] = {}
                    # Save any pending field
                    if current_field and current_field_data:
                        config_data['fields'].append(current_field_data.copy())
                        current_field = None
                        current_field_data = {}
                elif '**field_' in line and line.strip().endswith(':**') and in_fields_section:
                    # Save previous field if exists
                    if current_field and current_field_data:
                        config_data['fields'].append(current_field_data.copy())
                    
                    # Start new field
                    current_field = self._extract_field_number(line)
                    current_field_data = {}
                elif ('**set' in line or '**read' in line) and in_commands_section:

                    command_type, command_value = self._parse_command(line)
                    if command_type and command_value:

                    if '**name:**' in line:
                        current_field_data['name'] = self._extract_value(line)
                    elif '**label:**' in line:
                        current_field_data['label'] = self._extract_value(line)
                    elif '**ui_element:**' in line:

                    elif '**validation:**' in line:
                        current_field_data['validation'] = self._extract_value(line)
                    elif '**unit:**' in line:
                        current_field_data['unit'] = self._extract_value(line)
                    elif '**options:**' in line:
                        current_field_data['options'] = self._parse_options(line)
                    elif '**visibility_condition:**' in line:
                        current_field_data['visibility_condition'] = self._extract_value(line)
                elif not in_fields_section and not in_commands_section:
                    # Handle main config properties (not in fields or commands section)
                    if '**name:**' in line:
                        config_data['name'] = self._extract_value(line)
                    elif '**label:**' in line:
                        config_data['label'] = self._extract_value(line)
                    elif '**description:**' in line:

                    elif '**visibility_condition:**' in line:
                        config_data['visibility_condition'] = self._extract_value(line)

        # Don't forget the last field
        if current_field and current_field_data and in_fields_section:
            if 'fields' not in config_data:
                config_data['fields'] = []
            config_data['fields'].append(current_field_data)

    def _extract_value(self, line: str) -> str:
        """Extract value from a markdown line."""
        # Remove markdown formatting and extract the value
        line = re.sub(r'\*\*[^*]+:\*\*', '', line)  # Remove **key:** part
        line = line.strip('* ').strip()
        # Remove quotes if present
        if line.startswith('"') and line.endswith('"'):
            line = line[1:-1]
        return line
    
    def _parse_options(self, line: str) -> List[str]:

        # Extract the array part

        if match:

            # Split by comma and clean up
            options = [opt.strip().strip('"') for opt in options_str.split(',')]
            return options
        return []

    def _parse_command(self, line: str) -> tuple[Optional[str], Optional[str]]:
        """Parse command from a markdown line."""

"""
Profile Converter - Format standardization and compatibility for profile files.

This module provides format conversion and standardization for profile files including:
1. Converting raw profile data to standard ProfileData format
2. Legacy profile format migration utilities
3. Parameter name mapping and value format conversion
4. Integration with existing ProfileData structure
5. Backward compatibility with different profile versions

# Import existing data models
from services.profile_data_models import ProfileData, ProfileMetadata, ValidationInfo


class ProfileFormatVersion(Enum):
    """Supported profile format versions."""
    LEGACY_V0_9 = "0.9"
    CURRENT_V1_0 = "1.0"
    CURRENT_V1_1 = "1.1"
    FUTURE_V1_2 = "1.2"


class ConversionErrorType(Enum):
    """Types of conversion errors."""
    UNSUPPORTED_VERSION = "unsupported_version"
    MISSING_REQUIRED_DATA = "missing_required_data"
    INVALID_PARAMETER_FORMAT = "invalid_parameter_format"
    PARAMETER_MAPPING_ERROR = "parameter_mapping_error"
    VALUE_CONVERSION_ERROR = "value_conversion_error"


@dataclass
class ConversionError:
    """Conversion error information."""
    error_type: ConversionErrorType
    parameter_name: Optional[str]
    original_value: Optional[str]
    error_message: str
    suggestion: Optional[str] = None


@dataclass
class ConversionResult:
    """Result of profile conversion operation."""
    success: bool
    profile_data: Optional[ProfileData] = None
    errors: List[ConversionError] = field(default_factory=list)

    migration_log: List[str] = field(default_factory=list)
    errors: List[ConversionError] = field(default_factory=list)


class ProfileConverter:
    """
    Format standardization and compatibility for profile files.
    
    Features:
    - Converting raw profile data to standard ProfileData format
    - Legacy profile format migration utilities
    - Parameter name mapping and value format conversion
    - Integration with existing ProfileData structure
    """
    
    def __init__(self, logger: Optional[logging.Logger] = None):
        """
        Initialize the profile converter.
        
        Args:
            logger: Optional logger instance for detailed logging
        """
        self.logger = logger or logging.getLogger(__name__)
        self.current_version = ProfileFormatVersion.CURRENT_V1_0.value
        
        # Initialize parameter mappings
        self._init_parameter_mappings()
        
        # Initialize value converters
        self._init_value_converters()
        
        # Initialize legacy format handlers
        self._init_legacy_format_handlers()
        
        # Initialize section mappings
        self._init_section_mappings()

    def convert_to_standard_format(self, raw_data: Dict[str, Any]) -> ConversionResult:
        """
        Convert raw profile data to standard ProfileData format.
        
        Args:
            raw_data: Raw profile data dictionary (could be from ConfigParser or other sources)
            
        Returns:
            ConversionResult with standardized ProfileData or conversion errors
        """
        self.logger.info("Converting raw data to standard format")
        
        errors = []
        warnings = []
        parameters_converted = 0
        parameters_failed = 0
        
        try:
            # Detect format version
            format_version = self._detect_format_version(raw_data)
            self.logger.debug(f"Detected format version: {format_version}")
            
            # Extract metadata
            metadata_result = self._extract_and_convert_metadata(raw_data, format_version)
            if not metadata_result['success']:
                errors.extend(metadata_result['errors'])
                if not metadata_result['metadata']:
                    return ConversionResult(
                        success=False,
                        errors=errors,
                        conversion_summary="Failed to extract profile metadata"
                    )
            
            metadata = metadata_result['metadata']
            warnings.extend(metadata_result.get('warnings', []))
            
            # Extract and convert device parameters
            parameters_result = self._extract_and_convert_parameters(raw_data, format_version)
            device_parameters = parameters_result['parameters']
            errors.extend(parameters_result['errors'])
            warnings.extend(parameters_result.get('warnings', []))
            parameters_converted = parameters_result['converted_count']
            parameters_failed = parameters_result['failed_count']
            
            # Create validation info
            validation_info = ValidationInfo(
                is_validated=True,
                validation_date=datetime.now().isoformat(),
                validation_errors=[]
            )
            
            # Create standardized ProfileData
            profile_data = ProfileData(
                metadata=metadata,
                device_parameters=device_parameters,
                validation_info=validation_info,
                format_version=self.current_version
            )
            
            # Update parameter count in metadata
            profile_data.metadata.parameter_count = len(device_parameters)
            
            success = len([e for e in errors if e.error_type in [
                ConversionErrorType.UNSUPPORTED_VERSION,
                ConversionErrorType.MISSING_REQUIRED_DATA
            ]]) == 0
            
            conversion_summary = self._generate_conversion_summary(
                parameters_converted, parameters_failed, len(warnings), format_version
            )
            
            self.logger.info(f"Conversion completed: {conversion_summary}")
            
            return ConversionResult(
                success=success,
                profile_data=profile_data,
                errors=errors,
                warnings=warnings,
                conversion_summary=conversion_summary,
                parameters_converted=parameters_converted,
                parameters_failed=parameters_failed
            )
            
        except Exception as e:
            error_msg = f"Unexpected error during conversion: {str(e)}"
            self.logger.error(error_msg, exc_info=True)
            
            return ConversionResult(
                success=False,
                errors=[ConversionError(
                    error_type=ConversionErrorType.INVALID_PARAMETER_FORMAT,
                    parameter_name=None,
                    original_value=None,
                    error_message=error_msg
                )],
                conversion_summary="Conversion failed due to unexpected error"
            )

    def migrate_legacy_format(self, legacy_data: Dict[str, Any], version: str) -> MigrationResult:
        """
        Migrate legacy profile format to current standard format.
        
        Args:
            legacy_data: Legacy profile data
            version: Version of the legacy format
            
        Returns:
            MigrationResult with migrated ProfileData or migration errors
        """
        self.logger.info(f"Migrating legacy format from version {version} to {self.current_version}")
        
        migration_log = []
        errors = []
        
        try:
            # Check if version is supported for migration
            if version not in self.legacy_format_handlers:
                error = ConversionError(
                    error_type=ConversionErrorType.UNSUPPORTED_VERSION,
                    parameter_name=None,
                    original_value=version,
                    error_message=f"Unsupported legacy version: {version}",
                    suggestion=f"Supported versions: {', '.join(self.legacy_format_handlers.keys())}"
                )
                return MigrationResult(
                    success=False,
                    original_version=version,
                    target_version=self.current_version,
                    errors=[error]
                )
            
            # Get migration handler for the version
            migration_handler = self.legacy_format_handlers[version]
            migration_log.append(f"Starting migration from version {version}")
            
            # Apply version-specific migrations
            migrated_data = migration_handler(legacy_data, migration_log)
            
            # Convert migrated data to standard format
            conversion_result = self.convert_to_standard_format(migrated_data)
            
            if not conversion_result.success:
                errors.extend(conversion_result.errors)
                migration_log.append("Migration failed during format conversion")
                return MigrationResult(
                    success=False,
                    original_version=version,
                    target_version=self.current_version,
                    migration_log=migration_log,
                    errors=errors
                )
            
            migration_log.append(f"Migration completed successfully: {conversion_result.parameters_converted} parameters converted")
            
            return MigrationResult(
                success=True,
                original_version=version,
                target_version=self.current_version,
                migrated_data=conversion_result.profile_data,
                migration_log=migration_log,
                errors=conversion_result.errors
            )
            
        except Exception as e:
            error_msg = f"Migration failed: {str(e)}"
            self.logger.error(error_msg, exc_info=True)
            migration_log.append(error_msg)
            
            return MigrationResult(
                success=False,
                original_version=version,
                target_version=self.current_version,
                migration_log=migration_log,
                errors=[ConversionError(
                    error_type=ConversionErrorType.INVALID_PARAMETER_FORMAT,
                    parameter_name=None,
                    original_value=None,
                    error_message=error_msg
                )]
            )

    def handle_error(self, error: Exception, operation_context: str,
                    parameter_name: Optional[str] = None,
                    attempt_number: int = 0) -> ErrorHandlingResult:
        """
        Handle an error with appropriate categorization and response.
        
        Args:
            error: The exception that occurred
            operation_context: Context of the operation (e.g., "load_from_device")
            parameter_name: Name of parameter being processed (if applicable)
            attempt_number: Current attempt number (0-based)
            
        Returns:
            ErrorHandlingResult with handling decision and user messages
        """
        category = self._categorize_error(error)
        severity = self._determine_severity(error, category)
        
        # Create context key for retry tracking
        context_key = f"{operation_context}:{parameter_name or 'general'}"
        
        # Determine if we should retry
        should_retry = self._should_retry_error(error, category, attempt_number)
        retry_delay = self._calculate_retry_delay(attempt_number) if should_retry else 0.0
        
        # Generate user-friendly messages
        user_message = self._create_user_message(error, category, parameter_name)
        technical_message = self._create_technical_message(error, operation_context, parameter_name)
        suggestions = self._get_resolution_suggestions(error, category, parameter_name)
        
        # Log the error
        self._log_error(error, category, severity, operation_context, parameter_name, attempt_number)
        
        # Update retry count
        if should_retry:
            self.retry_counts[context_key] = attempt_number + 1
        else:
            self.retry_counts.pop(context_key, None)
        
        return ErrorHandlingResult(
            should_retry=should_retry,
            retry_delay=retry_delay,
            user_message=user_message,
            technical_message=technical_message,
            suggestions=suggestions,
            category=category,
            severity=severity,
            context={
                "operation": operation_context,
                "parameter": parameter_name,
                "attempt": attempt_number,
                "error_type": type(error).__name__
            }
        )

    def _categorize_error(self, error: Exception) -> ProfileErrorCategory:
        """Categorize an error based on its type and message."""
        if isinstance(error, ProfileError):
            return error.category
        
        error_str = str(error).lower()
        error_type = type(error).__name__.lower()
        
        # Device connection errors
        if any(keyword in error_str for keyword in [
            "connection", "port", "access denied", "permission denied", 
            "device not found", "no such file or directory"
        ]) and any(keyword in error_str for keyword in ["com", "serial", "usb"]):
            return ProfileErrorCategory.DEVICE_CONNECTION
        
        # Device communication errors
        elif any(keyword in error_str for keyword in [
            "timeout", "read", "write", "serial", "communication", "busy", "device"
        ]):
            return ProfileErrorCategory.DEVICE_COMMUNICATION
        
        # Parameter validation errors
        elif any(keyword in error_str for keyword in [
            "validation", "invalid", "format", "range", "required"
        ]) or "validation" in error_type:
            return ProfileErrorCategory.PARAMETER_VALIDATION
        
        # Profile file errors
        elif any(keyword in error_str for keyword in [
            "file", "directory", "path", "json", "encoding"
        ]):
            return ProfileErrorCategory.PROFILE_FILE
        
        # Parameter specification errors
        elif any(keyword in error_str for keyword in [
            "specification", "parsing", "markdown"
        ]) and not any(keyword in error_str for keyword in ["validation", "invalid", "format"]):
            return ProfileErrorCategory.PARAMETER_SPECIFICATION
        
        # Operation cancelled
        elif any(keyword in error_str for keyword in [
            "cancelled", "canceled", "interrupted", "stopped"
        ]):
            return ProfileErrorCategory.OPERATION_CANCELLED
        
        # Default to system error
        return ProfileErrorCategory.SYSTEM_ERROR   
 
    def _determine_severity(self, error: Exception, category: ProfileErrorCategory) -> ProfileErrorSeverity:
        """Determine the severity of an error."""
        if isinstance(error, ProfileError):
            return error.severity
        
        # Critical errors that prevent operation
        if category in [ProfileErrorCategory.DEVICE_CONNECTION, ProfileErrorCategory.SYSTEM_ERROR]:
            return ProfileErrorSeverity.CRITICAL
        
        # Errors that affect operation but may be recoverable
        elif category in [ProfileErrorCategory.DEVICE_COMMUNICATION, ProfileErrorCategory.PROFILE_FILE]:
            return ProfileErrorSeverity.ERROR
        
        # Validation errors are typically user-correctable
        elif category == ProfileErrorCategory.PARAMETER_VALIDATION:
            return ProfileErrorSeverity.ERROR  # Changed from WARNING to ERROR
        
        # Default to error
        return ProfileErrorSeverity.ERROR
    
    def _should_retry_error(self, error: Exception, category: ProfileErrorCategory, 
                           attempt: int) -> bool:
        """Determine if an error should be retried."""
        # Don't retry if we've exceeded max attempts
        if attempt >= self.max_retries:
            return False
        
        # Don't retry validation or file errors
        if category in [ProfileErrorCategory.PARAMETER_VALIDATION, 
                       ProfileErrorCategory.PROFILE_FILE,
                       ProfileErrorCategory.PARAMETER_SPECIFICATION]:
            return False
        
        # Don't retry cancelled operations
        if category == ProfileErrorCategory.OPERATION_CANCELLED:
            return False
        
        # Retry communication errors
        if category == ProfileErrorCategory.DEVICE_COMMUNICATION:
            error_str = str(error).lower()
            # Retry timeouts and busy errors
            return any(keyword in error_str for keyword in ["timeout", "busy", "temporary"])
        
        # Don't retry connection errors by default (they need manual intervention)
        if category == ProfileErrorCategory.DEVICE_CONNECTION:
            return False
        
        # Default to not retrying
        return False
    
    def _calculate_retry_delay(self, attempt: int) -> float:
        """Calculate retry delay using exponential backoff."""
        delay = self.base_delay * (2 ** attempt)
        return min(delay, self.max_delay)

                    "count": 0,
                    "errors": [],

            error_severity = self._determine_severity(error, category)
            if error_severity.value == "critical":

            elif error_severity.value == "error" and categories[category_name]["severity"] != "critical":

            elif error_severity.value == "warning" and categories[category_name]["severity"] not in ["critical", "error"]:
                categories[category_name]["severity"] = "warning"
            
            # Collect suggestions
            suggestions = self._get_resolution_suggestions(error, category)
            all_suggestions.update(suggestions)
        
        return {
            "total_errors": len(errors),
            "categories": categories,
            "suggestions": list(all_suggestions),
            "operation_context": operation_context
        }

    def handle_error(self, error_type: ErrorType, message: str, 
                    context: Optional[ProfileOperationContext] = None,
                    details: str = "", exception: Optional[Exception] = None) -> bool:
        """
        Handle a profile operation error.
        
        Args:
            error_type: Type of error that occurred
            message: Human-readable error message
            context: Operation context information
            details: Additional error details
            exception: Original exception if available
            
        Returns:
            True if operation should continue, False if it should stop
        """
        # Determine error category
        category = self.error_categories.get(error_type, ErrorCategory.CRITICAL)
        
        # Create error object
        error = ProfileError(
            error_type=error_type,
            category=category,
            message=message,
            details=details,
            context=context,
            exception=exception,
            suggested_actions=self.error_suggestions.get(error_type, []),
            recovery_possible=category != ErrorCategory.CRITICAL
        )
        
        # Track the error
        self._track_error(error)
        
        # Handle based on category
        if category == ErrorCategory.CRITICAL:
            return self._handle_critical_error(error)
        elif category == ErrorCategory.NON_CRITICAL:
            return self._handle_non_critical_error(error)
        elif category == ErrorCategory.USER_CONFIRMATION:
            return self._handle_user_confirmation_error(error)
        
        return False
    
    def _track_error(self, error: ProfileError):
        """Track an error for the operation."""
        with self._error_lock:
            operation_id = error.context.operation_id if error.context else "unknown"
            
            if operation_id not in self._operation_errors:
                self._operation_errors[operation_id] = []
            
            self._operation_errors[operation_id].append(error)
    
    def _handle_critical_error(self, error: ProfileError) -> bool:
        """
        Handle a critical error that stops the operation.
        
        Args:
            error: The critical error
            
        Returns:
            False (operation should stop)
        """
        # Log the critical error
        self.logger.error(f"CRITICAL ERROR: {error.message}")
        if error.details:
            self.logger.error(f"Details: {error.details}")
        if error.exception:
            self.logger.error(f"Exception: {str(error.exception)}")
            self.logger.error(f"Traceback: {traceback.format_exc()}")
        
        # Show error dialog
        self._show_critical_error_dialog(error)
        
        return False
    
    def _handle_non_critical_error(self, error: ProfileError) -> bool:
        """
        Handle a non-critical error by logging it.
        
        Args:
            error: The non-critical error
            
        Returns:
            True (operation should continue)
        """
        # Log the non-critical error
        self.logger.warning(f"NON-CRITICAL ERROR: {error.message}")
        if error.details:
            self.logger.warning(f"Details: {error.details}")
        
        # Log suggested actions

            self.logger.info(f"Suggested actions: {', '.join(error.suggested_actions)}")
        
        return True
    
    def _handle_user_confirmation_error(self, error: ProfileError) -> bool:
        """
        Handle an error that requires user confirmation.
        
        Args:
            error: The error requiring confirmation
            
        Returns:
            True if user chooses to continue, False otherwise

        # Log the error
        self.logger.warning(f"USER CONFIRMATION REQUIRED: {error.message}")
        if error.details:
            self.logger.warning(f"Details: {error.details}")
        
        # Check confirmation cache to avoid duplicate dialogs
        cache_key = f"{error.error_type.value}_{error.message}"
        if cache_key in self._confirmation_cache:
            return self._confirmation_cache[cache_key]
        
        # Show confirmation dialog
        result = self._show_confirmation_dialog(error)
        
        # Cache the result
        self._confirmation_cache[cache_key] = result
        
        return result

    def save_profile_to_file(self, profile: UniversalProfile, file_path: str,

            create_backup: Whether to create a backup if file exists

            # Update profile metadata before saving
            profile.modified_at = datetime.now()
            profile.metadata.profile_format_version = self.CURRENT_FORMAT_VERSION
            
            # Convert profile to dictionary for saving (without runtime metadata)
            profile_data = profile.to_dict(for_save=True)
            
            # Add file metadata
            file_metadata = {
                "file_format_version": self.CURRENT_FORMAT_VERSION,
                "saved_at": datetime.now().isoformat(),
                "saved_by": "LT32 PRO Universal Profile System"
            }
            profile_data["_file_metadata"] = file_metadata
            
            # Write to temporary file first for atomic operation
            temp_file_path = file_path + ".tmp"
            
            with open(temp_file_path, 'w', encoding='utf-8') as file:
                json.dump(profile_data, file, indent=2, ensure_ascii=False)
            
            # Verify the written file
            verification_result = self._verify_saved_file(temp_file_path, profile)
            if not verification_result.success:
                os.remove(temp_file_path)
                return OperationResult(
                    operation_type="save_profile",
                    success=False,
                    error_message=f"File verification failed: {verification_result.error_message}",
                    duration_seconds=(datetime.now() - operation_start).total_seconds()
                )
            
            # Move temporary file to final location (atomic operation)
            shutil.move(temp_file_path, file_path)
            
            # Log successful save
            self.logger.info(f"Profile '{profile.name}' saved successfully to {file_path}")
            
            return OperationResult(
                operation_type="save_profile",
                success=True,
                message=f"Profile saved to {file_path}",
                duration_seconds=(datetime.now() - operation_start).total_seconds(),
                details={
                    "file_path": file_path,
                    "profile_name": profile.name,
                    "parameter_count": len(profile.parameters),
                    "file_size_bytes": os.path.getsize(file_path)
                }
            )
            
        except PermissionError as e:
            error_msg = f"Permission denied writing to {file_path}: {str(e)}"
            self.logger.error(error_msg)
            return OperationResult(
                operation_type="save_profile",
                success=False,
                error_message=error_msg,
                duration_seconds=(datetime.now() - operation_start).total_seconds()
            )
            
        except OSError as e:
            error_msg = f"OS error writing file {file_path}: {str(e)}"
            self.logger.error(error_msg)
            return OperationResult(
                operation_type="save_profile",
                success=False,
                error_message=error_msg,
                duration_seconds=(datetime.now() - operation_start).total_seconds()
            )
            
        except Exception as e:
            error_msg = f"Unexpected error saving profile: {str(e)}"
            self.logger.error(error_msg)
            return OperationResult(
                operation_type="save_profile",
                success=False,
                error_message=error_msg,
                duration_seconds=(datetime.now() - operation_start).total_seconds()
            )
    
    def load_profile_from_file(self, file_path: str, 
                             auto_migrate: bool = True) -> Tuple[Optional[UniversalProfile], OperationResult]:
        """
        Load a universal profile from a JSON file.
        
        Args:
            file_path: Path to the profile file to load
            auto_migrate: Whether to automatically migrate older format versions
            
        Returns:
            Tuple of (loaded profile or None, operation result)
        """
        operation_start = datetime.now()
        
        try:
            # Check if file exists
            if not os.path.exists(file_path):
                error_msg = f"Profile file not found: {file_path}"
                return None, OperationResult(
                    operation_type="load_profile",
                    success=False,
                    error_message=error_msg,
                    duration_seconds=(datetime.now() - operation_start).total_seconds()
                )
            
            # Check file permissions
            if not os.access(file_path, os.R_OK):
                error_msg = f"Profile file is not readable: {file_path}"
                return None, OperationResult(
                    operation_type="load_profile",
                    success=False,
                    error_message=error_msg,
                    duration_seconds=(datetime.now() - operation_start).total_seconds()
                )
            
            # Check file size
            file_size = os.path.getsize(file_path)
            if file_size == 0:
                error_msg = "Profile file is empty"
                return None, OperationResult(
                    operation_type="load_profile",
                    success=False,
                    error_message=error_msg,
                    duration_seconds=(datetime.now() - operation_start).total_seconds()
                )
            
            # Read and parse JSON file
            with open(file_path, 'r', encoding='utf-8') as file:
                profile_data = json.load(file)

            # Normalize parameter values that were produced by device dumps and may contain
            # Python-style stringified dicts (single quotes) or literal values.
            # This makes the loader tolerant of the two sample files you provided.
            try:
                import ast
                import json as _json
                if isinstance(profile_data, dict) and "parameters" in profile_data and isinstance(profile_data["parameters"], dict):
                    for p_name, p_data in profile_data["parameters"].items():
                        # Expect param entry to be a dict with a "value" key in most cases
                        if not isinstance(p_data, dict):
                            continue
                        if "value" not in p_data:
                            continue
                        val = p_data["value"]
                        # Only attempt parsing for string values that look like structured data or literals
                        if isinstance(val, str):
                            v = val.strip()
                            parsed = None
                            try:
                                # Use ast.literal_eval first ÔÇö safe for Python-style dicts/literals
                                if (v.startswith("{") and v.endswith("}")) or (v.startswith("[") and v.endswith("]")) or v in ("True", "False", "None") or v.replace('.', '', 1).isdigit():
                                    parsed = ast.literal_eval(v)
                            except Exception:
                                parsed = None
                            if parsed is None:
                                # Try a JSON-safe fallback by converting single quotes to double quotes
                                try:
                                    if (v.startswith("{") and v.endswith("}")) or (v.startswith("[") and v.endswith("]")):
                                        parsed = _json.loads(v.replace("'", "\""))
                                except Exception:
                                    parsed = None
                            if parsed is not None:
                                p_data["value"] = parsed

                        # Targeted normalization for known problematic device dump structures
                        # - server_settings: ensure port is an integer and address is clean
                        # - tcp_ack and other single-field toggle params: unwrap dicts like {'tcp_ack': 'Disabled'} to 'Disabled'
                        try:
                            # Handle server_settings entries which may appear as 'network_parameters.server_settings' or similar
                            if isinstance(p_data.get("value"), dict) and p_name.endswith("server_settings"):
                                sv = p_data["value"]
                                # Normalize port
                                if "port" in sv:
                                    port_val = sv["port"]
                                    if isinstance(port_val, str):
                                        port_str = port_val.strip()
                                        if port_str.isdigit():
                                            sv["port"] = int(port_str)
                                        else:
                                            # try extracting digits
                                            import re
                                            digits = re.findall(r'\d+', port_str)
                                            sv["port"] = int(digits[-1]) if digits else 0
                                    elif isinstance(port_val, (int, float)):
                                        sv["port"] = int(port_val)
                                # Clean address string
                                if "address" in sv and isinstance(sv["address"], str):
                                    sv["address"] = sv["address"].strip()
                                p_data["value"] = sv

                            # Unwrap single-key dict values for single-value parameters like tcp_ack
                            if isinstance(p_data.get("value"), dict) and len(p_data["value"]) == 1:
                                sole_key = next(iter(p_data["value"]))
                                sole_val = p_data["value"][sole_key]
                                # If key matches the parameter short name, unwrap it
                                short_name = p_name.split('.')[-1]
                                if sole_key == short_name:
                                    # Map boolean-like strings to normalized labels (trim)
                                    if isinstance(sole_val, str):
                                        sv = sole_val.strip()
                                        # Accept 'True'/'False' or '1'/'0' or 'Enabled'/'Disabled'
                                        if sv.lower() in ("true", "1"):
                                            p_data["value"] = True
                                        elif sv.lower() in ("false", "0"):
                                            p_data["value"] = False
                                        else:
                                            p_data["value"] = sv
                                    else:
                                        p_data["value"] = sole_val
                        except Exception:
                            # Ignore normalization errors per-profile entry
                            pass
            except Exception:
                # Keep original data if normalization fails for any reason
                pass
            
            # Validate file format
            validation_result = self._validate_profile_data(profile_data)
            if not validation_result.is_valid:
                error_msg = f"Invalid profile format: {'; '.join(validation_result.errors)}"
                return None, OperationResult(
                    operation_type="load_profile",
                    success=False,
                    error_message=error_msg,
                    duration_seconds=(datetime.now() - operation_start).total_seconds(),
                    details={"validation_errors": validation_result.errors}
                )
            
            # Check format version and migrate if needed
            file_format_version = self._get_format_version(profile_data)
            migration_performed = False
            
            if file_format_version != self.CURRENT_FORMAT_VERSION:
                if auto_migrate:
                    migration_result = self._migrate_profile_data(profile_data, file_format_version)
                    if migration_result.success:
                        profile_data = migration_result.details["migrated_data"]
                        migration_performed = True
                        self.logger.info(f"Profile migrated from version {file_format_version} to {self.CURRENT_FORMAT_VERSION}")
                    else:
                        error_msg = f"Profile migration failed: {migration_result.error_message}"
                        return None, OperationResult(
                            operation_type="load_profile",
                            success=False,
                            error_message=error_msg,
                            duration_seconds=(datetime.now() - operation_start).total_seconds()
                        )
                else:
                    self.logger.warning(f"Profile format version {file_format_version} differs from current {self.CURRENT_FORMAT_VERSION}")
            
            # Create profile from data
            profile = UniversalProfile.from_dict(profile_data)
            
            # Log successful load
            self.logger.info(f"Profile '{profile.name}' loaded successfully from {file_path}")
            
            return profile, OperationResult(
                operation_type="load_profile",
                success=True,
                message=f"Profile loaded from {file_path}",
                duration_seconds=(datetime.now() - operation_start).total_seconds(),
                details={
                    "file_path": file_path,
                    "profile_name": profile.name,
                    "parameter_count": len(profile.parameters),
                    "file_size_bytes": file_size,
                    "migration_performed": migration_performed,
                    "format_version": file_format_version
                }
            )
            
        except json.JSONDecodeError as e:
            error_msg = f"Invalid JSON format in profile file: {str(e)}"
            self.logger.error(error_msg)
            return None, OperationResult(
                operation_type="load_profile",
                success=False,
                error_message=error_msg,
                duration_seconds=(datetime.now() - operation_start).total_seconds()
            )
            
        except UnicodeDecodeError as e:
            error_msg = f"File encoding error: {str(e)}"
            self.logger.error(error_msg)
            return None, OperationResult(
                operation_type="load_profile",
                success=False,
                error_message=error_msg,
                duration_seconds=(datetime.now() - operation_start).total_seconds()
            )
            
        except Exception as e:
            error_msg = f"Unexpected error loading profile: {str(e)}"
            self.logger.error(error_msg)
            return None, OperationResult(
                operation_type="load_profile",
                success=False,
                error_message=error_msg,
                duration_seconds=(datetime.now() - operation_start).total_seconds()
            )
    
    def validate_profile_file(self, file_path: str) -> ValidationResult:
        """
        Validate a profile file without loading it completely.
        
        Args:
            file_path: Path to the profile file to validate
            
        Returns:
            ValidationResult: Validation result with detailed information
        """
        result = ValidationResult(is_valid=True)
        
        try:
            # Check file existence and permissions
            if not os.path.exists(file_path):
                result.add_error(f"Profile file not found: {file_path}")
                return result
            
            if not os.access(file_path, os.R_OK):
                result.add_error(f"Profile file is not readable: {file_path}")
                return result
            
            # Check file size
            file_size = os.path.getsize(file_path)
            if file_size == 0:
                result.add_error("Profile file is empty")
                return result
            
            if file_size > 10 * 1024 * 1024:  # 10MB limit
                result.add_warning(f"Profile file is very large: {file_size} bytes")
            
            # Validate JSON format
            try:
                with open(file_path, 'r', encoding='utf-8') as file:
                    profile_data = json.load(file)
            except json.JSONDecodeError as e:
                result.add_error(f"Invalid JSON format: {str(e)}")
                return result
            except UnicodeDecodeError as e:
                result.add_error(f"File encoding error: {str(e)}")
                return result
            
            # Validate profile data structure
            data_validation = self._validate_profile_data(profile_data)
            result.errors.extend(data_validation.errors)
            result.warnings.extend(data_validation.warnings)
            
            # Check format version
            format_version = self._get_format_version(profile_data)
            if format_version not in self.SUPPORTED_VERSIONS:
                result.add_error(f"Unsupported profile format version: {format_version}")
            elif format_version != self.CURRENT_FORMAT_VERSION:
                result.add_warning(f"Profile format version {format_version} is older than current {self.CURRENT_FORMAT_VERSION}")
            
            # Validate required fields
            required_fields = ["name", "version", "created_at", "modified_at", "metadata", "parameters"]
            for field in required_fields:
                if field not in profile_data:
                    result.add_error(f"Missing required field: {field}")
            
            # Validate metadata structure
            if "metadata" in profile_data:
                metadata = profile_data["metadata"]
                if not isinstance(metadata, dict):
                    result.add_error("Metadata must be a dictionary")
                else:
                    required_metadata_fields = ["parameter_spec_version", "device_model", "profile_format_version"]
                    for field in required_metadata_fields:
                        if field not in metadata:
                            result.add_warning(f"Missing metadata field: {field}")
            
            # Validate parameters structure
            if "parameters" in profile_data:
                parameters = profile_data["parameters"]
                if not isinstance(parameters, dict):
                    result.add_error("Parameters must be a dictionary")
                else:
                    # Validate parameter structure
                    for param_name, param_data in parameters.items():
                        if not isinstance(param_data, dict):
                            result.add_error(f"Parameter '{param_name}' data must be a dictionary")
                            continue
                        
                        required_param_fields = ["value", "source", "timestamp"]
                        for field in required_param_fields:
                            if field not in param_data:
                                result.add_warning(f"Parameter '{param_name}' missing field: {field}")
            
        except Exception as e:
            result.add_error(f"Unexpected error during validation: {str(e)}")
        
        return result 
   
    def migrate_profile_file(self, file_path: str, backup: bool = True) -> OperationResult:
        """
        Migrate a profile file to the current format version.
        
        Args:
            file_path: Path to the profile file to migrate
            backup: Whether to create a backup before migration
            
        Returns:
            OperationResult: Result of the migration operation
        """
        operation_start = datetime.now()
        
        try:
            # Load the profile
            profile, load_result = self.load_profile_from_file(file_path, auto_migrate=False)
            if not load_result.success:
                return OperationResult(
                    operation_type="migrate_profile",
                    success=False,
                    error_message=f"Failed to load profile for migration: {load_result.error_message}",
                    duration_seconds=(datetime.now() - operation_start).total_seconds()
                )
            
            # Check if migration is needed
            with open(file_path, 'r', encoding='utf-8') as file:
                profile_data = json.load(file)
            
            current_version = self._get_format_version(profile_data)
            if current_version == self.CURRENT_FORMAT_VERSION:
                return OperationResult(
                    operation_type="migrate_profile",
                    success=True,
                    message="Profile is already at current format version",
                    duration_seconds=(datetime.now() - operation_start).total_seconds(),
                    details={"format_version": current_version}
                )
            
            # Create backup if requested
            if backup:
                backup_result = self._create_backup(file_path)
                if not backup_result.success:
                    self.logger.warning(f"Backup creation failed: {backup_result.error_message}")
            
            # Perform migration
            migration_result = self._migrate_profile_data(profile_data, current_version)
            if not migration_result.success:
                return OperationResult(
                    operation_type="migrate_profile",
                    success=False,
                    error_message=f"Migration failed: {migration_result.error_message}",
                    duration_seconds=(datetime.now() - operation_start).total_seconds()
                )
            
            # Create migrated profile
            migrated_data = migration_result.details["migrated_data"]
            migrated_profile = UniversalProfile.from_dict(migrated_data)
            
            # Save migrated profile
            save_result = self.save_profile_to_file(migrated_profile, file_path, create_backup=False)
            if not save_result.success:
                return OperationResult(
                    operation_type="migrate_profile",
                    success=False,
                    error_message=f"Failed to save migrated profile: {save_result.error_message}",
                    duration_seconds=(datetime.now() - operation_start).total_seconds()
                )
            
            self.logger.info(f"Profile migrated from version {current_version} to {self.CURRENT_FORMAT_VERSION}")
            
            return OperationResult(
                operation_type="migrate_profile",
                success=True,
                message=f"Profile migrated from version {current_version} to {self.CURRENT_FORMAT_VERSION}",
                duration_seconds=(datetime.now() - operation_start).total_seconds(),
                details={
                    "old_version": current_version,
                    "new_version": self.CURRENT_FORMAT_VERSION,
                    "backup_created": backup
                }
            )
            
        except Exception as e:
            error_msg = f"Unexpected error during migration: {str(e)}"
            self.logger.error(error_msg)
            return OperationResult(
                operation_type="migrate_profile",
                success=False,
                error_message=error_msg,
                duration_seconds=(datetime.now() - operation_start).total_seconds()
            )
    
    def get_profile_info(self, file_path: str) -> Dict[str, Any]:
        """
        Get basic information about a profile file without loading it completely.
        
        Args:
            file_path: Path to the profile file
            
        Returns:
            Dict[str, Any]: Profile information dictionary
        """
        info = {
            "file_path": file_path,
            "exists": False,
            "readable": False,
            "file_size_bytes": 0,
            "format_version": None,
            "profile_name": None,
            "created_at": None,
            "modified_at": None,
            "parameter_count": 0,
            "validation_status": "unknown"
        }
        
        try:
            if os.path.exists(file_path):
                info["exists"] = True
                info["readable"] = os.access(file_path, os.R_OK)
                info["file_size_bytes"] = os.path.getsize(file_path)
                
                if info["readable"] and info["file_size_bytes"] > 0:
                    try:
                        with open(file_path, 'r', encoding='utf-8') as file:
                            profile_data = json.load(file)
                        
                        info["format_version"] = self._get_format_version(profile_data)
                        info["profile_name"] = profile_data.get("name")
                        info["created_at"] = profile_data.get("created_at")
                        info["modified_at"] = profile_data.get("modified_at")
                        
                        if "parameters" in profile_data and isinstance(profile_data["parameters"], dict):
                            info["parameter_count"] = len(profile_data["parameters"])
                        
                        # Quick validation
                        validation_result = self._validate_profile_data(profile_data)
                        if validation_result.is_valid:
                            info["validation_status"] = "valid"
                        elif validation_result.errors:
                            info["validation_status"] = "invalid"
                        else:
                            info["validation_status"] = "warnings"
                            
                    except (json.JSONDecodeError, UnicodeDecodeError):
                        info["validation_status"] = "invalid_format"
                    except Exception:
                        info["validation_status"] = "error"
        
        except Exception:
            info["validation_status"] = "error"
        
        return info
    
    def list_profile_files(self, directory: str, recursive: bool = False) -> List[Dict[str, Any]]:
        """
        List all profile files in a directory.
        
        Args:
            directory: Directory to search for profile files
            recursive: Whether to search subdirectories recursively
            
        Returns:
            List[Dict[str, Any]]: List of profile file information dictionaries
        """
        profile_files = []
        
        try:
            if not os.path.exists(directory):
                return profile_files
            
            if recursive:
                for root, dirs, files in os.walk(directory):
                    for file in files:
                        if file.endswith(self.PROFILE_EXTENSION):
                            file_path = os.path.join(root, file)
                            profile_info = self.get_profile_info(file_path)
                            profile_files.append(profile_info)
            else:
                for file in os.listdir(directory):
                    if file.endswith(self.PROFILE_EXTENSION):
                        file_path = os.path.join(directory, file)
                        if os.path.isfile(file_path):
                            profile_info = self.get_profile_info(file_path)
                            profile_files.append(profile_info)
        
        except Exception as e:
            self.logger.error(f"Error listing profile files in {directory}: {str(e)}")
        
        return profile_files
    
    def _ensure_profile_extension(self, file_path: str) -> str:
        """Ensure file path has the correct profile extension."""
        if not file_path.endswith(self.PROFILE_EXTENSION):
            return file_path + self.PROFILE_EXTENSION
        return file_path
    
    def _create_backup(self, file_path: str) -> OperationResult:
        """Create a backup of an existing profile file."""
        try:
            if not os.path.exists(file_path):
                return OperationResult(
                    operation_type="create_backup",
                    success=False,
                    error_message="File does not exist"
                )
            
            # Create backup directory
            directory = os.path.dirname(file_path)
            backup_dir = os.path.join(directory, self.BACKUP_DIR)
            os.makedirs(backup_dir, exist_ok=True)
            
            # Generate backup filename with timestamp
            filename = os.path.basename(file_path)
            name, ext = os.path.splitext(filename)
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            backup_filename = f"{name}_backup_{timestamp}{ext}"
            backup_path = os.path.join(backup_dir, backup_filename)
            
            # Copy file to backup location
            shutil.copy2(file_path, backup_path)
            
            return OperationResult(
                operation_type="create_backup",
                success=True,
                message=f"Backup created at {backup_path}",
                details={"backup_path": backup_path}
            )
            
        except Exception as e:
            return OperationResult(
                operation_type="create_backup",
                success=False,
                error_message=f"Backup creation failed: {str(e)}"
            )
    
    def _verify_saved_file(self, file_path: str, original_profile: UniversalProfile) -> OperationResult:
        """Verify that a saved file can be loaded and matches the original profile."""
        try:
            # Try to load the saved file
            with open(file_path, 'r', encoding='utf-8') as file:
                saved_data = json.load(file)
            
            # Basic structure validation
            validation_result = self._validate_profile_data(saved_data)
            if not validation_result.is_valid:
                return OperationResult(
                    operation_type="verify_file",
                    success=False,
                    error_message=f"Saved file validation failed: {'; '.join(validation_result.errors)}"
                )
            
            # Create profile from saved data and compare key fields
            loaded_profile = UniversalProfile.from_dict(saved_data)
            
            if loaded_profile.name != original_profile.name:
                return OperationResult(
                    operation_type="verify_file",
                    success=False,
                    error_message="Profile name mismatch after save"
                )
            
            if len(loaded_profile.parameters) != len(original_profile.parameters):
                return OperationResult(
                    operation_type="verify_file",
                    success=False,
                    error_message="Parameter count mismatch after save"
                )
            
            return OperationResult(
                operation_type="verify_file",
                success=True,
                message="File verification successful"
            )
            
        except Exception as e:
            return OperationResult(
                operation_type="verify_file",
                success=False,
                error_message=f"File verification error: {str(e)}"
            )
    
    def _validate_profile_data(self, profile_data: Dict[str, Any]) -> ValidationResult:
        """Validate the structure of profile data."""
        result = ValidationResult(is_valid=True)
        
        if not isinstance(profile_data, dict):
            result.add_error("Profile data must be a dictionary")
            return result
        
        # Check required top-level fields
        required_fields = ["name", "version", "created_at", "modified_at", "metadata", "parameters"]
        for field in required_fields:
            if field not in profile_data:
                result.add_error(f"Missing required field: {field}")
        
        # Validate field types
        if "name" in profile_data and not isinstance(profile_data["name"], str):
            result.add_error("Profile name must be a string")
        
        if "version" in profile_data and not isinstance(profile_data["version"], str):
            result.add_error("Profile version must be a string")
        
        if "metadata" in profile_data and not isinstance(profile_data["metadata"], dict):
            result.add_error("Profile metadata must be a dictionary")
        
        if "parameters" in profile_data and not isinstance(profile_data["parameters"], dict):
            result.add_error("Profile parameters must be a dictionary")
        
        # Validate timestamp formats
        for timestamp_field in ["created_at", "modified_at"]:
            if timestamp_field in profile_data:
                try:
                    datetime.fromisoformat(profile_data[timestamp_field])
                except (ValueError, TypeError):
                    result.add_error(f"Invalid timestamp format for {timestamp_field}")
        
        return result
    
    def _get_format_version(self, profile_data: Dict[str, Any]) -> str:

    total_entries: int

    def __init__(self, log_directory: str = "logs"):

        self._error_patterns = {

    def analyze_logs(self, hours: int = 24) -> LogAnalysisResult:
        """
        Perform comprehensive log analysis.
        
        Args:
            hours: Number of hours to analyze
            
        Returns:
            Analysis results
        """
        cutoff_time = datetime.now() - timedelta(hours=hours)
        
        # Analyze different log files
        operation_logs = self._analyze_operation_logs(cutoff_time)
        performance_logs = self._analyze_performance_logs(cutoff_time)
        error_logs = self._analyze_error_logs(cutoff_time)
        
        # Identify common issues
        common_issues = self._identify_common_issues(operation_logs, error_logs)
        
        # Generate performance insights

            recommendations=recommendations

    def _analyze_operation_logs(self, cutoff_time: datetime) -> List[Dict[str, Any]]:
        """Analyze profile operation logs."""
        log_file = self.log_directory / 'profile_operations.log'
        if not log_file.exists():
            return []
        
        logs = []
        try:
            with open(log_file, 'r', encoding='utf-8') as f:
                for line in f:
                    try:
                        # Parse log line
                        parts = line.strip().split(' - ', 3)
                        if len(parts) >= 4:
                            timestamp_str = parts[0]
                            timestamp = datetime.strptime(timestamp_str, '%Y-%m-%d %H:%M:%S,%f')
                            
                            if timestamp >= cutoff_time:
                                logs.append({
                                    'timestamp': timestamp,
                                    'logger': parts[1],
                                    'level': parts[2],
                                    'message': parts[3]
                                })
                    except Exception:
                        continue  # Skip malformed lines
        except Exception as e:
            self.logger.warning(f"Failed to read operation logs: {e}")
        
        return logs
    
    def _analyze_performance_logs(self, cutoff_time: datetime) -> List[Dict[str, Any]]:
        """Analyze performance logs."""
        log_file = self.log_directory / 'profile_performance.log'
        if not log_file.exists():
            return []
        
        logs = []
        try:
            with open(log_file, 'r', encoding='utf-8') as f:
                for line in f:
                    try:
                        data = json.loads(line.strip())
                        timestamp = datetime.fromisoformat(data['timestamp'])
                        
                        if timestamp >= cutoff_time:
                            logs.append(data)
                    except Exception:
                        continue  # Skip malformed lines
        except Exception as e:
            self.logger.warning(f"Failed to read performance logs: {e}")
        
        return logs
    
    def _analyze_error_logs(self, cutoff_time: datetime) -> List[Dict[str, Any]]:

    def _identify_common_issues(self, operation_logs: List[Dict[str, Any]], 
                              error_logs: List[Dict[str, Any]]) -> List[CommonIssue]:
        """Identify common issues from logs."""
        issues = []
        all_logs = operation_logs + error_logs
        
        for issue_type, pattern_info in self._error_patterns.items():
            pattern = re.compile(pattern_info['pattern'], re.IGNORECASE)
            matching_logs = []
            affected_operations = set()
            
            for log in all_logs:
                message = log.get('message', '')
                if pattern.search(message):
                    matching_logs.append(log)
                    
                    # Extract operation ID if present
                    op_match = re.search(r'OPERATION_\w+: (\w+_\d+)', message)
                    if op_match:
                        affected_operations.add(op_match.group(1))
            
            if matching_logs:
                issues.append(CommonIssue(
                    issue_type=issue_type,
                    description=pattern_info['description'],
                    frequency=len(matching_logs),
                    first_occurrence=min(log['timestamp'] for log in matching_logs),
                    last_occurrence=max(log['timestamp'] for log in matching_logs),
                    affected_operations=list(affected_operations),
                    suggested_solution=pattern_info['solution'],
                    severity=pattern_info['severity']
                ))
        
        # Sort by frequency (most common first)
        issues.sort(key=lambda x: x.frequency, reverse=True)
        return issues
    
    def _generate_performance_insights(self, performance_logs: List[Dict[str, Any]]) -> Dict[str, Any]:
        """Generate performance insights from logs."""
        if not performance_logs:
            return {'message': 'No performance data available'}
        
        # Calculate statistics
        durations = [log.get('duration_ms', 0) for log in performance_logs if log.get('duration_ms')]
        memory_usage = [log.get('memory_usage_mb', 0) for log in performance_logs if log.get('memory_usage_mb') is not None]
        cpu_usage = [log.get('cpu_usage_percent', 0) for log in performance_logs if log.get('cpu_usage_percent') is not None]
        
        # Operation type analysis
        operation_stats = defaultdict(list)
        for log in performance_logs:
            op_type = log.get('operation_type', 'unknown')
            if log.get('duration_ms'):
                operation_stats[op_type].append(log['duration_ms'])
        
        # Identify slow operations
        slow_operations = [
            log for log in performance_logs
            if log.get('duration_ms', 0) > self._performance_thresholds['slow_operation_ms']
        ]
        
        # Identify high resource usage
        high_memory_ops = [
            log for log in performance_logs
            if (log.get('memory_usage_mb') or 0) > self._performance_thresholds['high_memory_usage_mb']
        ]
        
        high_cpu_ops = [
            log for log in performance_logs
            if (log.get('cpu_usage_percent') or 0) > self._performance_thresholds['high_cpu_usage_percent']
        ]
        
        insights = {
            'total_operations': len(performance_logs),
            'average_duration_ms': round(sum(durations) / len(durations), 2) if durations else 0,
            'median_duration_ms': round(sorted(durations)[len(durations)//2], 2) if durations else 0,
            'max_duration_ms': max(durations) if durations else 0,
            'min_duration_ms': min(durations) if durations else 0,
            'slow_operations_count': len(slow_operations),
            'slow_operations_percentage': round(len(slow_operations) / len(performance_logs) * 100, 2),
            'average_memory_usage_mb': round(sum(memory_usage) / len(memory_usage), 2) if memory_usage else 0,
            'max_memory_usage_mb': max(memory_usage) if memory_usage else 0,
            'high_memory_operations_count': len(high_memory_ops),
            'average_cpu_usage_percent': round(sum(cpu_usage) / len(cpu_usage), 2) if cpu_usage else 0,
            'max_cpu_usage_percent': max(cpu_usage) if cpu_usage else 0,
            'high_cpu_operations_count': len(high_cpu_ops),
            'operation_type_performance': {}
        }
        
        # Add per-operation-type statistics
        for op_type, durations in operation_stats.items():
            insights['operation_type_performance'][op_type] = {
                'count': len(durations),
                'average_duration_ms': round(sum(durations) / len(durations), 2),
                'max_duration_ms': max(durations),
                'min_duration_ms': min(durations)
            }
        
        return insights

        if performance_insights.get('slow_operations_percentage', 0) > 20:

        # General recommendations
        if not recommendations:
