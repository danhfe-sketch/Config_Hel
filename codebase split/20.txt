        self._on_cancel()

                    self.dialog.grab_release()

        """Check for timeout and close dialog if stuck."""
        try:
            if self.is_destroying:
                return
                
            if not hasattr(self, 'dialog') or self.dialog is None or not self.dialog.winfo_exists():
                return

                self._close_dialog()

class SimpleProgressDialog:
    """Simple, thread-safe progress dialog with language support."""
    
    def __init__(self, parent, title_key: str, total_steps: int, i18n_service=None):
        """
        Initialize the progress dialog.
        
        Args:
            parent: Parent window
            title_key: Translation key for dialog title
            total_steps: Total number of steps for progress calculation
            i18n_service: Translation service instance
        """
        self.parent = parent
        self.title_key = title_key
        self.total_steps = max(1, total_steps)  # Ensure at least 1 to avoid division by zero
        self.i18n = i18n_service
        self.current_step = 0
        self.is_cancelled = False
        self.is_closed = False
        self.cancel_callback = None
        self.dialog = None
        
        # Create dialog in main thread
        self._create_dialog()
        
    def _create_dialog(self):
        """Create the progress dialog using basic Tkinter components."""
        try:
            # Create toplevel window
            self.dialog = tk.Toplevel(self.parent)
            
            # Set title using translation service
            title = self._get_translated_text(self.title_key, "Progress")
            self.dialog.title(title)
            
            self.dialog.geometry("400x150")
            self.dialog.resizable(False, False)
            
            # Set colors to match application theme
            self.dialog.configure(bg='#2b2b2b')
            
            # Center the dialog
            self._center_dialog()
            
            # Make modal
            self.dialog.transient(self.parent)
            self.dialog.grab_set()
            
            # Create main container with padding
            main_frame = tk.Frame(self.dialog, bg='#2b2b2b', padx=20, pady=20)
            main_frame.pack(fill="both", expand=True)

            self.status_label.pack(pady=(0, 15))

            style.configure("Custom.Horizontal.TProgressbar", 
                          background='#0078d4',

                          lightcolor='#0078d4',

            self.progress_var = tk.DoubleVar()

                main_frame,

        try:
            self.dialog.update_idletasks()
            
            # Get parent position and size
            parent_x = self.parent.winfo_x()
            parent_y = self.parent.winfo_y()
            parent_width = self.parent.winfo_width()
            parent_height = self.parent.winfo_height()

            message_key: Translation key for status message
            param_name: Optional parameter name to include in message

            self.current_step = current_step
            
            # Calculate progress percentage
            progress_percentage = min(100, (current_step / self.total_steps) * 100)
            
            # Update status message
            status_text = self._get_translated_text(message_key, f"Step {current_step}")

            if hasattr(self, 'status_label') and self.status_label:
                self.status_label.config(text=status_text)

            if hasattr(self, 'progress_var') and self.progress_var:
                self.progress_var.set(progress_percentage)
            
            # Update percentage label
            percentage_text = f"{int(progress_percentage)}%"
            if hasattr(self, 'percentage_label') and self.percentage_label:
                self.percentage_label.config(text=percentage_text)
                
        except Exception as e:
            print(f"Progress update error: {e}")

        """Set callback function for cancel operation."""
        self.cancel_callback = callback

        """Handle cancel button click."""

            self.is_cancelled = True
            try:
                self.cancel_callback()
            except Exception as e:
                print(f"Cancel callback error: {e}")

        self._on_cancel()

    def _validate_resource_path(self, path: str) -> Optional[str]:
        """
        Validate and resolve a resource path with comprehensive error handling.
        
        Args:
            path: Relative path to validate
            
        Returns:
            Absolute path if valid, None otherwise
        """
        if not path:
            logger.warning("Empty or None path provided for validation")
            return None
        
        try:
            # Step 1: Resolve the path
            logger.debug(f"Resolving resource path: {path}")
            resolved_path = resource_path(path)
            logger.debug(f"Path resolved to: {resolved_path}")
            
            # Step 2: Check if path exists
            if not os.path.exists(resolved_path):
                logger.debug(f"Path does not exist: {resolved_path}")
                
                # Try to provide helpful debugging information
                parent_dir = os.path.dirname(resolved_path)
                if os.path.exists(parent_dir):
                    try:
                        siblings = os.listdir(parent_dir)
                        logger.debug(f"Parent directory exists with contents: {siblings}")
                        
                        # Look for similar files
                        filename = os.path.basename(path)
                        similar_files = [f for f in siblings if filename.lower() in f.lower() or f.lower() in filename.lower()]
                        if similar_files:
                            logger.debug(f"Similar files found: {similar_files}")
                    except Exception as e:
                        logger.debug(f"Could not list parent directory contents: {e}")
                else:
                    logger.debug(f"Parent directory does not exist: {parent_dir}")
                
                return None
            
            # Step 3: Verify it's a file (not a directory)
            if not os.path.isfile(resolved_path):
                logger.warning(f"Path exists but is not a file: {resolved_path}")
                return None
            
            # Step 4: Check file permissions
            if not os.access(resolved_path, os.R_OK):
                logger.warning(f"File exists but is not readable: {resolved_path}")
                return None
            
            # Step 5: Get file information for debugging
            try:
                file_size = os.path.getsize(resolved_path)
                logger.debug(f"Ô£ô Valid resource found: {resolved_path} ({file_size} bytes)")
            except Exception as e:
                logger.debug(f"Could not get file size for {resolved_path}: {e}")
            
            return resolved_path
            
        except ValueError as e:
            logger.warning(f"Invalid path format '{path}': {e}")
            return None
            
        except OSError as e:
            logger.warning(f"OS error resolving path '{path}': {e}")
            return None
            
        except Exception as e:
            logger.error(f"Unexpected error resolving path '{path}': {type(e).__name__}: {e}")
            return None

        paths_to_try = [self.config.image_path] + self.config.fallback_image_paths
        
        logger.info(f"Starting cascading image fallback mechanism with {len(paths_to_try)} possible paths")
        logger.debug(f"Fallback order: {' -> '.join(paths_to_try)} -> text-based fallback")
        
        fallback_results = []  # Track results for debugging
        
        for i, path in enumerate(paths_to_try):
            fallback_level = "primary" if i == 0 else f"fallback-{i}"
            logger.info(f"[{fallback_level}] Attempting to load image: {path}")
            
            # Step 1: Validate resource path
            resolved_path = self._validate_resource_path(path)
            if resolved_path is None:
                error_msg = f"Path resolution failed for {path}"
                logger.warning(f"[{fallback_level}] {error_msg}")
                fallback_results.append({
                    'path': path,
                    'level': fallback_level,
                    'stage': 'path_resolution',
                    'success': False,
                    'error': error_msg
                })
                continue
            
            # Step 2: Attempt to load image file
            try:
                logger.debug(f"[{fallback_level}] Opening image file: {resolved_path}")
                img = Image.open(resolved_path)
                logger.info(f"[{fallback_level}] Successfully opened image file")
                logger.debug(f"[{fallback_level}] Original image size: {img.size}, mode: {img.mode}")
                
                # Step 3: Validate image properties
                if img.size[0] <= 0 or img.size[1] <= 0:
                    raise ValueError(f"Invalid image dimensions: {img.size}")
                
                # Step 4: Process image (resize if needed)
                processed_img = self._process_image_for_display(img, fallback_level)
                if processed_img is None:
                    raise ValueError("Image processing failed")
                
                # Step 5: Convert to PhotoImage
                logger.debug(f"[{fallback_level}] Converting to PhotoImage")
                photo_image = ImageTk.PhotoImage(processed_img)
                
                # Success!
                logger.info(f"[{fallback_level}] Ô£ô Image successfully loaded and processed: {resolved_path}")
                fallback_results.append({
                    'path': path,
                    'level': fallback_level,
                    'stage': 'complete',
                    'success': True,
                    'resolved_path': resolved_path,
                    'original_size': img.size,
                    'final_size': processed_img.size
                })
                
                # Log successful fallback summary
                self._log_fallback_success(fallback_results)
                return photo_image
                
            except FileNotFoundError as e:
                error_msg = f"Image file not found: {e}"
                logger.warning(f"[{fallback_level}] {error_msg}")
                fallback_results.append({
                    'path': path,
                    'level': fallback_level,
                    'stage': 'file_access',
                    'success': False,
                    'error': error_msg
                })
                
            except PermissionError as e:
                error_msg = f"Permission denied accessing image: {e}"
                logger.warning(f"[{fallback_level}] {error_msg}")
                fallback_results.append({
                    'path': path,
                    'level': fallback_level,
                    'stage': 'file_access',
                    'success': False,
                    'error': error_msg
                })
                
            except (OSError, IOError) as e:
                error_msg = f"I/O error loading image: {e}"
                logger.warning(f"[{fallback_level}] {error_msg}")
                fallback_results.append({
                    'path': path,
                    'level': fallback_level,
                    'stage': 'file_io',
                    'success': False,
                    'error': error_msg
                })
                
            except ValueError as e:
                error_msg = f"Invalid image data: {e}"
                logger.warning(f"[{fallback_level}] {error_msg}")
                fallback_results.append({
                    'path': path,
                    'level': fallback_level,
                    'stage': 'image_validation',
                    'success': False,
                    'error': error_msg
                })
                
            except Exception as e:
                error_msg = f"Unexpected error loading image: {type(e).__name__}: {e}"
                logger.error(f"[{fallback_level}] {error_msg}")
                fallback_results.append({
                    'path': path,
                    'level': fallback_level,
                    'stage': 'unexpected_error',
                    'success': False,
                    'error': error_msg
                })
        
        # All image loading attempts failed
        logger.error("Ô£ù All image loading attempts failed, will use text-based fallback")
        self._log_fallback_failure(fallback_results)
        return None
    
    def _process_image_for_display(self, img: Image.Image, fallback_level: str) -> Optional[Image.Image]:
        """
        Process an image for display with proper error handling.
        
        Args:
            img: PIL Image to process
            fallback_level: Current fallback level for logging
            
        Returns:
            Processed Image or None if processing fails
        """
        try:
            # Ensure image is in a compatible mode
            if img.mode not in ('RGB', 'RGBA', 'L'):
                logger.debug(f"[{fallback_level}] Converting image from {img.mode} to RGB")
                img = img.convert('RGB')
            
            # Get original dimensions
            width, height = img.size
            max_width, max_height = self.config.max_image_size
            
            logger.debug(f"[{fallback_level}] Image dimensions: {width}x{height}, max allowed: {max_width}x{max_height}")
            
            # Resize if needed while maintaining aspect ratio
            if width > max_width or height > max_height:
                ratio = min(max_width / width, max_height / height)
                new_width = int(width * ratio)
                new_height = int(height * ratio)
                
                logger.debug(f"[{fallback_level}] Resizing image: {width}x{height} -> {new_width}x{new_height} (ratio: {ratio:.3f})")
                
                # Use high-quality resampling
                img = img.resize((new_width, new_height), Image.LANCZOS)
                logger.debug(f"[{fallback_level}] Image resized successfully")
            else:
                logger.debug(f"[{fallback_level}] Image size is within limits, no resizing needed")
            
            return img
            
        except Exception as e:
            logger.error(f"[{fallback_level}] Error processing image: {e}")
            return None
    
    def _log_fallback_success(self, fallback_results: list):
        """Log successful fallback information for debugging."""
        successful_result = fallback_results[-1]  # Last result should be successful
        failed_attempts = len(fallback_results) - 1
        
        if failed_attempts > 0:
            logger.info(f"Image loaded successfully after {failed_attempts} failed attempt(s)")
            logger.debug("Failed attempts summary:")
            for result in fallback_results[:-1]:
                logger.debug(f"  - {result['level']}: {result['path']} failed at {result['stage']} - {result['error']}")
        
        logger.info(f"Final image: {successful_result['resolved_path']} "
                   f"({successful_result['original_size']} -> {successful_result['final_size']})")
    
    def _log_fallback_failure(self, fallback_results: list):
        """Log detailed failure information for debugging."""
        logger.error("Detailed fallback failure analysis:")
        logger.error(f"Total attempts: {len(fallback_results)}")
        
        # Group failures by stage
        failure_stages = {}
        for result in fallback_results:
            stage = result['stage']
            if stage not in failure_stages:
                failure_stages[stage] = []
            failure_stages[stage].append(result)
        
        for stage, failures in failure_stages.items():
            logger.error(f"  {stage.replace('_', ' ').title()} failures: {len(failures)}")
            for failure in failures:
                logger.error(f"    - {failure['level']}: {failure['path']} - {failure['error']}")
        
        # Provide troubleshooting suggestions
        logger.error("Troubleshooting suggestions:")
        logger.error("  1. Verify image files exist in the assets/images/ directory")
        logger.error("  2. Check file permissions for image files")
        logger.error("  3. Ensure images are in supported formats (PNG, JPG, ICO)")
        logger.error("  4. Verify PyInstaller is bundling assets correctly if running as executable")
        logger.error("  5. Check that image files are not corrupted")

    def _load_splash_content(self):
        """
        Load the splash screen content with robust cascading fallback mechanism.
        
        Fallback sequence:
        1. Try to load primary splash image (splash.png)
        2. Try fallback images (logo.png, icon.ico, etc.)
        3. Create styled text-based fallback
        4. Create minimal text fallback (last resort)
        """
        logger.info("Starting splash content loading with cascading fallback mechanism")
        
        try:
            # Phase 1: Attempt image loading with fallbacks
            logger.debug("Phase 1: Attempting image loading")
            splash_image = self._load_image_with_fallbacks()
            
            if splash_image is not None:
                # Successfully loaded an image
                logger.info("Ô£ô Phase 1 successful: Image-based splash screen created")
                self.splash_image = splash_image
                self.splash_label = tk.Label(self.root, image=self.splash_image, bd=0)
                self.splash_label.pack()
                self._log_content_success("image")
                return
            
            # Phase 2: Image loading failed, create text-based fallback
            logger.info("Phase 1 failed, proceeding to Phase 2: Text-based fallback")
            self._create_text_fallback()
            logger.info("Ô£ô Phase 2 successful: Text-based splash screen created")
            self._log_content_success("text")
            return
                
        except Exception as e:
            logger.error(f"Phase 2 failed with critical error: {e}")
            
            # Phase 3: Last resort - minimal fallback
            try:
                logger.warning("Proceeding to Phase 3: Minimal fallback (last resort)")
                self._create_minimal_fallback()
                logger.info("Ô£ô Phase 3 successful: Minimal splash screen created")
                self._log_content_success("minimal")
                return
                
            except Exception as fallback_error:
                logger.critical(f"Phase 3 failed: {fallback_error}")
                logger.critical("All splash screen fallback mechanisms have failed")
                
                # Log system information for debugging
                self._log_system_debug_info()
                
                # Re-raise with context
                raise RuntimeError("Complete splash screen failure: all fallback mechanisms exhausted") from fallback_error

    def _center_window(self):

        try:
            self.root.update_idletasks()

            # Get screen dimensions
            screen_width = self.root.winfo_screenwidth()

    def _fade_in(self, step=None, delay=None, current=0.0):
        """
        Gradually fade in the splash screen with smooth timing.
        
        Args:
            step: Alpha increment per step (calculated from config if None)
            delay: Delay between steps in milliseconds (calculated from config if None)
            current: Current alpha value
        """
        try:
            # Calculate smooth fade parameters based on config
            if step is None:
                total_steps = self.config.fade_in_duration // 16  # ~60fps
                step = 1.0 / max(1, total_steps)
            if delay is None:
                delay = 16  # ~60fps for smooth animation
                
            current += step
            if current <= 1.0:
                self.root.attributes('-alpha', current)
                self.root.after(delay, lambda: self._fade_in(step, delay, current))
            else:
                self.root.attributes('-alpha', 1.0)  # Ensure fully opaque
                logger.debug("Fade-in animation completed")
        except Exception as e:
            logger.error(f"Error during fade-in animation: {e}")
            # Set to fully opaque if animation fails
            try:
                self.root.attributes('-alpha', 1.0)
            except:
                pass
    
    def _fade_out(self, step=None, delay=None, current=1.0):
        """
        Gradually fade out the splash screen with smooth timing.
        
        Args:
            step: Alpha decrement per step (calculated from config if None)
            delay: Delay between steps in milliseconds (calculated from config if None)
            current: Current alpha value
        """
        try:
            # Calculate smooth fade parameters based on config
            if step is None:
                total_steps = self.config.fade_out_duration // 16  # ~60fps
                step = 1.0 / max(1, total_steps)
            if delay is None:
                delay = 16  # ~60fps for smooth animation
                
            current -= step
            if current >= 0:
                self.root.attributes('-alpha', current)
                self.root.after(delay, lambda: self._fade_out(step, delay, current))
            else:
                logger.debug("Fade-out animation completed, destroying window")
                self._animation_running = False
                self.root.destroy()
        except Exception as e:
            logger.error(f"Error during fade-out animation: {e}")
            # Force destroy if animation fails
            try:
                self._animation_running = False
                self.root.destroy()
            except:
                pass
    
    def _start_progress_animation(self):
        """Start the smooth progress bar animation."""
        if not self._animation_running:
            self._animation_running = True
            self._animate_progress_smooth()
    
    def _animate_progress_smooth(self):
        """
        Animate the progress bar with smooth transitions and proper timing control.
        This replaces the old time-based animation with a more responsive system.
        """
        try:
            if not hasattr(self, 'root') or not self.root.winfo_exists():
                logger.debug("Root window no longer exists, stopping progress animation")
                self._animation_running = False
                return
            
            if not hasattr(self, 'progress_bar') or self._close_requested:
                logger.debug("Progress bar not available or close requested, stopping animation")
                self._animation_running = False
                return
                
            total_width = self.root.winfo_width()
            if total_width <= 0:
                # Window not ready yet, try again later
                if self._animation_running:
                    self.root.after(16, self._animate_progress_smooth)  # ~60fps
                return
            
            # Smooth progress animation towards target
            current_time = time.time()
            time_delta = current_time - self.progress_state.last_update_time
            
            # Calculate smooth progress increment
            if self.progress_state.current < self.progress_state.target:
                # Animate towards target
                progress_diff = self.progress_state.target - self.progress_state.current
                increment = min(progress_diff, self.progress_state.animation_speed * (time_delta * 60))  # Scale by time
                self.progress_state.current = min(self.progress_state.target, 
                                                self.progress_state.current + increment)
            elif self.progress_state.current > self.progress_state.target:
                # Animate backwards if needed (rare case)
                progress_diff = self.progress_state.current - self.progress_state.target
                decrement = min(progress_diff, self.progress_state.animation_speed * (time_delta * 60))
                self.progress_state.current = max(self.progress_state.target,
                                                self.progress_state.current - decrement)
            
            # Update progress state
            self.progress_state.is_complete = self.progress_state.current >= 1.0
            self.progress_state.last_update_time = current_time
            
            # Update the progress bar visual
            new_width = int(self.progress_state.current * total_width)
            self.progress_bar.coords(self.progress_rect, 0, 0, new_width, 5)
            
            # Add subtle color transition as progress increases
            if hasattr(self, 'progress_rect'):
                # Gradually brighten the progress color as it fills
                brightness = 0.7 + (0.3 * self.progress_state.current)  # 70% to 100% brightness
                color_value = int(255 * brightness)
                progress_color = f"#{color_value:02x}{color_value:02x}ff"  # Blue with varying brightness
                self.progress_bar.itemconfig(self.progress_rect, fill=progress_color)
            
            logger.debug(f"Smooth progress updated: {self.progress_state.current:.3f} -> {self.progress_state.target:.3f} "
                        f"(width: {new_width}/{total_width})")
            
            # Continue animation if running and not complete
            if self._animation_running and not self._close_requested:
                self.root.after(16, self._animate_progress_smooth)  # ~60fps for smooth animation
            else:
                logger.debug("Smooth progress animation stopped")
                self._animation_running = False
                
        except Exception as e:
            logger.error(f"Error during smooth progress animation: {e}")
            self._animation_running = False
    
    def update_progress(self, value: float, message: str = None, animate: bool = True):
        """
        Update the progress bar with a specific value using smooth animation.
        
        Args:
            value: Progress value between 0.0 and 1.0
            message: Optional progress message
            animate: Whether to animate to the new value (True) or jump immediately (False)
        """
        try:
            # Validate input
            value = max(0.0, min(1.0, value))
            
            if hasattr(self, 'root') and self.root.winfo_exists():
                # Update progress state
                if animate:
                    # Set target for smooth animation
                    self.progress_state.target = value
                    # Start animation if not already running
                    if not self._animation_running:
                        self._start_progress_animation()
                else:
                    # Jump immediately to value
                    self.progress_state.current = value
                    self.progress_state.target = value
                    
                    # Update visual immediately if progress bar exists
                    if hasattr(self, 'progress_bar'):
                        total_width = self.root.winfo_width()
                        if total_width > 0:
                            new_width = int(value * total_width)
                            self.progress_bar.coords(self.progress_rect, 0, 0, new_width, 5)
                
                # Update message and completion state
                if message:
                    self.progress_state.message = message
                self.progress_state.is_complete = value >= 1.0
                self.progress_state.last_update_time = time.time()
                
                logger.debug(f"Progress {'animated' if animate else 'immediately'} updated to: {value:.2%}" + 
                           (f" - {message}" if message else ""))
                
                # Auto-close when progress reaches 100% if configured
                if self.progress_state.is_complete and not self._close_requested:
                    logger.debug("Progress completed, scheduling auto-close")
                    # Small delay to let the animation finish
                    self.root.after(200, self._auto_close_on_completion)
                    
        except Exception as e:
            logger.error(f"Error updating progress: {e}")
    
    def _auto_close_on_completion(self):
        """Automatically close splash screen when progress is complete."""
        try:
            if self.progress_state.is_complete and not self._close_requested:
                logger.debug("Auto-closing splash screen on progress completion")
                self.close()
        except Exception as e:
            logger.error(f"Error during auto-close: {e}")
    
    def close(self):
        """Close the splash screen with proper timing controls and fade-out effect."""
        try:
            if self._close_requested:
                logger.debug("Close already requested, ignoring duplicate close call")
                return
                
            self._close_requested = True
            
            # Ensure minimum display time
            elapsed = time.time() * 1000 - self.start_time
            logger.debug(f"Closing splash screen. Elapsed time: {elapsed}ms, "
                        f"Minimum required: {self.config.min_display_time}ms")
            
            if elapsed < self.config.min_display_time:
                remaining = int(self.config.min_display_time - elapsed)
                logger.debug(f"Waiting additional {remaining}ms before closing")
                self.root.after(remaining, self._perform_close)
            else:
                logger.debug("Minimum display time met, proceeding with close")
                self._perform_close()
                
        except Exception as e:
            logger.error(f"Error closing splash screen: {e}")
            # Force close if normal close fails
            self._force_close()

        """Perform the actual close operation with fade-out."""
        try:
            # Stop progress animation
            self._animation_running = False
            
            # Complete progress bar if not already complete
            if not self.progress_state.is_complete:
                self.update_progress(1.0, "Complete", animate=False)
            
            logger.debug("Starting fade-out animation")
            self._fade_out()
            
        except Exception as e:
            logger.error(f"Error performing close: {e}")
            self._force_close()
    
    def _force_close(self):
        """Force close the splash screen without animations."""
        try:
            self._animation_running = False
            self._close_requested = True
            if hasattr(self, 'root'):
                self.root.destroy()
            logger.debug("Splash screen force closed")
        except Exception as e:
            logger.error(f"Error force closing splash screen: {e}")

            # Update UI with the field values
            self.app.after(0, self._update_ui_with_values, ui_field_values, populate_result)

            from models.operation_result import ProfileOperationResult, OperationType
            error_result = ProfileOperationResult(
                operation_type=OperationType.LOAD_FROM_FILE,
                success=False
            )
            error_result.add_error(str(e))

This is a simplified version that avoids complex widget management and threading issues.
"""

from tkinter import ttk
import threading
import time
from typing import Callable, Optional
from dataclasses import dataclass


@dataclass
class UltraSimpleProgressInfo:
    """Ultra simple progress information."""
    progress_percentage: float = 0.0
    status_message: str = ""
    current_parameter: int = 0
    total_parameters: int = 0
    is_complete: bool = False
    has_error: bool = False
    error_message: str = ""


class UltraSimpleProgressDialog:
    """Ultra simple, crash-resistant progress dialog using basic Tkinter."""
    
    def __init__(self, parent, operation_type: str = "device_operation"):

        self.operation_type = operation_type

        self.is_closed = False
        self.cancel_callback = None
        self.dialog = None
        
        # Create dialog in main thread
        self._create_dialog()

        try:
            # Create toplevel window
            self.dialog = tk.Toplevel(self.parent)
            self.dialog.title("Opera├º├úo do Dispositivo")
            self.dialog.geometry("500x200")
            self.dialog.resizable(False, False)
            
            # Set dark theme colors to match application
            self.dialog.configure(bg='#2b2b2b')
            
            # Center the dialog
            self._center_dialog()
            
            # Make modal
            self.dialog.transient(self.parent)
            self.dialog.grab_set()
            
            # Create main container with padding
            main_frame = tk.Frame(self.dialog, bg='#2b2b2b', padx=30, pady=30)
            main_frame.pack(fill="both", expand=True)

                font=("Segoe UI", 14, "bold"),
                fg='white',
                bg='#2b2b2b'
            )
            title_label.pack()
            
            # Status message
            self.status_label = tk.Label(

                relief='flat',

            # Create progress bar fill
            self.progress_fill = tk.Frame(
                self.progress_container,
                bg='#0078d4',
                height=8,

            x = parent_x + (parent_width - dialog_width) // 2

        """Update progress - thread safe."""
        if self.is_closed or not self.dialog:
            return

        try:
            self.dialog.after_idle(self._do_update, progress_info)
        except Exception:
            pass
            
    def _do_update(self, progress_info: UltraSimpleProgressInfo):
        """Perform the actual update on main thread."""
        try:
            if self.is_closed or not self.dialog:
                return
                
            # Update status message
            status_text = progress_info.status_message
            if progress_info.total_parameters > 0 and progress_info.current_parameter > 0:
                status_text += f"\nPar├ómetro {progress_info.current_parameter}/{progress_info.total_parameters}"
                
            if hasattr(self, 'status_label') and self.status_label:
                self.status_label.config(text=status_text)
            
            # Update progress bar
            if hasattr(self, 'progress_container') and hasattr(self, 'progress_fill'):
                # Get container width
                container_width = 440  # Fixed width for reliability
                progress_width = max(1, int((progress_info.progress_percentage / 100.0) * container_width))
                
                # Update the progress fill frame
                self.progress_fill.place(width=progress_width)
                
                # Change color based on progress
                if progress_info.progress_percentage >= 100:
                    self.progress_fill.config(bg='#00aa00')  # Green when complete
                else:
                    self.progress_fill.config(bg='#0078d4')  # Blue for progress
            
            # Update percentage label
            percentage_text = f"{int(progress_info.progress_percentage)}%"
            if hasattr(self, 'percentage_label') and self.percentage_label:
                self.percentage_label.config(text=percentage_text)
            
            # Handle completion
            if progress_info.is_complete:
                if progress_info.has_error:
                    self._show_error(progress_info.error_message)
                else:
                    self._show_success()
                    
        except Exception as e:
            print(f"Update error: {e}")
            
    def _show_success(self):
        """Show success state."""
        try:
            if hasattr(self, 'status_label') and self.status_label:
                self.status_label.config(text="Opera├º├úo conclu├¡da com sucesso!")
            if hasattr(self, 'cancel_button') and self.cancel_button:
                self.cancel_button.config(text="Fechar", bg="lightgreen")
        except Exception:
            pass
            
    def _show_error(self, error_message: str):
        """Show error state."""
        try:
            if hasattr(self, 'status_label') and self.status_label:
                self.status_label.config(text=f"Opera├º├úo falhou: {error_message}")
            if hasattr(self, 'cancel_button') and self.cancel_button:
                self.cancel_button.config(text="Fechar", bg="lightcoral")
        except Exception:
            pass
            
    def set_cancel_callback(self, callback: Callable):
        """Set cancel callback."""
        self.cancel_callback = callback
        
    def _on_cancel(self):
        """Handle cancel."""
        if self.cancel_callback and not self.is_cancelled:
            self.is_cancelled = True
            try:
                self.cancel_callback()
            except Exception:
                pass
        self.close()
        
    def _on_close(self):
        """Handle window close."""
        self._on_cancel()
        
    def show(self):
        """Show dialog."""
        try:
            if self.dialog and not self.is_closed:
                self.dialog.deiconify()
                self.dialog.lift()
        except Exception:
            pass
            
    def hide(self):
        """Hide dialog."""
        try:
            if self.dialog and not self.is_closed:
                self.dialog.withdraw()
        except Exception:
            pass
            
    def close(self):
        """Close dialog."""
        if self.is_closed:
            return
            
        self.is_closed = True
        
        try:
            if self.dialog:
                self.dialog.grab_release()
                self.dialog.destroy()
                self.dialog = None
        except Exception:
            pass
            
        # Clear references
        self.status_label = None
        self.progress_bar = None
        self.progress_var = None
        self.percentage_label = None
        self.cancel_button = None


def create_ultra_simple_progress_info(current_param: int, total_params: int,

    def update_progress(self, progress_info: UnifiedProgressInfo):
        """
        Update the progress dialog with new information - thread safe.
        
        Args:
            progress_info: Progress information
        """
        if self.is_closed or not self.dialog:
            return
            
        # Schedule update on main thread
        try:
            if self.dialog.winfo_exists():
                self.dialog.after_idle(self._do_update, progress_info)
        except Exception:
            # Dialog may have been destroyed
            pass
            
    def update_translations(self):
        """Update all translatable text in the dialog."""
        if self.is_closed or not self.dialog:

        except Exception:

    def _do_update_translations(self):
        """Update translations on main thread."""
        try:
            if self.is_closed or not self.dialog:
                return
                
            # Update dialog title
            title = self._get_operation_title()
            self.dialog.title(title)
            
            # Update title label
            if hasattr(self, 'title_label') and self.title_label:
                self.title_label.configure(text=title)

            if hasattr(self, 'cancel_button') and self.cancel_button:

            if hasattr(self, 'close_button') and self.close_button:
                close_text = self._get_translated_text("close", "Close")
                self.close_button.configure(text=close_text)
                
            # Update status label if it has translatable text
            if hasattr(self, 'status_label') and self.status_label:
                current_text = self.status_label.cget("text")
                
                # Check for various translatable status messages
                translatable_patterns = {
                    # Initializing messages
                    ("Initializing...", "Inicializando...", "Inicializando..."): ("initializing", "Initializing..."),
                    # Processing step messages
                    ("Processing step", "Processando passo", "Procesando paso"): ("processing_step", "Processing step"),
                    # Connection messages
                    ("Connecting to device", "Conectando ao dispositivo", "Conectando al dispositivo"): ("connecting_to_device", "Connecting to device..."),
                    # Reading messages
                    ("Reading profile file", "Lendo arquivo de perfil", "Leyendo archivo de perfil"): ("reading_profile_file", "Reading profile file..."),
                    # Finalizing messages
                    ("Finalizing operation", "Finalizando opera├º├úo", "Finalizando operaci├│n"): ("finalizing_operation", "Finalizing operation..."),
                    ("Writing profile file", "Escrevendo arquivo de perfil", "Escribiendo archivo de perfil"): ("writing_profile_file", "Writing profile file..."),
                    ("Applying configuration", "Aplicando configura├º├úo", "Aplicando configuraci├│n"): ("applying_configuration", "Applying configuration..."),
                }
                
                # Check if current text matches any translatable pattern
                for patterns, (key, default) in translatable_patterns.items():
                    if any(pattern in current_text for pattern in patterns):
                        # Extract any dynamic parts (like step numbers)
                        import re
                        step_match = re.search(r'(\d+)', current_text)
                        if step_match and key == "processing_step":
                            step_num = step_match.group(1)
                            translated_text = f"{self._get_translated_text(key, default)} {step_num}..."
                        else:
                            translated_text = self._get_translated_text(key, default)

    def _do_update(self, progress_info: UnifiedProgressInfo):
        """Perform the actual update on main thread."""
        try:
            if self.is_closed or not self.dialog:
                return
                
            # Start timeout monitoring on first progress update
            if self.operation_start_time is None and progress_info.current_step > 0:
                self._start_timeout_monitoring()
                
            # Update status message
            if progress_info.status_message:
                self.status_label.configure(text=progress_info.status_message)
                
            # Update progress bar
            progress_value = max(0, min(1, progress_info.progress_percentage))
            self.progress_bar.set(progress_value)
            
            # Update percentage display
            percentage = int(progress_value * 100)
            self.percentage_label.configure(text=f"{percentage}%")
            
            # Update progress text with step and parameter info
            progress_parts = []
            if progress_info.total_steps > 0:
                step_text = self._get_translated_text("step", "Step")
                progress_parts.append(f"{step_text} {progress_info.current_step}/{progress_info.total_steps}")
                
            if progress_info.total_parameters > 0 and progress_info.parameter_count > 0:
                param_text = self._get_translated_text("parameter", "Parameter")
                progress_parts.append(f"{param_text} {progress_info.parameter_count}/{progress_info.total_parameters}")
                
            self.progress_text.configure(text=" ÔÇó ".join(progress_parts))
            
            # Update current parameter display
            if progress_info.current_parameter:
                param_text = self._get_translated_text("processing_parameter", "Processing parameter")
                parameter_display = f"{param_text}: {progress_info.current_parameter}"
                self.parameter_label.configure(text=parameter_display)
            else:
                self.parameter_label.configure(text="")
                
            # Handle completion
            if progress_info.is_complete:
                self._handle_completion(progress_info)
                
            # Handle cancellation
            if progress_info.is_cancelled:
                self._handle_cancellation()
                
        except Exception as e:
            print(f"Progress update error: {e}")
            
    def _handle_completion(self, progress_info: UnifiedProgressInfo):
        """Handle operation completion."""
        # Stop timeout monitoring
        self._stop_timeout_monitoring()
        
        # Update UI for completion
        if progress_info.has_error:
            # Show error state
            error_text = self._get_translated_text("operation_failed", "Operation failed")
            if progress_info.error_message:
                error_text += f": {progress_info.error_message}"
            self.status_label.configure(text=error_text)
            
            # Change progress bar color to red
            self.progress_bar.configure(progress_color=("#cc0000", "#cc0000"))
            
        else:
            # Show success state
            success_text = self._get_translated_text("operation_completed", "Operation completed successfully")
            self.status_label.configure(text=success_text)
            
            # Change progress bar color to green
            self.progress_bar.configure(progress_color=("#00aa00", "#00aa00"))
            self.progress_bar.set(1.0)
            self.percentage_label.configure(text="100%")
            
        # Hide cancel button and show close button
        self.cancel_button.grid_remove()
        self.close_button.grid(row=0, column=0, pady=0, sticky="e")
        
        # Call completion callback if set (only once)
        if self.completion_callback and not self.completion_called:
            self.completion_called = True
            try:
                self.completion_callback(progress_info)
            except Exception as e:
                print(f"Completion callback error: {e}")
                
    def _handle_cancellation(self):
        """Handle operation cancellation."""
        # Stop timeout monitoring
        self._stop_timeout_monitoring()
        
        # Update UI for cancellation
        cancelled_text = self._get_translated_text("operation_cancelled", "Operation cancelled")
        self.status_label.configure(text=cancelled_text)
        
        # Change progress bar color to orange
        self.progress_bar.configure(progress_color=("#ff8c00", "#ff8c00"))
        
        # Hide cancel button and show close button
        self.cancel_button.grid_remove()
        self.close_button.grid(row=0, column=0, pady=0, sticky="e")
        
    def _start_timeout_monitoring(self):
        """Start monitoring for operation timeout."""
        self.operation_start_time = time.time()
        self._check_timeout()
        
    def _check_timeout(self):
        """Check if operation has timed out."""
        if self.is_closed or not self.dialog:
            return
            
        if self.operation_start_time:
            elapsed = time.time() - self.operation_start_time
            if elapsed > self.operation_timeout:
                # Operation timed out
                timeout_text = self._get_translated_text("operation_timeout", "Operation timed out")
                self.status_label.configure(text=timeout_text)
                
                # Trigger cancellation
                self._on_cancel()
                return
                
        # Schedule next timeout check
        self.timeout_after_id = self.dialog.after(5000, self._check_timeout)
        
    def _stop_timeout_monitoring(self):
        """Stop timeout monitoring."""
        if self.timeout_after_id:
            try:
                self.dialog.after_cancel(self.timeout_after_id)
            except Exception:
                pass
            self.timeout_after_id = None
            
    def set_cancel_callback(self, callback: Callable):
        """Set callback function for cancel operation."""
        self.cancel_callback = callback
        
    def set_completion_callback(self, callback: Callable):
        """Set callback function for operation completion."""
        self.completion_callback = callback
        
    def _on_cancel(self):
        """Handle cancel button click."""
        if not self.is_cancelled and self.cancel_callback:
            self.is_cancelled = True
            try:
                self.cancel_callback()
            except Exception as e:
                print(f"Cancel callback error: {e}")
        self.close_dialog()
        
    def _on_close(self):
        """Handle close button click."""
        self.close_dialog()
        
    def _on_window_close(self):
        """Handle window close event."""
        if not self.is_cancelled:
            self._on_cancel()
        else:
            self.close_dialog()
            
    def show(self):
        """Show the dialog."""
        try:
            if self.dialog and not self.is_closed:
                self.dialog.deiconify()
                self.dialog.lift()
                self.dialog.focus()
        except Exception:
            pass
            
    def hide(self):
        """Hide the dialog."""
        try:
            if self.dialog and not self.is_closed:
                self.dialog.withdraw()
        except Exception:
            pass
            
    def close_dialog(self):
        """Close the dialog safely."""
        if self.is_closed:
            return
            
        self.is_closed = True
        
        # Stop timeout monitoring
        self._stop_timeout_monitoring()
        
        try:
            if self.dialog:
                self.dialog.grab_release()
                self.dialog.destroy()
                self.dialog = None
        except Exception:
            pass
            
        # Clear references
        self.status_label = None
        self.progress_bar = None
        self.progress_text = None
        self.percentage_label = None
        self.parameter_label = None
        self.cancel_button = None
        self.close_button = None

# -*- coding: utf-8 -*-

import sys
import os
import logging

# Set up logging for helpers
logger = logging.getLogger(__name__)

def resource_path(relative_path):
    """ 
    Get absolute path to an asset, which works for development and for PyInstaller.
    Enhanced with better error handling and debugging information.
    
    Args:
        relative_path: Relative path to the resource
        
    Returns:
        Absolute path to the resource
        
    Raises:
        ValueError: If relative_path is None or empty
        OSError: If there are issues with path resolution
    """
    if not relative_path:
        raise ValueError("relative_path cannot be None or empty")
    
    try:
        # PyInstaller creates a temp folder and stores path in _MEIPASS
        if hasattr(sys, '_MEIPASS'):
            base_path = sys._MEIPASS
            logger.debug(f"Using PyInstaller base path: {base_path}")
        else:
            base_path = os.path.abspath(".")
            logger.debug(f"Using development base path: {base_path}")
        
        # Join the paths
        full_path = os.path.join(base_path, relative_path)
        
        # Normalize the path (resolve .. and . components)
        normalized_path = os.path.normpath(full_path)
        
        logger.debug(f"Resource path resolution: {relative_path} -> {normalized_path}")
        
        return normalized_path
        
    except Exception as e:
        logger.error(f"Error resolving resource path for '{relative_path}': {e}")
        raise OSError(f"Failed to resolve resource path: {relative_path}") from e

def validate_resource_exists(relative_path):
    """
    Validate that a resource exists and return its absolute path.
    
    Args:
        relative_path: Relative path to validate
        
    Returns:
        Absolute path if resource exists, None otherwise
    """
    try:
        full_path = resource_path(relative_path)
        
        if os.path.exists(full_path):
            logger.debug(f"Resource exists: {full_path}")
            return full_path
        else:
            logger.debug(f"Resource does not exist: {full_path}")
            return None
            
    except Exception as e:
        logger.warning(f"Error validating resource '{relative_path}': {e}")
        return None

def get_resource_info(relative_path):
    """
    Get detailed information about a resource.
    
    Args:
        relative_path: Relative path to analyze
        
    Returns:
        Dictionary with resource information
    """
    info = {
        'relative_path': relative_path,
        'absolute_path': None,
        'exists': False,
        'is_file': False,
        'is_directory': False,
        'size': None,
        'error': None
    }
    
    try:
        full_path = resource_path(relative_path)
        info['absolute_path'] = full_path
        
        if os.path.exists(full_path):
            info['exists'] = True
            info['is_file'] = os.path.isfile(full_path)
            info['is_directory'] = os.path.isdir(full_path)
            
            if info['is_file']:
                try:
                    info['size'] = os.path.getsize(full_path)
                except OSError as e:
                    info['error'] = f"Could not get file size: {e}"
                    
    except Exception as e:
        info['error'] = str(e)
        logger.error(f"Error getting resource info for '{relative_path}': {e}")
    
    return info

"""
UI Responsiveness Helper

This module provides utilities to keep the UI responsive during long operations.
"""

import threading
import time
from typing import Callable, Optional


class UIResponsivenessHelper:
    """Helper class to maintain UI responsiveness during operations."""
    
    @staticmethod
    def run_with_ui_updates(operation_func: Callable, 
                           ui_update_func: Optional[Callable] = None,
                           update_interval: float = 0.1):
        """
        Run an operation while keeping the UI responsive.
        
        Args:
            operation_func: Function to run in background
            ui_update_func: Function to call for UI updates
            update_interval: How often to update UI (seconds)
        """
        result = {'completed': False, 'result': None, 'error': None}
        
        def worker():
            try:
                result['result'] = operation_func()
            except Exception as e:
                result['error'] = str(e)
            finally:
                result['completed'] = True
        
        # Start operation in background thread
        thread = threading.Thread(target=worker, daemon=True)
        thread.start()
        
        # Keep UI responsive while waiting
        while not result['completed']:
            if ui_update_func:
                try:
                    ui_update_func()
                except:
                    pass  # Ignore UI update errors
            time.sleep(update_interval)
        
        # Wait for thread to complete
        thread.join(timeout=1.0)
        
        if result['error']:
            raise Exception(result['error'])
        
        return result['result']
    
    @staticmethod
    def safe_ui_call(ui_func: Callable, *args, **kwargs):
        """Safely call a UI function, ignoring errors."""
        try:
            return ui_func(*args, **kwargs)
        except Exception as e:
            print(f"UI call error (ignored): {e}")
            return None

"""
Widget Parent Resolution Utility

This module provides utilities for safely resolving parent widgets from various
UI element types, including custom UI elements that don't follow standard tkinter
patterns.
"""

from typing import Any, Optional, List, Dict
from dataclasses import dataclass
import logging
import traceback
import sys
from datetime import datetime


@dataclass
class ParentResolutionResult:
    """Result of parent widget resolution attempt"""
    parent: Optional[Any]
    method_used: str
    success: bool
    fallback_available: bool


@dataclass
class WidgetIntrospectionInfo:
    """Information about widget structure and available attributes"""
    widget_type: str
    available_attributes: List[str]
    parent_access_methods: List[str]
    hierarchy_depth: int


def get_widget_parent(widget: Any) -> Optional[Any]:
    """
    Safely get the parent container of a widget using multiple strategies.
    
    This function attempts various methods to find the parent widget, handling
    different UI element types including custom elements that don't follow
    standard tkinter patterns.
    
    Args:
        widget: The widget to find the parent for
        
    Returns:
        Parent widget or None if not found
    """
    if widget is None:
        return None
    
    # Strategy 1: Try standard tkinter master attribute
    try:
        if hasattr(widget, 'master'):
            parent = widget.master
            if parent is not None and parent != "" and parent is not False:
                return parent
    except (AttributeError, Exception):
        pass
    
    # Strategy 2: Try winfo_parent() for tkinter widgets
    try:
        if hasattr(widget, 'winfo_parent'):
            parent_name = widget.winfo_parent()
            if parent_name and parent_name != "":
                # Get the actual parent widget from the name
                if hasattr(widget, 'nametowidget'):
                    parent = widget.nametowidget(parent_name)
                    if parent is not None:
                        return parent
    except (AttributeError, Exception):
        pass
    
    # Strategy 3: Try custom _parent attribute
    try:
        if hasattr(widget, '_parent'):
            parent = widget._parent
            if parent is not None and parent != "" and parent is not False:
                return parent
    except (AttributeError, Exception):
        pass
    
    # Strategy 4: Try container attribute (for parameter UI elements)
    try:
        if hasattr(widget, 'container'):
            parent = widget.container
            if parent is not None and parent != "" and parent is not False:
                return parent
    except (AttributeError, Exception):
        pass
    
    # Strategy 5: Try container_frame attribute (for BaseUIElement derivatives)
    try:
        if hasattr(widget, 'container_frame'):
            parent = widget.container_frame
            if parent is not None and parent != "" and parent is not False:
                return parent
    except (AttributeError, Exception):
        pass
    
    # Strategy 6: Try parent attribute (alternative naming)
    try:
        if hasattr(widget, 'parent'):
            parent = widget.parent
            if parent is not None and parent != "" and parent is not False:
                return parent
    except (AttributeError, Exception):
        pass
    
    # Strategy 7: For custom UI elements, try accessing the actual tkinter widget
    try:
        if hasattr(widget, 'widget'):
            actual_widget = widget.widget
            if actual_widget is not None and hasattr(actual_widget, 'master'):
                parent = actual_widget.master
                if parent is not None and parent != "" and parent is not False:
                    return parent
    except (AttributeError, Exception):
        pass
    
    # Strategy 8: Try winfo_parent on the wrapped widget
    try:
        if hasattr(widget, 'widget'):
            actual_widget = widget.widget
            if actual_widget is not None and hasattr(actual_widget, 'winfo_parent'):
                parent_name = actual_widget.winfo_parent()
                if parent_name and parent_name != "":
                    if hasattr(actual_widget, 'nametowidget'):
                        parent = actual_widget.nametowidget(parent_name)
                        if parent is not None:
                            return parent
    except (AttributeError, Exception):
        pass
    
    # No parent found
    return None


def get_widget_parent_with_details(widget: Any) -> ParentResolutionResult:
    """
    Get widget parent with detailed information about the resolution process.
    
    Args:
        widget: The widget to find the parent for
        
    Returns:
        ParentResolutionResult with details about the resolution attempt
    """
    if widget is None:
        return ParentResolutionResult(
            parent=None,
            method_used="none",
            success=False,
            fallback_available=False
        )
    
    # Try each strategy and record which one worked
    def _safe_get_attr(w, attr):
        """Safely get attribute and check for valid values"""
        if not hasattr(w, attr):
            return None
        value = getattr(w, attr, None)
        if value is not None and value != "" and value is not False:
            return value
        return None
    
    strategies = [
        ("master", lambda w: _safe_get_attr(w, 'master')),
        ("winfo_parent", _try_winfo_parent),
        ("_parent", lambda w: _safe_get_attr(w, '_parent')),
        ("container", lambda w: _safe_get_attr(w, 'container')),
        ("container_frame", lambda w: _safe_get_attr(w, 'container_frame')),
        ("parent", lambda w: _safe_get_attr(w, 'parent')),
        ("widget.master", _try_widget_master),
        ("widget.winfo_parent", _try_widget_winfo_parent),
    ]
    
    for method_name, strategy_func in strategies:
        try:
            parent = strategy_func(widget)
            if parent is not None:
                return ParentResolutionResult(
                    parent=parent,
                    method_used=method_name,
                    success=True,
                    fallback_available=True
                )
        except Exception:
            continue
    
    return ParentResolutionResult(
        parent=None,
        method_used="none",
        success=False,
        fallback_available=False
    )


def _try_winfo_parent(widget: Any) -> Optional[Any]:
    """Try to get parent using winfo_parent method"""
    if not hasattr(widget, 'winfo_parent'):
        return None
    
    parent_name = widget.winfo_parent()
    if not parent_name or parent_name == "":
        return None
    
    if hasattr(widget, 'nametowidget'):
        parent = widget.nametowidget(parent_name)
        if parent is not None:
            return parent
    
    return None


def _try_widget_master(widget: Any) -> Optional[Any]:
    """Try to get parent from wrapped widget's master attribute"""
    if not hasattr(widget, 'widget'):
        return None
    
    actual_widget = widget.widget
    if actual_widget is None:
        return None
    
    if hasattr(actual_widget, 'master'):
        parent = actual_widget.master
        if parent is not None and parent != "" and parent is not False:
            return parent
    
    return None


def _try_widget_winfo_parent(widget: Any) -> Optional[Any]:
    """Try to get parent from wrapped widget's winfo_parent method"""
    if not hasattr(widget, 'widget'):
        return None
    
    actual_widget = widget.widget
    if actual_widget is None:
        return None
    
    return _try_winfo_parent(actual_widget)


def inspect_widget_attributes(widget: Any) -> WidgetIntrospectionInfo:
    """
    Analyze widget structure and available attributes for debugging.
    
    Args:
        widget: The widget to inspect
        
    Returns:
        WidgetIntrospectionInfo with analysis results
    """
    if widget is None:
        return WidgetIntrospectionInfo(
            widget_type="None",
            available_attributes=[],
            parent_access_methods=[],
            hierarchy_depth=0
        )
    
    widget_type = type(widget).__name__
    available_attributes = [attr for attr in dir(widget) if not attr.startswith('__')]
    
    # Check which parent access methods are available
    parent_access_methods = []
    
    if hasattr(widget, 'master'):
        parent_access_methods.append('master')
    if hasattr(widget, 'winfo_parent'):
        parent_access_methods.append('winfo_parent')
    if hasattr(widget, '_parent'):
        parent_access_methods.append('_parent')
    if hasattr(widget, 'container'):
        parent_access_methods.append('container')
    if hasattr(widget, 'container_frame'):
        parent_access_methods.append('container_frame')
    if hasattr(widget, 'parent'):
        parent_access_methods.append('parent')
    if hasattr(widget, 'widget'):
        parent_access_methods.append('widget (wrapped)')
        # Check wrapped widget
        try:
            wrapped = widget.widget
            if wrapped and hasattr(wrapped, 'master'):
                parent_access_methods.append('widget.master')
            if wrapped and hasattr(wrapped, 'winfo_parent'):
                parent_access_methods.append('widget.winfo_parent')
        except:
            pass
    
    # Calculate hierarchy depth
    hierarchy_depth = _calculate_hierarchy_depth(widget)
    
    return WidgetIntrospectionInfo(
        widget_type=widget_type,
        available_attributes=available_attributes,
        parent_access_methods=parent_access_methods,
        hierarchy_depth=hierarchy_depth
    )


def _calculate_hierarchy_depth(widget: Any, max_depth: int = 10) -> int:
    """Calculate widget hierarchy depth"""
    depth = 0
    current = widget
    
    while current and depth < max_depth:
        parent = get_widget_parent(current)
        if parent is None or parent == current:
            break
        current = parent
        depth += 1
    
    return depth


def get_widget_hierarchy_info(widget: Any, max_levels: int = 5) -> Dict[str, Any]:
    """
    Get detailed information about widget hierarchy for debugging.
    
    Args:
        widget: The widget to analyze
        max_levels: Maximum levels to traverse up the hierarchy
        
    Returns:
        Dictionary with hierarchy information
    """
    hierarchy = []
    current = widget
    level = 0
    
    while current and level < max_levels:
        info = inspect_widget_attributes(current)
        hierarchy.append({
            'level': level,
            'widget_type': info.widget_type,
            'available_methods': info.parent_access_methods,
            'widget_id': id(current)
        })
        
        parent = get_widget_parent(current)
        if parent is None or parent == current:
            break
        
        current = parent
        level += 1
    
    return {
        'hierarchy': hierarchy,
        'total_levels': level,
        'root_found': level < max_levels
    }


def log_widget_access_debug(widget: Any, context: str = "") -> None:
    """
    Log detailed debugging information about widget access attempts.
    
    Args:
        widget: The widget being accessed
        context: Additional context for the log message
    """
    logger = logging.getLogger(__name__)
    
    if widget is None:
        logger.warning(f"Widget access debug [{context}]: Widget is None")
        return
    
    info = inspect_widget_attributes(widget)
    resolution_result = get_widget_parent_with_details(widget)
    
    logger.debug(f"Widget access debug [{context}]:")
    logger.debug(f"  Widget type: {info.widget_type}")
    logger.debug(f"  Available parent methods: {info.parent_access_methods}")
    logger.debug(f"  Resolution success: {resolution_result.success}")
    logger.debug(f"  Method used: {resolution_result.method_used}")
    logger.debug(f"  Parent found: {resolution_result.parent is not None}")
    
    if not resolution_result.success:
        logger.warning(f"Failed to resolve parent for {info.widget_type} widget in context: {context}")
        hierarchy_info = get_widget_hierarchy_info(widget)
        logger.debug(f"  Hierarchy info: {hierarchy_info}")


def log_widget_access_failure(widget: Any, context: str, exception: Exception = None, 
                             fallback_used: bool = False, fallback_description: str = "") -> None:
    """
    Log detailed information about widget access failures with comprehensive error details.
    
    Args:
        widget: The widget that failed to be accessed
        context: Context where the failure occurred
        exception: The exception that occurred (if any)
        fallback_used: Whether a fallback strategy was used
        fallback_description: Description of the fallback strategy used
    """
    logger = logging.getLogger(__name__)
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    
    # Create detailed error report
    error_report = [
        f"=== Widget Access Failure Report [{timestamp}] ===",
        f"Context: {context}",
        f"Fallback Used: {fallback_used}",
    ]
    
    if fallback_used and fallback_description:
        error_report.append(f"Fallback Strategy: {fallback_description}")
    
    # Widget information
    if widget is not None:
        try:
            info = inspect_widget_attributes(widget)
            error_report.extend([
                f"Widget Type: {info.widget_type}",
                f"Available Attributes: {len(info.available_attributes)} attributes",
                f"Parent Access Methods: {info.parent_access_methods}",
                f"Hierarchy Depth: {info.hierarchy_depth}",
            ])
            
            # Try to get more detailed widget information
            widget_details = _get_detailed_widget_info(widget)
            if widget_details:
                error_report.append(f"Widget Details: {widget_details}")
                
        except Exception as info_error:
            error_report.append(f"Failed to get widget info: {str(info_error)}")
    else:
        error_report.append("Widget: None")
    
    # Exception information
    if exception:
        error_report.extend([
            f"Exception Type: {type(exception).__name__}",
            f"Exception Message: {str(exception)}",
        ])
        
        # Add traceback if available
        if hasattr(exception, '__traceback__') and exception.__traceback__:
            tb_lines = traceback.format_tb(exception.__traceback__)
            error_report.append("Traceback:")
            error_report.extend([f"  {line.strip()}" for line in tb_lines])
    
    # Resolution attempt details
    if widget is not None:
        try:
            resolution_result = get_widget_parent_with_details(widget)
            error_report.extend([
                f"Resolution Attempted: {resolution_result.method_used}",
                f"Resolution Success: {resolution_result.success}",
                f"Fallback Available: {resolution_result.fallback_available}",
            ])
        except Exception as res_error:
            error_report.append(f"Failed to get resolution details: {str(res_error)}")
    
    error_report.append("=" * 50)
    
    # Log the complete error report
    full_report = "\n".join(error_report)
    
    if fallback_used:
        logger.warning(f"Widget access failure with fallback in context '{context}':\n{full_report}")
    else:
        logger.error(f"Widget access failure in context '{context}':\n{full_report}")
    
    # Also print to console for immediate visibility during development
    print(f"WIDGET ACCESS FAILURE [{context}]: {exception if exception else 'Unknown error'}")
    if fallback_used:
        print(f"  Fallback strategy used: {fallback_description}")


def log_widget_access_warning(widget: Any, context: str, warning_message: str, 
                             fallback_strategy: str = "") -> None:
    """
    Log warning messages for widget access issues that don't prevent functionality.
    
    Args:
        widget: The widget being accessed
        context: Context where the warning occurred
        warning_message: The warning message
        fallback_strategy: Description of fallback strategy being used
    """
    logger = logging.getLogger(__name__)
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    
    warning_details = [
        f"Widget Access Warning [{timestamp}]",
        f"Context: {context}",
        f"Warning: {warning_message}",
    ]
    
    if fallback_strategy:
        warning_details.append(f"Fallback Strategy: {fallback_strategy}")
    
    if widget is not None:
        try:
            info = inspect_widget_attributes(widget)
            warning_details.extend([
                f"Widget Type: {info.widget_type}",
                f"Available Methods: {info.parent_access_methods}",
            ])
        except Exception:
            warning_details.append("Widget Info: Unable to retrieve")
    
    full_warning = " | ".join(warning_details)
    logger.warning(full_warning)
    
    # Print abbreviated warning to console
    print(f"WARNING [{context}]: {warning_message}")
    if fallback_strategy:
        print(f"  Using fallback: {fallback_strategy}")


def log_widget_hierarchy_analysis(widget: Any, context: str = "", max_levels: int = 5) -> None:
    """
    Log detailed widget hierarchy analysis for debugging complex widget structures.
    
    Args:
        widget: The widget to analyze
        context: Context for the analysis
        max_levels: Maximum hierarchy levels to analyze
    """
    logger = logging.getLogger(__name__)
    
    if widget is None:
        logger.debug(f"Hierarchy analysis [{context}]: Widget is None")
        return
    
    try:
        hierarchy_info = get_widget_hierarchy_info(widget, max_levels)
        
        analysis_report = [
            f"=== Widget Hierarchy Analysis [{context}] ===",
            f"Total Levels Found: {hierarchy_info['total_levels']}",
            f"Root Found: {hierarchy_info['root_found']}",
            "Hierarchy Details:",
        ]
        
        for level_info in hierarchy_info['hierarchy']:
            level = level_info['level']
            widget_type = level_info['widget_type']
            methods = level_info['available_methods']
            widget_id = level_info['widget_id']
            
            analysis_report.append(
                f"  Level {level}: {widget_type} (ID: {widget_id}) - Methods: {methods}"
            )
        
        analysis_report.append("=" * 50)
        
        full_analysis = "\n".join(analysis_report)
        logger.debug(full_analysis)
        
    except Exception as e:
        logger.error(f"Failed to analyze widget hierarchy for context '{context}': {str(e)}")


def _get_detailed_widget_info(widget: Any) -> str:
    """
    Get detailed information about a widget for error reporting.
    
    Args:
        widget: The widget to analyze
        
    Returns:
        String with detailed widget information
    """
    details = []
    
    try:
        # Basic widget information
        details.append(f"Type: {type(widget).__name__}")
        details.append(f"ID: {id(widget)}")
        
        # Try to get widget name/title if available
        for attr in ['_name', 'name', 'title', '_title']:
            if hasattr(widget, attr):
                try:
                    value = getattr(widget, attr)
                    if value:
                        details.append(f"{attr}: {value}")
                        break
                except Exception:
                    continue
        
        # Check for common widget properties
        for prop in ['width', 'height', 'x', 'y']:
            if hasattr(widget, prop):
                try:
                    value = getattr(widget, prop)
                    details.append(f"{prop}: {value}")
                except Exception:
                    continue
        
        # Check if widget is visible/mapped
        for visibility_attr in ['winfo_viewable', 'winfo_ismapped', 'winfo_exists']:
            if hasattr(widget, visibility_attr):
                try:
                    value = getattr(widget, visibility_attr)()
                    details.append(f"{visibility_attr}: {value}")
                except Exception:
                    continue
        
        return " | ".join(details) if details else "No details available"
        
    except Exception as e:
        return f"Error getting details: {str(e)}"


def ensure_application_continues(operation_name: str, critical: bool = False) -> callable:
    """
    Decorator to ensure application continues functioning even when UI enhancements fail.
    
    Args:
        operation_name: Name of the operation being performed
        critical: Whether this operation is critical to application functionality
        
    Returns:
        Decorator function
    """
    def decorator(func):
        def wrapper(*args, **kwargs):
            try:
                return func(*args, **kwargs)
            except Exception as e:
                # Log the error with full details
                context = f"{operation_name}_operation"
                
                # Get widget from args if available
                widget = None
                if args and hasattr(args[0], '__class__'):
                    # First arg might be self, second might be widget
                    if len(args) > 1:
                        widget = args[1]
                    elif len(args) > 0 and hasattr(args[0], 'widget'):
                        widget = args[0].widget
                
                log_widget_access_failure(
                    widget=widget,
                    context=context,
                    exception=e,
                    fallback_used=True,
                    fallback_description="Operation skipped to maintain application stability"
                )
                
                # Also report to error handling service if available
                try:
                    from services.widget_error_handling_service import widget_error_service, ErrorSeverity
                    widget_error_service.report_widget_error(
                        widget=widget,
                        context=context,
                        exception=e,
                        severity=ErrorSeverity.CRITICAL if critical else ErrorSeverity.WARNING,
                        fallback_used=True,
                        fallback_description="Operation skipped to maintain application stability",
                        user_impact="UI enhancement unavailable but core functionality preserved"
                    )
                except ImportError:
                    pass  # Error service not available
                
                if critical:
                    # For critical operations, re-raise the exception
                    raise
                else:
                    # For non-critical operations, return None and continue
                    print(f"Non-critical operation '{operation_name}' failed but application continues")
                    return None
        
        return wrapper
    return decorator
