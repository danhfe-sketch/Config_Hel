        'heartbeat_ign_off': '300',

        'timer_ign_off': '300',
        'vibration': '128',
        'sensitivity': '5',
        'movement_alert': 'ON',
        'power_alert': 'ON',
        'battery_alert': 'ON',

        'protocol': '2',
        'tcp_ack': 'OFF',
        'GT06SEL': '2',  # Default to CRX protocol
        'POSBYACK': '0'  # Default to disabled TCP ACK

    def __init__(self):
        self.validation_issues: List[ValidationIssue] = []

    def validate_profile_file(self, file_path: str) -> Tuple[bool, List[ValidationIssue]]:
        """
        Validate a complete profile file.
        
        Args:
            file_path: Path to the profile file to validate
            
        Returns:
            Tuple of (is_valid, list_of_issues)
        """
        self.validation_issues.clear()
        
        if not os.path.exists(file_path):
            self.validation_issues.append(ValidationIssue(
                ValidationSeverity.ERROR, "File", "existence",
                f"Profile file does not exist: {file_path}"
            ))
            return False, self.validation_issues
        
        try:
            config = configparser.ConfigParser()
            config.read(file_path, encoding='utf-8')
            
            self._validate_file_format(config)
            self._validate_required_sections(config)
            self._validate_parameter_values(config)
            self._check_parameter_completeness(config)
            
        except Exception as e:
            self.validation_issues.append(ValidationIssue(
                ValidationSeverity.ERROR, "File", "parsing",
                f"Failed parsing profile file: {str(e)}"
            ))
            return False, self.validation_issues
        
        has_errors = any(issue.severity == ValidationSeverity.ERROR for issue in self.validation_issues)
        return not has_errors, self.validation_issues

    def _validate_file_format(self, config: configparser.ConfigParser) -> None:
        """Validate basic file format."""
        if not config.sections():
            self.validation_issues.append(ValidationIssue(
                ValidationSeverity.ERROR, "File", "format",
                "Profile file contains no sections"
            ))

    def _validate_required_sections(self, config: configparser.ConfigParser) -> None:
        """Validate required sections and parameters."""
        for section_name, required_params in self.REQUIRED_SECTIONS.items():
            if not config.has_section(section_name):
                self.validation_issues.append(ValidationIssue(
                    ValidationSeverity.ERROR, section_name, "section",
                    f"Required section missing: {section_name}"
                ))
                continue
            
            section = config[section_name]
            for param in required_params:
                if param not in section:
                    self.validation_issues.append(ValidationIssue(
                        ValidationSeverity.ERROR, section_name, param,
                        f"Required parameter missing: {param}"
                    ))

    def _validate_parameter_values(self, config: configparser.ConfigParser) -> None:
        """Validate parameter values."""
        # Create a case-insensitive mapping of validation rules
        validation_rules_lower = {k.lower(): k for k in self.VALIDATION_RULES.keys()}
        
        for section_name in config.sections():
            if section_name not in self.REQUIRED_SECTIONS and section_name not in self.OPTIONAL_SECTIONS:
                continue
                
            section = config[section_name]
            for param_name, value in section.items():
                # Check for case-insensitive match
                if param_name in validation_rules_lower:
                    # Use the original case name for validation
                    original_param_name = validation_rules_lower[param_name]
                    self._validate_single_parameter(section_name, original_param_name, value)

    def _validate_single_parameter(self, section: str, param: str, value: str) -> None:
        """Validate a single parameter value."""
        if not value.strip():
            if self.VALIDATION_RULES[param].get('required', False):
                self.validation_issues.append(ValidationIssue(
                    ValidationSeverity.ERROR, section, param,
                    f"Required parameter cannot be empty: {param}"
                ))
            return
        
        rules = self.VALIDATION_RULES[param]
        param_type = rules['type']
        
        try:
            if param_type == 'ip_address':
                ipaddress.ip_address(value)
            elif param_type == 'port':
                port = int(value)
                min_val, max_val = rules['range']
                if not (min_val <= port <= max_val):
                    raise ValueError(f"Port must be between {min_val} and {max_val}")
            elif param_type == 'integer':
                int_val = int(value)
                if 'range' in rules:
                    min_val, max_val = rules['range']
                    if not (min_val <= int_val <= max_val):
                        raise ValueError(f"Value must be between {min_val} and {max_val}")
            elif param_type == 'boolean':
                if value.upper() not in ['ON', 'OFF', 'TRUE', 'FALSE', 'ENABLED', 'DISABLED']:
                    raise ValueError("Boolean value must be ON/OFF, TRUE/FALSE, or ENABLED/DISABLED")
            elif param_type == 'enum':
                if value not in rules['values']:
                    raise ValueError(f"Value must be one of: {', '.join(rules['values'])}")
            elif param_type == 'string':
                if 'max_length' in rules and len(value) > rules['max_length']:
                    raise ValueError(f"String too long (max {rules['max_length']} characters)")
                if 'pattern' in rules and not re.match(rules['pattern'], value):
                    raise ValueError(f"String does not match required pattern")
            elif param_type == 'imei':
                if not self._validate_imei(value):
                    raise ValueError("Invalid IMEI format (must be 15 digits)")
            elif param_type == 'iccid':
                if not self._validate_iccid(value):
                    raise ValueError("Invalid ICCID format (must be 19-20 digits)")
            elif param_type == 'version':
                if not re.match(r'^\d+\.\d+(\.\d+)?$', value):
                    raise ValueError("Invalid version format (expected X.Y or X.Y.Z)")
                    
        except ValueError as e:
            self.validation_issues.append(ValidationIssue(
                ValidationSeverity.ERROR, section, param,
                f"Invalid value for {param}: {str(e)}", current_value=value
            ))

    def _validate_imei(self, imei: str) -> bool:
        """Validate IMEI format."""
        return re.match(r'^\d{15}$', imei) is not None

    def _validate_iccid(self, iccid: str) -> bool:
        """Validate ICCID format."""
        return re.match(r'^\d{19,20}$', iccid) is not None

    def _check_parameter_completeness(self, config: configparser.ConfigParser) -> None:
        """Check for missing optional parameters."""
        # Check optional sections
        for section_name, params in self.OPTIONAL_SECTIONS.items():
            if not config.has_section(section_name):
                self.validation_issues.append(ValidationIssue(
                    ValidationSeverity.INFO, section_name, "section",
                    f"Optional section missing: {section_name}"
                ))
                continue
            
            section = config[section_name]
            for param in params:
                if param not in section and param in self.DEFAULT_VALUES:
                    self.validation_issues.append(ValidationIssue(
                        ValidationSeverity.INFO, section_name, param,
                        f"Optional parameter missing: {param}",
                        suggested_value=self.DEFAULT_VALUES[param]
                    ))
        
        # Check required sections for optional parameters with default values
        for section_name, params in self.REQUIRED_SECTIONS.items():
            if config.has_section(section_name):
                section = config[section_name]
                # Get all parameters that could be in this section
                all_section_params = []
                if section_name == 'Profile_Info':
                    all_section_params = ['profile_version']  # Only profile_version has a default
                elif section_name == 'APN_Configuration':
                    all_section_params = ['apn_user', 'apn_password']
                
                for param in all_section_params:
                    if param not in section and param in self.DEFAULT_VALUES:
                        self.validation_issues.append(ValidationIssue(
                            ValidationSeverity.INFO, section_name, param,
                            f"Optional parameter missing: {param}",
                            suggested_value=self.DEFAULT_VALUES[param]
                        ))

    def get_validation_summary(self) -> Dict[str, int]:

"""
Non-Blocking Profile Manager - Executes profile operations in background threads.

This module provides the NonBlockingProfileManager class that executes profile operations
in background threads to maintain UI responsiveness. It includes thread-safe progress
callbacks, operation cancellation, and proper cleanup.

Key Features:
- Background thread execution for all profile operations
- Thread-safe progress updates using tkinter's after() method
- Operation cancellation with proper cleanup and thread termination
- Operation context management for maintaining state across threads
- Comprehensive error handling and recovery
"""

import threading
import time
import logging
from datetime import datetime
from dataclasses import dataclass
from typing import Dict, Any, Optional, Callable, List
from enum import Enum


class OperationType(Enum):
    """Enumeration of supported profile operation types."""
    LOAD_FROM_DEVICE = "load_from_device"
    LOAD_TO_DEVICE = "load_to_device"
    LOAD_TO_UI = "load_to_ui"
    SAVE_FROM_UI = "save_from_ui"


@dataclass
class OperationContext:
    """Context information for profile operations."""
    operation_type: OperationType
    operation_id: str
    file_path: Optional[str] = None
    profile_name: Optional[str] = None
    parameters: Optional[Dict[str, str]] = None
    progress_callback: Optional[Callable] = None
    completion_callback: Optional[Callable] = None
    cancellation_token: Optional[threading.Event] = None
    start_time: Optional[datetime] = None
    metadata: Optional[Dict[str, Any]] = None


@dataclass
class ProgressInfo:
    """Progress information for profile operations."""
    operation_id: str
    operation_type: OperationType
    current_step: int
    total_steps: int
    current_parameter: str
    status_message: str
    can_cancel: bool
    is_complete: bool
    success: bool = False
    error_message: Optional[str] = None


@dataclass
class OperationResult:
    """Result of a profile operation."""
    operation_id: str
    operation_type: OperationType
    success: bool
    error_message: Optional[str] = None
    parameters_processed: int = 0
    failed_parameters: List[str] = None
    operation_duration: float = 0.0
    metadata: Optional[Dict[str, Any]] = None


class NonBlockingProfileManager:
    """
    Non-blocking profile manager that executes operations in background threads.
    
    This class provides thread-safe execution of profile operations while maintaining
    UI responsiveness through background thread execution and thread-safe callbacks.
    """
    
    def __init__(self, app_instance):
        """
        Initialize the non-blocking profile manager.
        
        Args:
            app_instance: Main application instance with UI and services
        """
        self.app = app_instance
        self.i18n = getattr(app_instance, 'i18n_service', None) or getattr(app_instance, 'i18n', None)
        
        # Initialize logger
        self.logger = logging.getLogger(__name__)
        
        # Operation management
        self.active_operations: Dict[str, OperationContext] = {}
        self.operation_threads: Dict[str, threading.Thread] = {}
        self.operation_counter = 0
        self.operation_lock = threading.Lock()
        
        # Default callbacks
        self.default_progress_callback: Optional[Callable] = None
        self.default_completion_callback: Optional[Callable] = None
        
        # Operation timeout settings (in seconds)
        self.operation_timeouts = {
            OperationType.LOAD_FROM_DEVICE: 300,  # 5 minutes
            OperationType.LOAD_TO_DEVICE: 300,    # 5 minutes
            OperationType.LOAD_TO_UI: 30,         # 30 seconds
            OperationType.SAVE_FROM_UI: 30        # 30 seconds
        }
        
        # Optimized thread pool settings
        import os
        cpu_count = os.cpu_count() or 2
        self.max_concurrent_operations = min(4, max(2, cpu_count // 2))
        
        # Performance optimization settings
        self.progress_update_interval = 0.2  # 200ms between updates
        self.last_progress_update = {}  # Track last update time per operation
        self.memory_cleanup_threshold = 100  # MB
        self.parameter_batch_size = 10  # Process parameters in batches
        
        # Performance monitoring
        self.performance_monitor = None
        try:
            from .profile_performance_monitor import ProfilePerformanceMonitor
            self.performance_monitor = ProfilePerformanceMonitor()
            self.performance_monitor.start_monitoring()
        except ImportError:
            self.logger.warning("Performance monitor not available")
        
        self.logger.info(f"NonBlockingProfileManager initialized with {self.max_concurrent_operations} max concurrent operations")
    
    def execute_profile_operation(self, operation_type: OperationType, **kwargs) -> str:
        """
        Execute a profile operation in a background thread with performance monitoring.
        
        Args:
            operation_type: Type of operation to execute
            **kwargs: Operation-specific arguments
            
        Returns:
            str: Operation ID for tracking the operation
            
        Raises:
            RuntimeError: If too many operations are running concurrently
        """
        # Check concurrent operation limit
        with self.operation_lock:
            if len(self.active_operations) >= self.max_concurrent_operations:
                # Clean up completed operations first
                self.cleanup_completed_operations()
                
                # Check again after cleanup
                if len(self.active_operations) >= self.max_concurrent_operations:
                    raise RuntimeError(f"Maximum concurrent operations ({self.max_concurrent_operations}) exceeded")
            
            # Generate unique operation ID
            self.operation_counter += 1
            operation_id = f"{operation_type.value}_{self.operation_counter}_{int(time.time())}"
        
        # Create operation context
        context = OperationContext(
            operation_type=operation_type,
            operation_id=operation_id,
            file_path=kwargs.get('file_path'),
            profile_name=kwargs.get('profile_name'),
            parameters=kwargs.get('parameters'),
            progress_callback=kwargs.get('progress_callback', self.default_progress_callback),
            completion_callback=kwargs.get('completion_callback', self.default_completion_callback),
            cancellation_token=threading.Event(),
            start_time=datetime.now(),
            metadata=kwargs.get('metadata', {})
        )
        
        # Store operation context
        with self.operation_lock:
            self.active_operations[operation_id] = context
        
        # Start performance monitoring for this operation
        if self.performance_monitor:
            expected_params = kwargs.get('expected_parameters', 0)
            self.performance_monitor.start_operation_monitoring(
                operation_id, operation_type.value, expected_params
            )
        
        # Create and start operation thread
        operation_thread = threading.Thread(
            target=self._execute_operation_thread,
            args=(context,),
            name=f"ProfileOperation_{operation_id}",
            daemon=True
        )
        
        self.operation_threads[operation_id] = operation_thread
        operation_thread.start()
        
        self.logger.info(f"Started profile operation: {operation_id} ({operation_type.value})")
        return operation_id
    
    def cancel_operation(self, operation_id: str) -> bool:
        """
        Cancel a running profile operation.
        
        Args:
            operation_id: ID of the operation to cancel
            
        Returns:
            bool: True if cancellation was initiated, False if operation not found
        """
        with self.operation_lock:
            context = self.active_operations.get(operation_id)
            if not context:
                self.logger.warning(f"Cannot cancel operation {operation_id}: not found")
                return False
            
            # Signal cancellation
            if context.cancellation_token:
                context.cancellation_token.set()
                self.logger.info(f"Cancellation requested for operation: {operation_id}")
                return True
            
        return False
    
    def get_operation_status(self, operation_id: str) -> Optional[OperationContext]:
        """
        Get the status of a profile operation.
        
        Args:
            operation_id: ID of the operation
            
        Returns:
            OperationContext or None if operation not found
        """
        with self.operation_lock:
            return self.active_operations.get(operation_id)
    
    def get_active_operations(self) -> List[str]:
        """
        Get list of active operation IDs.
        
        Returns:
            List of active operation IDs
        """
        with self.operation_lock:
            return list(self.active_operations.keys())
    
    def set_progress_callback(self, callback: Callable[[ProgressInfo], None]) -> None:
        """
        Set the default progress callback for all operations.
        
        Args:
            callback: Function to call with progress updates
        """
        self.default_progress_callback = callback
        self.logger.debug("Default progress callback set")
    
    def set_completion_callback(self, callback: Callable[[OperationResult], None]) -> None:
        """
        Set the default completion callback for all operations.
        
        Args:
            callback: Function to call when operations complete
        """
        self.default_completion_callback = callback
        self.logger.debug("Default completion callback set")
    
    def cleanup_completed_operations(self) -> int:
        """
        Clean up completed operation threads and contexts.
        
        Returns:
            int: Number of operations cleaned up
        """
        cleaned_count = 0
        
        with self.operation_lock:
            completed_operations = []
            
            for operation_id, thread in self.operation_threads.items():
                if not thread.is_alive():
                    completed_operations.append(operation_id)
            
            for operation_id in completed_operations:
                # Remove from active operations
                if operation_id in self.active_operations:
                    del self.active_operations[operation_id]
                
                # Remove thread reference
                if operation_id in self.operation_threads:
                    del self.operation_threads[operation_id]
                
                cleaned_count += 1
        
        if cleaned_count > 0:
            self.logger.debug(f"Cleaned up {cleaned_count} completed operations")
        
        return cleaned_count
    
    def shutdown(self, timeout: float = 30.0) -> bool:
        """
        Shutdown the manager and wait for all operations to complete.
        
        Args:
            timeout: Maximum time to wait for operations to complete
            
        Returns:
            bool: True if all operations completed within timeout
        """
        self.logger.info("Shutting down NonBlockingProfileManager")
        
        # Cancel all active operations
        with self.operation_lock:
            operation_ids = list(self.active_operations.keys())
        
        for operation_id in operation_ids:
            self.cancel_operation(operation_id)
        
        # Wait for all threads to complete
        start_time = time.time()
        while time.time() - start_time < timeout:
            with self.operation_lock:
                if not self.operation_threads:
                    break
                
                # Check if any threads are still alive
                alive_threads = [t for t in self.operation_threads.values() if t.is_alive()]
                if not alive_threads:
                    break
            
            time.sleep(0.1)
        
        # Force cleanup
        self.cleanup_completed_operations()
        
        # Check if shutdown was successful
        with self.operation_lock:
            remaining_operations = len(self.active_operations)
            if remaining_operations > 0:
                self.logger.warning(f"Shutdown incomplete: {remaining_operations} operations still active")
                return False
        
        # Stop performance monitoring
        if self.performance_monitor:
            try:
                self.performance_monitor.stop_monitoring()
            except Exception as e:
                self.logger.error(f"Error stopping performance monitor: {e}")
        
        self.logger.info("NonBlockingProfileManager shutdown complete")
        return True
    
    def _execute_operation_thread(self, context: OperationContext) -> None:
        """
        Execute a profile operation in a background thread.
        
        Args:
            context: Operation context with all necessary information
        """
        operation_id = context.operation_id
        operation_type = context.operation_type
        
        try:
            self.logger.info(f"Executing operation {operation_id} ({operation_type.value})")
            
            # Execute the appropriate operation
            if operation_type == OperationType.LOAD_FROM_DEVICE:
                result = self._execute_load_from_device(context)
            elif operation_type == OperationType.LOAD_TO_DEVICE:
                result = self._execute_load_to_device(context)
            elif operation_type == OperationType.LOAD_TO_UI:
                result = self._execute_load_to_ui(context)
            elif operation_type == OperationType.SAVE_FROM_UI:
                result = self._execute_save_from_ui(context)
            else:
                raise ValueError(f"Unsupported operation type: {operation_type}")
            
            # Calculate operation duration
            if context.start_time:
                result.operation_duration = (datetime.now() - context.start_time).total_seconds()
            
            self.logger.info(f"Operation {operation_id} completed: success={result.success}")
            
        except Exception as e:
            self.logger.error(f"Operation {operation_id} failed with exception: {str(e)}")
            
            # Create error result
            result = OperationResult(
                operation_id=operation_id,
                operation_type=operation_type,
                success=False,
                error_message=str(e),
                operation_duration=(datetime.now() - context.start_time).total_seconds() if context.start_time else 0.0
            )
        
        finally:
            # Finish performance monitoring
            if self.performance_monitor:
                try:
                    performance_profile = self.performance_monitor.finish_operation_monitoring(
                        operation_id, result.success, result.error_message
                    )
                    
                    # Log performance summary
                    self.logger.info(f"Operation {operation_id} performance: "
                                   f"duration={performance_profile.total_duration:.2f}s, "
                                   f"params={performance_profile.parameters_processed}, "
                                   f"rate={performance_profile.parameters_per_second:.2f} params/sec, "
                                   f"peak_memory={performance_profile.peak_memory_usage:.1f}MB")
                    
                    # Log optimization recommendations
                    if performance_profile.optimization_recommendations:
                        self.logger.info(f"Optimization recommendations for {operation_id}: "
                                       f"{'; '.join(performance_profile.optimization_recommendations)}")
                        
                except Exception as perf_error:
                    self.logger.error(f"Performance monitoring error for {operation_id}: {perf_error}")
            
            # Clean up progress update tracking
            if operation_id in self.last_progress_update:
                del self.last_progress_update[operation_id]
            
            # Call completion callback on main thread if provided
            if context.completion_callback:
                self._schedule_completion_callback(context.completion_callback, result)
            
            # Note: Don't automatically clean up operation context here
            # Let cleanup_completed_operations() handle it to allow status checking
    
    def _execute_load_from_device(self, context: OperationContext) -> OperationResult:
        """
        Execute load from device operation.
        
        Args:
            context: Operation context
            
        Returns:
            OperationResult with operation results
        """
        operation_id = context.operation_id
        file_path = context.file_path
        
        if not file_path:
            raise ValueError("file_path is required for load_from_device operation")
        
        # Check if operation was cancelled before starting
        if context.cancellation_token and context.cancellation_token.is_set():
            return OperationResult(
                operation_id=operation_id,
                operation_type=OperationType.LOAD_FROM_DEVICE,
                success=False,
                error_message="Operation cancelled before execution"
            )
        
        # Send initial progress update
        self._send_progress_update(context, 0, 100, "", "Initializing device read...")
        
        try:
            # Check device connection
            if not self._is_device_connected():
                raise RuntimeError("Device not connected")
            
            self._send_progress_update(context, 5, 100, "", "Reading device parameters...")
            
            # Get all parameter commands
            if not hasattr(self.app, 'configuration_reader_service') or not self.app.configuration_reader_service:
                raise RuntimeError("Configuration reader service not available")
            
            config_reader = self.app.configuration_reader_service
            all_commands = config_reader._get_all_parameter_commands()
            
            if not all_commands:
                raise RuntimeError("No parameter commands available")
            
            # Read parameters with optimized batching and progress updates
            parameters = {}
            successful_reads = 0
            failed_parameters = []
            total_commands = len(all_commands)
            
            # Process parameters in optimized batches
            for batch_start in range(0, total_commands, self.parameter_batch_size):
                batch_end = min(batch_start + self.parameter_batch_size, total_commands)
                batch_commands = all_commands[batch_start:batch_end]
                
                # Check for cancellation before each batch
                if context.cancellation_token and context.cancellation_token.is_set():
                    return OperationResult(
                        operation_id=operation_id,
                        operation_type=OperationType.LOAD_FROM_DEVICE,
                        success=False,
                        error_message="Operation cancelled by user",
                        parameters_processed=successful_reads,
                        failed_parameters=failed_parameters
                    )
                
                # Process batch
                for i, (param_name, command, command_type) in enumerate(batch_commands):
                    param_index = batch_start + i
                    
                    # Update progress (optimized frequency)
                    progress = 10 + int((param_index / total_commands) * 80)  # 10-90% for parameter reading
                    self._send_progress_update(context, progress, 100, param_name, f"Reading parameter: {param_name}")
                    
                    # Record parameter processing start time
                    param_start_time = time.time()
                    
                    try:
                        result = config_reader.read_specific_parameter(param_name, command, command_type, max_retries=1)
                        
                        if result.success and result.value is not None:
                            parameters[param_name] = str(result.value)
                            successful_reads += 1
                        else:
                            failed_parameters.append(param_name)
                            parameters[param_name] = ""  # Store empty value for failed parameters
                            
                    except Exception as param_error:
                        self.logger.warning(f"Failed to read parameter {param_name}: {str(param_error)}")
                        failed_parameters.append(param_name)
                        parameters[param_name] = ""
                    
                    # Record parameter processing time for performance monitoring
                    param_processing_time = time.time() - param_start_time
                    if self.performance_monitor:
                        self.performance_monitor.record_parameter_processed(
                            operation_id, param_name, param_processing_time
                        )
                
                # Trigger memory cleanup if needed
                if self.performance_monitor and self.performance_monitor.should_trigger_memory_cleanup():
                    self.performance_monitor.trigger_memory_cleanup()
                
                # Small delay between batches to prevent overwhelming the device
                if batch_end < total_commands:
                    time.sleep(0.05)  # 50ms delay between batches
            
            # Save to profile file using SimplifiedProfileFileHandler
            self._send_progress_update(context, 90, 100, "", "Saving profile file...")
            
            # Use SimplifiedProfileFileHandler to write the parameters directly
            from services.simplified_profile_file_handler import SimplifiedProfileFileHandler
            file_handler = SimplifiedProfileFileHandler()
            
            success, error_message = file_handler.write_universal_profile(file_path, parameters)
            
            if not success:
                raise RuntimeError(f"Failed to save profile file: {error_message}")
            
            self.logger.info(f"Successfully saved profile file: {file_path} with {len(parameters)} parameters")
            
            # Final progress update
            self._send_progress_update(context, 100, 100, "", "Operation completed successfully")
            
            return OperationResult(
                operation_id=operation_id,
                operation_type=OperationType.LOAD_FROM_DEVICE,
                success=True,
                parameters_processed=successful_reads,
                failed_parameters=failed_parameters,
                metadata={'total_parameters': total_commands, 'file_path': file_path}
            )
            
        except Exception as e:
            self._send_progress_update(context, 0, 100, "", f"Error: {str(e)}")
            raise
    
    def _execute_load_to_device(self, context: OperationContext) -> OperationResult:
        """
        Execute load to device operation.
        
        Args:
            context: Operation context
            
        Returns:
            OperationResult with operation results
        """
        operation_id = context.operation_id
        file_path = context.file_path
        parameters = context.parameters
        
        if not file_path and not parameters:
            raise ValueError("Either file_path or parameters is required for load_to_device operation")
        
        # Check if operation was cancelled before starting
        if context.cancellation_token and context.cancellation_token.is_set():
            return OperationResult(
                operation_id=operation_id,
                operation_type=OperationType.LOAD_TO_DEVICE,
                success=False,
                error_message="Operation cancelled before execution"
            )
        
        # Send initial progress update
        self._send_progress_update(context, 0, 100, "", "Loading profile file...")
        
        try:
            # Check device connection
            if not self._is_device_connected():
                raise RuntimeError("Device not connected")
            
            # If parameters not provided, read from file
            if not parameters:
                self._send_progress_update(context, 10, 100, "", "Reading profile file...")
                
                # Use SimplifiedProfileFileHandler to read the profile
                from services.simplified_profile_file_handler import SimplifiedProfileFileHandler
                file_handler = SimplifiedProfileFileHandler()
                parameters, error = file_handler.read_universal_profile(file_path)
                
                if error:
                    raise RuntimeError(f"Failed to read profile file: {error}")
                
                if not parameters:
                    raise RuntimeError("No parameters found in profile file")
            
            self._send_progress_update(context, 20, 100, "", "Preparing device communication...")
            
            # Get device communication manager directly
            if not hasattr(self.app, 'device_communication_manager') or not self.app.device_communication_manager:
                raise RuntimeError("Device communication manager not available")
            
            device_manager = self.app.device_communication_manager
            
            # Create progress callback for device operations
            def device_progress_callback(current_step, total_steps, current_param, status_msg):
                # Map device progress to overall progress (20-95%)
                if total_steps > 0:
                    device_progress = (current_step / total_steps) * 75  # 75% of total progress
                    overall_progress = 20 + device_progress
                    self._send_progress_update(context, int(overall_progress), 100, current_param, status_msg)
            
            self._send_progress_update(context, 25, 100, "", "Writing parameters to device...")
            
            # Write parameters to device using DeviceCommunicationManager
            import asyncio
            loop = asyncio.new_event_loop()
            asyncio.set_event_loop(loop)
            
            try:
                write_results = loop.run_until_complete(
                    device_manager.write_all_parameters(parameters, device_progress_callback)
                )
                
                # Check for cancellation after device operation
                if context.cancellation_token and context.cancellation_token.is_set():
                    return OperationResult(
                        operation_id=operation_id,
                        operation_type=OperationType.LOAD_TO_DEVICE,
                        success=False,
                        error_message="Operation cancelled during device writing",
                        parameters_processed=len([r for r in write_results.values() if r])
                    )
                
                # Process results
                successful_writes = sum(1 for success in write_results.values() if success)
                failed_parameters = [param for param, success in write_results.items() if not success]
                
                self._send_progress_update(context, 95, 100, "", "Finalizing operation...")
                
                # Determine success based on write results
                if successful_writes > 0:
                    self._send_progress_update(context, 100, 100, "", f"Successfully wrote {successful_writes}/{len(parameters)} parameters")
                    
                    return OperationResult(
                        operation_id=operation_id,
                        operation_type=OperationType.LOAD_TO_DEVICE,
                        success=True,
                        parameters_processed=successful_writes,
                        failed_parameters=failed_parameters,
                        metadata={'file_path': file_path, 'total_parameters': len(parameters)}
                    )
                else:
                    raise RuntimeError(f"Failed to write any parameters to device. Failed parameters: {failed_parameters}")
                    
            finally:
                loop.close()
                
        except Exception as e:
            self._send_progress_update(context, 0, 100, "", f"Error: {str(e)}")
            raise

    def _send_progress_update(self, context: OperationContext, current_step: int, total_steps: int, 
                            current_parameter: str, status_message: str) -> None:
        """
        Send progress update using thread-safe callback with optimized frequency.
        
        Args:
            context: Operation context
            current_step: Current step number
            total_steps: Total number of steps
            current_parameter: Name of current parameter being processed
            status_message: Status message to display
        """
        if not context.progress_callback:
            return
        
        # Optimize progress update frequency to prevent UI flooding
        current_time = time.time()
        operation_id = context.operation_id
        
        # Check if enough time has passed since last update
        if operation_id in self.last_progress_update:
            time_since_last = current_time - self.last_progress_update[operation_id]
            if time_since_last < self.progress_update_interval and current_step < total_steps:
                return  # Skip this update to prevent flooding
        
        self.last_progress_update[operation_id] = current_time
        
        # Record progress update for performance monitoring
        if self.performance_monitor:
            self.performance_monitor.record_progress_update(operation_id, current_parameter)
        
        # Create progress info
        progress_info = ProgressInfo(
            operation_id=context.operation_id,
            operation_type=context.operation_type,
            current_step=current_step,
            total_steps=total_steps,
            current_parameter=current_parameter,
            status_message=status_message,
            can_cancel=True,
            is_complete=(current_step >= total_steps),
            success=(current_step >= total_steps)
        )
        
        # Schedule callback on main thread
        self._schedule_progress_callback(context.progress_callback, progress_info)
    
    def _schedule_progress_callback(self, callback: Callable, progress_info: ProgressInfo) -> None:
        """
        Schedule progress callback on main thread using tkinter's after() method.
        
        Args:
            callback: Progress callback function
            progress_info: Progress information to send
        """
        try:
            if hasattr(self.app, 'root') and self.app.root and hasattr(self.app.root, 'after_idle'):
                self.app.root.after_idle(lambda: self._safe_callback_execution(callback, progress_info))
            else:
                # Fallback: call directly (not thread-safe but better than nothing)
                self._safe_callback_execution(callback, progress_info)
        except Exception as e:
            self.logger.error(f"Error scheduling progress callback: {str(e)}")
    
    def _schedule_completion_callback(self, callback: Callable, result: OperationResult) -> None:
        """
        Schedule completion callback on main thread using tkinter's after() method.
        
        Args:
            callback: Completion callback function
            result: Operation result to send
        """

                self._safe_callback_execution(callback, result)
        except Exception as e:

            *args: Arguments to pass to callback

            callback(*args)

    def create_progress_dialog(self, title: str, operation_name: str, 
                              progress_type: str = PROGRESS_TYPE_DETERMINATE,
                              total_steps: int = 100,
                              cancelable: bool = True) -> str:
        """
        Create a progress dialog for a long-running operation.
        
        Args:
            title: The dialog title
            operation_name: Name of the operation
            progress_type: Type of progress indicator (determinate/indeterminate)
            total_steps: Total number of steps for determinate progress
            cancelable: Whether the operation can be canceled
            
        Returns:
            Dialog ID for updating the progress
        """
        # Generate a unique ID for this dialog
        dialog_id = f"progress_{len(self.progress_dialogs) + 1}_{int(time.time())}"
        
        # Create the dialog window
        dialog = ctk.CTkToplevel(self.app)
        dialog.title(title)
        dialog.geometry("400x170")
        dialog.resizable(False, False)
        dialog.transient(self.app)  # Make it float on top of the main window
        dialog.grab_set()  # Make it modal
        
        # Center the dialog on the main window
        if hasattr(self.app, 'winfo_x') and hasattr(self.app, 'winfo_y'):
            x = self.app.winfo_x() + (self.app.winfo_width() // 2) - 200
            y = self.app.winfo_y() + (self.app.winfo_height() // 2) - 75
            dialog.geometry(f"+{x}+{y}")
        
        # Create dialog content
        dialog.grid_columnconfigure(0, weight=1)
        
        # Operation name label
        operation_label = ctk.CTkLabel(dialog, text=operation_name, font=ctk.CTkFont(size=14, weight="bold"))
        operation_label.grid(row=0, column=0, padx=20, pady=(20, 10), sticky="w")
        
        # Progress bar
        progress_bar = ctk.CTkProgressBar(dialog)
        progress_bar.grid(row=1, column=0, padx=20, pady=(0, 10), sticky="ew")
        
        if progress_type == self.PROGRESS_TYPE_INDETERMINATE:
            # For indeterminate progress, we'll animate the progress bar
            progress_bar.set(0)
            self.stop_progress[dialog_id] = False
            self._start_indeterminate_animation(dialog_id, progress_bar)
        else:
            # For determinate progress, initialize to 0
            progress_bar.set(0)
        
        # Status label
        status_label = ctk.CTkLabel(dialog, text="Starting operation...")
        status_label.grid(row=2, column=0, padx=20, pady=(0, 10), sticky="w")

        cancel_button = None
        if cancelable:
            cancel_button = ctk.CTkButton(
                dialog, 
                text="Cancel", 
                command=lambda: self._cancel_operation(dialog_id)
            )
            cancel_button.grid(row=3, column=0, padx=20, pady=(0, 20))

        self.progress_dialogs[dialog_id] = {
            "dialog": dialog,
            "progress_bar": progress_bar,
            "status_label": status_label,
            "operation_name": operation_name,
            "progress_type": progress_type,
            "total_steps": total_steps,
            "current_step": 0,
            "cancelable": cancelable,
            "cancel_callback": None,
            "cancel_button": cancel_button
        }
        
        return dialog_id
    
    def update_progress(self, dialog_id: str, step: int = None, 
                       status_text: Optional[str] = None,
                       increment: int = None) -> None:
        """
        Update the progress of an operation.
        
        Args:
            dialog_id: The dialog ID returned from create_progress_dialog
            step: Current step (for determinate progress)
            status_text: Optional status text to display
            increment: Optional increment to add to current step
        """
        if dialog_id not in self.progress_dialogs:
            return

        dialog_info = self.progress_dialogs[dialog_id]
        dialog = dialog_info["dialog"]
        # Check if dialog and widgets still exist
        if not dialog.winfo_exists() or not dialog_info["progress_bar"].winfo_exists() or not dialog_info["status_label"].winfo_exists():
            return

        # Update step if provided
        if step is not None:
            dialog_info["current_step"] = min(step, dialog_info["total_steps"])
        elif increment is not None:
            dialog_info["current_step"] = min(
                dialog_info["current_step"] + increment,
                dialog_info["total_steps"]
            )

        # Calculate progress value (0.0 to 1.0)
        if dialog_info["progress_type"] == self.PROGRESS_TYPE_DETERMINATE:
            progress_value = dialog_info["current_step"] / dialog_info["total_steps"]
            if dialog_info["progress_bar"].winfo_exists():
                dialog_info["progress_bar"].set(progress_value)

        # Update status text if provided
        if status_text and dialog_info["status_label"].winfo_exists():
            dialog_info["status_label"].configure(text=status_text)
    
    def complete_progress(self, dialog_id: str, success: bool = True, 
                         completion_message: Optional[str] = None) -> None:
        """
        Complete a progress operation and close the dialog.
        
        Args:
            dialog_id: The dialog ID returned from create_progress_dialog
            success: Whether the operation completed successfully
            completion_message: Optional completion message
        """
        if dialog_id not in self.progress_dialogs:
            return

        dialog_info = self.progress_dialogs[dialog_id]
        dialog = dialog_info["dialog"]
        # Check if dialog and widgets still exist
        if not dialog.winfo_exists() or not dialog_info["progress_bar"].winfo_exists() or not dialog_info["status_label"].winfo_exists():
            return

        # Stop any animation
        if dialog_id in self.stop_progress:
            self.stop_progress[dialog_id] = True

        # Set progress to 100% if successful
        if success and dialog_info["progress_type"] == self.PROGRESS_TYPE_DETERMINATE:
            if dialog_info["progress_bar"].winfo_exists():
                dialog_info["progress_bar"].set(1.0)

        # Update status text
        if completion_message and dialog_info["status_label"].winfo_exists():
            dialog_info["status_label"].configure(text=completion_message)
        elif success and dialog_info["status_label"].winfo_exists():
            dialog_info["status_label"].configure(text="Operation completed successfully.")
        elif dialog_info["status_label"].winfo_exists():
            # Check if this was a cancellation vs. a failure
            if completion_message and "canceled" in completion_message.lower():
                dialog_info["status_label"].configure(text="Operation was canceled.")
            else:
                dialog_info["status_label"].configure(text="Operation failed or was canceled.")

        # Show completion message in log
        operation_name = dialog_info["operation_name"]
        if success:
            self.show_success_notification(f"{operation_name} completed successfully")
        else:
            if hasattr(self.app, 'log_frame'):
                # Check if this was a cancellation vs. a failure
                if completion_message and "canceled" in completion_message.lower():
                    self.app.log_frame.log_message(f"INFO: {operation_name} was canceled by user\n")
                else:
                    self.app.log_frame.log_message(f"WARNING: {operation_name} was not completed\n")

        # Close the dialog after a short delay
        self.app.after(1500, lambda: self._close_progress_dialog(dialog_id))
    
    def set_cancel_callback(self, dialog_id: str, callback: Callable) -> None:
        """
        Set a callback function to be called when the operation is canceled.
        
        Args:
            dialog_id: The dialog ID returned from create_progress_dialog
            callback: The callback function
        """
        if dialog_id in self.progress_dialogs:
            self.progress_dialogs[dialog_id]["cancel_callback"] = callback
    
    def _cancel_operation(self, dialog_id: str) -> None:
        """
        Cancel an operation and close its progress dialog.
        
        Args:
            dialog_id: The dialog ID to cancel
        """
        if dialog_id not in self.progress_dialogs:
            return
            
        dialog_info = self.progress_dialogs[dialog_id]
        
        # Disable cancel button to prevent double-clicks
        if dialog_info.get("cancel_button") and dialog_info["cancel_button"].winfo_exists():
            dialog_info["cancel_button"].configure(state="disabled")
        
        # Call the cancel callback if provided
        if dialog_info["cancel_callback"]:
            dialog_info["cancel_callback"]()
        
        # Stop any animation
        if dialog_id in self.stop_progress:
            self.stop_progress[dialog_id] = True
        
        # Update status text
        dialog_info["status_label"].configure(text="Operation canceled.")
        
        # Log the cancellation
        operation_name = dialog_info["operation_name"]
        if hasattr(self.app, 'log_frame'):
            self.app.log_frame.log_message(f"INFO: {operation_name} was canceled by user\n")
        
        # Close the dialog after a short delay
        self.app.after(1000, lambda: self._close_progress_dialog(dialog_id))
    
    def _close_progress_dialog(self, dialog_id: str) -> None:
        """
        Close a progress dialog and clean up resources.
        
        Args:
            dialog_id: The dialog ID to close
        """
        if dialog_id not in self.progress_dialogs:
            return
            
        # Get the dialog window
        dialog = self.progress_dialogs[dialog_id]["dialog"]
        
        # Release grab and destroy the dialog
        dialog.grab_release()
        dialog.destroy()
        
        # Clean up resources
        if dialog_id in self.progress_threads:
            self.progress_threads.pop(dialog_id, None)
        if dialog_id in self.stop_progress:
            self.stop_progress.pop(dialog_id, None)
        
        # Remove from progress dialogs
        self.progress_dialogs.pop(dialog_id, None)
    
    def _start_indeterminate_animation(self, dialog_id: str, progress_bar) -> None:
        """
        Start an animation for indeterminate progress.
        
        Args:
            dialog_id: The dialog ID
            progress_bar: The progress bar widget
        """
        # Create a thread to animate the progress bar
        def animate_progress():
            direction = 1  # 1 for increasing, -1 for decreasing
            value = 0.0
            
            while not self.stop_progress.get(dialog_id, True):
                # Update progress value
                value += 0.02 * direction
                
                # Reverse direction at the ends
                if value >= 1.0:
                    value = 1.0
                    direction = -1
                elif value <= 0.0:
                    value = 0.0
                    direction = 1
                
                # Update the progress bar
                self.app.after(0, lambda v=value: progress_bar.set(v))
                
                # Sleep briefly
                time.sleep(0.05)
        
        # Start the animation thread
        thread = threading.Thread(target=animate_progress, daemon=True)
        self.progress_threads[dialog_id] = thread
        thread.start()
    
    def show_operation_progress_in_log(self, operation: str, progress: int, total: int) -> None:
        """
        Show operation progress in the console.

            # Redirect progress messages to console instead of UI log
            percentage = int((progress / total) * 100)

            filled_length = int(bar_length * progress / total)
            bar = 'Ôûê' * filled_length + 'Ôûæ' * (bar_length - filled_length)
            
            print(f"INFO: {operation}: {progress}/{total} ({percentage}%) [{bar}]")

"""
Parameter Error Feedback Service

This service provides user feedback for parameter processing errors,
including summary feedback when multiple parameters fail to process.
"""

import logging
from typing import Dict, List, Optional
from dataclasses import dataclass
from enum import Enum

from services.parameter_processor import ProcessingSummary, ErrorSeverity

logger = logging.getLogger(__name__)


class FeedbackLevel(Enum):
    """Levels of user feedback."""
    INFO = "info"
    WARNING = "warning"
    ERROR = "error"
    CRITICAL = "critical"


@dataclass
class UserFeedback:
    """User feedback message."""
    level: FeedbackLevel
    title: str
    message: str
    details: Optional[List[str]] = None
    parameter_count: Optional[int] = None


class ParameterErrorFeedbackService:
    """
    Provides user feedback for parameter processing errors and warnings.
    
    This service analyzes parameter processing results and generates appropriate
    user feedback messages, including summary feedback for multiple failures.
    """
    
    def __init__(self):
        """Initialize the feedback service."""
        self.feedback_history = []
    
    def generate_processing_feedback(self, summary: ProcessingSummary) -> List[UserFeedback]:
        """
        Generate user feedback based on parameter processing summary.
        
        Args:
            summary: ProcessingSummary from parameter processing operation
            
        Returns:
            List of UserFeedback messages for the user
            
        Requirements: 5.4 - Provide summary feedback for multiple failures
        """
        feedback_messages = []
        
        # Generate overall summary feedback
        if summary.failed_parameters == 0:
            # All parameters processed successfully
            if summary.total_parameters > 0:
                feedback_messages.append(UserFeedback(
                    level=FeedbackLevel.INFO,
                    title="Parameter Processing Complete",
                    message=f"Successfully processed all {summary.total_parameters} parameters from device.",
                    parameter_count=summary.total_parameters
                ))
        else:
            # Some or all parameters failed
            success_rate = (summary.successful_parameters / summary.total_parameters) * 100
            
            if summary.successful_parameters == 0:
                # Complete failure
                feedback_messages.append(UserFeedback(
                    level=FeedbackLevel.ERROR,
                    title="Parameter Processing Failed",
                    message=f"Failed to process any of the {summary.total_parameters} parameters from device.",
                    details=summary.errors[:5] if summary.errors else None,  # Limit to first 5 errors
                    parameter_count=summary.total_parameters
                ))
            else:
                # Partial failure
                feedback_level = FeedbackLevel.WARNING if success_rate >= 50 else FeedbackLevel.ERROR
                
                feedback_messages.append(UserFeedback(
                    level=feedback_level,
                    title="Parameter Processing Partially Complete",
                    message=f"Processed {summary.successful_parameters} of {summary.total_parameters} parameters successfully ({success_rate:.1f}% success rate).",
                    details=self._format_failure_details(summary),
                    parameter_count=summary.total_parameters
                ))
        
        # Generate specific warnings if present
        if summary.warnings:
            warning_count = len(summary.warnings)
            if warning_count <= 3:
                # Show individual warnings for small numbers
                for warning in summary.warnings:
                    feedback_messages.append(UserFeedback(
                        level=FeedbackLevel.WARNING,
                        title="Parameter Processing Warning",
                        message=warning
                    ))
            else:
                # Summarize warnings for large numbers
                feedback_messages.append(UserFeedback(
                    level=FeedbackLevel.WARNING,
                    title="Parameter Processing Warnings",
                    message=f"Encountered {warning_count} warnings during parameter processing.",
                    details=summary.warnings[:5]  # Show first 5 warnings
                ))
        
        # Store feedback in history
        self.feedback_history.extend(feedback_messages)
        
        return feedback_messages
    
    def _format_failure_details(self, summary: ProcessingSummary) -> List[str]:
        """
        Format failure details for user feedback.
        
        Args:
            summary: ProcessingSummary with error information
            
        Returns:
            List of formatted error details
        """
        details = []
        
        # Add error summary
        if summary.errors:
            error_count = len(summary.errors)
            if error_count <= 5:
                details.extend(summary.errors)
            else:
                details.extend(summary.errors[:3])
                details.append(f"... and {error_count - 3} more errors")
        
        # Add warning summary if there are many warnings
        if summary.warnings and len(summary.warnings) > 3:
            details.append(f"Additionally, {len(summary.warnings)} warnings were encountered")
        
        return details
    
    def generate_ui_update_error_feedback(self, parameter_name: str, error_message: str) -> UserFeedback:
        """
        Generate feedback for UI update errors.
        
        Args:
            parameter_name: Name of parameter that failed to update in UI

        self.feedback_history.append(feedback)

    def validate_connection_before_operation(self, operation_name: str = "parameter operation") -> ConnectionValidationResult:
        """
        Validate the connection before performing parameter operations.
        
        Args:
            operation_name: Name of the operation being performed
            
        Returns:
            ConnectionValidationResult with validation details
        """
        if not self.connection_validation_enabled:
            return ConnectionValidationResult(
                is_valid=True,
                error_message=None,
                connection_quality=100,
                last_activity=datetime.now(),
                suggested_actions=[]
            )

        # Enhanced connection validation with comprehensive checks
        validation_issues = []
        suggested_actions = []
        connection_quality = 100
        
        # Check basic connection
        if not hasattr(self.app, 'serial_service') or not self.app.serial_service.is_connected:
            validation_issues.append("Device is not connected")
            connection_quality = 0
            suggested_actions.extend([
                "Connect to the device using the Connection tab",
                "Verify the device is powered on and in configuration mode",
                "Check the USB cable connection is secure",
                "Select the correct COM port from the dropdown",
                "Try a different USB port if connection fails"
            ])
        else:
            # Check serial port status
            try:
                if hasattr(self.app.serial_service, 'serial_port') and self.app.serial_service.serial_port:
                    if not self.app.serial_service.serial_port.is_open:
                        validation_issues.append("Serial port is not open")
                        connection_quality = 0
                        suggested_actions.extend([
                            "Reconnect to the device",
                            "Check if another application is using the COM port",
                            "Restart the application if the port appears stuck"
                        ])
            except Exception as e:
                validation_issues.append(f"Serial port error: {str(e)}")
                connection_quality = 0
                suggested_actions.append("Check the device connection and try reconnecting")

        # Check connection quality
        if connection_quality > 0:
            connection_quality = self._assess_connection_quality()
        
        # Check last activity with more detailed analysis
        last_activity = getattr(self.app.serial_service, 'last_activity_time', None)
        if last_activity and isinstance(last_activity, datetime):
            try:
                time_since_activity = datetime.now() - last_activity
                
                if time_since_activity > timedelta(minutes=10):
                    validation_issues.append(f"No communication for {int(time_since_activity.total_seconds() / 60)} minutes")
                    connection_quality = max(0, connection_quality - 40)
                    suggested_actions.extend([
                        "Send a test command to verify device responsiveness",
                        "Check if the device has entered sleep mode",
                        "Try power cycling the device",
                        "Verify the device is still physically connected"
                    ])
                elif time_since_activity > timedelta(minutes=5):
                    validation_issues.append(f"Limited recent communication ({int(time_since_activity.total_seconds() / 60)} minutes ago)")
                    connection_quality = max(0, connection_quality - 20)
                    suggested_actions.extend([
                        "Test the connection by sending a simple command",
                        "Check if the device is still responsive"
                    ])
            except (TypeError, AttributeError):
                # Handle cases where last_activity is not a datetime object
                pass

        # Check for recent failures with detailed analysis
        try:
            consecutive_failures = int(self.consecutive_failures) if self.consecutive_failures is not None else 0
            
            if consecutive_failures >= 5:
                validation_issues.append(f"High failure rate detected ({consecutive_failures} consecutive failures)")
                connection_quality = max(0, connection_quality - 50)
                suggested_actions.extend([
                    "Check the device connection stability",
                    "Try restarting the device completely",
                    "Verify there's no electromagnetic interference",
                    "Consider using a different USB port or cable",
                    "Check if the device firmware needs updating"
                ])
            elif consecutive_failures >= 3:
                validation_issues.append(f"Multiple recent failures ({consecutive_failures} consecutive)")
                connection_quality = max(0, connection_quality - 30)
                suggested_actions.extend([
                    "Check the device connection stability",
                    "Try restarting the device",
                    "Verify there's no interference"
                ])
        except (TypeError, ValueError):
            # Handle cases where consecutive_failures is not a number
            pass

        # Check device responsiveness if we have a way to test it
        if connection_quality > 0 and hasattr(self.app, 'serial_service'):
            responsiveness_check = self._check_device_responsiveness()
            if not responsiveness_check['responsive']:
                validation_issues.append("Device not responding to test commands")
                connection_quality = max(0, connection_quality - 30)
                suggested_actions.extend(responsiveness_check['suggested_actions'])

        # Determine overall validation result
        is_valid = len(validation_issues) == 0 or connection_quality >= 50
        
        # Create comprehensive error message
        error_message = None
        if validation_issues:
            error_message = f"Connection validation failed for {operation_name}: " + "; ".join(validation_issues)
        
        # Log validation results
        self._log_connection_validation_results(operation_name, is_valid, connection_quality, validation_issues)
        
        return ConnectionValidationResult(
            is_valid=is_valid,
            error_message=error_message,
            connection_quality=connection_quality,
            last_activity=last_activity,
            suggested_actions=suggested_actions
        )
    
    def _check_device_responsiveness(self) -> Dict[str, Any]:
        """
        Check if the device is responsive by analyzing recent communication patterns.
        
        Returns:
            Dictionary with responsiveness status and suggested actions
        """
        try:
            # Check if we have recent successful operations
            recent_successes = []
            recent_timeouts = []
            
            try:
                recent_successes = [e for e in self.error_history if 
                                  hasattr(e, 'timestamp') and isinstance(e.timestamp, datetime) and
                                  (datetime.now() - e.timestamp).total_seconds() < 60 and
                                  hasattr(e, 'error_type') and
                                  e.error_type not in [ParameterErrorType.COMMUNICATION_TIMEOUT, 
                                                     ParameterErrorType.DEVICE_NOT_RESPONDING]]
                
                # Check if we have recent timeouts
                recent_timeouts = [e for e in self.error_history if 
                                 hasattr(e, 'timestamp') and isinstance(e.timestamp, datetime) and
                                 (datetime.now() - e.timestamp).total_seconds() < 300 and
                                 hasattr(e, 'error_type') and
                                 e.error_type == ParameterErrorType.COMMUNICATION_TIMEOUT]
            except (TypeError, AttributeError):
                pass
            
            if len(recent_timeouts) > 2:
                return {
                    'responsive': False,
                    'suggested_actions': [
                        "Device appears unresponsive - try power cycling",
                        "Check if device is in sleep mode or low power state",
                        "Verify device is in configuration mode",
                        "Try increasing communication timeout settings"
                    ]
                }
            elif len(recent_successes) > 0:
                return {
                    'responsive': True,
                    'suggested_actions': []
                }
            else:
                return {
                    'responsive': True,  # Assume responsive if no recent data
                    'suggested_actions': []
                }
                
        except Exception as e:
            return {
                'responsive': True,  # Default to responsive on error
                'suggested_actions': []
            }

    def handle_parameter_error(self, error_type: ParameterErrorType, parameter_name: str,
                             command_sent: str, response_received: str = "",
                             context: Dict[str, Any] = None) -> ParameterError:
        """
        Handle a parameter operation error with comprehensive logging and user feedback.
        
        Args:
            error_type: Type of error that occurred
            parameter_name: Name of the parameter being processed
            command_sent: Command that was sent to the device
            response_received: Response received from device (if any)
            context: Additional context information
            
        Returns:
            ParameterError object with error details
        """
        if context is None:
            context = {}

        # Determine error severity
        severity = self._determine_error_severity(error_type, context)
        
        # Create error object
        error = ParameterError(
            error_type=error_type,
            severity=severity,
            parameter_name=parameter_name,
            command_sent=command_sent,
            response_received=response_received,
            error_message=self._format_error_message(error_type, parameter_name, command_sent, response_received, context),
            suggested_actions=self.corrective_actions.get(error_type, []),
            timestamp=datetime.now(),
            context=context
        )

        # Add to error history
        self.error_history.append(error)
        
        # Update failure tracking
        self._update_failure_tracking(error_type)
        
        # Log the error with appropriate detail level
        self._log_parameter_error(error)
        
        # Show user notification based on severity
        self._show_error_notification(error)
        
        # Trigger automatic recovery if appropriate
        self._attempt_automatic_recovery(error)
        
        return error

    def handle_communication_timeout(self, parameter_name: str, command_sent: str, 
                                   timeout_duration: float) -> ParameterError:
        """
        Handle communication timeout errors with specific timeout information.
        
        Args:
            parameter_name: Name of the parameter
            command_sent: Command that timed out
            timeout_duration: How long we waited before timing out
            
        Returns:
            ParameterError object
        """
        context = {
            'timeout_duration': timeout_duration,
            'timeout_threshold': self.timeout_threshold,
            'retry_suggested': timeout_duration < self.timeout_threshold * 2,
            'connection_check_needed': timeout_duration >= self.timeout_threshold
        }
        
        # Enhanced timeout handling with specific corrective actions
        error = self.handle_parameter_error(
            ParameterErrorType.COMMUNICATION_TIMEOUT,
            parameter_name,
            command_sent,
            "",
            context
        )
        
        # Add timeout-specific corrective actions
        timeout_actions = [
            f"Wait {int(self.timeout_threshold)}s and retry the parameter operation",
            "Check if the device is responding to other commands",
            "Verify the device is not in sleep mode or low power state",
            "Try increasing the communication timeout in settings",
            "Check for interference on the communication line"
        ]
        
        # Replace generic actions with timeout-specific ones
        error.suggested_actions = timeout_actions
        
        # Log detailed timeout information
        self._log_timeout_details(parameter_name, command_sent, timeout_duration)
        
        return error

    def handle_invalid_response(self, parameter_name: str, command_sent: str,
                              response_received: str, expected_format: str = "",
                              parsing_error: str = "") -> ParameterError:
        """
        Handle invalid device response errors with detailed response information.
        
        Args:
            parameter_name: Name of the parameter
            command_sent: Command that was sent
            response_received: Invalid response received
            expected_format: Expected response format
            parsing_error: Specific parsing error message
            
        Returns:
            ParameterError object
        """
        context = {
            'expected_format': expected_format,
            'parsing_error': parsing_error,
            'response_length': len(response_received),
            'response_type': self._analyze_response_type(response_received),
            'possible_causes': self._analyze_response_issues(response_received, expected_format)
        }
        
        # Enhanced invalid response handling
        error = self.handle_parameter_error(
            ParameterErrorType.INVALID_RESPONSE,
            parameter_name,
            command_sent,
            response_received,
            context
        )
        
        # Add response-specific corrective actions
        response_actions = self._get_response_specific_actions(response_received, expected_format, parsing_error)
        error.suggested_actions = response_actions
        
        # Log detailed response analysis
        self._log_response_analysis(parameter_name, command_sent, response_received, expected_format, parsing_error)
        
        return error
    
    def _analyze_response_type(self, response: str) -> str:
        """
        Analyze the type of response received to provide better error context.
        
        Args:
            response: Response received from device
            
        Returns:
            String describing the response type
        """
        if not response:
            return "empty_response"
        elif response.startswith("ERROR"):
            return "error_response"
        elif response.startswith("OK"):
            return "ok_response"
        elif ":" in response:
            return "formatted_response"
        elif response.isdigit():
            return "numeric_response"
        elif len(response) < 5:
            return "short_response"
        else:
            return "unknown_response"
    
    def _analyze_response_issues(self, response: str, expected_format: str) -> List[str]:
        """
        Analyze potential issues with the response.
        
        Args:
            response: Response received
            expected_format: Expected format
            
        Returns:
            List of possible causes for the invalid response
        """
        issues = []
        
        if not response:
            issues.append("Device sent empty response")
        elif response.startswith("ERROR"):
            issues.append("Device returned error response")
        elif expected_format and ":" in expected_format and ":" not in response:
            issues.append("Response missing expected delimiter ':'")
        elif expected_format and "=" in expected_format and "=" not in response:
            issues.append("Response missing expected delimiter '='")
        elif len(response) < 3:
            issues.append("Response too short to be valid")
        elif response.count(",") != expected_format.count(",") if expected_format else False:
            issues.append("Response has wrong number of comma-separated values")
        
        return issues

        """

    def _assess_connection_quality(self) -> int:
        """
        Assess the current connection quality based on recent activity and errors.
        
        Returns:
            Connection quality score (0-100)
        """
        base_quality = 100
        
        # Reduce quality based on recent errors
        try:
            recent_errors = [e for e in self.error_history if 
                            hasattr(e, 'timestamp') and isinstance(e.timestamp, datetime) and
                            (datetime.now() - e.timestamp).total_seconds() < 300]  # Last 5 minutes
            
            quality_reduction = min(len(recent_errors) * 10, 50)
            base_quality -= quality_reduction
        except (TypeError, AttributeError):
            pass
        
        # Reduce quality based on consecutive failures
        try:
            consecutive_failures = int(self.consecutive_failures) if self.consecutive_failures is not None else 0
            consecutive_penalty = min(consecutive_failures * 15, 60)
            base_quality -= consecutive_penalty
        except (TypeError, ValueError):
            pass
        
        # Check if we have error handler with connection quality
        if hasattr(self.app, 'error_handler') and hasattr(self.app.error_handler, 'connection_quality'):
            try:
                # Use the main error handler's quality assessment
                main_quality = self.app.error_handler.connection_quality
                if isinstance(main_quality, (int, float)):
                    base_quality = min(base_quality, main_quality)
            except (TypeError, AttributeError):
                # Handle cases where connection_quality is not a number
                pass
        
        return max(0, base_quality)

    def _determine_error_severity(self, error_type: ParameterErrorType, context: Dict[str, Any]) -> ParameterErrorSeverity:
        """
        Determine the severity of an error based on type and context.
        
        Args:
            error_type: Type of error
            context: Error context
            
        Returns:
            Error severity level
        """
        # Critical errors that prevent all operations
        if error_type in [ParameterErrorType.CONNECTION_LOST]:
            return ParameterErrorSeverity.CRITICAL
        
        # Errors that affect multiple operations
        if error_type in [ParameterErrorType.DEVICE_NOT_RESPONDING, ParameterErrorType.DEVICE_BUSY]:
            return ParameterErrorSeverity.ERROR
        
        # Errors that affect single operations but may be recoverable
        if error_type in [ParameterErrorType.COMMUNICATION_TIMEOUT, ParameterErrorType.INVALID_RESPONSE]:
            # Check retry count to escalate severity
            try:
                retry_count = int(context.get('retry_count', 0))
                max_retries = int(self.max_retry_attempts)
                if retry_count >= max_retries:
                    return ParameterErrorSeverity.ERROR
                else:
                    return ParameterErrorSeverity.WARNING
            except (TypeError, ValueError):

        return ParameterErrorSeverity.WARNING

            # The actual retry would be handled by the calling service

        if log_entry.success:

        else:

            # Error categorization
            if log_entry.error_message:
                error_category = self._categorize_error(log_entry.error_message)
                self.error_stats[error_category] += 1

        param_stats = self.parameter_stats[log_entry.parameter_name]

            param_stats['reads'] += 1
        elif log_entry.operation_type in ['write', 'bulk_write']:
            param_stats['writes'] += 1
        
        if not log_entry.success:

    def _categorize_error(self, error_message: str) -> str:

            return 'format'

            return 'permission'
        elif 'validation' in error_lower or 'range' in error_lower:
            return 'validation'
        else:

                'operation_id': operation_id,
                'parameter': parameter_name,
                'success': success,
                'error': error_message,

                'successful': operation['successful'],
                'failed': operation['failed']

        """
        Log completion of a bulk operation and return summary.
        
        Args:
            operation_id: ID of the bulk operation
            
        Returns:
            Dictionary with operation summary
        """
        with self._lock:
            if operation_id not in self.active_operations:
                return {}
            
            operation = self.active_operations[operation_id]
            end_time = datetime.now()
            duration = (end_time - operation['start_time']).total_seconds() * 1000
            
            summary = {
                'operation_id': operation_id,
                'type': operation['type'],
                'total_parameters': len(operation['parameters']),
                'processed': operation['processed'],
                'successful': operation['successful'],
                'failed': operation['failed'],
                'duration_ms': duration,
                'success_rate': (operation['successful'] / operation['processed'] * 100) if operation['processed'] > 0 else 0
            }
            
            # Remove from active operations
            del self.active_operations[operation_id]
            
            self.log_system_event(f"Bulk operation completed: {operation['type']}", summary)
            
            return summary

    def get_operation_statistics(self, time_range: Optional[timedelta] = None) -> Dict[str, Any]:
        """
        Get comprehensive operation statistics.
        
        Args:
            time_range: Optional time range to filter statistics (e.g., last hour)
            
        Returns:
            Dictionary with detailed statistics
        """
        with self._lock:
            # Filter logs by time range if specified
            if time_range:
                cutoff_time = datetime.now() - time_range
                filtered_logs = [log for log in self.operation_logs if log.timestamp >= cutoff_time]
            else:
                filtered_logs = list(self.operation_logs)
            
            if not filtered_logs:
                return {
                    'total_operations': 0,
                    'successful_operations': 0,
                    'failed_operations': 0,
                    'success_rate': 0.0,
                    'average_duration_ms': 0.0,
                    'operation_types': {},
                    'error_categories': {},
                    'parameter_stats': {},
                    'time_range': str(time_range) if time_range else 'all_time'
                }
            
            # Calculate statistics
            total_ops = len(filtered_logs)
            successful_ops = sum(1 for log in filtered_logs if log.success)
            failed_ops = total_ops - successful_ops
            
            # Operation types
            op_types = defaultdict(int)
            for log in filtered_logs:
                op_types[log.operation_type] += 1
            
            # Error categories
            error_categories = defaultdict(int)
            for log in filtered_logs:
                if not log.success and log.error_message:
                    category = self._categorize_error(log.error_message)
                    error_categories[category] += 1
            
            # Parameter statistics
            param_stats = defaultdict(lambda: {'reads': 0, 'writes': 0, 'errors': 0, 'avg_duration_ms': 0.0})
            param_durations = defaultdict(list)
            
            for log in filtered_logs:
                param = log.parameter_name
                param_durations[param].append(log.duration_ms)
                
                if log.operation_type in ['read', 'bulk_read']:
                    param_stats[param]['reads'] += 1
                elif log.operation_type in ['write', 'bulk_write']:
                    param_stats[param]['writes'] += 1
                
                if not log.success:
                    param_stats[param]['errors'] += 1
            
            # Calculate average durations
            for param, durations in param_durations.items():
                param_stats[param]['avg_duration_ms'] = sum(durations) / len(durations)
            
            # Overall average duration
            all_durations = [log.duration_ms for log in filtered_logs]
            avg_duration = sum(all_durations) / len(all_durations) if all_durations else 0.0
            
            return {
                'total_operations': total_ops,
                'successful_operations': successful_ops,
                'failed_operations': failed_ops,
                'success_rate': (successful_ops / total_ops * 100) if total_ops > 0 else 0.0,
                'average_duration_ms': avg_duration,
                'operation_types': dict(op_types),
                'error_categories': dict(error_categories),
                'parameter_stats': dict(param_stats),
                'time_range': str(time_range) if time_range else 'all_time',
                'session_id': self.current_session_id,
                'log_count': total_ops
            }
